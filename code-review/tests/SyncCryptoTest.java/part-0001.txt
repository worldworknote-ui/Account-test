4: /** 10.0: تشفير المزامنة والساعة المنطقية (بلا قاعدة بيانات). */
5: public class SyncCryptoTest {
6:     static int fails, passes;
7: 
8:     static void check(boolean c, String m) {
9:         if (c) passes++;
10:         else fails++;

12:     }

14:     public static void main(String[] a) throws Exception {
15:         byte[] vk = SyncCrypto.random(32);
16:         SyncCrypto.Keys k = new SyncCrypto.Keys("v123", vk);
17:         byte[] msg = "{\"ops\":[1,2,3],\"name\":\"النور\"}".getBytes(StandardCharsets.UTF_8);
18:         String name = k.fileName("p-", "dev:1");
19:         byte[] sealed = k.sealFile("p", name, msg);
20:         check(sealed.length % 4096 == 0, "الحجم مضاعف لـ 4KB: " + sealed.length);
21:         check(new String(k.openFile("p", name, sealed), StandardCharsets.UTF_8).contains("النور"), "يُفك بالمفتاح نفسه");
22:         boolean tamper = false;
23:         byte[] bad = sealed.clone();
24:         bad[100] ^= 1;
25:         try {
26:             k.openFile("p", name, bad);
27:         } catch (Exception e) {
28:             tamper = true;
29:         }
30:         check(tamper, "تغيير بايت واحد يُرفض");
31:         boolean moved = false;
32:         try {
33:             k.openFile("p", k.fileName("p-", "dev:2"), sealed);
34:         } catch (Exception e) {
35:             moved = true;
36:         }
37:         check(moved, "نقل الملف إلى اسم آخر (إعادة ترتيب) يُرفض");

40:             new SyncCrypto.Keys("v123", SyncCrypto.random(32)).openFile("p", name, sealed);

46:         check(name.matches("p-[0-9a-f]{32}\\.sab") && name.equals(k.fileName("p-", "dev:1")), "اسم معتم ثابت");

58:         System.out.println("\n" + passes + " نجح، " + fails + " فشل");
