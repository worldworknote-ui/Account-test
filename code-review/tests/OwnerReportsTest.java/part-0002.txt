187:         TimeZone.setDefault(TimeZone.getTimeZone("Asia/Damascus"));   // تواريخ الملف بتوقيت المكتب
188:         JSONObject j = new JSONObject(new String(Files.readAllBytes(p), StandardCharsets.UTF_8));
189:         OwnerReports.Input in = new OwnerReports.Input();
190:         in.now = 1791838800000L;   // 2026-10-13 00:00 بتوقيت دمشق
191:         JSONArray cs = j.getJSONArray("currencies");
192:         for (int i = 0; i < cs.length(); i++) in.decimals.put(cs.getJSONObject(i).getString("code"), cs.getJSONObject(i).getInt("decimals"));
193:         JSONArray os = j.getJSONArray("offices");
194:         for (int i = 0; i < os.length(); i++) in.names.put(os.getJSONObject(i).getLong("id"), os.getJSONObject(i).getString("name"));
195:         JSONArray ts = j.getJSONArray("tx");

200:             t.uuid = o.optString("uuid");

202:             t.type = o.getString("type");

217:         for (int i = 0; ob != null && i < ob.length(); i++) {
218:             JSONObject o = ob.getJSONObject(i);

226:     static void real(Path p, String out) throws Exception {
227:         System.out.println("== بيانات حقيقية: " + p.getFileName());
228:         OwnerReports.Input in = load(p);
229:         long t0 = System.currentTimeMillis();
230:         Report r = OwnerReports.build(in, "", true);
231:         long ms = System.currentTimeMillis() - t0;
232:         for (String h : OwnerReports.headlines(in)) System.out.println("    " + h);
233:         check(ms < 5000, "كل التقارير على " + in.txs.size() + " حركة في " + ms + " ms");
234:         boolean sums = true;
235:         for (boolean us : new boolean[]{true, false})
236:             for (OwnerReports.Aging x : OwnerReports.aging(in, us)) {
237:                 long s = 0;
238:                 for (long b : x.buckets) s += b;
239:                 sums &= s == x.balance && x.balance > 0;
240:             }
241:         check(sums, "أعمار الديون: مجموع الفئات = الرصيد لكل مكتب وعملة");
242:         // ما لنا وما علينا في التقرير = رصيد حساب المكتب في دفتر القيود نفسه (journal_lines) لكل مكتب وعملة
243:         Map<String, Long> ledger = new HashMap<>();
244:         JSONArray jl = lastFile.getJSONArray("journal_lines");
245:         for (int i = 0; i < jl.length(); i++) {
246:             JSONObject l = jl.getJSONObject(i);
247:             if (!"OFFICE".equals(l.optString("account"))) continue;
248:             ledger.merge(l.getLong("office_id") + "|" + l.getString("currency"), l.optLong("debit_minor") - l.optLong("credit_minor"), Long::sum);
249:         }
250:         ledger.values().removeIf(v -> v == 0);
251:         Map<String, Long> fromAging = new HashMap<>();
252:         for (OwnerReports.Aging x : OwnerReports.aging(in, true)) fromAging.put(x.office + "|" + x.currency, x.balance);
253:         for (OwnerReports.Aging x : OwnerReports.aging(in, false)) fromAging.put(x.office + "|" + x.currency, -x.balance);
254:         check(ledger.equals(fromAging), "أرصدة التقرير = أرصدة المكاتب في دفتر القيود (عدد الأرصدة " + ledger.size() + ")");
255:         ByteArrayOutputStream bo = new ByteArrayOutputStream();
256:         XlsxWriter.write(r, bo, null, TimeZone.getTimeZone("UTC"));
257:         Files.write(Paths.get(out, "owner-reports-real.xlsx"), bo.toByteArray());
258:     }
