317:     static byte[] derToRaw(byte[] der) {
318:         int i = 2;
319:         if ((der[1] & 0x80) != 0) i += der[1] & 0x7f;
320:         byte[] out = new byte[64];
321:         for (int part = 0; part < 2; part++) {
322:             int len = der[i + 1];
323:             int start = i + 2;
324:             int skip = len > 32 ? len - 32 : 0;
325:             System.arraycopy(der, start + skip, out, part * 32 + (32 - (len - skip)), len - skip);
326:             i = start + len;
327:         }
328:         return out;
329:     }

331:     static byte[] rawToDer(byte[] raw) {
332:         byte[][] parts = new byte[2][];
333:         for (int p = 0; p < 2; p++) {
334:             int s = p * 32;
335:             while (s < p * 32 + 31 && raw[s] == 0) s++;
336:             int len = p * 32 + 32 - s;
337:             boolean pad = (raw[s] & 0x80) != 0;
338:             byte[] v = new byte[len + (pad ? 1 : 0)];
339:             System.arraycopy(raw, s, v, pad ? 1 : 0, len);
340:             parts[p] = v;
341:         }
342:         int total = 2 + parts[0].length + 2 + parts[1].length;
343:         byte[] der = new byte[2 + total];
344:         der[0] = 0x30;
345:         der[1] = (byte) total;
346:         int i = 2;
347:         for (byte[] v : parts) {
348:             der[i++] = 0x02;
349:             der[i++] = (byte) v.length;
350:             System.arraycopy(v, 0, der, i, v.length);
351:             i += v.length;
352:         }
353:         return der;
354:     }
