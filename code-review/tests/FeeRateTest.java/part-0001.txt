1: import com.smart.hawala.*;
2: import java.math.BigDecimal;
3: /** 7.1.2: الأجرة كنسبة من المبلغ الأصلي (بالمئة أو بالألف). */
4: public class FeeRateTest {
5:   static int fails = 0;
6:   static void check(boolean c, String m) { System.out.println((c ? "  PASS " : "  FAIL ") + m); if (!c) fails++; }
7:   static boolean throwsIAE(Runnable r) { try { r.run(); return false; } catch (IllegalArgumentException e) { return true; } }
8:   public static void main(String[] a) {
9:     BigDecimal one = Model.parseRate("1"), oneHalf = Model.parseRate("1.5");
10:     check(one != null && one.compareTo(BigDecimal.ONE) == 0, "parse 1");
11:     check(oneHalf != null && oneHalf.compareTo(new BigDecimal("1.5")) == 0, "parse 1.5");
12:     check(new BigDecimal("1.5").compareTo(Model.parseRate("١٫٥")) == 0, "Arabic digits and Arabic decimal sign");
13:     check(new BigDecimal("1.5").compareTo(Model.parseRate("1,5")) == 0, "comma as decimal");
14:     check(new BigDecimal("1.5").compareTo(Model.parseRate(" 1.5 % ")) == 0, "percent sign and spaces ignored");
15:     check(new BigDecimal("2").compareTo(Model.parseRate("2‰")) == 0, "per-mille sign ignored");
16:     check(new BigDecimal("0.25").compareTo(Model.parseRate(".25")) == 0, "leading dot");
17:     for (String bad : new String[]{"", "0", "0.0", "-1", "abc", "1.2.3", "1.23456", "12345"})
18:       check(Model.parseRate(bad) == null, "rejects [" + bad + "]");
19: 
20:     // USD بمنزلتين عشريتين: الوحدة الصغرى سنت
21:     check(Model.feeFromRate(500000, one, Model.PER_CENT) == 5000, "5000.00 × 1% = 50.00");
22:     check(Model.feeFromRate(200000, oneHalf, Model.PER_MILLE) == 300, "2000.00 × 1.5‰ = 3.00");
23:     check(Model.feeFromRate(55000, one, Model.PER_CENT) == 550 && !Model.feeRounded(55000, one, Model.PER_CENT), "550.00 × 1% = 5.50 exact");
24:     // ليرة سورية بلا منازل عشرية
25:     check(Model.feeFromRate(1000000, oneHalf, Model.PER_MILLE) == 1500, "1,000,000 SYP × 1.5‰ = 1500");
26:     // تقريب نصف للأعلى لأقرب وحدة صغرى
27:     check(Model.feeFromRate(5050, one, Model.PER_CENT) == 51 && Model.feeRounded(5050, one, Model.PER_CENT), "50.50 × 1% = 0.505 -> 0.51 (rounded)");
28:     check(Model.feeFromRate(33333, one, Model.PER_CENT) == 333 && Model.feeRounded(33333, one, Model.PER_CENT), "333.33 × 1% = 3.3333 -> 3.33");
29:     check(Model.feeFromRate(100, one, Model.PER_MILLE) == 0, "100 SYP × 1‰ = 0.1 -> 0 (caller refuses a zero fee)");
30:     check(Model.feeFromRate(99999999999L, new BigDecimal("3"), Model.PER_CENT) == 3000000000L, "large amount: no overflow");
31:     check(Model.feeFromRate(100000, new BigDecimal("100"), Model.PER_CENT) == 100000, "100% = the amount itself");
32:     check(throwsIAE(() -> Model.feeFromRate(100000, new BigDecimal("101"), Model.PER_CENT)), "rate above 100% refused");
33:     check(throwsIAE(() -> Model.feeFromRate(100000, new BigDecimal("1001"), Model.PER_MILLE)), "rate above 1000‰ refused");
34:     check(throwsIAE(() -> Model.feeFromRate(0, one, Model.PER_CENT)), "zero amount refused");
35:     check(throwsIAE(() -> Model.feeFromRate(100, one, 10)), "unknown base refused");
36:     check(Model.rateLabel(new BigDecimal("1.50"), Model.PER_MILLE).equals("1.5‰") && Model.rateLabel(one, Model.PER_CENT).equals("1٪"), "labels");
37:     // الأجرة بالعملة نفسها: نفس الوحدة الصغرى، فالمجموع مع المبلغ دقيق
38:     long amt = 123456789, fee = Model.feeFromRate(amt, oneHalf, Model.PER_CENT);
39:     check(fee == 1851852, "1,234,567.89 × 1.5% = 18,518.52 (got " + fee + ")");
40:     System.out.println(fails == 0 ? "ALL FEE RATE TESTS PASSED" : fails + " FAILED");
41:     System.exit(fails == 0 ? 0 : 1);
42:   }
43: }
