58:   function readBook(file) {
59:     return file.arrayBuffer().then(function (ab) {
60:       var buf = new Uint8Array(ab);
61:       var wb;
62:       var lower = file.name.toLowerCase();
63:       if (/\.(csv|txt|tsv)$/.test(lower)) {
64:         var text = decodeText(buf).replace(/^\uFEFF/, "");
65:         var semi = (text.split("\n")[0].match(/;/g) || []).length > (text.split("\n")[0].match(/,/g) || []).length;
66:         wb = XLSX.read(text, { type: "string", raw: true, FS: lower.endsWith(".tsv") ? "\t" : (semi ? ";" : undefined) });
67:       } else {
68:         wb = XLSX.read(buf, { type: "array", cellNF: true, cellDates: false });
69:       }
70:       return { name: file.name, sheets: wb.SheetNames.map(function (n) { return { name: n, rows: C.sheetRows(wb.Sheets[n], XLSX) }; }) };
