1: // كشوفات حساب نموذجية بالشكل نفسه الذي يصدّره البرنامج القديم: ملف لكل حساب وعملة،
2: // أعمدة متباعدة (E, K, O, U, AA)، «Pre Balance»، صف عناوين يتكرر كل صفحة، وسطر «اجمالي الرصيد».
3: "use strict";
4: function fmt(n) {
5:   const neg = n < 0;
6:   const s = Math.abs(n).toFixed(2).replace(/\B(?=(\d{3})+(?!\d))/g, ",");
7:   return (neg ? "-" : "") + s;
8: }
9: function row(cells) {
10:   const r = new Array(30).fill(null);
11:   Object.keys(cells).forEach((k) => { r[+k] = cells[k]; });
12:   return r;
13: }
14: const HEAD = row({ 4: "البيان", 10: "الرصيد", 14: "دائن / علينا", 20: "مدين / لنا", 26: "تاريخ الحركة" });
15: /**
16:  * spec: { account, currency, from, to, pre: {bal, credit, debit} | null, rows: [[date, desc, credit, debit, bal]], foot: [amount, sideText] | null,
17:  *         pageEvery, sheets (عدد الأوراق) }
18:  */
19: function statementBook(XLSX, spec) {
20:   const top = [
21:     row({}),
22:     row({ 15: spec.from, 22: ":", 23: "تاريخ البداية" }),
23:     row({ 15: spec.to, 22: ":", 23: "تاريخ النهاية" }),
24:     row({}), row({}),
25:     row({ 3: spec.account }),
26:     row({ 5: spec.currency }),
27:     row({}),
28:     HEAD
29:   ];
30:   const body = [];
31:   if (spec.pre) body.push(row({ 4: "Pre Balance", 10: fmt(spec.pre.bal), 14: fmt(spec.pre.credit), 20: fmt(spec.pre.debit), 26: spec.from }));
32:   spec.rows.forEach((r, i) => {
33:     if (spec.pageEvery && i > 0 && i % spec.pageEvery === 0) body.push(HEAD);
34:     body.push(row({ 4: r[1], 10: fmt(r[4]), 14: fmt(r[2]), 20: fmt(r[3]), 26: r[0] }));
35:   });
36:   const end = [];
37:   if (spec.foot) end.push(row({ 1: "مبلغ بالحروف", 9: fmt(spec.foot[0]), 11: ":", 12: "اجمالي الرصيد", 19: spec.foot[1], 24: ":", 25: "نوع الرصيد" }));
38:   end.push(row({}), row({ 2: "شاكرين حسن تعاونكم" }), row({ 17: "مدير" }));
39:   const wb = XLSX.utils.book_new();
40:   const all = top.concat(body).concat(end);
41:   const n = spec.sheets || 1;
42:   const per = Math.ceil(all.length / n);
43:   for (let i = 0; i < n; i++) {
44:     const part = all.slice(i * per, (i + 1) * per);
45:     if (i > 0) part.unshift(HEAD);
46:     XLSX.utils.book_append_sheet(wb, XLSX.utils.aoa_to_sheet(part), "Page " + (i + 1));
47:   }
48:   return wb;
49: }
