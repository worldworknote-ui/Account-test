455:   /** قسم الأرصدة الافتتاحية: صف لكل مكتب (وعملة). */
456:   function buildOpenings(model, cfg, sec, sheets) {
457:     sheets.forEach(function (sh) {
458:       var rows = sh.rows;
459:       for (var r = sec.headerRow + 1; r < rows.length; r++) {
460:         var row = rows[r];
461:         var src = { sheet: sh.name, row: r + 1 };
462:         if (isBlankRow(row)) continue;
463:         model.rowsRead++;
464:         var name = text(row[sec.cols.office]);
465:         var label = rowLabel(row);
466:         if (TOTAL_RE.test(name) || (!name && TOTAL_RE.test(label))) { skip(model, src, "صف مجموع"); continue; }
467:         var values = [];
468:         if (sec.layout === "wide") {
469:           (sec.wide || []).forEach(function (w) {
470:             if (w.code && w.code !== "__skip") values.push({ code: w.code, v: row[w.col], signed: true });
471:           });
472:         } else {
473:           var code = rowCurrency(cfg, sec, row, model, src);
474:           if (code === null) continue;
475:           if (code === "__skip") { skip(model, src, "عملة مستبعدة"); continue; }
476:           if (sec.layout === "split") {
477:             values.push({ code: code, v: row[sec.cols.debit], dir: "D" });
478:             values.push({ code: code, v: row[sec.cols.credit], dir: "C" });
479:           } else values.push({ code: code, v: row[sec.cols.amount], signed: true });
480:         }
481:         var any = false, bad = false, net = {};
482:         values.forEach(function (x) {
483:           var p = parseAmount(x.v);
484:           if (p.empty) return;
485:           if (!p.ok) { err(model, src, "مبلغ غير مفهوم: «" + text(x.v) + "» — " + p.reason); bad = true; return; }
486:           if (p.zero) return;
487:           if (!x.signed && p.neg) { err(model, src, "قيمة سالبة في عمود " + (x.dir === "D" ? "لنا/مدين" : "علينا/دائن") + ": «" + text(x.v) + "»"); bad = true; return; }
488:           var m = amountMinor(cfg, model, src, p, x.code, "المبلغ");
489:           if (m === null) { bad = true; return; }
490:           var pos = sec.positiveIs || "D";
491:           var dir = x.signed ? (p.neg ? (pos === "D" ? "C" : "D") : pos) : x.dir;
492:           dir = flip(dir, cfg);
493:           net[x.code] = (net[x.code] || 0n) + (dir === "D" ? m : -m);
494:           any = true;
495:         });
496:         if (bad) continue;
497:         if (!name) {
498:           if (any) err(model, src, "يوجد رصيد بلا اسم مكتب");
499:           else skip(model, src, "صف بلا مكتب ولا رصيد");
500:           continue;
501:         }
502:         var oi = officeFor(model, name, sec.cols.phone !== undefined ? text(row[sec.cols.phone]) : "", cfg);
503:         if (!any) { skip(model, src, "مكتب بلا رصيد (أُضيف المكتب فقط)"); model.offices[oi].emptyOk = true; continue; }
504:         Object.keys(net).forEach(function (code) {
505:           if (net[code] === 0n) return;
506:           model.openings.push({ office: oi, code: code, minor: net[code], when: sec.date || null, src: src });
507:           model.offices[oi].openings++;
508:           model.used[code] = true;
509:         });
510:       }
511:     });
512:   }
