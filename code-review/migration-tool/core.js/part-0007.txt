1075:   function uuid(rand) {
1076:     var b = rand(16);
1077:     b[6] = (b[6] & 0x0f) | 0x40;
1078:     b[8] = (b[8] & 0x3f) | 0x80;
1079:     var h = Array.prototype.map.call(b, function (x) { return (x < 16 ? "0" : "") + x.toString(16); }).join("");
1080:     return h.slice(0, 8) + "-" + h.slice(8, 12) + "-" + h.slice(12, 16) + "-" + h.slice(16, 20) + "-" + h.slice(20);
1081:   }
