1290:   async function encrypt(plain, password, opts, subtle, rand) {
1291:     opts = opts || {};
1292:     var iterations = opts.iterations || 310000;
1293:     var chunk = opts.chunk || 262144;
1294:     var salt = opts.salt || rand(16);
1295:     var prefix = opts.prefix || rand(8);
1296:     var header = concat([utf8("SMARTBK5"), be32(1), be32(iterations), be32(16), salt, be32(8), prefix, be32(chunk)]);
1297:     var base = await subtle.importKey("raw", utf8(password), "PBKDF2", false, ["deriveKey"]);
1298:     var key = await subtle.deriveKey({ name: "PBKDF2", hash: "SHA-256", salt: salt, iterations: iterations }, base,
1299:       { name: "AES-GCM", length: 256 }, false, ["encrypt"]);
1300:     var out = [header];
1301:     var v = new Uint8Array(await subtle.encrypt({ name: "AES-GCM", iv: iv(prefix, 0xffffffff), additionalData: concat([header, utf8("VERIFY")]), tagLength: 128 },
1302:       key, new Uint8Array(16)));
1303:     out.push(be32(v.length), v);
1304:     var count = Math.max(1, Math.ceil(plain.length / chunk));
1305:     for (var i = 0; i < count; i++) {
1306:       var last = i === count - 1 ? 1 : 0;
1307:       var part = plain.subarray(i * chunk, Math.min(plain.length, (i + 1) * chunk));
1308:       var aad = new Uint8Array(header.length + 9);
1309:       aad.set(header, 0);
1310:       // رقم الجزء 8 بايت (long) ثم علامة «الأخير»
1311:       aad.set(be32(0), header.length);
1312:       aad.set(be32(i), header.length + 4);
1313:       aad[header.length + 8] = last;
1314:       var ct = new Uint8Array(await subtle.encrypt({ name: "AES-GCM", iv: iv(prefix, i), additionalData: aad, tagLength: 128 }, key, part));
1315:       out.push(new Uint8Array([last]), be32(ct.length), ct);
1316:     }
1317:     return concat(out);
