654:   function recordBalance(model, cfg, sec, officeName, row, balCell, src, curFn) {
655:     var p = parseAmount(balCell);
656:     if (!p.ok || p.empty) return;

659:     var def = curDef(cfg, cur);
660:     if (!def) return;

662:     if (m === null) {
663:       var f = p.frac.slice(0, def.dec);
664:       m = toMinor({ int: p.int, frac: f }, def.dec);
665:     }

667:     var signed = (p.neg ? (pos === "D" ? -m : m) : (pos === "D" ? m : -m));
668:     if (cfg.reverse) signed = -signed;
669:     var key = keyOf(cfg, officeName) + "|" + cur;
670:     model.lastBalance = model.lastBalance || {};
671:     model.lastBalance[key] = { minor: signed, src: src };
672:   }
