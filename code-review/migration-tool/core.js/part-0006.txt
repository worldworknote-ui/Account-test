867:   /** يحوّل مبلغًا مقروءًا إلى وحدة صغرى بإشارته، أو يرمي سببًا مفهومًا. */
868:   function signedMinor(p, dec, what) {
869:     if (p.empty) return 0n;
870:     if (!p.ok) throw new Error(what + " غير مفهوم (" + p.reason + ")");
871:     var m = toMinor(p, dec);
872:     if (m === null) throw new Error(what + " فيه منازل عشرية أكثر من المسموح للعملة (" + dec + ")");
873:     return p.neg ? -m : m;
874:   }

876:   /**
877:    * يتحقق من كشف واحد ويحوّله إلى رصيد سابق وحركات. الإشارة الداخلية: موجب = «علينا» (دائن − مدين)،
878:    * ثم تُقلب عند الإخراج لأن التطبيق يحسب «لنا» موجبًا (مدين − دائن).
879:    * يُجرَّب تفسيرا عمود الرصيد (موجبه علينا أو لنا)، ويُقبل فقط التفسير الذي يطابق كل الصفوف والإجمالي.
880:    */
881:   function checkStatement(st, dec, fallbackPositive) {
882:     var rows = st.rows.map(function (x) {
883:       var d = signedMinor(x.d, dec, "مبلغ «مدين/لنا»"), c = signedMinor(x.c, dec, "مبلغ «دائن/علينا»");
884:       if (d < 0n || c < 0n) throw new Error("قيمة سالبة في عمود مدين أو دائن (صف " + x.src.row + ")");
885:       return { x: x, d: d, c: c, b: x.b.empty ? null : signedMinor(x.b, dec, "الرصيد") };
886:     });
887:     var preB = st.pre ? signedMinor(st.pre.b, dec, "الرصيد السابق") : 0n;
888:     var cum = null;
889:     if (st.pre && !st.pre.d.empty && !st.pre.c.empty) {
890:       var pd = signedMinor(st.pre.d, dec, "مجموع مدين السابق"), pc = signedMinor(st.pre.c, dec, "مجموع دائن السابق");
891:       if (pd !== 0n || pc !== 0n) cum = pc - pd;
892:     }
893:     var foot = null;
894:     if (st.footer) foot = { m: signedMinor(st.footer.p, dec, "اجمالي الرصيد"), side: st.footer.side };
895:     function attempt(mode) {
896:       var s = mode === "C" ? preB : -preB;
897:       var res = { mode: mode, s0: s, bad: null, checked: 0 };
898:       if (cum !== null && cum !== s) { res.bad = { src: st.pre.src, shown: preB, want: cum, what: "الرصيد السابق لا يساوي (دائن − مدين) في سطره" }; return res; }
899:       for (var i = 0; i < rows.length; i++) {
900:         s += rows[i].c - rows[i].d;
901:         if (rows[i].b !== null) {
902:           var want = mode === "C" ? s : -s;
903:           if (rows[i].b !== want) { res.bad = { src: rows[i].x.src, shown: rows[i].b, want: want, what: "الرصيد المكتوب لا يساوي الرصيد المحسوب" }; return res; }
904:           res.checked++;
905:         }
906:       }
907:       if (foot) {
908:         var abs = foot.m < 0n ? -foot.m : foot.m;
909:         var sAbs = s < 0n ? -s : s;
910:         var sideOk = foot.side === null || foot.side === "0" ? true : (foot.side === "C" ? s >= 0n : s <= 0n);
911:         var signedFoot = foot.m < 0n && foot.side === null ? (mode === "C" ? foot.m : -foot.m) : null;
912:         var okFoot = signedFoot !== null ? signedFoot === s : (abs === sAbs && (s === 0n || sideOk));
913:         if (!okFoot) { res.bad = { src: st.footer.src, shown: foot.m, want: s, what: "اجمالي الرصيد في آخر الكشف لا يساوي الرصيد المحسوب" }; return res; }
914:       }
915:       res.final = s;
916:       return res;
917:     }
918:     var A = attempt("C"), B = attempt("D");
919:     var pick = null, ambiguous = false;
920:     if (!A.bad && !B.bad) {
921:       // التفسيران متطابقان: لا حركات تغيّر الرصيد ولا مجاميع ولا نوع رصيد في الإجمالي
922:       if (A.final === B.final) pick = A;
923:       else { pick = fallbackPositive === "D" ? B : A; ambiguous = true; }
924:     } else if (!A.bad) pick = A;
925:     else if (!B.bad) pick = B;
926:     if (!pick) {
927:       var first = A.bad && B.bad ? ((A.bad.src.row || 0) >= (B.bad.src.row || 0) ? A.bad : B.bad) : (A.bad || B.bad);
928:       return { ok: false, bad: first, rows: rows };
929:     }
930:     return { ok: true, mode: pick.mode, s0: pick.s0, final: pick.final, checked: pick.checked, ambiguous: ambiguous, rows: rows, footer: !!foot };
931:   }
