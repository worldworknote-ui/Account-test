741:   /** معلومات الحساب والعملة والفترة من الأسطر التي فوق صف العناوين. */
742:   function stmtMeta(lines) {
743:     var meta = { account: null, curText: "", from: null, to: null, title: [] };
744:     var texts = [];
745:     lines.forEach(function (row) {
746:       var cells = row.filter(function (c) { return c && text(c); });
747:       var label = cells.map(text).join(" ");
748:       var isFrom = /تاريخ\s*(البدايه|البداية)|من\s*تاريخ|^من\b/.test(label), isTo = /تاريخ\s*(النهايه|النهاية)|(الى|إلى)\s*تاريخ/.test(label);
749:       cells.forEach(function (c) {
750:         var t = text(c);
751:         var dt = parseDate(c, "ymd", null);
752:         if (dt && /\d{1,4}\s*[\/\-.]\s*\d{1,2}\s*[\/\-.]\s*\d{1,4}/.test(normDigits(t))) {
753:           if (isFrom && !meta.from) meta.from = dt;
754:           else if (isTo && !meta.to) meta.to = dt;
755:           return;
756:         }
757:         if (META_SKIP_RE.test(clean(t))) return;
758:         texts.push(t);
759:       });
760:     });
761:     // العملة: نص يُعرف كعملة؛ الحساب: نص فيه رمز رقمي واسم، وإلا أول نص ليس عملة
762:     var curIdx = -1;
763:     for (var i = 0; i < texts.length; i++) if (guessCurrency(texts[i])) { curIdx = i; break; }
764:     var accIdx = -1;
765:     for (var j = 0; j < texts.length; j++) {
766:       if (j === curIdx) continue;
767:       var a = splitAccount(texts[j]);
768:       if (a.code && a.name) { accIdx = j; break; }
769:     }
770:     if (accIdx < 0) for (var k = 0; k < texts.length; k++) if (k !== curIdx) { accIdx = k; break; }
771:     if (accIdx >= 0) meta.account = splitAccount(texts[accIdx]);
772:     if (curIdx >= 0) meta.curText = clean(texts[curIdx]);
773:     else {
774:       // عملة غير معروفة (ذهب، عملة محلية…): أول نص بعد الحساب يُعرض للمطابقة
775:       for (var n = accIdx + 1; n < texts.length; n++) if (n !== accIdx) { meta.curText = clean(texts[n]); break; }
776:     }
777:     meta.title = texts.filter(function (x, idx) { return idx !== accIdx && idx !== curIdx; });
778:     return meta;
779:   }

781:   function footerOf(row) {
782:     var cells = row.filter(function (c) { return c && text(c); });
783:     if (!cells.some(function (c) { return FOOT_RE.test(clean(text(c)).replace(/\s*:\s*$/, "")); })) return null;
784:     var amount = null, side = null;
785:     cells.forEach(function (c) {
786:       var t = clean(text(c));
787:       if (FOOT_RE.test(t) || t === ":" || /نوع\s*الرصيد/.test(t)) return;
788:       var p = parseAmount(c);
789:       if (p.ok && !p.hadText && amount === null) { amount = p; return; }
790:       var k = looseKey(t);
791:       if (/دائن|علينا|credit|\bcr\b/i.test(t) || k === looseKey("له")) side = side || "C";
792:       else if (/مدين|لنا|debit|\bdr\b/i.test(t) || k === looseKey("عليه")) side = side || "D";
793:       else if (/متوازن|متعادل|صفر|balanced/i.test(t)) side = side || "0";
794:     });
795:     return amount ? { p: amount, side: side } : null;
796:   }

807:   function parseStatements(fileName, sheets) {
808:     var blocks = [];

811:     sheets.forEach(function (sh) {

813:         if (isBlankRow(row)) return;

816:         if (hc) {
