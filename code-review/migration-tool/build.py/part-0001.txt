1: #!/usr/bin/env python3
2: """يجمع أداة النقل في ملف HTML واحد يعمل بدون إنترنت: app/src/main/assets/migration-tool.html"""
3: import os
4: import sys
5: 
6: HERE = os.path.dirname(os.path.abspath(__file__))
7: OUT = sys.argv[1] if len(sys.argv) > 1 else os.path.join(HERE, "..", "app", "src", "main", "assets", "migration-tool.html")
8: 
9: 
10: def read(p):
11:     with open(os.path.join(HERE, p), encoding="utf-8") as f:
12:         return f.read()
13: 
14: 
15: def safe_js(s):
16:     # لا يجوز أن يظهر «</script» داخل السكربت المضمّن، و«<!--» قد يربك المحلل
17:     return s.replace("</script", "<\\/script").replace("<!--", "<\\!--")
18: 
19: 
20: html = read("template.html")
21: html = html.replace("{{STYLE}}", read("style.css"))
22: html = html.replace("{{XLSX}}", safe_js(read("vendor/xlsx.full.min.js")))
23: html = html.replace("{{CORE}}", safe_js(read("core.js")))
24: html = html.replace("{{UI}}", safe_js(read("ui.js")))
25: os.makedirs(os.path.dirname(os.path.abspath(OUT)), exist_ok=True)
26: with open(OUT, "w", encoding="utf-8") as f:
27:     f.write(html)
28: print("wrote", os.path.abspath(OUT), len(html.encode("utf-8")), "bytes")
