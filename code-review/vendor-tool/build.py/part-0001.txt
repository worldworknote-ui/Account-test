1: #!/usr/bin/env python3
2: """يجمع أداة البائع في ملف HTML واحد يعمل بدون إنترنت.
3: الأداة للبائع وحده: لا توضع داخل التطبيق ولا تُرسل للعملاء (لا تحوي مفاتيح خاصة، لكنها أداة إصدار الرخص).
4: الاستعمال: python3 vendor-tool/build.py [مسار الملف الناتج]"""
5: import os
6: import sys
7: 
8: HERE = os.path.dirname(os.path.abspath(__file__))
9: OUT = sys.argv[1] if len(sys.argv) > 1 else os.path.join(HERE, "dist", "أداة-البائع-المحاسب-الذكي.html")
10: 
11: 
12: def read(p):
13:     with open(p, encoding="utf-8") as f:
14:         return f.read()
15: 
16: 
17: def safe_js(s):
18:     return s.replace("</script", "<\\/script").replace("<!--", "<\\!--")
19: 
20: 
21: html = read(os.path.join(HERE, "template.html"))
22: html = html.replace("{{STYLE}}", read(os.path.join(HERE, "..", "smart-import", "style.css")) + "\n" + read(os.path.join(HERE, "style.css")))
23: html = html.replace("{{CORE}}", safe_js(read(os.path.join(HERE, "core.js")) + "\n" + read(os.path.join(HERE, "qr.js"))))
24: html = html.replace("{{UI}}", safe_js(read(os.path.join(HERE, "ui.js"))))
25: os.makedirs(os.path.dirname(os.path.abspath(OUT)), exist_ok=True)
26: with open(OUT, "w", encoding="utf-8") as f:
27:     f.write(html)
28: print("wrote", os.path.abspath(OUT), len(html.encode("utf-8")), "bytes")
