196:   /** توقيع ECDSA بترميز DER ← r||s بطول ثابت (صيغة WebCrypto). */
197:   function derToRaw(sig, size) {
198:     var s = der(sig, 0), k = kids(s);
199:     if (k.length !== 2) throw new Error("توقيع ECDSA غير صالح");
200:     var out = new Uint8Array(size * 2);
201:     k.forEach(function (n, idx) {
202:       var v = body(n);
203:       while (v.length > size && v[0] === 0) v = v.subarray(1);
204:       if (v.length > size) throw new Error("توقيع ECDSA أطول من المنحنى");
205:       out.set(v, idx * size + (size - v.length));
206:     });
207:     return out;
208:   }

210:   /** r||s ← DER (التطبيق يتحقق بـ SHA256withECDSA في Java ويتوقع DER). */
211:   function rawToDer(raw) {
212:     var size = raw.length / 2, parts = [];
213:     for (var p = 0; p < 2; p++) {
214:       var v = raw.subarray(p * size, (p + 1) * size), s = 0;
215:       while (s < v.length - 1 && v[s] === 0) s++;
216:       v = v.subarray(s);
217:       if (v[0] & 0x80) {
218:         var w = new Uint8Array(v.length + 1);
219:         w.set(v, 1);
220:         v = w;
221:       }
222:       parts.push(v);
223:     }
224:     var len = 2 + parts[0].length + 2 + parts[1].length;
225:     var hdr = len < 128 ? [0x30, len] : [0x30, 0x81, len];
226:     var out = new Uint8Array(hdr.length + len), i = 0;
227:     hdr.forEach(function (x) { out[i++] = x; });
228:     parts.forEach(function (v) {
229:       out[i++] = 0x02;
230:       out[i++] = v.length;
231:       out.set(v, i);
232:       i += v.length;
233:     });
234:     return out;
235:   }

237:   async function verifySig(cert, issuer) {
238:     var a = SIG[cert.sigAlg];
239:     if (!a) throw new Error("خوارزمية توقيع غير مدعومة: " + cert.sigAlg);
240:     if (a[0] === "RSA") {
241:       if (issuer.keyAlg !== RSA_OID) return false;
242:       var rk = await subtle.importKey("spki", issuer.spki, { name: "RSASSA-PKCS1-v1_5", hash: a[1] }, false, ["verify"]);
243:       return subtle.verify("RSASSA-PKCS1-v1_5", rk, cert.sig, cert.tbs);
244:     }
245:     if (issuer.keyAlg !== EC_OID || !CURVES[issuer.curve]) return false;
246:     var cv = CURVES[issuer.curve];
247:     var ek = await subtle.importKey("spki", issuer.spki, { name: "ECDSA", namedCurve: cv[0] }, false, ["verify"]);
248:     var raw;
249:     try {
250:       raw = derToRaw(cert.sig, cv[1]);
251:     } catch (e) {
252:       return false;
253:     }
254:     return subtle.verify({ name: "ECDSA", hash: a[1] }, ek, raw, cert.tbs);
255:   }

311:   async function verifyChain(chainB64, opts) {
312:     opts = opts || {};
313:     var res = { ok: false, sigOk: true, rootTrusted: false, rootName: "", problems: [], kd: null, leafSpki: null, revoked: [] };
314:     if (!chainB64 || !chainB64.length) {
315:       res.problems.push("لا توجد سلسلة تصديق");
316:       return res;
317:     }
318:     var certs;
319:     try {
320:       certs = chainB64.map(function (c) { return parseCert(typeof c === "string" ? b64ToBytes(c) : c); });
321:     } catch (e) {
322:       res.sigOk = false;
323:       res.problems.push("شهادة غير مقروءة: " + e.message);
324:       return res;
325:     }
326:     res.leafSpki = certs[0].spki;
327:     for (var i = 0; i < certs.length; i++) {
328:       var issuer = certs[i + 1] || certs[i];
329:       var good = false;
330:       try {
331:         good = await verifySig(certs[i], issuer);
332:       } catch (e) {
333:         res.problems.push(e.message);
334:       }
335:       if (!good) {
336:         res.sigOk = false;
337:         res.problems.push("توقيع الشهادة " + (i + 1) + " من " + certs.length + " لا يتحقق");
338:       }
339:       if (i > 0 && certs[i].exts[KD_OID]) {
340:         res.sigOk = false;
341:         res.problems.push("شهادة وسيطة تحمل امتداد التصديق (سلسلة مركّبة من هاتف آخر)");
342:       }
343:     }
344:     var top = bytesToB64(certs[certs.length - 1].spki);
345:     GOOGLE_ROOTS.concat((opts.extraRoots || []).map(function (s) { return { name: "جذر مضاف", spki: s }; })).forEach(function (r) {
346:       if (r.spki === top) {
347:         res.rootTrusted = true;
348:         res.rootName = r.name;
349:       }
350:     });
351:     var st = opts.status && opts.status.entries;
352:     if (st) certs.forEach(function (c) {
353:       var e = st[c.serial] || st[c.serial.toLowerCase()];
354:       if (e && /REVOKED|SUSPENDED/i.test(e.status || "REVOKED")) res.revoked.push(c.serial + " (" + (e.reason || e.status || "") + ")");
355:     });
356:     if (!certs[0].exts[KD_OID]) res.problems.push("لا يوجد امتداد التصديق في شهادة المفتاح");
357:     else if (certs[0].extCount[KD_OID] > 1) {
358:       res.sigOk = false;
359:       res.problems.push("امتداد التصديق مكرر");
360:     } else {
361:       try {
362:         res.kd = parseKeyDescription(certs[0].exts[KD_OID]);
363:       } catch (e) {
364:         res.problems.push("امتداد التصديق غير مقروء: " + e.message);
365:       }
366:     }
367:     res.ok = res.sigOk && res.rootTrusted && !!res.kd && !res.revoked.length;
368:     return res;
369:   }

378:     var s = String(text || "");
379:     var at = s.indexOf(prefix);
380:     if (at < 0) return null;

383:     var chunks = m[0].split(/\s+/).filter(Boolean);

388:         if (v) return v;
389:       } catch (e) {
390:         // نجرب أقصر
391:       }
392:     }
393:     return null;
