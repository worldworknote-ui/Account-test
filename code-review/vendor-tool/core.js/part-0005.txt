712:   /** يتحقق من «SA8-REL:» بمفتاح الهاتف القديم المحفوظ في السجل. */
713:   async function verifyRelease(text, pubB64) {
714:     if (String(text || "").indexOf("SA8-REL:") < 0) return { ok: false, error: "هذا ليس إيصال تحرير (يبدأ بـ SA8-REL:)" };
715:     var t = extract(text, "SA8-REL:", "A-Za-z0-9_\\-.", function (c) {
716:       var d = c.indexOf(".");
717:       if (d < 1 || d !== c.lastIndexOf(".") || d === c.length - 1) return null;
718:       parseFields(fromUtf8(b64ToBytes(c.substring(0, d))));
719:       der(b64ToBytes(c.substring(d + 1)), 0);
720:       return c;
721:     });
722:     if (!t) return { ok: false, error: "الإيصال مبتور أو غير مقروء" };
723:     var dot = t.indexOf(".");
724:     try {
725:       var bodyB = b64ToBytes(t.substring(0, dot)), sig = b64ToBytes(t.substring(dot + 1));
726:       var f = parseFields(fromUtf8(bodyB));
727:       var pub = b64ToBytes(pubB64);
728:       var pk = await subtle.importKey("spki", pub, { name: "ECDSA", namedCurve: "P-256" }, false, ["verify"]);
729:       var good = await subtle.verify({ name: "ECDSA", hash: "SHA-256" }, pk, derToRaw(sig, 32), bodyB);
730:       if (!good) return { ok: false, error: "توقيع الإيصال لا يطابق مفتاح الهاتف المسجّل لهذه الرخصة" };
731:       if (f.v !== "1" || f.t !== "release") return { ok: false, error: "صيغة الإيصال غير معروفة" };
732:       if (f.dk !== (await deviceFingerprint(pub))) return { ok: false, error: "الإيصال من هاتف آخر" };
733:       return { ok: true, lid: f.lid, dk: f.dk, at: Number(f.at), nonce: f.n };
734:     } catch (e) {
735:       return { ok: false, error: "الإيصال غير مقروء" };
736:     }
737:   }
