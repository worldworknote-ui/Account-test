27:     var z = 0;
28:     for (var i = 7; i >= 0; i--) {
29:       z = (z << 1) ^ ((z >>> 7) * 0x11D);

31:     }

35:   function rsDivisor(deg) {
36:     var r = [];
37:     for (var i = 0; i < deg - 1; i++) r.push(0);
38:     r.push(1);
39:     var root = 1;
40:     for (var k = 0; k < deg; k++) {
41:       for (var j = 0; j < r.length; j++) {
42:         r[j] = mul(r[j], root);
43:         if (j + 1 < r.length) r[j] ^= r[j + 1];
44:       }

46:     }

171:     for (var y = 0; y < size; y++) for (var x = 0; x < size; x++) {

178:         case 3: inv = (x + y) % 3 === 0; break;
179:         case 4: inv = (Math.floor(x / 3) + Math.floor(y / 2)) % 2 === 0; break;
180:         case 5: inv = x * y % 2 + x * y % 3 === 0; break;
181:         case 6: inv = (x * y % 2 + x * y % 3) % 2 === 0; break;
182:         default: inv = ((x + y) % 2 + x * y % 3) % 2 === 0;
