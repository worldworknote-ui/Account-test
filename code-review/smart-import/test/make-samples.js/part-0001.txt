10: const serial = (y, m, d) => (Date.UTC(y, m - 1, d) - Date.UTC(1899, 11, 30)) / 86400000;

14: /** صفوف ← ورقة: التواريخ أرقام Excel بتنسيق تاريخ، والمعادلات بقيمتها المحفوظة. */
15: function sheet(rows) {
16:   const plain = rows.map((r) => r.map((v) => (v && typeof v === "object" ? null : v)));
17:   const ws = XLSX.utils.aoa_to_sheet(plain);
18:   rows.forEach((r, ri) => r.forEach((v, ci) => {
19:     if (!v || typeof v !== "object") return;
20:     const a = XLSX.utils.encode_cell({ r: ri, c: ci });
21:     if (v.__date !== undefined) ws[a] = { t: "n", v: v.__date, z: "dd/mm/yyyy" };
22:     else if (v.__f !== undefined) ws[a] = { t: typeof v.v === "number" ? "n" : "s", v: v.v, f: v.__f };
23:   }));
24:   const rg = XLSX.utils.decode_range(ws["!ref"]);
25:   rows.forEach((r, ri) => { rg.e.r = Math.max(rg.e.r, ri); rg.e.c = Math.max(rg.e.c, r.length - 1); });
26:   ws["!ref"] = XLSX.utils.encode_range(rg);
27:   return ws;
28: }
29: function save(name, sheets) {
30:   const wb = XLSX.utils.book_new();
31:   sheets.forEach(([n, rows]) => XLSX.utils.book_append_sheet(wb, sheet(rows), n));
32:   fs.writeFileSync(path.join(out, name), Buffer.from(XLSX.write(wb, { type: "array", bookType: "xlsx" })));
33: }
