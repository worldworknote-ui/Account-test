28: function load(names) {
29:   return names.map((n) => {
30:     const buf = fs.readFileSync(path.join(SAMPLES, n));
31:     let wb;
32:     if (/\.(csv|txt|tsv)$/i.test(n)) {
33:       const txt = new TextDecoder("utf-8").decode(buf).replace(/^﻿/, "");
34:       const first = txt.split("\n")[0];
35:       const semi = (first.match(/;/g) || []).length > (first.match(/,/g) || []).length;
36:       wb = XLSX.read(txt, { type: "string", raw: true, FS: semi ? ";" : undefined });
37:     } else wb = XLSX.read(buf, { type: "buffer", cellNF: true, cellDates: false, cellFormula: true, bookFiles: true });
38:     return { name: n, wb, sheets: wb.SheetNames.map((s) => ({ name: s, rows: X.sheetRows(wb.Sheets[s], XLSX) })) };
