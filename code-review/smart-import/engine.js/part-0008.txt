1365:   function splitArgs(s) {
1366:     var out = [], depth = 0, q = false, cur = "";
1367:     for (var i = 0; i < s.length; i++) {
1368:       var ch = s.charAt(i);
1369:       if (ch === '"') { q = !q; cur += ch; continue; }
1370:       if (!q && ch === "(") depth++;
1371:       if (!q && ch === ")") depth--;
1372:       if (!q && depth === 0 && (ch === "," || ch === ";")) { out.push(cur.trim()); cur = ""; continue; }
1373:       cur += ch;
1374:     }
1375:     out.push(cur.trim());
1376:     return out;
