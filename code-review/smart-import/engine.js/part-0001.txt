16:   function colName(i) { var s = ""; i++; while (i > 0) { var m = (i - 1) % 26; s = String.fromCharCode(65 + m) + s; i = Math.floor((i - 1) / 26); } return s; }

18:   function colIndex(letters) {
19:     var n = 0;
20:     for (var i = 0; i < letters.length; i++) n = n * 26 + (letters.charCodeAt(i) - 64);

22:   }

24:   /** «B8:H5279» ← {s:{r,c}, e:{r,c}} (صفر أولًا). */
25:   function decodeRef(ref) {
26:     var m = String(ref || "").replace(/\$/g, "").match(/^([A-Z]+)(\d+)(?::([A-Z]+)(\d+))?$/i);
27:     if (!m) return null;
28:     var s = { r: +m[2] - 1, c: colIndex(m[1].toUpperCase()) };
29:     var e = m[3] ? { r: +m[4] - 1, c: colIndex(m[3].toUpperCase()) } : { r: s.r, c: s.c };

51:   function resolvePath(dir, target) {

117:   function guessDir(t) {
118:     var s = clean(normDigits(t)).toLowerCase();
119:     if (!s || s.length > 40) return null;
120:     var toks = s.split(/[^a-z؀-ۿ]+/).filter(Boolean).map(function (w) { return C.looseKey(w); });
121:     if (!toks.length) return s === "+" ? "D" : s === "-" ? "C" : null;
122:     var d = 0, c = 0;
123:     toks.forEach(function (w) {
124:       if (C_WORDS.indexOf(w) >= 0) c++;
125:       else if (D_WORDS.indexOf(w) >= 0) d++;
126:     });
127:     if (d && !c) return "D";
128:     if (c && !d) return "C";
129:     return null;
130:   }
