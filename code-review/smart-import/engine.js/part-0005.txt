762:   /**
763:    * قيمة خلية «الرصيد» بالوحدة الصغرى وإشارتها كما في الملف. الرصيد المحسوب بمعادلة قد يحمل كسرًا عائمًا ضئيلًا
764:    * (70987.00000000001) فيُقرّب إلى منازل العملة فقط إن كان الفرق أقل من جزء من ألف من الوحدة الصغرى.
765:    * «لنا 500» / «علينا 500» تُقرأ بإشارتها؛ نص لا يُفهم اتجاهه ← null.
766:    */
767:   function balanceMinor(cell, dec) {
768:     if (!cell) return null;
769:     if (cell.n !== undefined && typeof cell.n === "number") {
770:       if (!isFinite(cell.n)) return null;
771:       var f = Math.pow(10, dec), x = cell.n * f, r = Math.round(x);
772:       if (Math.abs(x - r) > 1e-3 || Math.abs(r) > 9e15) return null;
773:       return BigInt(r);
774:     }
775:     var p = C.parseAmount(cell);
776:     if (!p.ok || p.empty) return null;
777:     var m = C.toMinor(p, dec);
778:     if (m === null) return null;
779:     var neg = p.neg;
780:     if (p.hadText) {
781:       var g = guessDir(clean(normDigits(text(cell))).replace(/[\d.,()\-+]/g, " "));
782:       if (!g) return null;
783:       if (g === "C") neg = !neg;
784:     }
785:     return neg ? -m : m;
786:   }
