72:   function readBook(file) {
73:     return file.arrayBuffer().then(function (ab) {
74:       var buf = new Uint8Array(ab);
75:       var lower = file.name.toLowerCase();
76:       var wb;
77:       try {
78:         if (/\.(csv|txt|tsv)$/.test(lower)) {
79:           var txt = decodeText(buf).replace(/^﻿/, "");
80:           var first = txt.split("\n")[0];
81:           var semi = (first.match(/;/g) || []).length > (first.match(/,/g) || []).length;
82:           var tab = lower.endsWith(".tsv") || (first.match(/\t/g) || []).length > (first.match(/,/g) || []).length;
83:           wb = XLSX.read(txt, { type: "string", raw: true, FS: tab ? "\t" : (semi ? ";" : undefined) });
84:         } else {
85:           wb = XLSX.read(buf, { type: "array", cellNF: true, cellDates: false, cellFormula: true, bookFiles: true });
86:         }
87:       } catch (e) {
88:         var msg = String(e && e.message || e);
89:         if (/password|encrypt/i.test(msg)) throw new Error("«" + file.name + "» محمي بكلمة سر. افتحه في Excel واحفظه بدون كلمة سر ثم اختره مجددًا.");
90:         throw new Error("تعذر فتح «" + file.name + "»: " + msg);
91:       }
92:       return { name: file.name, wb: wb, sheets: wb.SheetNames.map(function (nm) { return { name: nm, rows: X.sheetRows(wb.Sheets[nm], XLSX) }; }) };
