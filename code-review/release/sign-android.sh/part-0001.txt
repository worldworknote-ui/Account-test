1: #!/bin/bash
2: # 10.11: توقيع النسخة النهائية للهاتف بمفتاح 8.0 على جهاز صاحب المفتاح (المفتاح لا يغادر جهازه ولا يوضع في المستودع).
3: #   bash release/sign-android.sh <app-release-unsigned.apk> [ملف الإخراج.apk]
4: # يحتاج: keystore.properties ومجلد signing في مجلد المشروع (كما في keystore.properties.example)، وapksigner من Android SDK build-tools
5: # (أو حدّد APKSIGNER=<المسار>). يتحقق بعد التوقيع أن بصمة الشهادة = BuildInfo.SIGNING_CERT_SHA256، وإلا يحذف الناتج:
6: # نسخة بمفتاح آخر لا تُثبَّت فوق النسخة الحالية ولا تعمل عليها الرخص.
7: set -euo pipefail
8: HERE="$(cd "$(dirname "$0")" && pwd)"
9: P="$(dirname "$HERE")"
10: IN="${1:?حدد ملف app-release-unsigned.apk}"
11: VERSION=$(grep -o 'VERSION = "[^"]*"' "$P/app/src/main/java/com/smart/hawala/BuildInfo.java" | cut -d'"' -f2)
12: OUT="${2:-$(dirname "$IN")/SmartAccountant-$VERSION.apk}"
13: WANT=$(grep -o 'SIGNING_CERT_SHA256 = "[^"]*"' "$P/app/src/main/java/com/smart/hawala/BuildInfo.java" | cut -d'"' -f2 | tr 'A-F' 'a-f')
14: PROPS="$P/keystore.properties"
15: [ -f "$PROPS" ] || { echo "لا يوجد $PROPS (انظر keystore.properties.example)"; exit 1; }
16: prop() { grep -E "^$1=" "$PROPS" | head -1 | cut -d= -f2-; }
17: KS="$P/$(prop storeFile)"
18: [ -f "$KS" ] || { echo "ملف المفتاح غير موجود: $KS"; exit 1; }
19: if [ -z "${APKSIGNER:-}" ]; then
20:   SDK="${ANDROID_HOME:-${ANDROID_SDK_ROOT:-$HOME/Android/Sdk}}"
21:   APKSIGNER=$(ls -d "$SDK"/build-tools/*/apksigner 2>/dev/null | sort -V | tail -1 || true)
22: fi
23: [ -x "${APKSIGNER:-}" ] || { echo "حدّد APKSIGNER (من Android SDK build-tools)"; exit 1; }
24: 
25: # كلمات المرور تُمرَّر بمتغيرات بيئة لا في سطر الأوامر
26: export SA_KS_PASS="$(prop storePassword)" SA_KEY_PASS="$(prop keyPassword)"
27: "$APKSIGNER" sign --ks "$KS" --ks-key-alias "$(prop keyAlias)" --ks-pass env:SA_KS_PASS --key-pass env:SA_KEY_PASS --out "$OUT" "$IN"
28: unset SA_KS_PASS SA_KEY_PASS
29: 
30: GOT=$("$APKSIGNER" verify --print-certs "$OUT" | grep -m1 'certificate SHA-256 digest' | awk '{print $NF}' | tr 'A-F' 'a-f')
31: if [ "$GOT" != "$WANT" ]; then
32:   rm -f "$OUT"
33:   echo "بصمة الشهادة $GOT لا تطابق مفتاح 8.0 ($WANT): حُذف الناتج."
34:   exit 1
35: fi
36: "$APKSIGNER" verify --verbose "$OUT" | grep -E "Verified|v[0-9] scheme" || true
37: sha256sum "$OUT"
38: echo "SIGNED OK: $OUT"
