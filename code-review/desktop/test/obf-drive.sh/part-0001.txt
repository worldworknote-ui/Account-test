1: #!/bin/bash
2: # 12.6: يشغّل اختبارات الواجهة الآلية (PhoneShot وPhoneDrive وLookDrive) على النسخة المعمّاة نفسها التي تُشحن:
3: # الاختبارات تُعمّى مع البرنامج بالخريطة نفسها (-applymapping) فتُنادي الأسماء الجديدة، ويُتأكد أن أسماء أصناف البرنامج
4: # مطابقة لنسخة الشحن. R8_JAR=<r8.jar> bash desktop/test/obf-drive.sh
5: set -e
6: HERE="$(cd "$(dirname "$0")" && pwd)"
7: D="$(dirname "$HERE")"
8: P="$(dirname "$D")"
9: LIBS="$D/lib/sqlite-jdbc.jar:$D/lib/org.json.jar:$D/lib/flatlaf.jar:$D/lib/zxing-core.jar"
10: W=$(mktemp -d /tmp/sa-obf-XXXXXX)
11: bash "$D/build.sh" >/dev/null
12: cp -r "$D/build/classes" "$W/plain"
13: mkdir -p "$W/tests"
14: javac -nowarn -encoding UTF-8 -cp "$W/plain:$LIBS" -d "$W/tests" "$HERE/PhoneShot.java" "$HERE/PhoneDrive.java" "$HERE/LookDrive.java" 2>&1 | grep -v "JAVA_TOOL\|^Note" || true
15: bash "$D/obfuscate.sh" "$W/plain" "$W/ship" "$W/map.txt"
16: cp -r "$W/plain" "$W/both" && cp -r "$W/tests/"* "$W/both/"
17: cat > "$W/keep-tests.pro" <<'R'
18: -keep class com.smart.hawala.PhoneShot** { *; }
19: -keep class com.smart.hawala.PhoneDrive** { *; }
20: -keep class com.smart.hawala.desk.LookDrive** { *; }
21: R
22: bash "$D/obfuscate.sh" "$W/both" "$W/drive" "$W/map2.txt" "$W/map.txt" "$W/keep-tests.pro"
23: a=$(cd "$W/ship" && find . -name '*.class' | sort | md5sum); b=$(cd "$W/drive" && find . -name '*.class' ! -path '*PhoneShot*' ! -path '*PhoneDrive*' ! -path '*LookDrive*' | sort | md5sum)
24: [ "$a" = "$b" ] || { echo "أسماء أصناف نسخة الاختبار تختلف عن نسخة الشحن"; exit 1; }
25: echo "ok   أسماء الأصناف المعمّاة مطابقة لنسخة الشحن"
26: grep -q "com.smart.hawala.Entitlements -> com.smart.hawala.Entitlements" "$W/map.txt" && { echo "منطق الترخيص لم يُعمَّ"; exit 1; }
27: echo "ok   منطق الترخيص والحدود معمّى ($(grep -E '^com.smart.hawala.(Entitlements|License|LicenseToken|QuotaState|WaQuota) ->' "$W/map.txt" | tr '\n' ' '))"
28: PW=$(python3 -c "import json;print(json.load(open('$P/tests/data/expect.json'))['D']['password'])")
29: UCP="$W/drive:$LIBS"
30: java -Djava.awt.headless=true -Dstdout.encoding=UTF-8 -cp "$UCP" com.smart.hawala.PhoneShot "$W/ui" "$P/tests/data/D.sba" "$PW" > "$W/shot.log" 2>&1 || { tail -30 "$W/shot.log"; echo "FAIL PhoneShot (معمّاة)"; exit 1; }
31: echo "ok   PhoneShot على النسخة المعمّاة"
32: xvfb-run -a -s '-screen 0 1440x1000x24' java -Dstdout.encoding=UTF-8 -cp "$UCP" com.smart.hawala.PhoneDrive "$W/drive-out" "$P/tests/data/D.sba" "$PW" > "$W/drive.log" 2>&1 || { grep -E "FAIL|Exception" "$W/drive.log" | head -30; echo "FAIL PhoneDrive (معمّاة)"; exit 1; }
33: grep -c "^ok" "$W/drive.log" | sed 's/^/ok   PhoneDrive على النسخة المعمّاة: /'
34: LH=$(mktemp -d /tmp/sa-look-XXXXXX)
35: xvfb-run -a -s '-screen 0 1600x1100x24' java -Dsa.home="$LH/home" -Dstdout.encoding=UTF-8 -cp "$UCP" com.smart.hawala.desk.LookDrive "$W/look" > "$W/look.log" 2>&1 || { tail -20 "$W/look.log"; echo "FAIL LookDrive (معمّاة)"; exit 1; }
36: echo "ok   LookDrive على النسخة المعمّاة"
37: rm -rf "$LH" "$W"
38: echo "OBF DRIVE OK"
