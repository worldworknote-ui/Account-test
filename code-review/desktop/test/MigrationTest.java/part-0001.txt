35:     static Db.TxInput tx(long office, boolean us, long amount, String cur, String person) {

57:     static void copy(File from, File to) throws Exception {
58:         to.mkdirs();
59:         for (File f : from.listFiles()) {
60:             File t = new File(to, f.getName());
61:             if (f.isDirectory()) copy(f, t);
62:             else Files.copy(f.toPath(), t.toPath());
63:         }
64:     }
