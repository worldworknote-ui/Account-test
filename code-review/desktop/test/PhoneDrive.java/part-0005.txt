747:     /** يبحث عن نص ظاهر في النوافذ (الأعلى أولًا)، ويرجع مركزه على الشاشة. */
748:     static Point find(String text, boolean bottom, boolean hint) throws Exception {
749:         final Point[] res = {null};
750:         SwingUtilities.invokeAndWait(() -> {
751:             List<ViewRootImpl> roots = WindowStack.roots();
752:             roots.sort((x, y) -> Integer.compare(y.layer(), x.layer()));
753:             for (ViewRootImpl r : roots) {
754:                 List<View> hits = new ArrayList<>();
755:                 collect(r.getView(), text, hint, hits);
756:                 if (hits.isEmpty()) continue;
757:                 List<View> exact = new ArrayList<>();
758:                 for (View v : hits) {
759:                     CharSequence t = hint ? ((TextView) v).getHint() : ((TextView) v).getText();
760:                     if (t.toString().trim().equals(text)) exact.add(v);
761:                 }
762:                 if (!exact.isEmpty()) hits = exact;
763:                 View best = hits.get(0);
764:                 for (View v : hits) {
765:                     int[] l = new int[2], m = new int[2];
766:                     v.getLocationOnScreen(l);
767:                     best.getLocationOnScreen(m);
768:                     if (bottom ? l[1] > m[1] : l[1] < m[1]) best = v;
769:                 }
770:                 res[0] = phys(best, best.getWidth() / 2f, best.getHeight() / 2f);
771:                 return;
772:             }
773:         });
774:         if (res[0] == null) { fails++; log.add("NOT FOUND: " + text); }
775:         return res[0];
776:     }
777: 
778:     static void collect(View v, String text, boolean hint, List<View> hits) {
779:         if (!v.isShown()) return;
780:         if (v instanceof TextView) {
781:             CharSequence t = hint ? ((TextView) v).getHint() : ((TextView) v).getText();
782:             if (t != null && t.toString().contains(text) && v.getWidth() > 0) hits.add(v);
783:         }
784:         if (v instanceof ViewGroup) for (int i = 0; i < ((ViewGroup) v).getChildCount(); i++) collect(((ViewGroup) v).getChildAt(i), text, hint, hits);
785:     }
