808:     static Point phys(View v, float x, float y) {
809:         float[] o = {x, y};
810:         v.mapToRoot(o);
811:         ViewRootImpl root = null;
812:         for (ViewRootImpl r : WindowStack.roots()) if (v.getRootView() == r.getView()) root = r;
813:         if (root == null) return null;
814:         Point s = root.pane.getLocationOnScreen();
815:         float sc = ViewRootImpl.scale();
816:         return new Point(s.x + Math.round(o[0] * sc), s.y + Math.round(o[1] * sc));
817:     }
