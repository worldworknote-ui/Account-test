161:     static <T> T find(Container c, Class<T> k) {
162:         List<T> l = all(c, k);
163:         return l.isEmpty() ? null : l.get(0);
164:     }
165: 
166:     static <T> List<T> all(Component c, Class<T> k) {
167:         List<T> o = new ArrayList<>();
168:         if (k.isInstance(c)) o.add(k.cast(c));
169:         if (c instanceof Container) for (Component x : ((Container) c).getComponents()) o.addAll(all(x, k));
170:         if (c instanceof Window) for (Window w : ((Window) c).getOwnedWindows()) o.addAll(all(w, k));
171:         return o;
172:     }

174:     /** مركز خانة كتابة تبدأ عبارتها الإرشادية بالنص المعطى، بإحداثيات الشاشة. */
175:     static Point hintPoint(String hint) throws Exception {
176:         final Point[] r = {null};
177:         SwingUtilities.invokeAndWait(() -> {
178:             ViewRootImpl root = null;
179:             View v = null;
180:             for (ViewRootImpl x : WindowStack.roots()) {
181:                 View f = findHint(x.getView(), hint);
182:                 if (f != null) { root = x; v = f; }
183:             }
184:             if (v == null) { System.out.println("roots: " + WindowStack.roots().size()); return; }
185:             float[] o = {v.getWidth() / 2f, v.getHeight() / 2f};
186:             v.mapToRoot(o);
187:             Point s = root.pane.getLocationOnScreen();
188:             float sc = ViewRootImpl.scale();
189:             r[0] = new Point(s.x + Math.round(o[0] * sc), s.y + Math.round(o[1] * sc));
190:         });
191:         return r[0];
192:     }
