1: #!/bin/bash
2: # 12.6: تعمية أصناف البرنامج بـ R8 (من أدوات أندرويد: cmdline-tools/lib/r8.jar) — قواعدها في obfuscate.pro.
3: #   R8_JAR=<r8.jar>  bash desktop/obfuscate.sh <مجلد الأصناف> <مجلد الإخراج> <ملف الخريطة> [خريطة سابقة تُطبَّق] [قواعد إضافية]
4: # الخريطة (mapping) تبقى خارج الحزم دائمًا: تُحفظ مع الإصدار لقراءة سجلات الانهيار فقط.
5: set -e
6: HERE="$(cd "$(dirname "$0")" && pwd)"
7: IN="$1"; OUTD="$2"; MAP="$3"; APPLY="$4"; EXTRA="$5"
8: [ -f "$R8_JAR" ] || { echo "حدّد R8_JAR"; exit 1; }
9: JH="${JAVA_HOME:-$(dirname "$(dirname "$(readlink -f "$(command -v java)")")")}"
10: T=$(mktemp -d)
11: ( cd "$IN" && jar cf "$T/in.jar" $(find . -name '*.class' | sed 's|^\./||') )
12: CONF="$T/rules.pro"
13: cp "$HERE/obfuscate.pro" "$CONF"
14: [ -n "$APPLY" ] && echo "-applymapping $APPLY" >> "$CONF"
15: [ -n "$EXTRA" ] && cat "$EXTRA" >> "$CONF"
16: java -cp "$R8_JAR" com.android.tools.r8.R8 --classfile --release --lib "$JH" \
17:   --classpath "$HERE/lib/sqlite-jdbc.jar" --classpath "$HERE/lib/org.json.jar" --classpath "$HERE/lib/flatlaf.jar" --classpath "$HERE/lib/zxing-core.jar" \
18:   --pg-conf "$CONF" --pg-map-output "$MAP" --output "$T/out.jar" "$T/in.jar" 2>&1 | grep -v "JAVA_TOOL\|^Info in\|does not match anything\|^  \*;\|^}\|<init>(...);\|^Proguard configuration" || true
19: [ -f "$T/out.jar" ] || { echo "فشلت التعمية"; exit 1; }
20: rm -rf "$OUTD"; mkdir -p "$OUTD"
21: ( cd "$OUTD" && unzip -qo "$T/out.jar" && rm -rf META-INF )
22: # الموارد غير الأصناف كما هي
23: ( cd "$IN" && find . -type f ! -name '*.class' -exec cp --parents {} "$OUTD/" \; )
24: rm -rf "$T"
25: echo "OBFUSCATE OK: $(grep -c ' -> ' "$MAP") اسمًا في الخريطة"
