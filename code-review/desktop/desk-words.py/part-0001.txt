1: #!/usr/bin/env python3
2: """11.2: نسخة الكمبيوتر تقول «الكمبيوتر/الجهاز» بدل «الهاتف» في نصوص البرنامج نفسها.
3: 
4: يعمل عند البناء فقط وعلى نسخة من ملفات تطبيق الهاتف (لا يغيّر ملفات أندرويد)، وداخل النصوص الثابتة في الكود فقط،
5: فلا يمس بيانات المستخدم أبدًا (ملاحظة فيها «الهاتف» تبقى كما هي).
6: لا يغيّر: أرقام الهواتف ("رقم الهاتف"، "هاتف المكتب")، والنصوص التي تشرح أجهزة متعددة (فيها الماك أو الويندوز أو أندرويد)،
7: وملفات قراءة رسائل الحوالات (كلماتها للمطابقة لا للعرض).
8: 
9: الاستعمال: desk-words.py <مجلد ملفات الهاتف> <مجلد الناتج>
10: """
11: import os
12: import re
13: import sys
14: 
15: SKIP_FILES = {"MessageParser.java", "SyncHelp.java"}
16: SKIP_IF = ["الماك", "ماك", "الويندوز", "ويندوز", "أندرويد", "الكمبيوتر", "رقم الهاتف", "هاتف المكتب", "تلفون", "تليفون", "phone"]
17: # عبارات محددة فقط (كلمة «الهاتف» وحدها قد تعني رقم الهاتف فتبقى كما هي)
18: RULES = [
19:     ("بالبصمة أو بقفل الشاشة", "ببصمة الجهاز أو كلمة سره"),
20:     ("قفل شاشة الهاتف", "كلمة سر الجهاز"),
21:     ("بلا بصمة ولا قفل شاشة", "بلا بصمة ولا كلمة سر للجهاز"),
22:     ("ولا قفل شاشة", "ولا كلمة سر للجهاز"),
23:     ("أو قفل الشاشة", "أو كلمة سر الجهاز"),
24:     ("قفل الشاشة", "كلمة سر الجهاز"),
25:     ("إعدادات الهاتف", "إعدادات النظام"),
26:     ("هذا الهاتف", "هذا الكمبيوتر"),
27:     ("هاتفك", "جهازك"),
28:     ("هاتفًا آخر", "جهازًا آخر"),
29:     ("هاتف آخر", "جهاز آخر"),
30:     ("هاتف جديد", "جهاز جديد"),
31:     ("حسب الهاتف", "حسب الجهاز"),
32:     ("خط الهاتف", "خط النظام"),
33:     ("ذاكرة الهاتف", "ذاكرة الكمبيوتر"),
34:     ("خارج الهاتف", "خارج الكمبيوتر"),
35:     ("عن الهاتف", "عن الكمبيوتر"),
36:     ("داخل الهاتف", "داخل الكمبيوتر"),
37:     ("في الهاتف", "في الكمبيوتر"),
38:     ("على الهاتف", "على الكمبيوتر"),
39:     ("من الهاتف", "من الكمبيوتر"),
40:     ("ضاع الهاتف", "ضاع الكمبيوتر"),
41:     ("حامل الهاتف", "حامل الكمبيوتر"),
42:     ("يحمل الهاتف", "يحمل الكمبيوتر"),
43:     ("يفتح الهاتف", "يفتح الكمبيوتر"),
44:     ("بصمة الهاتف", "بصمة الكمبيوتر"),
45:     ("مفتاح الهاتف", "مفتاح الكمبيوتر"),
46:     ("يرسل الهاتف", "يرسل الكمبيوتر"),
47:     ("الهاتف الأصلي", "الجهاز الأصلي"),
48:     ("الهاتف القديم", "الجهاز القديم"),
49:     ("الهاتف الجديد", "الجهاز الجديد"),
50:     ("الهاتف وحده", "الكمبيوتر وحده"),
51:     ("الهاتف وقفله", "الكمبيوتر وقفله"),
52:     ("الهاتف بلا بصمة", "الكمبيوتر بلا بصمة"),
53: ]
54: LIT = re.compile(r'"(?:[^"\\\n]|\\.)*"')
55: 
56: 
57: def fix(lit):
58:     if not any(w in lit for w in ("هاتف", "قفل شاشة", "قفل الشاشة")):
59:         return lit
60:     if any(w in lit for w in SKIP_IF):
61:         return lit
62:     # تسمية رقم الهاتف قبل قيمته: "...الهاتف: " + phone
63:     if lit.rstrip('"').rstrip().endswith("الهاتف:"):
64:         return lit
65:     for a, b in RULES:
66:         lit = lit.replace(a, b)
67:     return lit
68: 
69: 
70: def main():
71:     src, out = sys.argv[1], sys.argv[2]
72:     os.makedirs(out, exist_ok=True)
73:     changed = 0
74:     for n in sorted(os.listdir(src)):
75:         if not n.endswith(".java"):
76:             continue
77:         s = open(os.path.join(src, n), encoding="utf-8").read()
78:         t = s if n in SKIP_FILES else LIT.sub(lambda m: fix(m.group(0)), s)
79:         if t != s:
80:             changed += 1
81:         open(os.path.join(out, n), "w", encoding="utf-8").write(t)
82:     print("desk-words: %d files adjusted" % changed)
83: 
84: 
85: if __name__ == "__main__":
86:     main()
