1: #!/bin/bash
2: # يبني «المحاسب الذكي» للماك (والاختبارات على أي نظام فيه JDK 17+):
3: #   bash desktop/build.sh            ← desktop/build/smart-accountant.jar
4: # الحسابات وقاعدة البيانات والنسخ الاحتياطي والتقارير من ملفات الهاتف نفسها (app/src/main/java) دون نسخها،
5: # فلا يختلف رقم ولا صيغة نسخة بين الهاتف والماك.
6: set -e
7: HERE="$(cd "$(dirname "$0")" && pwd)"
8: P="$(dirname "$HERE")"
9: SH="$P/app/src/main/java/com/smart/hawala"
10: OUT="$HERE/build"
11: rm -rf "$OUT/classes"; mkdir -p "$OUT/classes"
12: 
13: # 11.0: كل ملفات تطبيق الهاتف (الواجهات نفسها) تُبنى فوق طبقة توافق أندرويد في desktop/src/android،
14: # عدا خمسة ملفات لها بدائل كمبيوتر في desktop/src/com/smart/hawala (النسخ الاحتياطي، المفاتيح، الترخيص، التطبيق، Google).
15: EXCL=" BackupManager Crypto License SmartApp DriveAuth DbCipher "
16: python3 "$HERE/gen-r.py" "$P/app/src/main/res" "$OUT/gen/com/smart/hawala/R.java"
17: SRCS="$OUT/gen/com/smart/hawala/R.java"
18: # 11.2: نسخة الكمبيوتر تقول «الكمبيوتر/الجهاز» بدل «الهاتف» (في نصوص البرنامج الثابتة فقط، على نسخة من الملفات)
19: rm -rf "$OUT/desk-src"
20: python3 "$HERE/desk-words.py" "$SH" "$OUT/desk-src"
21: for f in "$OUT/desk-src"/*.java; do n=$(basename "$f" .java); case "$EXCL" in *" $n "*) continue;; esac; SRCS="$SRCS $f"; done
22: DESK=$(find "$HERE/src" -name '*.java')
23: CP="$HERE/lib/sqlite-jdbc.jar:$HERE/lib/org.json.jar:$HERE/lib/flatlaf.jar:$HERE/lib/zxing-core.jar"
24: set +e
25: javac -nowarn -encoding UTF-8 --release 17 -Xmaxerrs 3000 -cp "$CP" -d "$OUT/classes" $SRCS $DESK > "$OUT/javac.log" 2>&1
26: RC=$?
27: set -e
28: grep -v "^Note:\|JAVA_TOOL" "$OUT/javac.log" || true
29: [ $RC -eq 0 ] || { echo "COMPILE FAILED"; exit 1; }
30: [ -n "$COMPILE_ONLY" ] && { echo "COMPILE OK"; exit 0; }
31: # 12.6: نسخة التوزيع معمّاة (OBFUSCATE=1 R8_JAR=…): أصناف بأسماء بلا معنى، والخريطة في build/obf-map.txt خارج الحزمة
32: if [ -n "$OBFUSCATE" ]; then
33:   bash "$HERE/obfuscate.sh" "$OUT/classes" "$OUT/obf-classes" "$OUT/obf-map.txt"
34:   rm -rf "$OUT/classes" && mv "$OUT/obf-classes" "$OUT/classes"
35: fi
36: 
37: # الموارد: الخطوط والأيقونات
38: mkdir -p "$OUT/classes/res"
39: cp "$P"/app/src/main/res/font/*.ttf "$OUT/classes/res/" 2>/dev/null || true
40: # 11.0: رسومات الهاتف المتجهة (res/drawable/*.xml) تُرسم كما هي على الكمبيوتر
41: mkdir -p "$OUT/classes/res/drawable" && cp "$P"/app/src/main/res/drawable/*.xml "$OUT/classes/res/drawable/"
42: [ -d "$HERE/res" ] && cp -r "$HERE"/res/* "$OUT/classes/res/"
43: # 10.16: أداتا النقل (صفحتا HTML تعملان في المتصفح بلا إنترنت) لمساعد النقل على الكمبيوتر، من ملفات الهاتف نفسها
44: cp "$P"/app/src/main/assets/migration-tool.html "$P"/app/src/main/assets/smart-import.html "$OUT/classes/res/" 2>/dev/null || true
45: # 10.2: معرّف عميل Google للكمبيوتر (اختياري): GOOGLE_OAUTH_JSON=ملف client_secret_….json المنزّل من Google Cloud
46: [ -n "$GOOGLE_OAUTH_JSON" ] && [ -f "$GOOGLE_OAUTH_JSON" ] && cp "$GOOGLE_OAUTH_JSON" "$OUT/classes/res/google-oauth.json"
47: 
48: # jar واحد يضم المكتبات (تشغيل بلا classpath) — 10.4: ومعها FlatLaf (Apache 2.0) للمظهر الحديث
49: rm -rf "$OUT/merge"; mkdir -p "$OUT/merge"
50: ( cd "$OUT/merge" && unzip -qo "$HERE/lib/sqlite-jdbc.jar" && unzip -qo "$HERE/lib/org.json.jar" && unzip -qo "$HERE/lib/flatlaf.jar" -x 'module-info.class' && unzip -qo "$HERE/lib/zxing-core.jar" -x 'module-info.class' && rm -rf META-INF/*.SF META-INF/*.RSA META-INF/*.DSA META-INF/MANIFEST.MF )
51: cp -r "$OUT/classes/"* "$OUT/merge/"
52: printf "Manifest-Version: 1.0\nMain-Class: com.smart.hawala.desk.Main\n" > "$OUT/manifest.txt"
53: rm -f "$OUT/smart-accountant.jar"
54: ( cd "$OUT/merge" && jar --create --file "$OUT/smart-accountant.jar" --manifest "$OUT/manifest.txt" . )
55: echo "BUILD OK: $OUT/smart-accountant.jar ($(du -h "$OUT/smart-accountant.jar" | cut -f1))"
