1: #!/bin/bash
2: # يبني نسخة ويندوز x64 من «المحاسب الذكي» (مجلد جاهز مضغوط zip) على هذا الجهاز (لينكس):
3: #   RT_WIN=<بيئة جافا مصغّرة للويندوز من jlink بوحدات Temurin ويندوز>  JPKG_WIN=<jmods/jdk.jpackage.jmod من JDK ويندوز>
4: #   SQLITE_FULL=<sqlite-jdbc الكامل (فيه مكتبة ويندوز)>  bash package-win.sh [مجلد الإخراج]
5: # بناء RT_WIN (إصدار jlink يطابق إصدار الوحدات تمامًا):
6: #   jlink --module-path <jdk-win>/jmods --add-modules java.base,java.desktop,java.sql,jdk.crypto.ec,jdk.unsupported,jdk.crypto.mscapi \
7: #         --strip-java-debug-attributes --no-header-files --no-man-pages --compress=zip-0 --output rt-win
8: # المشغّل هو مشغّل jpackage الأصلي لويندوز (بلا نافذة أوامر) بتوقيع Adoptium كما هو دون تعديل؛ الأيقونة في الاختصارات (ملف .ico).
9: set -e
10: HERE="$(cd "$(dirname "$0")" && pwd)"
11: P="$(dirname "$HERE")"
12: OUT="${1:-$HERE/dist}"
13: VERSION=$(grep -o 'VERSION = "[^"]*"' "$P/app/src/main/java/com/smart/hawala/BuildInfo.java" | cut -d'"' -f2)
14: [ -x "$RT_WIN/bin/java.exe" ] || { echo "حدّد RT_WIN"; exit 1; }
15: [ -f "$JPKG_WIN" ] || { echo "حدّد JPKG_WIN"; exit 1; }
16: [ -f "$SQLITE_FULL" ] || { echo "حدّد SQLITE_FULL"; exit 1; }
17: # 12.6: مكتبة القاعدة المشفّرة (io.github.willena:sqlite-jdbc، SQLite3MultipleCiphers) لا xerial العادية
18: unzip -l "$SQLITE_FULL" | grep -q 'org/sqlite/mc/SQLiteMCSqlCipherConfig.class' || { echo "SQLITE_FULL ليست مكتبة التشفير (io.github.willena:sqlite-jdbc)"; exit 1; }
19: [ -f "$R8_JAR" ] || { echo "حدّد R8_JAR (cmdline-tools/lib/r8.jar من أدوات أندرويد) لتعمية النسخة"; exit 1; }
20: 
21: # 12.6: نسخة التوزيع معمّاة دائمًا (الخريطة في desktop/build/obf-map.txt، لا تدخل الحزمة)
22: OBFUSCATE=1 bash "$HERE/build.sh"
23: mkdir -p "$OUT"
24: ICONS="$OUT/icons"; rm -rf "$ICONS"
25: java -cp "$HERE/build/smart-accountant.jar" com.smart.hawala.desk.IconGen "$ICONS"
26: 
27: STAGE="$OUT/stage-win"
28: NAME="SmartAccountant-$VERSION-Windows"
29: R="$STAGE/$NAME"
30: rm -rf "$STAGE"; mkdir -p "$R/app"
31: python3 - "$JPKG_WIN" "$R/SmartAccountant.exe" <<'EOF'
32: import sys, zipfile, io
33: d = open(sys.argv[1], 'rb').read()[4:]
34: z = zipfile.ZipFile(io.BytesIO(d))
35: open(sys.argv[2], 'wb').write(z.read('classes/jdk/jpackage/internal/resources/jpackageapplauncherw.exe'))
36: EOF
37: file "$R/SmartAccountant.exe" | grep -q 'PE32+ executable (GUI) x86-64' || { echo "المشغّل ليس ويندوز x64 بواجهة رسومية"; exit 1; }
38: 
39: cp "$HERE/win/SmartAccountant.cfg" "$R/app/"
40: J="$R/app/smart-accountant.jar"
41: cp "$HERE/build/smart-accountant.jar" "$J"
42: zip -qd "$J" "org/sqlite/native/Linux/*" "org/sqlite/native/Mac/*" >/dev/null || true
43: TMPN="$OUT/native-win"; rm -rf "$TMPN"; mkdir -p "$TMPN"
44: ( cd "$TMPN" && unzip -qo "$SQLITE_FULL" 'org/sqlite/native/Windows/x86_64/*' && zip -qr "$J" org )
45: rm -rf "$TMPN"
46: unzip -l "$J" | grep -q 'org/sqlite/native/Windows/x86_64/sqlitejdbc.dll' || { echo "مكتبة sqlite لويندوز مفقودة"; exit 1; }
47: unzip -l "$J" | grep -q 'org/sqlite/native/\(Linux\|Mac\)/' && { echo "بقيت مكتبات غير ويندوز"; exit 1; }
48: 
49: cp -a "$RT_WIN" "$R/runtime"
50: 
51: python3 - "$ICONS" "$R/SmartAccountant.ico" <<'EOF'
52: import sys, os, struct
53: sizes = [16, 24, 32, 48, 64, 128, 256]
54: imgs = [open(os.path.join(sys.argv[1], "icon_%d.png" % s), "rb").read() for s in sizes]
55: head = struct.pack("<HHH", 0, 1, len(sizes))
56: off = 6 + 16 * len(sizes)
57: ents = b""
58: for s, d in zip(sizes, imgs):
59:     ents += struct.pack("<BBBBHHII", s % 256, s % 256, 0, 0, 1, 32, len(d), off)
60:     off += len(d)
61: open(sys.argv[2], "wb").write(head + ents + b"".join(imgs))
62: EOF
63: 
64: # سكربتات PowerShell بترميز UTF-8 مع BOM (يلزم Windows PowerShell 5.1 لقراءة العربية)، وملفات cmd بنص لاتيني فقط ونهايات CRLF
65: bom() { printf '\xef\xbb\xbf' > "$2"; sed 's/$/\r/' "$1" >> "$2"; }
66: bom "$HERE/win/install.ps1" "$R/install.ps1"
67: bom "$HERE/win/uninstall.ps1" "$R/uninstall.ps1"
68: bom "$HERE/win/اقرأني-ويندوز.txt" "$R/اقرأني-ويندوز.txt"
69: printf '@echo off\r\nchcp 65001 >nul\r\npowershell.exe -NoProfile -ExecutionPolicy Bypass -File "%%~dp0install.ps1"\r\n' > "$R/تثبيت.cmd"
70: printf '@echo off\r\nchcp 65001 >nul\r\ncopy /y "%%~dp0uninstall.ps1" "%%TEMP%%\\sa-uninstall.ps1" >nul\r\npowershell.exe -NoProfile -ExecutionPolicy Bypass -File "%%TEMP%%\\sa-uninstall.ps1"\r\n' > "$R/إلغاء-التثبيت.cmd"
71: 
72: ZIP="$OUT/$NAME.zip"
73: rm -f "$ZIP"
74: # zipfile في بايثون يضع علامة UTF-8 للأسماء العربية فيفكّها مستكشف ويندوز صحيحة
75: python3 - "$STAGE" "$ZIP" <<'EOF'
76: import sys, os, zipfile
77: base, out = sys.argv[1], sys.argv[2]
78: with zipfile.ZipFile(out, "w", zipfile.ZIP_DEFLATED, compresslevel=9) as z:
79:     for root, dirs, files in os.walk(base):
80:         dirs.sort()
81:         for f in sorted(files):
82:             p = os.path.join(root, f)
83:             z.write(p, os.path.relpath(p, base))
84: EOF
85: echo "PACKAGE OK: $ZIP ($(du -h "$ZIP" | cut -f1))"
