104:         String t = sql.trim();
105:         if (t.length() < 6) return false;
106:         String h = t.substring(0, 6).toUpperCase(java.util.Locale.ROOT);
107:         if (!h.startsWith("SELECT") && !h.startsWith("WITH")) return false;
108:         String low = t.toLowerCase(java.util.Locale.ROOT);
109:         if (low.contains("temp.") || low.contains("del_entries")) return false;

113:     private Cursor readerQuery(String sql, String[] args) {
114:         readerLock.lock();
115:         try {
116:             if (reader == null || reader.isClosed()) {
117:                 reader = connect(path, password);
118:                 try (Statement s = reader.createStatement()) {
119:                     s.execute("PRAGMA query_only=1");
120:                     s.execute("PRAGMA cache_size=-16384");
121:                 }
122:             }
123:             try (PreparedStatement p = reader.prepareStatement(sql)) {
124:                 if (args != null) for (int i = 0; i < args.length; i++) {
125:                     if (args[i] == null) p.setNull(i + 1, java.sql.Types.NULL);
126:                     else p.setString(i + 1, args[i]);
127:                 }
128:                 boolean has = p.execute();
129:                 READER_HITS.incrementAndGet();
130:                 if (!has) return new Cursor(new String[0], new ArrayList<>());
131:                 try (ResultSet rs = p.getResultSet()) {
132:                     return read(rs);
133:                 }
134:             }
135:         } catch (SQLException e) {
136:             throw wrap(e, sql);
137:         } catch (Exception e) {
138:             throw new SQLiteException(e.getMessage(), e);
139:         } finally {
