288:                 if (args[i] == null) p.setNull(i + 1, java.sql.Types.NULL);

303:     private static Cursor read(ResultSet rs) throws SQLException {
304:         ResultSetMetaData md = rs.getMetaData();
305:         int n = md.getColumnCount();
306:         String[] cols = new String[n];
307:         for (int i = 0; i < n; i++) cols[i] = md.getColumnLabel(i + 1);
308:         List<Object[]> rows = new ArrayList<>();
309:         while (rs.next()) {
310:             Object[] r = new Object[n];
311:             for (int i = 0; i < n; i++) {
312:                 Object o = rs.getObject(i + 1);
313:                 if (o instanceof Integer || o instanceof Short || o instanceof Byte) o = ((Number) o).longValue();
314:                 else if (o instanceof Float) o = ((Float) o).doubleValue();
315:                 r[i] = o;
316:             }
317:             rows.add(r);
318:         }
319:         return new Cursor(cols, rows);
320:     }

415:         if (args == null) return;
416:         for (int i = 0; i < args.length; i++) {

420:             else if (o instanceof TextArg) {

422:                 else p.setString(k, ((TextArg) o).s);
423:             } else if (o instanceof Long || o instanceof Integer || o instanceof Short || o instanceof Byte) p.setLong(k, ((Number) o).longValue());
424:             else if (o instanceof Boolean) p.setLong(k, (Boolean) o ? 1 : 0);
425:             else if (o instanceof Double || o instanceof Float) p.setDouble(k, ((Number) o).doubleValue());
426:             else if (o instanceof byte[]) p.setBytes(k, (byte[]) o);
