105:     /** سطر واحد جاهز للرسم بالخطوط البديلة والاتجاه الصحيح؛ null لنص فارغ. */
106:     public TextLayout layout(String s) {
107:         if (s == null || s.isEmpty()) return null;
108:         Typeface t = typeface == null ? Typeface.DEFAULT : typeface;
109:         boolean bold = t.isBold() || (flags & FAKE_BOLD_TEXT_FLAG) != 0;
110:         Font base = Typeface.base(t.familyName(), bold);
111:         AttributedString as = new AttributedString(s);
112:         FontKit.apply(as, s, 0, s.length(), base, textSize, bold);
113:         as.addAttribute(TextAttribute.RUN_DIRECTION, firstStrongRtl(s) ? TextAttribute.RUN_DIRECTION_RTL : TextAttribute.RUN_DIRECTION_LTR);
114:         if (letterSpacing != 0) as.addAttribute(TextAttribute.TRACKING, letterSpacing);
115:         if ((flags & STRIKE_THRU_TEXT_FLAG) != 0) as.addAttribute(TextAttribute.STRIKETHROUGH, TextAttribute.STRIKETHROUGH_ON);
116:         if ((flags & UNDERLINE_TEXT_FLAG) != 0) as.addAttribute(TextAttribute.UNDERLINE, TextAttribute.UNDERLINE_ON);
117:         return new TextLayout(as.getIterator(), FRC);
118:     }

120:     public static boolean firstStrongRtl(CharSequence s) {
121:         for (int i = 0; i < s.length(); i++) {
122:             byte d = Character.getDirectionality(s.charAt(i));
123:             if (d == Character.DIRECTIONALITY_RIGHT_TO_LEFT || d == Character.DIRECTIONALITY_RIGHT_TO_LEFT_ARABIC) return true;
124:             if (d == Character.DIRECTIONALITY_LEFT_TO_RIGHT) return false;
125:         }
126:         return false;
127:     }
