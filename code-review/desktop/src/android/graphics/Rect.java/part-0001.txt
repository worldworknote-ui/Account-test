12:     public int height() { return bottom - top; }

14:     public int centerY() { return (top + bottom) >> 1; }

16:     public float exactCenterY() { return (top + bottom) * 0.5f; }
17:     public boolean isEmpty() { return left >= right || top >= bottom; }
18:     public boolean contains(int x, int y) { return left < right && top < bottom && x >= left && x < right && y >= top && y < bottom; }
19:     public void offset(int dx, int dy) { left += dx; right += dx; top += dy; bottom += dy; }
20:     public void offsetTo(int x, int y) { right += x - left; bottom += y - top; left = x; top = y; }
21:     public void inset(int dx, int dy) { left += dx; top += dy; right -= dx; bottom -= dy; }
22:     public boolean intersect(Rect r) { return intersect(r.left, r.top, r.right, r.bottom); }
23:     public boolean intersect(int l, int t, int r, int b) {
24:         if (left < r && l < right && top < b && t < bottom) {
25:             if (left < l) left = l;
26:             if (top < t) top = t;
27:             if (right > r) right = r;
28:             if (bottom > b) bottom = b;
29:             return true;
30:         }
31:         return false;
32:     }
33:     public void union(Rect r) {
34:         if (r.isEmpty()) return;
35:         if (isEmpty()) { set(r); return; }
36:         left = Math.min(left, r.left); top = Math.min(top, r.top); right = Math.max(right, r.right); bottom = Math.max(bottom, r.bottom);
37:     }
38:     @Override public boolean equals(Object o) {
39:         if (!(o instanceof Rect)) return false;
40:         Rect r = (Rect) o;
41:         return left == r.left && top == r.top && right == r.right && bottom == r.bottom;
42:     }
43:     @Override public int hashCode() { return ((left * 31 + top) * 31 + right) * 31 + bottom; }
44:     @Override public String toString() { return "Rect(" + left + ", " + top + " - " + right + ", " + bottom + ")"; }
45: }
