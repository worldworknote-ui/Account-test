83:     /** شكل الرسم داخل المستطيل r. */
84:     private Shape shapeIn(float l, float t, float r, float b) {
85:         float ww = r - l, hh = b - t;
86:         if (shape == OVAL) return new Ellipse2D.Float(l, t, ww, hh);
87:         if (shape == LINE) return new Line2D.Float(l, t + hh / 2, r, t + hh / 2);
88:         if (radii != null) return Path.roundRect(l, t, r, b, radii);
89:         if (radius > 0) {
90:             float rr = Math.min(radius, Math.min(ww, hh) / 2f);
91:             return new RoundRectangle2D.Float(l, t, ww, hh, 2 * rr, 2 * rr);
92:         }
93:         return new Rectangle2D.Float(l, t, ww, hh);
94:     }

101:     @Override public void draw(Canvas c) {
102:         Rect bb = getBounds();
103:         if (bb.isEmpty()) return;
104:         float half = strokeWidth / 2f;
105:         java.awt.Graphics2D g = c.graphics();
106:         float a = c.alpha * alpha / 255f;
107:         int[] st = getState();
108:         if (shape != LINE) {
109:             // التعبئة داخل نصف عرض الحد (كما في أندرويد)
110:             Shape fill = shapeIn(bb.left + half, bb.top + half, bb.right - half, bb.bottom - half);
111:             java.awt.Paint p = null;
112:             if (colors != null && colors.length > 0) {
113:                 p = gradient(bb, a);
114:             } else if (solid != null) {
115:                 int col = solid.getColorForState(st, solid.getDefaultColor());
116:                 if (hasTint()) col = tintColor();
117:                 int al = Math.round(Color.alpha(col) * a);
118:                 if (al > 0) p = new java.awt.Color(Color.red(col), Color.green(col), Color.blue(col), al);
119:             } else if (hasTint()) {
120:                 int col = tintColor();
121:                 p = new java.awt.Color(Color.red(col), Color.green(col), Color.blue(col), Math.round(Color.alpha(col) * a));
122:             }
123:             if (p != null) {
124:                 g.setPaint(p);
125:                 g.fill(fill);
126:             }
127:         }
128:         if (strokeWidth > 0 && strokeColor != null) {
129:             int col = strokeColor.getColorForState(st, strokeColor.getDefaultColor());
130:             int al = Math.round(Color.alpha(col) * a);
131:             if (al > 0) {
132:                 Shape s = shapeIn(bb.left + half, bb.top + half, bb.right - half, bb.bottom - half);
133:                 java.awt.BasicStroke bs = dashW > 0
134:                         ? new java.awt.BasicStroke(strokeWidth, java.awt.BasicStroke.CAP_BUTT, java.awt.BasicStroke.JOIN_ROUND, 4f, new float[]{dashW, Math.max(0.01f, dashGap)}, 0)
135:                         : new java.awt.BasicStroke(strokeWidth, java.awt.BasicStroke.CAP_BUTT, java.awt.BasicStroke.JOIN_ROUND);
136:                 g.setPaint(new java.awt.Color(Color.red(col), Color.green(col), Color.blue(col), al));
137:                 g.fill(bs.createStrokedShape(s));
138:             }
139:         }
140:     }

142:     private java.awt.Paint gradient(Rect b, float a) {
143:         int al = Math.round(255 * a);
144:         if (colors.length == 1) {
145:             int c0 = colors[0];
146:             return new java.awt.Color(Color.red(c0), Color.green(c0), Color.blue(c0), Color.alpha(c0) * al / 255);
147:         }
148:         if (gradientType == RADIAL_GRADIENT) {
149:             float rr = gradRadius <= 1f ? gradRadius * Math.min(b.width(), b.height()) : gradRadius;
150:             return new RadialGradient(b.left + cx * b.width(), b.top + cy * b.height(), Math.max(1, rr), colors, offsets, Shader.TileMode.CLAMP).awt(al);
151:         }
152:         float x0, y0, x1, y1;
153:         float l = b.left, t = b.top, r = b.right, bo = b.bottom;
154:         switch (orientation) {
155:             case TOP_BOTTOM: x0 = l; y0 = t; x1 = l; y1 = bo; break;
156:             case TR_BL: x0 = r; y0 = t; x1 = l; y1 = bo; break;
157:             case RIGHT_LEFT: x0 = r; y0 = t; x1 = l; y1 = t; break;
158:             case BR_TL: x0 = r; y0 = bo; x1 = l; y1 = t; break;
159:             case BOTTOM_TOP: x0 = l; y0 = bo; x1 = l; y1 = t; break;
160:             case BL_TR: x0 = l; y0 = bo; x1 = r; y1 = t; break;
161:             case LEFT_RIGHT: x0 = l; y0 = t; x1 = r; y1 = t; break;
162:             default: x0 = l; y0 = t; x1 = r; y1 = bo; break;
163:         }
164:         return new LinearGradient(x0, y0, x1, y1, colors, offsets, Shader.TileMode.CLAMP).awt(al);
165:     }
