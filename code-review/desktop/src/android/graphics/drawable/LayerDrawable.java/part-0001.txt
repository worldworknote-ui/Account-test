37:     @Override protected void onBoundsChange(Rect b) {
38:         for (Layer l : layers) {
39:             if (l.d == null) continue;
40:             int L = b.left + l.l, T = b.top + l.t, R = b.right - l.r, B = b.bottom - l.b;
41:             if (l.w >= 0 || l.h >= 0) {
42:                 int w = l.w >= 0 ? l.w : R - L, h = l.h >= 0 ? l.h : B - T;
43:                 int g = l.gravity;
44:                 int x = (g & android.view.Gravity.HORIZONTAL_GRAVITY_MASK) == android.view.Gravity.RIGHT ? R - w
45:                         : (g & android.view.Gravity.HORIZONTAL_GRAVITY_MASK) == android.view.Gravity.CENTER_HORIZONTAL ? (L + R - w) / 2 : L;
46:                 int y = (g & android.view.Gravity.VERTICAL_GRAVITY_MASK) == android.view.Gravity.BOTTOM ? B - h
47:                         : (g & android.view.Gravity.VERTICAL_GRAVITY_MASK) == android.view.Gravity.CENTER_VERTICAL ? (T + B - h) / 2 : T;
48:                 l.d.setBounds(x, y, x + w, y + h);
49:             } else l.d.setBounds(L, T, R, B);
50:         }
51:     }
