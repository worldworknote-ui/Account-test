5: public class ColorDrawable extends Drawable {
6:     private int color;
7:     private int alpha = 255;
8:     public ColorDrawable() {}
9:     public ColorDrawable(int c) { color = c; }
10:     public int getColor() { return color; }
11:     public void setColor(int c) { color = c; invalidateSelf(); }
12:     @Override public void setAlpha(int a) { alpha = a; invalidateSelf(); }
13:     @Override public int getAlpha() { return alpha; }
14:     @Override public void draw(Canvas c) {
15:         int col = hasTint() ? tintColor() : color;
16:         int a = Color.alpha(col) * alpha / 255;
17:         if (a == 0) return;
18:         Paint p = new Paint();
19:         p.setColor((a << 24) | (col & 0xFFFFFF));
20:         c.drawRect(getBounds(), p);
21:     }
22: }
