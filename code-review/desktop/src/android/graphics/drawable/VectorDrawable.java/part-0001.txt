32:     private static Grad grad(Element attr) {
33:         NodeList gl = attr.getElementsByTagName("gradient");
34:         if (gl.getLength() == 0) return null;
35:         Element g = (Element) gl.item(0);
36:         Grad d = new Grad();
37:         d.radial = "radial".equals(g.getAttribute("android:type"));
38:         d.x1 = num(g.getAttribute("android:startX"), 0);
39:         d.y1 = num(g.getAttribute("android:startY"), 0);
40:         d.x2 = num(g.getAttribute("android:endX"), 0);
41:         d.y2 = num(g.getAttribute("android:endY"), 0);
42:         d.cx = num(g.getAttribute("android:centerX"), 0);
43:         d.cy = num(g.getAttribute("android:centerY"), 0);
44:         d.r = num(g.getAttribute("android:gradientRadius"), 1);
45:         List<float[]> st = new ArrayList<>();
46:         List<Integer> cl = new ArrayList<>();
47:         NodeList items = g.getElementsByTagName("item");
48:         for (int i = 0; i < items.getLength(); i++) {
49:             Element it = (Element) items.item(i);
50:             st.add(new float[]{num(it.getAttribute("android:offset"), i / (float) Math.max(1, items.getLength() - 1))});
51:             cl.add(color(it.getAttribute("android:color")));
52:         }
53:         if (cl.isEmpty()) {
54:             if (g.hasAttribute("android:startColor")) { st.add(new float[]{0}); cl.add(color(g.getAttribute("android:startColor"))); }
55:             if (g.hasAttribute("android:centerColor")) { st.add(new float[]{0.5f}); cl.add(color(g.getAttribute("android:centerColor"))); }
56:             if (g.hasAttribute("android:endColor")) { st.add(new float[]{1}); cl.add(color(g.getAttribute("android:endColor"))); }
57:         }
58:         if (cl.isEmpty()) return null;
59:         if (cl.size() == 1) { st.add(new float[]{1}); cl.add(cl.get(0)); }
60:         d.stops = new float[st.size()];
61:         d.colors = new int[cl.size()];
62:         float last = -1;
63:         for (int i = 0; i < st.size(); i++) {
64:             // الإيقافات يجب أن تزيد زيادة صارمة في جافا
65:             float o = Math.max(0, Math.min(1, st.get(i)[0]));
66:             if (o <= last) o = Math.min(1, last + 0.0001f);
67:             d.stops[i] = o;
68:             last = o;
69:             d.colors[i] = cl.get(i);
70:         }
71:         for (int i = 1; i < d.stops.length; i++) if (d.stops[i] <= d.stops[i - 1]) return null;
72:         return d;
73:     }

104:     private static VectorDrawable parse(String name) {

116:             v.ih = Math.round(num(root.getAttribute("android:height"), 24));
117:             String tint = root.getAttribute("android:tint");

126:     private static void walk(VectorDrawable v, Element e, AffineTransform at) {
127:         NodeList ch = e.getChildNodes();
128:         for (int i = 0; i < ch.getLength(); i++) {
129:             if (!(ch.item(i) instanceof Element)) continue;
130:             Element c = (Element) ch.item(i);
131:             if ("group".equals(c.getTagName())) {
132:                 AffineTransform t = new AffineTransform(at);
133:                 float px = num(c.getAttribute("android:pivotX"), 0), py = num(c.getAttribute("android:pivotY"), 0);
134:                 float tx = num(c.getAttribute("android:translateX"), 0), ty = num(c.getAttribute("android:translateY"), 0);
135:                 float sx = num(c.getAttribute("android:scaleX"), 1), sy = num(c.getAttribute("android:scaleY"), 1);
136:                 float rot = num(c.getAttribute("android:rotation"), 0);
137:                 t.translate(tx + px, ty + py);
138:                 t.rotate(Math.toRadians(rot));
139:                 t.scale(sx, sy);
140:                 t.translate(-px, -py);
141:                 walk(v, c, t);
142:             } else if ("path".equals(c.getTagName())) {
143:                 P p = new P();
144:                 p.path = SvgPath.parse(c.getAttribute("android:pathData"));
145:                 p.fill = c.hasAttribute("android:fillColor") ? color(c.getAttribute("android:fillColor")) : 0;
146:                 p.stroke = c.hasAttribute("android:strokeColor") ? color(c.getAttribute("android:strokeColor")) : 0;
147:                 p.sw = num(c.getAttribute("android:strokeWidth"), 0);
148:                 p.fillAlpha = num(c.getAttribute("android:fillAlpha"), 1);
149:                 p.strokeAlpha = num(c.getAttribute("android:strokeAlpha"), 1);
150:                 String cap = c.getAttribute("android:strokeLineCap"), join = c.getAttribute("android:strokeLineJoin");
151:                 p.cap = "round".equals(cap) ? java.awt.BasicStroke.CAP_ROUND : "square".equals(cap) ? java.awt.BasicStroke.CAP_SQUARE : java.awt.BasicStroke.CAP_BUTT;
152:                 p.join = "round".equals(join) ? java.awt.BasicStroke.JOIN_ROUND : "bevel".equals(join) ? java.awt.BasicStroke.JOIN_BEVEL : java.awt.BasicStroke.JOIN_MITER;
153:                 p.evenOdd = "evenOdd".equals(c.getAttribute("android:fillType"));
154:                 NodeList kids = c.getChildNodes();
155:                 for (int k = 0; k < kids.getLength(); k++) {
156:                     if (!(kids.item(k) instanceof Element)) continue;

162:                 }

164:                 p.t = at;
