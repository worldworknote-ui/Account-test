84:     public boolean op(Path path, Op op) { return op(this, path, op); }
85:     public boolean op(Path a, Path b, Op op) {
86:         Area x = new Area(a.p), y = new Area(b.p);
87:         switch (op) {
88:             case DIFFERENCE: x.subtract(y); break;
89:             case INTERSECT: x.intersect(y); break;
90:             case UNION: x.add(y); break;
91:             case XOR: x.exclusiveOr(y); break;
92:             case REVERSE_DIFFERENCE: y.subtract(x); x = y; break;
93:         }
94:         p = new Path2D.Float(x);
95:         return true;
96:     }

98:     /** مستطيل بزوايا مختلفة: radii = [tlx,tly, trx,try, brx,bry, blx,bly]. */
99:     public static Path2D.Float roundRect(float l, float t, float r, float b, float[] radii) {
100:         float w = r - l, h = b - t;
101:         float[] q = new float[8];
102:         for (int i = 0; i < 8; i++) q[i] = radii == null || i >= radii.length ? 0 : Math.max(0, radii[i]);
103:         // تصغير متناسب إن تجاوزت الأنصاف الضلع (كما في أندرويد)
104:         float s = 1f;
105:         s = Math.min(s, fit(w, q[0] + q[2]));
106:         s = Math.min(s, fit(w, q[6] + q[4]));
107:         s = Math.min(s, fit(h, q[1] + q[7]));
108:         s = Math.min(s, fit(h, q[3] + q[5]));
109:         for (int i = 0; i < 8; i++) q[i] *= s;
110:         final float k = 0.5522847f;
111:         Path2D.Float p = new Path2D.Float();
112:         p.moveTo(l + q[0], t);
113:         p.lineTo(r - q[2], t);
114:         if (q[2] > 0 || q[3] > 0) p.curveTo(r - q[2] + q[2] * k, t, r, t + q[3] - q[3] * k, r, t + q[3]);
115:         p.lineTo(r, b - q[5]);
116:         if (q[4] > 0 || q[5] > 0) p.curveTo(r, b - q[5] + q[5] * k, r - q[4] + q[4] * k, b, r - q[4], b);
117:         p.lineTo(l + q[6], b);
118:         if (q[6] > 0 || q[7] > 0) p.curveTo(l + q[6] - q[6] * k, b, l, b - q[7] + q[7] * k, l, b - q[7]);
119:         p.lineTo(l, t + q[1]);
120:         if (q[0] > 0 || q[1] > 0) p.curveTo(l, t + q[1] - q[1] * k, l + q[0] - q[0] * k, t, l + q[0], t);
121:         p.closePath();
122:         return p;
123:     }
