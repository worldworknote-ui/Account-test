20:         try {

38:         return null;
39:     }

45:     }

47:     public Builder buildUpon() { return new Builder(s); }
48:     /** الملف المحلي لهذا العنوان (file://). */
49:     public File file() {
50:         // 11.2: ملفات المشاركة الداخلية (content://<الحزمة>.files/<الاسم>) = مجلد cache/shared للتطبيق
51:         if (s.startsWith("content://")) {
52:             String rest = s.substring("content://".length());
53:             int sl = rest.indexOf('/');
54:             String auth = sl < 0 ? rest : rest.substring(0, sl);
55:             String name = sl < 0 ? "" : decode(rest.substring(sl + 1));
56:             android.content.Context c = android.content.Context.app;
57:             if (auth.endsWith(".files") && c != null && !name.isEmpty() && !name.contains("/") && !name.contains("..")) return new File(new File(c.getCacheDir(), "shared"), name);
58:         }
59:         String x = s;
60:         int q = x.indexOf('?');
61:         if (q >= 0) x = x.substring(0, q);
62:         try { return new File(new java.net.URI(x)); } catch (Exception e) { return new File(decode(x.replaceFirst("^file:/+", "/"))); }
63:     }
64:     public boolean isFile() { return s.startsWith("file:") || (s.startsWith("content://") && file() != null && s.substring(10).split("/")[0].endsWith(".files")); }

79:         public Builder encodedPath(String p) { s = s + p; return this; }
80:         public Builder fragment(String f) { s = s + "#" + f; return this; }

82:         public Uri build() { return new Uri(s); }
