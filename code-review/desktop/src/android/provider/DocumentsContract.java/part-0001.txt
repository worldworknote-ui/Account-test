29:     public static Uri createDocument(ContentResolver r, Uri parent, String mime, String name) throws java.io.FileNotFoundException {
30:         File dir = parent.file();
31:         dir.mkdirs();
32:         File f = new File(dir, name);
33:         if (Document.MIME_TYPE_DIR.equals(mime)) { f.mkdirs(); return Uri.fromFile(f); }
34:         try {
35:             if (!f.exists() && !f.createNewFile()) throw new java.io.FileNotFoundException(f.getPath());
36:         } catch (java.io.IOException e) {
37:             throw new java.io.FileNotFoundException(e.getMessage());
38:         }

47:         File f = doc.file(), t = new File(f.getParentFile(), name);

52:     /** للاستعلام من ContentResolver. */
53:     public static Cursor query(Uri u, String[] proj) {
54:         String[] cols = proj != null ? proj : new String[]{Document.COLUMN_DOCUMENT_ID, Document.COLUMN_DISPLAY_NAME, Document.COLUMN_MIME_TYPE,
55:                 Document.COLUMN_LAST_MODIFIED, Document.COLUMN_SIZE};
56:         MatrixCursor c = new MatrixCursor(cols);
57:         File f = u.file();
58:         if (u.toString().endsWith(CHILDREN)) {
59:             File[] l = f.listFiles();
60:             if (l != null) for (File x : l) if (!x.getName().startsWith(".")) c.addRow(row(cols, x));
61:         } else if (f.exists()) c.addRow(row(cols, f));
62:         return c;
63:     }
64:     private static Object[] row(String[] cols, File f) {
65:         Object[] r = new Object[cols.length];
66:         for (int i = 0; i < cols.length; i++) {
67:             switch (cols[i]) {
68:                 case Document.COLUMN_DOCUMENT_ID: r[i] = f.getAbsolutePath(); break;
69:                 case Document.COLUMN_DISPLAY_NAME: r[i] = f.getName(); break;
70:                 case Document.COLUMN_MIME_TYPE: r[i] = f.isDirectory() ? Document.MIME_TYPE_DIR : "application/octet-stream"; break;
71:                 case Document.COLUMN_LAST_MODIFIED: r[i] = f.lastModified(); break;
72:                 case Document.COLUMN_SIZE: r[i] = f.length(); break;
73:                 case Document.COLUMN_FLAGS: r[i] = 0x4F; break;
74:                 default: r[i] = null;
75:             }
76:         }
77:         return r;
78:     }
