56:         String user = System.getProperty("user.home");
57:         String os = System.getProperty("os.name", "").toLowerCase();
58:         if (os.contains("mac")) return new File(user, "Library/Application Support/SmartAccountant");
59:         if (os.contains("win")) {

119:     private static final java.util.Map<String, Object> services = new java.util.HashMap<>();
120:     public Object getSystemService(String name) {
121:         synchronized (services) {
122:             Object o = services.get(name);
123:             if (o != null) return o;
124:             switch (name) {
125:                 case CLIPBOARD_SERVICE: o = new ClipboardManager(); break;
126:                 case INPUT_METHOD_SERVICE: o = new android.view.inputmethod.InputMethodManager(); break;
127:                 case NOTIFICATION_SERVICE: o = new android.app.NotificationManager(); break;
128:                 case JOB_SCHEDULER_SERVICE: o = new android.app.job.JobScheduler(this); break;
129:                 case KEYGUARD_SERVICE: o = new android.app.KeyguardManager(); break;
130:                 case CONNECTIVITY_SERVICE: o = new android.net.ConnectivityManager(); break;
131:                 case POWER_SERVICE: o = new android.os.PowerManager(); break;
132:                 case USER_SERVICE: o = new android.os.UserManager(); break;
133:                 case WINDOW_SERVICE: o = new android.view.WindowManagerImpl(); break;
134:                 case ALARM_SERVICE: o = new android.app.AlarmManager(); break;
135:                 case VIBRATOR_SERVICE: o = new android.os.Vibrator(); break;
136:                 default: return null;
137:             }
138:             services.put(name, o);
139:             return o;
140:         }
141:     }
142:     @SuppressWarnings("unchecked")
143:     public final <T> T getSystemService(Class<T> c) {
144:         for (String n : new String[]{CLIPBOARD_SERVICE, INPUT_METHOD_SERVICE, NOTIFICATION_SERVICE, JOB_SCHEDULER_SERVICE, KEYGUARD_SERVICE, CONNECTIVITY_SERVICE, POWER_SERVICE, USER_SERVICE, WINDOW_SERVICE, ALARM_SERVICE, VIBRATOR_SERVICE}) {
145:             Object o = getSystemService(n);
