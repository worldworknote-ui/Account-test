34:     static synchronized Map<Integer, String[]> names() {
35:         if (names != null) return names;
36:         names = new HashMap<>();
37:         for (String cls : new String[]{"com.smart.hawala.R", "android.R"}) {
38:             try {
39:                 for (Class<?> k : Class.forName(cls).getDeclaredClasses()) {
40:                     for (java.lang.reflect.Field f : k.getFields()) {
41:                         if (f.getType() == int.class) names.put(f.getInt(null), new String[]{k.getSimpleName(), f.getName()});
42:                     }
43:                 }
44:             } catch (Throwable ignored) { }
45:         }
46:         return names;
47:     }

49:     public String getResourceEntryName(int id) { String[] n = names().get(id); if (n == null) throw new NotFoundException("#" + Integer.toHexString(id)); return n[1]; }
50:     public String getResourceTypeName(int id) { String[] n = names().get(id); if (n == null) throw new NotFoundException(); return n[0]; }
51:     public String getResourceName(int id) { String[] n = names().get(id); if (n == null) throw new NotFoundException(); return "com.smart.hawala:" + n[0] + "/" + n[1]; }
52:     public int getIdentifier(String name, String type, String pkg) {
53:         for (Map.Entry<Integer, String[]> e : names().entrySet()) if (e.getValue()[0].equals(type) && e.getValue()[1].equals(name)) return e.getKey();
54:         return 0;
55:     }

59:     public Drawable getDrawable(int id, Theme t) {
60:         String[] n = names().get(id);
61:         if (n == null) return null;
62:         if ("mipmap".equals(n[0])) return VectorDrawable.load("ic_logo");
63:         if ("drawable".equals(n[0])) {
64:             Drawable d = VectorDrawable.load(n[1]);
65:             if (d == null && "android.R".equals(null)) return null;
66:             return d != null ? d : VectorDrawable.load(androidFallback(n[1]));
67:         }
68:         return null;
69:     }

71:     private static String androidFallback(String name) {
72:         if (name.contains("share")) return "ic_share";
73:         if (name.contains("alert")) return "ic_warn";
74:         if (name.contains("search")) return "ic_search";
75:         if (name.contains("add")) return "ic_plus";
76:         if (name.contains("delete")) return "ic_trash";
77:         if (name.contains("sync")) return "ic_sync";
78:         if (name.contains("close")) return "ic_close";
79:         return "ic_info";

116:     /** طبقات السمة: الخط (FontPlex...) والوضع الليلي. */
117:     public void applyStyle(int style) {
118:         String[] n = names().get(style);
119:         if (n == null) return;
120:         switch (n[1]) {
121:             case "FontPlex": Typeface.themeFamily = "plex"; break;
122:             case "FontCairo": Typeface.themeFamily = "cairo"; break;
123:             case "FontKufi": Typeface.themeFamily = "kufi"; break;
124:             case "FontSystem": Typeface.themeFamily = "system"; break;
125:             default: break;
