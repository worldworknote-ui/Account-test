24:             return;
25:         }

27:             int r;
28:             try {
29:                 r = SystemAuth.verify(title == null ? "" : title);
30:             } finally {
31:                 running.set(false);
32:             }
33:             if (s != null && s.isCanceled()) return;
34:             if (r == 0) e.execute(() -> cb.onAuthenticationSucceeded(new AuthenticationResult()));
