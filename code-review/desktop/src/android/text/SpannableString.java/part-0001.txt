7:         text = s.toString();

9:                 : s instanceof SpannableStringBuilder ? ((SpannableStringBuilder) s).spans.copy(0, text.length()) : new SpanStore();

13:     @Override public void removeSpan(Object w) { spans.remove(w); }

16:     @Override public int getSpanEnd(Object w) { SpanStore.S x = spans.find(w); return x == null ? -1 : x.end; }
17:     @Override public int getSpanFlags(Object w) { SpanStore.S x = spans.find(w); return x == null ? 0 : x.flags; }
18:     @Override public int nextSpanTransition(int s, int l, Class t) { return spans.next(s, l, t); }
19:     @Override public int length() { return text.length(); }
20:     @Override public char charAt(int i) { return text.charAt(i); }
21:     @Override public CharSequence subSequence(int s, int e) { SpannableString r = new SpannableString(text.substring(s, e)); r.spans.list.addAll(spans.copy(s, e).list); return r; }
22:     @Override public String toString() { return text; }
23:     @Override public boolean equals(Object o) { return o instanceof CharSequence && text.equals(o.toString()); }
