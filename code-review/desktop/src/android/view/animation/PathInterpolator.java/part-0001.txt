3: /** منحنى بيزييه مكعب (مثل cubic-bezier في CSS). */
4: public class PathInterpolator extends BaseInterpolator {
5:     private final float x1, y1, x2, y2;
6:     public PathInterpolator(float cx, float cy) { this(cx * 2f / 3f, cy * 2f / 3f, 1f / 3f + cx * 2f / 3f, 1f / 3f + cy * 2f / 3f); }
7:     public PathInterpolator(float x1, float y1, float x2, float y2) { this.x1 = x1; this.y1 = y1; this.x2 = x2; this.y2 = y2; }
8:     public PathInterpolator(android.graphics.Path p) { this(0.4f, 0f, 0.2f, 1f); }
9:     private static float bez(float t, float a, float b) { float u = 1 - t; return 3 * u * u * t * a + 3 * u * t * t * b + t * t * t; }
10:     @Override public float getInterpolation(float x) {
11:         if (x <= 0) return 0;
12:         if (x >= 1) return 1;
13:         float lo = 0, hi = 1, t = x;
14:         for (int i = 0; i < 30; i++) {
15:             t = (lo + hi) / 2;
16:             float v = bez(t, x1, x2);
17:             if (Math.abs(v - x) < 1e-5f) break;
18:             if (v < x) lo = t; else hi = t;
19:         }
20:         return bez(t, y1, y2);
21:     }
22: }
