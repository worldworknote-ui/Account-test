3: public class Gravity {
4:     public static final int NO_GRAVITY = 0, AXIS_SPECIFIED = 1, AXIS_PULL_BEFORE = 2, AXIS_PULL_AFTER = 4, AXIS_CLIP = 8, AXIS_X_SHIFT = 0, AXIS_Y_SHIFT = 4;
5:     public static final int TOP = 0x30, BOTTOM = 0x50, LEFT = 0x03, RIGHT = 0x05, CENTER_VERTICAL = 0x10, CENTER_HORIZONTAL = 0x01,
6:             FILL_VERTICAL = 0x70, FILL_HORIZONTAL = 0x07, CENTER = 0x11, FILL = 0x77, CLIP_VERTICAL = 0x80, CLIP_HORIZONTAL = 0x08,
7:             RELATIVE_LAYOUT_DIRECTION = 0x00800000, START = RELATIVE_LAYOUT_DIRECTION | LEFT, END = RELATIVE_LAYOUT_DIRECTION | RIGHT,
8:             HORIZONTAL_GRAVITY_MASK = 0x07, VERTICAL_GRAVITY_MASK = 0x70, RELATIVE_HORIZONTAL_GRAVITY_MASK = START | END, DISPLAY_CLIP_VERTICAL = 0x10000000,
9:             DISPLAY_CLIP_HORIZONTAL = 0x01000000;

11:     /** يحوّل START/END إلى يسار/يمين حسب الاتجاه. */
12:     public static int getAbsoluteGravity(int g, int dir) {
13:         int r = g;
14:         if ((r & RELATIVE_LAYOUT_DIRECTION) > 0) {
15:             boolean rtl = dir == View.LAYOUT_DIRECTION_RTL;
16:             if ((r & START) == START) {
17:                 r &= ~START;
18:                 r |= rtl ? RIGHT : LEFT;
19:             } else if ((r & END) == END) {
20:                 r &= ~END;
21:                 r |= rtl ? LEFT : RIGHT;
22:             }
23:             r &= ~RELATIVE_LAYOUT_DIRECTION;
24:         }
25:         return r;
26:     }
27: 
28:     public static boolean isHorizontal(int g) { return g > 0 && (g & RELATIVE_HORIZONTAL_GRAVITY_MASK) != 0; }
29:     public static boolean isVertical(int g) { return g > 0 && (g & VERTICAL_GRAVITY_MASK) != 0; }
30: 
31:     public static void apply(int g, int w, int h, android.graphics.Rect container, android.graphics.Rect out, int dir) {
32:         apply(getAbsoluteGravity(g, dir), w, h, container, out);
33:     }
34: 
35:     public static void apply(int g, int w, int h, android.graphics.Rect c, android.graphics.Rect o) {
36:         int hg = g & HORIZONTAL_GRAVITY_MASK, vg = g & VERTICAL_GRAVITY_MASK;
37:         if (hg == RIGHT) { o.right = c.right; o.left = o.right - w; }
38:         else if (hg == CENTER_HORIZONTAL) { o.left = c.left + (c.width() - w) / 2; o.right = o.left + w; }
39:         else if (hg == FILL_HORIZONTAL) { o.left = c.left; o.right = c.right; }
40:         else { o.left = c.left; o.right = o.left + w; }
41:         if (vg == BOTTOM) { o.bottom = c.bottom; o.top = o.bottom - h; }
42:         else if (vg == CENTER_VERTICAL) { o.top = c.top + (c.height() - h) / 2; o.bottom = o.top + h; }
43:         else if (vg == FILL_VERTICAL) { o.top = c.top; o.bottom = c.bottom; }
44:         else { o.top = c.top; o.bottom = o.top + h; }
45:     }
46: }
