257:     /** 11.5: ما يفعله السحب الأفقي وزر «رجوع» في الفأرة (الرجوع للصفحة السابقة). */
258:     public static Runnable swipeBack;
259:     private static float swipeSum;
260:     private static long swipeAt, swipeDone;
261: 
262:     static void swipe(float d) {
263:         long now = System.currentTimeMillis();
264:         if (now - swipeAt > 350) swipeSum = 0;
265:         swipeAt = now;
266:         swipeSum += d;
267:         if (Math.abs(swipeSum) > 260 && now - swipeDone > 900 && swipeBack != null && android.app.WindowStack.top() != null && android.app.WindowStack.roots().size() == 1) {
268:             swipeDone = now;
269:             swipeSum = 0;
270:             swipeBack.run();
271:         }
272:     }

301:     private View hoveredEditor(MouseEvent e) {
302:         View v = deepest(lx(e), ly(e));
303:         while (v != null) {
304:             if (v.onCheckIsTextEditor()) return v;
305:             if (v.isClickable()) return null;
306:             v = v.parent instanceof View ? (View) v.parent : null;
307:         }
308:         return null;

311:     static float[] local(View v, float x, float y) {
312:         float[] o = {0, 0};

317:     private void hover(MouseEvent e) {
318:         View v = deepest(lx(e), ly(e));
319:         View target = null;
320:         boolean text = false;
321:         while (v != null) {
322:             if (v.onCheckIsTextEditor() && v.isEnabled()) { text = true; break; }
323:             if ((v.isClickable() || v.isLongClickable()) && v.isEnabled()) { target = v; break; }
324:             v = v.parent instanceof View ? (View) v.parent : null;
325:         }
326:         setHover(target);
327:         pane.setCursor(text ? Cursor.getPredefinedCursor(Cursor.TEXT_CURSOR) : target != null ? Cursor.getPredefinedCursor(Cursor.HAND_CURSOR) : Cursor.getDefaultCursor());
328:         View tip = deepest(lx(e), ly(e));
329:         String t = null;
330:         while (tip != null && t == null) {
331:             if (tip.getTooltipText() != null) t = tip.getTooltipText().toString();
332:             tip = tip.parent instanceof View ? (View) tip.parent : null;
333:         }
334:         pane.setToolTipText(t);
335:     }
