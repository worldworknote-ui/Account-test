91:     /** القياس والتخطيط ثم مستمعو التخطيط (حتى 4 مرات إن طلبوا تخطيطًا جديدًا). */
92:     public void traversal() {
93:         traversalScheduled = false;
94:         int w = width(), h = height();
95:         if (w <= 0 || h <= 0) return;
96:         for (int pass = 0; pass < 4 && layoutWanted; pass++) {
97:             layoutWanted = false;
98:             inLayout = true;
99:             try {
100:                 view.measure(View.MeasureSpec.makeMeasureSpec(w, View.MeasureSpec.EXACTLY), View.MeasureSpec.makeMeasureSpec(h, View.MeasureSpec.EXACTLY));
101:                 view.layout(0, 0, w, h);
102:             } finally {
103:                 inLayout = false;
104:             }
105:             ai.observer.dispatchOnGlobalLayout();
106:         }
107:         if (!firstFocusDone && view.getWidth() > 0) {
108:             firstFocusDone = true;
109:             if (focused == null) focusFirst(view);
110:         }
111:         TextOverlay.sync(this);
112:         pane.repaint();
113:     }

115:     /** أول خانة كتابة في النافذة تأخذ التركيز (كما في أندرويد). */
116:     private boolean focusFirst(View v) {
117:         if (v.getVisibility() != View.VISIBLE) return false;
118:         if (v.isFocusableInTouchMode() && v.isEnabled() && v.onCheckIsTextEditor()) { setFocused(v); return true; }
119:         if (v instanceof ViewGroup) {
120:             ViewGroup g = (ViewGroup) v;
121:             if (g.getDescendantFocusability() == ViewGroup.FOCUS_BLOCK_DESCENDANTS) return false;
122:             if (g.isFocusableInTouchMode() && g.getDescendantFocusability() == ViewGroup.FOCUS_BEFORE_DESCENDANTS) return true;
123:             for (int i = 0; i < g.getChildCount(); i++) if (focusFirst(g.getChildAt(i))) return true;
124:         }
125:         return false;
126:     }

129:     public void setFocused(View v) {
130:         if (v == focused) {
131:             if (v != null && v.onCheckIsTextEditor()) TextOverlay.show(this, v, -1, -1);
132:             return;
133:         }
134:         View old = focused;
135:         focused = v;
136:         if (old != null) old.setFocusedInternal(false);
137:         if (v != null) v.setFocusedInternal(true);
138:         ai.observer.dispatchFocus(old, v);
139:         if (v != null && v.onCheckIsTextEditor()) TextOverlay.show(this, v, -1, -1);
140:         else {
141:             TextOverlay.hide(this);
142:             if (pane.isShowing()) pane.requestFocusInWindow();
143:         }
144:         pane.repaint();
145:     }

151:     public void moveFocus(boolean forward) {
152:         java.util.List<View> all = new java.util.ArrayList<>();
153:         collect(view, all);
154:         if (all.isEmpty()) return;
155:         int i = all.indexOf(focused);
156:         int n = forward ? (i + 1) % all.size() : (i <= 0 ? all.size() - 1 : i - 1);
157:         View next = all.get(n);
158:         setFocused(next);
159:         next.requestRectangleOnScreen(new android.graphics.Rect(0, 0, next.getWidth(), next.getHeight()));
160:     }
161:     private void collect(View v, java.util.List<View> out) {
162:         if (v.getVisibility() != View.VISIBLE) return;
163:         if (v.onCheckIsTextEditor() && v.isEnabled() && v.isFocusable()) out.add(v);
164:         if (v instanceof ViewGroup) { ViewGroup g = (ViewGroup) v; for (int i = 0; i < g.getChildCount(); i++) collect(g.getChildAt(i), out); }
165:     }
