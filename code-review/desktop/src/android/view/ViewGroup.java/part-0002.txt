250:     List<View> drawOrder() {

252:         for (View c : children) if (c.getZ() != 0) { z = true; break; }
253:         if (!z) return children;

259:     @Override protected void dispatchDraw(Canvas c) {
260:         boolean clip = clipToPadding && (getPaddingLeft() | getPaddingTop() | getPaddingRight() | getPaddingBottom()) != 0;
261:         if (clip) {
262:             c.save();
263:             c.clipRect(mScrollX + getPaddingLeft(), mScrollY + getPaddingTop(), mScrollX + getWidth() - getPaddingRight(), mScrollY + getHeight() - getPaddingBottom());
264:         }
265:         java.awt.Rectangle vis = c.graphics().getClipBounds();
266:         for (View ch : new ArrayList<>(drawOrder())) {
267:             if (ch.getVisibility() != VISIBLE) continue;
268:             if (vis != null && ch.hasIdentityTransform() && ch.getZ() <= 0 && clipChildren) {
269:                 float l = ch.getLeft() + ch.getTranslationX(), t = ch.getTop() + ch.getTranslationY();
270:                 if (l > vis.x + vis.width || t > vis.y + vis.height || l + ch.getWidth() < vis.x || t + ch.getHeight() < vis.y) continue;
271:             }
272:             drawChild(c, ch, 0);
273:         }
274:         if (clip) c.restore();
275:     }

277:     protected boolean drawChild(Canvas c, View ch, long time) {
278:         float a = ch.getAlpha();
279:         if (a <= 0.003f) return false;
280:         c.save();
281:         c.translate(ch.getLeft() + ch.getTranslationX(), ch.getTop() + ch.getTranslationY());
282:         if (!ch.hasIdentityTransform()) {
283:             float pvx = ch.getPivotX(), pvy = ch.getPivotY();
284:             c.translate(pvx, pvy);
285:             if (ch.getRotation() != 0) c.rotate(ch.getRotation());
286:             c.scale(ch.getScaleX(), ch.getScaleY());
287:             c.translate(-pvx, -pvy);
288:         }
289:         float z = ch.getZ();
290:         if (z > 0.5f) Shadows.draw(c, ch, z, a);
291:         c.alpha *= a;
292:         if (clipChildren) c.clipRect(0, 0, ch.getWidth(), ch.getHeight());
293:         ch.draw(c);
294:         c.restore();
295:         return false;
296:     }

317:     @Override public boolean dispatchTouchEvent(MotionEvent e) {
318:         int act = e.getActionMasked();
319:         if (act == MotionEvent.ACTION_DOWN) {
320:             touchTarget = null;
321:             selfTouch = false;
322:             disallowIntercept = false;
323:         }
324:         if (selfTouch) return super.dispatchTouchEvent(e);
325:         boolean intercepted = false;
326:         if (act == MotionEvent.ACTION_DOWN || touchTarget != null) {
327:             intercepted = !disallowIntercept && onInterceptTouchEvent(e);
328:         }
329:         if (intercepted) {
330:             if (touchTarget != null) {
331:                 MotionEvent cancel = MotionEvent.obtain(e);
332:                 cancel.setAction(MotionEvent.ACTION_CANCEL);
333:                 deliver(touchTarget, cancel);
334:                 touchTarget = null;
335:             }
336:             selfTouch = true;
337:             boolean r = super.dispatchTouchEvent(e);
338:             if (act == MotionEvent.ACTION_UP || act == MotionEvent.ACTION_CANCEL) selfTouch = false;
339:             return r;
340:         }
341:         if (act == MotionEvent.ACTION_DOWN) {
342:             List<View> order = drawOrder();
343:             for (int i = order.size() - 1; i >= 0; i--) {
344:                 View c = order.get(i);
345:                 if (c.getVisibility() != VISIBLE || c.getAlpha() <= 0.003f) continue;
346:                 float[] q = {e.getX() + mScrollX, e.getY() + mScrollY};
347:                 c.parentToLocal(q);
