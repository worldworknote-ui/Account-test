348:                 if (q[0] < 0 || q[1] < 0 || q[0] >= c.getWidth() || q[1] >= c.getHeight()) continue;
349:                 MotionEvent ce = MotionEvent.obtain(e);
350:                 ce.setLocation(q[0], q[1]);
351:                 if (c.dispatchTouchEvent(ce)) {
352:                     touchTarget = c;
353:                     return true;
354:                 }
355:             }
356:             selfTouch = true;
357:             boolean r = super.dispatchTouchEvent(e);
358:             if (!r) selfTouch = false;
359:             return r;
360:         }
361:         if (touchTarget == null) return false;
362:         View t = touchTarget;
363:         if (act == MotionEvent.ACTION_UP || act == MotionEvent.ACTION_CANCEL) touchTarget = null;
364:         return deliver(t, e);
365:     }
366: 
367:     private boolean deliver(View c, MotionEvent e) {
368:         float[] q = {e.getX() + mScrollX, e.getY() + mScrollY};
369:         c.parentToLocal(q);
370:         MotionEvent ce = MotionEvent.obtain(e);
371:         ce.setLocation(q[0], q[1]);
372:         return c.dispatchTouchEvent(ce);
373:     }
