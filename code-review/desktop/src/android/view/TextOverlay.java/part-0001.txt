43:     /** يُظهر الخانة الحقيقية لهذا العنصر؛ (x,y) موضع النقرة بإحداثيات العنصر لوضع المؤشر (-1: لا تغيير). */
44:     public static void show(ViewRootImpl r, View v, float x, float y) {
45:         if (!(v instanceof TextHost)) return;
46:         TextHost h = (TextHost) v;
47:         Live l = map.get(r);
48:         if (l != null && (l.view != v || l.pwd != h.password() || l.single != h.singleLine())) { hide(r); l = null; }
49:         if (l == null) {
50:             l = create(r, v, h);
51:             map.put(r, l);
52:         }
53:         refresh(l);
54:         sync(r);
55:         final Live fl = l;
56:         if (x >= 0 && l.shown) {
57:             Rectangle b = l.field.getBounds();
58:             Point fp = SwingUtilities.convertPoint(r.pane, contentOrigin(v, x, y), l.clip);
59:             int pos = l.field.viewToModel2D(new Point(fp.x - b.x, fp.y - b.y));
60:             if (pos >= 0) l.field.setCaretPosition(Math.min(pos, l.field.getDocument().getLength()));
61:         } else if (x < 0) {
62:             int s = Math.max(0, Math.min(h.selStart(), l.field.getDocument().getLength())), e = Math.max(0, Math.min(h.selEnd(), l.field.getDocument().getLength()));
63:             fl.quiet = true;
64:             l.field.setCaretPosition(s);
65:             if (e != s) l.field.moveCaretPosition(e);
66:             fl.quiet = false;
67:         }
68:         SwingUtilities.invokeLater(() -> { if (fl.field.isShowing()) fl.field.requestFocusInWindow(); });
69:     }

83:     private static Live create(ViewRootImpl r, View v, TextHost h) {
84:         Live l = new Live();

88:         l.single = h.singleLine();

90:         if (l.pwd) {
91:             JPasswordField p = new JPasswordField();
92:             p.setUI(new javax.swing.plaf.basic.BasicPasswordFieldUI());

94:         } else if (l.single) {

97:             f = t;
98:         } else {
99:             JTextArea a = new JTextArea();
100:             a.setUI(new javax.swing.plaf.basic.BasicTextAreaUI());
101:             a.setLineWrap(true);
102:             a.setWrapStyleWord(true);
103:             f = a;
104:         }
105:         f.setOpaque(false);
106:         f.setBorder(BorderFactory.createEmptyBorder());
107:         f.setMargin(new Insets(0, 0, 0, 0));
108:         f.setBackground(new Color(0, 0, 0, 0));
109:         f.setFocusTraversalKeysEnabled(false);
110:         final Live fl = l;
111:         if (f.getDocument() instanceof AbstractDocument) ((AbstractDocument) f.getDocument()).setDocumentFilter(new Filter(h));
112:         f.getDocument().addDocumentListener(new DocumentListener() {
113:             void ch() {
114:                 if (fl.quiet) return;
115:                 String s = fl.field.getText();
116:                 fl.notifying = true;
117:                 try {
118:                     fl.host.userText(s, fl.field.getSelectionStart(), fl.field.getSelectionEnd());
119:                 } finally {
120:                     fl.notifying = false;
121:                 }
122:             }
123:             @Override public void insertUpdate(DocumentEvent e) { ch(); }
124:             @Override public void removeUpdate(DocumentEvent e) { ch(); }
125:             @Override public void changedUpdate(DocumentEvent e) {}
126:         });
127:         f.addCaretListener(e -> { if (!fl.quiet) fl.host.userSelection(Math.min(e.getDot(), e.getMark()), Math.max(e.getDot(), e.getMark())); });
128:         f.addKeyListener(new KeyAdapter() {
129:             @Override public void keyPressed(KeyEvent e) {
130:                 if (e.getKeyCode() == KeyEvent.VK_ENTER && (fl.single || e.isControlDown() || e.isMetaDown())) {
131:                     if (fl.host.enterPressed()) { e.consume(); return; }
132:                     if (fl.single) { r.moveFocus(true); e.consume(); return; }
133:                 }
134:                 if (e.getKeyCode() == KeyEvent.VK_ESCAPE || e.getKeyCode() == KeyEvent.VK_TAB
135:                         || ((e.isMetaDown() || e.isControlDown()) && e.getKeyCode() != KeyEvent.VK_C && e.getKeyCode() != KeyEvent.VK_V
136:                         && e.getKeyCode() != KeyEvent.VK_X && e.getKeyCode() != KeyEvent.VK_A && e.getKeyCode() != KeyEvent.VK_Z
137:                         && e.getKeyCode() != KeyEvent.VK_Y && e.getKeyCode() != KeyEvent.VK_LEFT && e.getKeyCode() != KeyEvent.VK_RIGHT
138:                         && e.getKeyCode() != KeyEvent.VK_BACK_SPACE && e.getKeyCode() != KeyEvent.VK_DELETE)) {
139:                     r.key(e);
140:                 }
141:             }
142:         });
143:         f.setComponentPopupMenu(menu(f));
144:         installUndo(f);
145:         JPanel clip = new JPanel(null);
146:         clip.setOpaque(false);
147:         clip.add(f);
148:         l.clip = clip;
149:         l.field = f;
150:         r.pane.add(clip);

153:     }

177:     private static void installUndo(JTextComponent f) {
178:         javax.swing.undo.UndoManager um = new javax.swing.undo.UndoManager();
179:         f.getDocument().addUndoableEditListener(um);
180:         int menu;
181:         try { menu = Toolkit.getDefaultToolkit().getMenuShortcutKeyMaskEx(); } catch (HeadlessException e) { menu = InputEvent.CTRL_DOWN_MASK; }
182:         f.getInputMap().put(KeyStroke.getKeyStroke(KeyEvent.VK_Z, menu), "sa-undo");
183:         f.getInputMap().put(KeyStroke.getKeyStroke(KeyEvent.VK_Z, menu | InputEvent.SHIFT_DOWN_MASK), "sa-redo");
184:         f.getInputMap().put(KeyStroke.getKeyStroke(KeyEvent.VK_Y, menu), "sa-redo");
185:         f.getActionMap().put("sa-undo", new AbstractAction() { @Override public void actionPerformed(ActionEvent e) { if (um.canUndo()) um.undo(); } });
186:         f.getActionMap().put("sa-redo", new AbstractAction() { @Override public void actionPerformed(ActionEvent e) { if (um.canRedo()) um.redo(); } });
