686:     void mapToParent(float[] p) {
687:         if (!hasIdentityTransform()) {
688:             float pvx = getPivotX(), pvy = getPivotY();
689:             float x = p[0] - pvx, y = p[1] - pvy;
690:             x *= sx; y *= sy;
691:             if (rot != 0) {
692:                 double a = Math.toRadians(rot);
693:                 float nx = (float) (x * Math.cos(a) - y * Math.sin(a)), ny = (float) (x * Math.sin(a) + y * Math.cos(a));
694:                 x = nx; y = ny;
695:             }
696:             p[0] = x + pvx; p[1] = y + pvy;
697:         }
698:         p[0] += mLeft + tx;
699:         p[1] += mTop + ty;
700:         if (parent instanceof View) {
701:             p[0] -= ((View) parent).mScrollX;
702:             p[1] -= ((View) parent).mScrollY;
703:         }
704:     }

708:         p[1] -= mTop + ty;
709:         if (!hasIdentityTransform()) {
710:             float pvx = getPivotX(), pvy = getPivotY();
711:             float x = p[0] - pvx, y = p[1] - pvy;
712:             if (rot != 0) {
713:                 double a = -Math.toRadians(rot);
714:                 float nx = (float) (x * Math.cos(a) - y * Math.sin(a)), ny = (float) (x * Math.sin(a) + y * Math.cos(a));
715:                 x = nx; y = ny;
716:             }
717:             if (sx != 0) x /= sx;
718:             if (sy != 0) y /= sy;
719:             p[0] = x + pvx; p[1] = y + pvy;
720:         }
721:         return true;

791:     public boolean onTouchEvent(MotionEvent e) {
792:         boolean can = clickable || longClickable;
793:         if (!enabled) return can;
794:         if (!can) return false;
795:         switch (e.getActionMasked()) {
796:             case MotionEvent.ACTION_DOWN:
797:                 longFired = false;
798:                 downX = e.getX(); downY = e.getY();
799:                 drawableHotspotChanged(e.getX(), e.getY());
800:                 setPressed(true);
801:                 if (longClickable && onLongClick != null) {
802:                     pendingLong = () -> {
803:                         if (isPressed() && performLongClick()) {
804:                             longFired = true;
805:                             setPressed(false);
806:                         }
807:                     };
808:                     postDelayed(pendingLong, ViewConfiguration.getLongPressTimeout());
809:                 }
810:                 return true;
811:             case MotionEvent.ACTION_MOVE:
812:                 drawableHotspotChanged(e.getX(), e.getY());
813:                 if (!pointInView(e.getX(), e.getY(), ViewConfiguration.TOUCH_SLOP)) {
814:                     cancelLong();
815:                     setPressed(false);
816:                 }
817:                 return true;
818:             case MotionEvent.ACTION_UP:
819:                 boolean was = isPressed();
820:                 cancelLong();
821:                 setPressed(false);
822:                 if (was && !longFired) {
823:                     if (attachInfo != null && attachInfo.root.focusOnClick(this)) { }
824:                     final MotionEvent copy = e;
825:                     performClickSafely();
826:                 }
827:                 return true;
