72:         public static final int UNSPECIFIED = 0, EXACTLY = 0x40000000, AT_MOST = 0x80000000;
73:         private static final int MODE_MASK = 0xC0000000;
74:         public static int makeMeasureSpec(int size, int mode) { return (Math.max(0, size) & ~MODE_MASK) | (mode & MODE_MASK); }
75:         public static int getMode(int spec) { return spec & MODE_MASK; }
76:         public static int getSize(int spec) { return spec & ~MODE_MASK; }
77:         public static String toString(int s) { return (getMode(s) == EXACTLY ? "EXACTLY " : getMode(s) == AT_MOST ? "AT_MOST " : "UNSPECIFIED ") + getSize(s); }
78:     }
