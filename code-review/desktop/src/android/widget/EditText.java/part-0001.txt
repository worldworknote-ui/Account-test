15:     /** النص القابل للتعديل: كل تعديل يمر بمراقبي النص كما في أندرويد. */
16:     private final class Buffer extends SpannableStringBuilder {
17:         @Override public SpannableStringBuilder replace(int st, int en, CharSequence s, int a, int b) {
18:             CharSequence src = s.subSequence(a, b);
19:             for (InputFilter f : getFilters()) {
20:                 CharSequence r = f.filter(src, 0, src.length(), this, st, en);
21:                 if (r != null) src = r;
22:             }
23:             int before = en - st, after = src.length();
24:             for (TextWatcher w : new java.util.ArrayList<>(watchers)) w.beforeTextChanged(this, st, before, after);
25:             super.replace(st, en, src, 0, src.length());
26:             int caret = st + after;
27:             selS = selE = Math.min(caret, length());
28:             textChangedInternal();
29:             for (TextWatcher w : new java.util.ArrayList<>(watchers)) w.onTextChanged(this, st, before, after);
30:             for (TextWatcher w : new java.util.ArrayList<>(watchers)) w.afterTextChanged(this);
31:             return this;
32:         }
33:     }

91:     @Override public void userText(String s, int a, int b) {
92:         String cur = buf.toString();
93:         if (s.equals(cur)) return;
94:         int pre = 0, max = Math.min(s.length(), cur.length());
95:         while (pre < max && s.charAt(pre) == cur.charAt(pre)) pre++;
96:         int suf = 0;
97:         while (suf < max - pre && s.charAt(s.length() - 1 - suf) == cur.charAt(cur.length() - 1 - suf)) suf++;
98:         buf.replace(pre, cur.length() - suf, s.substring(pre, s.length() - suf));
99:         selS = Math.min(a, buf.length());
100:         selE = Math.min(b, buf.length());
101:     }
