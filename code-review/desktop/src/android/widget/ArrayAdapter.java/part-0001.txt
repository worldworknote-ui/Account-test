7: /** محوّل القوائم البسيطة: سطر نصي لكل عنصر بنمط سطور أندرويد. */
8: public class ArrayAdapter<T> extends BaseAdapter implements Filterable {
9:     private final Context ctx;
10:     private final List<T> items;
11:     private int res, dropRes;
12:     private boolean notify = true;
13: 
14:     public ArrayAdapter(Context c, int res) { this(c, res, new ArrayList<T>()); }
15:     public ArrayAdapter(Context c, int res, int textId) { this(c, res, new ArrayList<T>()); }
16:     public ArrayAdapter(Context c, int res, T[] items) { this(c, res, new ArrayList<>(Arrays.asList(items))); }
17:     public ArrayAdapter(Context c, int res, int textId, T[] items) { this(c, res, items); }
18:     public ArrayAdapter(Context c, int res, List<T> items) { ctx = c; this.res = res; dropRes = res; this.items = items; }
19:     public ArrayAdapter(Context c, int res, int textId, List<T> items) { this(c, res, items); }
20:     public static ArrayAdapter<CharSequence> createFromResource(Context c, int arr, int res) { return new ArrayAdapter<>(c, res); }
21: 
22:     public Context getContext() { return ctx; }
23:     public void add(T t) { items.add(t); if (notify) notifyDataSetChanged(); }
24:     @SafeVarargs public final void addAll(T... t) { items.addAll(Arrays.asList(t)); if (notify) notifyDataSetChanged(); }
25:     public void addAll(Collection<? extends T> t) { items.addAll(t); if (notify) notifyDataSetChanged(); }
26:     public void insert(T t, int i) { items.add(i, t); if (notify) notifyDataSetChanged(); }
27:     public void remove(T t) { items.remove(t); if (notify) notifyDataSetChanged(); }
28:     public void clear() { items.clear(); if (notify) notifyDataSetChanged(); }
29:     public void sort(Comparator<? super T> c) { items.sort(c); if (notify) notifyDataSetChanged(); }
30:     public void setNotifyOnChange(boolean n) { notify = n; }
31:     @Override public void notifyDataSetChanged() { super.notifyDataSetChanged(); notify = true; }
32:     @Override public int getCount() { return items.size(); }
33:     @Override public T getItem(int pos) { return items.get(pos); }
34:     public int getPosition(T t) { return items.indexOf(t); }
35:     @Override public long getItemId(int pos) { return pos; }
36:     public void setDropDownViewResource(int r) { dropRes = r; }
37:     @Override public Filter getFilter() { return new Filter() {}; }
38: 
39:     @Override public View getView(int pos, View convert, ViewGroup parent) { return row(pos, convert, res, false); }
40:     @Override public View getDropDownView(int pos, View convert, ViewGroup parent) { return row(pos, convert, dropRes, true); }
41: 
42:     private View row(int pos, View convert, int layout, boolean drop) {
43:         TextView t = convert instanceof TextView ? (TextView) convert : new TextView(ctx);
44:         Object o = getItem(pos);
45:         t.setText(o instanceof CharSequence ? (CharSequence) o : String.valueOf(o));
46:         float d = ctx.getResources().getDisplayMetrics().density;
47:         boolean listRow = layout == android.R.layout.simple_list_item_1 || layout == android.R.layout.simple_spinner_dropdown_item || drop;
48:         if (layout == android.R.layout.simple_spinner_item && !drop) {
49:             t.setTextSize(16);
50:             t.setSingleLine(true);
51:             t.setEllipsize(android.text.TextUtils.TruncateAt.END);
52:             t.setPadding(0, 0, 0, 0);
53:         } else if (listRow) {
54:             t.setTextSize(16);
55:             t.setPadding(Math.round(16 * d), Math.round(4 * d), Math.round(16 * d), Math.round(4 * d));
56:             t.setMinHeight(Math.round(48 * d));
57:             if (layout == android.R.layout.simple_spinner_dropdown_item && !drop) { t.setSingleLine(true); t.setEllipsize(android.text.TextUtils.TruncateAt.END); }
58:         }
59:         t.setGravity(Gravity.CENTER_VERTICAL | Gravity.START);
60:         return t;
61:     }
