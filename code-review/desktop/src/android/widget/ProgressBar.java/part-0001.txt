8: public class ProgressBar extends View {

12:     private float angle;

16:     public ProgressBar(Context c) { this(c, null, 0); }

26:     public void setMin(int m) {}

38:     private int color() {
39:         if (tint != null) return tint.getDefaultColor();
40:         return getContext() instanceof android.app.Activity ? ((android.app.Activity) getContext()).themeAccent() : 0xFF0B6B53;
41:     }
42:     @Override protected void onMeasure(int ws, int hs) {
43:         float d = getResources().getDisplayMetrics().density;
44:         if (horizontal) setMeasuredDimension(resolveSize(Math.round(200 * d), ws), resolveSize(Math.round(16 * d), hs));
45:         else setMeasuredDimension(resolveSize(Math.round(48 * d), ws), resolveSize(Math.round(48 * d), hs));
46:     }
47:     @Override protected void onDraw(Canvas c) {
48:         float d = getResources().getDisplayMetrics().density;
49:         Paint p = new Paint(Paint.ANTI_ALIAS_FLAG);
50:         int col = color();

52:             float h = 4 * d, y = getHeight() / 2f - h / 2;
53:             p.setColor((col & 0xFFFFFF) | 0x40000000);
54:             c.drawRect(getPaddingLeft(), y, getWidth() - getPaddingRight(), y + h, p);
55:             p.setColor(col);
56:             float w = getWidth() - getPaddingLeft() - getPaddingRight();
57:             if (indeterminate) {
58:                 float t = ((System.currentTimeMillis() - start) % 1500) / 1500f;
59:                 float x0 = getPaddingLeft() + w * (t * 1.4f - 0.4f);
60:                 c.save();
61:                 c.clipRect(getPaddingLeft(), y, getWidth() - getPaddingRight(), y + h);
62:                 c.drawRect(x0, y, x0 + w * 0.4f, y + h, p);
63:                 c.restore();
64:                 postInvalidateDelayed(16);
65:             } else {
66:                 float pw = w * progress / max;
67:                 if (isRtl()) c.drawRect(getWidth() - getPaddingRight() - pw, y, getWidth() - getPaddingRight(), y + h, p);
68:                 else c.drawRect(getPaddingLeft(), y, getPaddingLeft() + pw, y + h, p);
69:             }
70:             return;
71:         }
72:         float size = Math.min(getWidth(), getHeight()) - 8 * d;
73:         float cx = getWidth() / 2f, cy = getHeight() / 2f, r = size / 2f - 2 * d;
74:         long t = System.currentTimeMillis() - start;
75:         float rot = (t % 1333) / 1333f * 360f;
76:         float sweep = 30 + 240 * (float) (0.5 - 0.5 * Math.cos((t % 1333) / 1333.0 * 2 * Math.PI));
77:         p.setStyle(Paint.Style.STROKE);
78:         p.setStrokeWidth(4 * d);
79:         p.setStrokeCap(Paint.Cap.ROUND);
80:         p.setColor(col);
81:         c.drawArc(new RectF(cx - r, cy - r, cx + r, cy + r), rot * 2 - 90, sweep, false, p);
82:         if (isShown()) postInvalidateDelayed(16);
83:     }
84: }
