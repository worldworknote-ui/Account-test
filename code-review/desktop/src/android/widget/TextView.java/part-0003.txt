355:     @Override protected void onMeasure(int ws, int hs) {
356:         int wm = MeasureSpec.getMode(ws), wsz = MeasureSpec.getSize(ws), hm = MeasureSpec.getMode(hs), hsz = MeasureSpec.getSize(hs);
357:         Drawable[] cd = absCompound();
358:         int padH = getPaddingLeft() + getPaddingRight() + cw(cd[0]) + cw(cd[2]);
359:         int padV = getPaddingTop() + getPaddingBottom() + ch(cd[1]) + ch(cd[3]);
360:         int width;
361:         if (autoMax > 0 && wm != MeasureSpec.UNSPECIFIED) autoFit(wsz - padH, hm == MeasureSpec.EXACTLY ? hsz - padV : -1);
362:         if (wm == MeasureSpec.EXACTLY) width = wsz;
363:         else {
364:             TextLines free = linesUnwrapped();
365:             int desired = (int) Math.ceil(free.maxWidth()) + padH;
366:             if (ems > 0) desired = Math.round(ems * paint.getTextSize()) + padH;
367:             for (Drawable d : new Drawable[]{cd[1], cd[3]}) if (d != null) desired = Math.max(desired, d.getBounds().width() + padH);
368:             desired = Math.max(desired, Math.max(minWidthPx, getSuggestedMinimumWidth()));
369:             desired = Math.min(desired, maxWidth == Integer.MAX_VALUE ? desired : maxWidth);
370:             width = wm == MeasureSpec.AT_MOST ? Math.min(desired, wsz) : desired;
371:         }
372:         int avail = Math.max(0, width - padH);
373:         TextLines tl = currentLines(avail);
374:         float textH = tl.count() == 0 ? oneLine() : tl.height();
375:         if (exactLines > 0 || minLines > 0) {
376:             float one = oneLine();
377:             int want = exactLines > 0 ? exactLines : minLines;
378:             if (tl.count() < want) textH = Math.max(textH, one * want);
379:         }
380:         int h = (int) Math.ceil(textH) + padV;
381:         for (Drawable d : new Drawable[]{cd[0], cd[2]}) if (d != null) h = Math.max(h, d.getBounds().height() + getPaddingTop() + getPaddingBottom());
382:         h = Math.max(h, Math.max(minHeightPx, getSuggestedMinimumHeight()));
383:         if (maxHeight != Integer.MAX_VALUE) h = Math.min(h, maxHeight);
384:         int height = hm == MeasureSpec.EXACTLY ? hsz : hm == MeasureSpec.AT_MOST ? Math.min(h, hsz) : h;
385:         setMeasuredDimension(width, height);
386:     }

389:     private TextLines linesUnwrapped() {
390:         if (unwrapped != null && unwrappedV == textVersion) return unwrapped;
391:         TextLines.Spec s = new TextLines.Spec();

395:         s.maxLines = Integer.MAX_VALUE; s.includePad = includePad; s.density = getResources().getDisplayMetrics().density;
396:         unwrapped = TextLines.build(t, s);
397:         unwrappedV = textVersion;

400:     private void autoFit(int w, int h) {
401:         float lo = autoMin, hi = autoMax, best = autoMin;
402:         float keep = paint.getTextSize();
403:         for (int i = 0; i < 8; i++) {
404:             float mid = (lo + hi) / 2f;
405:             paint.setTextSize(mid);
406:             textVersion++;
407:             TextLines.Spec s = new TextLines.Spec();
408:             s.paint = paint; s.width = singleLine ? -1 : w; s.maxLines = maxLines; s.singleLine = singleLine;
409:             TextLines l = TextLines.build(shown(), s);
410:             boolean fits = l.maxWidth() <= w && (h < 0 || l.height() <= h) && l.count() <= maxLines;
411:             if (fits) { best = mid; lo = mid; } else hi = mid;
412:         }
413:         paint.setTextSize(best);
414:         if (best != keep) textVersion++;
415:     }

422:     private int verticalOffset(TextLines tl) {
423:         int vg = gravity & Gravity.VERTICAL_GRAVITY_MASK;
424:         int boxH = getHeight() - getCompoundPaddingTop() - getCompoundPaddingBottom();
425:         float th = tl.count() == 0 ? oneLine() : tl.height();
426:         if (vg == Gravity.TOP) return 0;
427:         if (vg == Gravity.BOTTOM) return Math.max(0, Math.round(boxH - th));
428:         return Math.round((boxH - th) / 2f);
429:     }
