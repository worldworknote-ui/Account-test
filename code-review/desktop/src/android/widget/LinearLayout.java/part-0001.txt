74:     private boolean hasDividerBefore(int i) {
75:         if (divider == null) return false;
76:         if (i == 0) return (showDividers & SHOW_DIVIDER_BEGINNING) != 0;
77:         if (i == getChildCount()) return (showDividers & SHOW_DIVIDER_END) != 0;

98:         int weightedMaxW = 0, altMaxW = 0;
99:         boolean skipped = false;
100:         int largest = Integer.MIN_VALUE;
101:         for (int i = 0; i < count; i++) {
102:             View c = getChildAt(i);
103:             if (c.getVisibility() == GONE) continue;
104:             if (hasDividerBefore(i)) totalLength += divH();
105:             LayoutParams p = lp(c);
106:             total += p.weight;
107:             boolean useExcess = p.height == 0 && p.weight > 0;
108:             if (hm == MeasureSpec.EXACTLY && useExcess) {
109:                 totalLength = Math.max(totalLength, totalLength + p.topMargin + p.bottomMargin);
110:                 skipped = true;
111:             } else {
112:                 if (useExcess) p.height = LayoutParams.WRAP_CONTENT;
113:                 measureChildWithMargins(c, ws, 0, hs, total == 0 ? totalLength : 0);
114:                 if (useExcess) p.height = 0;
115:                 int ch = c.getMeasuredHeight();
116:                 totalLength = Math.max(totalLength, totalLength + ch + p.topMargin + p.bottomMargin);
117:                 if (useLargestChild) largest = Math.max(ch, largest);
118:             }
119:             boolean matchThis = false;
120:             if (wm != MeasureSpec.EXACTLY && p.width == LayoutParams.MATCH_PARENT) { matchW = true; matchThis = true; }
121:             int margin = p.leftMargin + p.rightMargin;
122:             int mw = c.getMeasuredWidth() + margin;
123:             maxW = Math.max(maxW, mw);
124:             state = combineMeasuredStates(state, c.getMeasuredState());
125:             allMatch = allMatch && p.width == LayoutParams.MATCH_PARENT;
126:             if (p.weight > 0) weightedMaxW = Math.max(weightedMaxW, matchThis ? margin : mw);
127:             else altMaxW = Math.max(altMaxW, matchThis ? margin : mw);
128:         }
