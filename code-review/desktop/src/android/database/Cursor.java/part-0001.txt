77:         Object o = v(i);

79:         if (o instanceof Long || o instanceof Integer || o instanceof Short || o instanceof Byte || o instanceof Boolean) return FIELD_TYPE_INTEGER;
80:         if (o instanceof Double || o instanceof Float) return FIELD_TYPE_FLOAT;
81:         if (o instanceof byte[]) return FIELD_TYPE_BLOB;

85:     public String getString(int i) {
86:         Object o = v(i);
87:         if (o == null) return null;
88:         if (o instanceof byte[]) return new String((byte[]) o, java.nio.charset.StandardCharsets.UTF_8);
89:         if (o instanceof Double) {
90:             double d = (Double) o;
91:             if (d == Math.rint(d) && !Double.isInfinite(d) && Math.abs(d) < 1e15) return Long.toString((long) d) + ".0";
92:             return Double.toString(d);
93:         }
94:         return o.toString();
95:     }

97:     public long getLong(int i) {
98:         Object o = v(i);
99:         if (o == null) return 0;
100:         if (o instanceof Number) return ((Number) o).longValue();
101:         if (o instanceof Boolean) return (Boolean) o ? 1 : 0;
102:         return parseLong(o.toString());

105:     /** كما يفعل SQLite: يقرأ أطول بادئة رقمية (نص غير رقمي = 0). */
106:     static long parseLong(String s) {
107:         s = s.trim();
108:         try {
109:             return Long.parseLong(s);
110:         } catch (NumberFormatException e) {
111:             try {
112:                 return (long) Double.parseDouble(s);
113:             } catch (NumberFormatException e2) {
114:                 int n = 0;
115:                 if (n < s.length() && (s.charAt(n) == '-' || s.charAt(n) == '+')) n++;
116:                 while (n < s.length() && Character.isDigit(s.charAt(n))) n++;
117:                 try {
118:                     return n == 0 ? 0 : Long.parseLong(s.substring(0, n));
119:                 } catch (NumberFormatException e3) {
120:                     return 0;
121:                 }
122:             }
123:         }
124:     }

134:         try {

136:         } catch (NumberFormatException e) {
