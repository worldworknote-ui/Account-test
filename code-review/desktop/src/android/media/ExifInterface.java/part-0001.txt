3: /** اتجاه الصورة من بيانات EXIF (JPEG) — قارئ صغير. */
4: public class ExifInterface {
5:     public static final String TAG_ORIENTATION = "Orientation";
6:     public static final int ORIENTATION_NORMAL = 1, ORIENTATION_FLIP_HORIZONTAL = 2, ORIENTATION_ROTATE_180 = 3, ORIENTATION_FLIP_VERTICAL = 4,
7:             ORIENTATION_TRANSPOSE = 5, ORIENTATION_ROTATE_90 = 6, ORIENTATION_TRANSVERSE = 7, ORIENTATION_ROTATE_270 = 8, ORIENTATION_UNDEFINED = 0;
8:     private int orientation = ORIENTATION_UNDEFINED;
9:     public ExifInterface(String path) throws java.io.IOException { this(new java.io.FileInputStream(path)); }
10:     public ExifInterface(java.io.File f) throws java.io.IOException { this(new java.io.FileInputStream(f)); }
11:     public ExifInterface(java.io.FileDescriptor fd) throws java.io.IOException { this(new java.io.FileInputStream(fd)); }
12:     public ExifInterface(java.io.InputStream in) throws java.io.IOException {
13:         try (java.io.InputStream s = in) { orientation = read(s.readNBytes(256 * 1024)); } catch (Exception e) { orientation = ORIENTATION_UNDEFINED; }
14:     }
15:     public int getAttributeInt(String tag, int def) { return TAG_ORIENTATION.equals(tag) && orientation != 0 ? orientation : def; }
16:     public String getAttribute(String tag) { return TAG_ORIENTATION.equals(tag) ? String.valueOf(orientation) : null; }
17:     public void setAttribute(String tag, String v) {}
18:     public void saveAttributes() {}

20:     private static int read(byte[] b) {
21:         if (b.length < 4 || (b[0] & 0xFF) != 0xFF || (b[1] & 0xFF) != 0xD8) return 0;
22:         int i = 2;
23:         while (i + 4 < b.length) {
24:             if ((b[i] & 0xFF) != 0xFF) return 0;
25:             int marker = b[i + 1] & 0xFF, len = ((b[i + 2] & 0xFF) << 8) | (b[i + 3] & 0xFF);
26:             if (marker == 0xE1 && i + 10 < b.length && b[i + 4] == 'E' && b[i + 5] == 'x' && b[i + 6] == 'i' && b[i + 7] == 'f') {
27:                 int t = i + 10;
28:                 boolean le = b[t] == 'I';
29:                 int ifd = u32(b, t + 4, le);
30:                 int p = t + ifd;
31:                 int n = u16(b, p, le);
32:                 for (int k = 0; k < n; k++) {
33:                     int e = p + 2 + k * 12;
34:                     if (e + 12 > b.length) break;
35:                     if (u16(b, e, le) == 0x0112) return u16(b, e + 8, le);
36:                 }
37:                 return 0;
38:             }
39:             i += 2 + len;
40:         }
41:         return 0;
42:     }
43:     private static int u16(byte[] b, int i, boolean le) { return le ? (b[i] & 0xFF) | ((b[i + 1] & 0xFF) << 8) : ((b[i] & 0xFF) << 8) | (b[i + 1] & 0xFF); }
44:     private static int u32(byte[] b, int i, boolean le) { return le ? u16(b, i, true) | (u16(b, i + 2, true) << 16) : (u16(b, i, false) << 16) | u16(b, i + 2, false); }
45: }
