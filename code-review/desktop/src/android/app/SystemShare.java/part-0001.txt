43:                     String line;
44:                     while ((line = r.readLine()) != null) {
45:                         if (line.contains("SA_SHOWN")) {
46:                             shown = true;
47:                             post(result, true);
48:                             break;

50:                         if (line.contains("SA_FAIL") || System.currentTimeMillis() > until) break;
51:                     }
52:                 }
53:                 if (!shown) {
54:                     p.destroy();

56:                 }
57:             } catch (Exception e) {
58:                 if (!shown) post(result, false);
59:             }
60:         }, "system-share");
61:         t.setDaemon(true);
62:         t.start();
63:     }
