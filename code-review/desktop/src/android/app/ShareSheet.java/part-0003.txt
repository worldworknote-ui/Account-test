402:     static boolean macCopyFiles(List<File> files) {

405:             StringBuilder js = new StringBuilder("ObjC.import('AppKit');var pb=$.NSPasteboard.generalPasteboard;pb.clearContents;var a=$.NSMutableArray.array;");

420:     static String jsString(String s) {
421:         StringBuilder b = new StringBuilder("\"");
422:         for (char c : s.toCharArray()) {
423:             if (c == '"' || c == '\\') b.append('\\').append(c);
424:             else if (c < 0x20 || c > 0x7e) b.append(String.format("\\u%04x", (int) c));
425:             else b.append(c);
426:         }
427:         return b.append('"').toString();
428:     }

430:     /** ينسخ الملفات إلى الحافظة (وللصورة الواحدة: الصورة نفسها أيضًا) للصق في واتساب وتلغرام والبريد. */
431:     static void copyFiles(List<File> files) {
432:         if (macCopyFiles(files)) return;
433:         final List<File> fl = new ArrayList<>(files);
434:         java.awt.Image img = null;
435:         if (fl.size() == 1 && fl.get(0).getName().toLowerCase().matches(".*\\.(png|jpe?g)$")) {
436:             try { img = javax.imageio.ImageIO.read(fl.get(0)); } catch (Exception ignored) { /* الملف وحده */ }
437:         }
438:         final java.awt.Image im = img;
439:         Transferable tr = new Transferable() {
440:             @Override public DataFlavor[] getTransferDataFlavors() {
441:                 return im != null ? new DataFlavor[]{DataFlavor.imageFlavor, DataFlavor.javaFileListFlavor} : new DataFlavor[]{DataFlavor.javaFileListFlavor};
442:             }
443:             @Override public boolean isDataFlavorSupported(DataFlavor f) {
444:                 for (DataFlavor x : getTransferDataFlavors()) if (x.equals(f)) return true;
445:                 return false;
446:             }
447:             @Override public Object getTransferData(DataFlavor f) throws UnsupportedFlavorException {
448:                 if (DataFlavor.imageFlavor.equals(f) && im != null) return im;
449:                 if (DataFlavor.javaFileListFlavor.equals(f)) return fl;
450:                 throw new UnsupportedFlavorException(f);
451:             }
452:         };
453:         Toolkit.getDefaultToolkit().getSystemClipboard().setContents(tr, null);
454:     }
