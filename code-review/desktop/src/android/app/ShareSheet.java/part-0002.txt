181:     static void run(Context ctx, String id, List<File> files, CharSequence text, String subject) {
182:         String t = text == null ? "" : text.toString();
183:         boolean hasFiles = !files.isEmpty();
184:         try {
185:             switch (id) {
186:                 case SYSTEM: {
187:                     int[] c = center();
188:                     // 11.5: الملف وحده إلى قائمة المشاركة (مع نص بجانبه كان واتساب في الماك يفتح نافذة كتابة فارغة)،
189:                     // والنص يُنسخ للصق بعد الملف
190:                     final boolean split = hasFiles && !t.isEmpty() && MAC;
191:                     if (split) copyText(t);
192:                     SystemShare.share(files, split ? "" : t, c[0], c[1], ok -> {
193:                         if (ok && split) Toast.makeText(ctx, "أُرسل الملف إلى قائمة المشاركة، والنص منسوخ: الصقه في المحادثة بعد الملف (⌘V)", Toast.LENGTH_LONG).show();
194:                         if (!ok) {
195:                             Toast.makeText(ctx, "تعذّر فتح مشاركة النظام على هذا الجهاز: انتقلنا إلى «نسخ ثم لصق» في واتساب", Toast.LENGTH_LONG).show();
196:                             run(ctx, WA_APP, files, text, subject);
197:                         }
198:                     });
199:                     return;
200:                 }
201:                 case WA_APP:
202:                 case WA_WEB:
203:                 case TG: {
204:                     boolean web = id.equals(WA_WEB);
205:                     boolean opened;
206:                     if (id.equals(TG)) opened = openApp("tg://msg?text=" + Uri.encode(t), "Telegram");
207:                     else if (!web) opened = openApp("whatsapp://send?text=" + Uri.encode(t), "WhatsApp");
208:                     else opened = openWebOnce(t);
209:                     if (!opened && !web) {
210:                         // التطبيق غير مثبّت: النسخة على الويب (مرة واحدة في الجلسة)
211:                         if (id.equals(TG)) Platform.browseQuiet("https://t.me/share/url?url=" + Uri.encode(t.isEmpty() ? " " : t));
212:                         else openWebOnce(t);
213:                         Toast.makeText(ctx, (id.equals(TG) ? "تطبيق تلغرام" : "تطبيق واتساب") + " غير موجود على هذا الكمبيوتر: فُتحت النسخة على الويب", Toast.LENGTH_LONG).show();
214:                     }
215:                     // الملفات (وصورة مع نصها) تُلصق على خطوات: لا يمكن أن تحمل الحافظة الصورة والنص معًا للصق واحد
216:                     if (hasFiles || web) pasteSteps(ctx, files, t, id.equals(TG) ? "تلغرام" : web ? "واتساب ويب" : "واتساب");
217:                     return;
218:                 }
219:                 case WA_DRAG: {
220:                     Platform.reveal(files.get(0));
221:                     if (!openApp("whatsapp://", "WhatsApp")) openWebOnce("");
222:                     if (!t.trim().isEmpty()) copyText(t);
223:                     new AlertDialog.Builder(ctx).setTitle("الإرسال إلى واتساب")
224:                             .setMessage("1) فُتح المجلد والملف محدد فيه، وفُتح واتساب.\n2) افتح المحادثة، ثم اسحب الملف من المجلد وأفلته فيها."
225:                                     + (t.trim().isEmpty() ? "" : "\n3) النص منسوخ: الصقه بعد الملف (" + (MAC ? "⌘V" : "Ctrl+V") + ")."))
226:                             .setPositiveButton("تم", null).show();
227:                     return;
228:                 }
229:                 case MAIL: {
230:                     if (hasFiles && MAC) {
231:                         List<String> cmd = new ArrayList<>();
232:                         cmd.add("open");
233:                         cmd.add("-a");
234:                         cmd.add("Mail");
235:                         for (File f : files) cmd.add(f.getAbsolutePath());
236:                         new ProcessBuilder(cmd).start();
237:                         if (!t.isEmpty()) {
238:                             ((android.content.ClipboardManager) ctx.getSystemService(Context.CLIPBOARD_SERVICE)).setText(t);
239:                             Toast.makeText(ctx, "فُتحت رسالة بالمرفق، والنص منسوخ للصق", Toast.LENGTH_LONG).show();
240:                         }
241:                         return;
242:                     }
243:                     Desktop.getDesktop().mail(new java.net.URI("mailto:?subject=" + Uri.encode(subject == null ? "" : subject) + "&body=" + Uri.encode(t)));
244:                     if (hasFiles) {
245:                         Platform.reveal(files.get(0));
246:                         Toast.makeText(ctx, "أرفق الملف من المجلد الذي فُتح", Toast.LENGTH_LONG).show();
247:                     }
248:                     return;
249:                 }
250:                 case COPY:
251:                     if (hasFiles && !t.trim().isEmpty()) {
252:                         pasteSteps(ctx, files, t, "البرنامج الذي تريده");
253:                         return;
254:                     }
255:                     if (hasFiles) copyFiles(files);
256:                     else ((android.content.ClipboardManager) ctx.getSystemService(Context.CLIPBOARD_SERVICE)).setText(t);
257:                     Toast.makeText(ctx, "نُسخ — الصقه حيث تريد", Toast.LENGTH_SHORT).show();
258:                     return;
259:                 case SAVE: {
260:                     if (!hasFiles) {
261:                         File out = Platform.saveDialog((subject == null || subject.isEmpty() ? "نص" : subject.replaceAll("[\\\\/:*?\"<>|]", "_")) + ".txt");
262:                         if (out != null) {
263:                             java.nio.file.Files.write(out.toPath(), t.getBytes(java.nio.charset.StandardCharsets.UTF_8));
264:                             Toast.makeText(ctx, "حُفظ: " + out.getName(), Toast.LENGTH_SHORT).show();
265:                         }
266:                     } else if (files.size() == 1) {
267:                         File out = Platform.saveDialog(files.get(0).getName());
268:                         if (out != null) {
269:                             java.nio.file.Files.copy(files.get(0).toPath(), out.toPath(), java.nio.file.StandardCopyOption.REPLACE_EXISTING);
270:                             Toast.makeText(ctx, "حُفظ: " + out.getName(), Toast.LENGTH_SHORT).show();
271:                         }
272:                     } else {
273:                         File dir = Platform.dirDialog();
274:                         if (dir != null) {
275:                             for (File f : files) java.nio.file.Files.copy(f.toPath(), new File(dir, f.getName()).toPath(), java.nio.file.StandardCopyOption.REPLACE_EXISTING);
276:                             Toast.makeText(ctx, "حُفظت " + files.size() + " ملفات", Toast.LENGTH_SHORT).show();
277:                         }
278:                     }
279:                     return;
280:                 }
281:                 case OPEN:
282:                     Desktop.getDesktop().open(files.get(0));
283:                     return;
284:                 case REVEAL:
285:                     Platform.reveal(files.get(0));
286:                     return;
287:                 default:
288:             }
289:         } catch (Exception e) {
290:             Toast.makeText(ctx, "تعذر: " + (e.getMessage() == null ? e.getClass().getSimpleName() : e.getMessage()), Toast.LENGTH_LONG).show();
291:         }
292:     }
