38:     // ------------------------------------------------------------------ التشغيل
39:     /** يشغّل شاشة جديدة من هذا الصنف (أول شاشة في البرنامج). */
40:     public static <T extends Activity> T launch(Class<T> cls, Bundle saved) {
41:         try {
42:             T a = cls.getDeclaredConstructor().newInstance();
43:             ((Activity) a).attachWindow();
44:             a.onCreate(saved);
45:             a.onPostCreate(saved);
46:             a.onStart();
47:             a.onResume();
48:             ((Activity) a).resumed = true;
49:             a.onPostResume();
50:             return a;
51:         } catch (RuntimeException e) {
52:             throw e;
53:         } catch (Exception e) {
54:             throw new RuntimeException(e);
55:         }
56:     }
57:     private void attachWindow() {
58:         live.add(this);
59:         root = WindowStack.add(decor, ViewRootImpl.LAYER_APP);
60:         root.onBack = this::backKey;
61:         root.keyHook = this::shortcutKey;
62:     }
63:     private void backKey() {
64:         for (int i = dialogs.size() - 1; i >= 0; i--) { Dialog d = dialogs.get(i); if (d.isShowing()) { d.backPressed(); return; } }
65:         onBackPressed();
66:     }

99:     public void resumeFromWindow() { if (!resumed && !destroyed) { onRestart(); onStart(); onResume(); resumed = true; onPostResume(); } }
100:     /** إغلاق البرنامج. */
101:     public void destroyFromWindow() {
102:         if (destroyed) return;

104:         destroyed = true;
105:         onDestroy();

128:     /** إعادة إنشاء الشاشة (بعد تغيير السمة أو الخط) كما في أندرويد: كائن جديد بالحالة المحفوظة. */
129:     public void recreate() {
130:         javax.swing.SwingUtilities.invokeLater(() -> {
131:             Bundle saved = new Bundle();
132:             onSaveInstanceState(saved);
133:             for (Dialog d : new ArrayList<>(dialogs)) d.dismiss();
134:             if (resumed) { onPause(); onStop(); resumed = false; }
135:             finishing = true;
136:             destroyed = true;
137:             onDestroy();
138:             live.remove(this);
139:             WindowStack.remove(root);
140:             Activity fresh = launch(getClass(), saved);
141:             fresh.intent = intent;
142:             if (Platform.onRecreate != null) Platform.onRecreate.accept(fresh);
143:         });
144:     }
