27:     static String keyFor(String account) {
28:         Db db = SmartApp.db;
29:         String cur = db == null ? "" : db.setting("drive_account", "");
30:         if (account == null || account.isEmpty()) account = cur;
31:         if (account.isEmpty()) return DeskDriveAuth.REFRESH;
32:         String k = "drive_refresh:" + account.trim().toLowerCase(java.util.Locale.ROOT);
33:         if (!DeskDriveAuth.has(k) && account.equalsIgnoreCase(cur) && DeskDriveAuth.has(DeskDriveAuth.REFRESH)) return DeskDriveAuth.REFRESH;
34:         return k;

118:     /** الدخول بحساب محدد: إن دخل للتو (اختيار الحساب) يُعتمد رمزه، وإلا يُفتح المتصفح. */
119:     static void connect(final Activity a, String account, final Done done) {
120:         final String acct = account == null ? "" : account;
121:         if (!acct.isEmpty() && acct.equalsIgnoreCase(pendingEmail) && DeskDriveAuth.has(PENDING)) {
122:             try {
123:                 DeskDriveAuth.promote(PENDING, keyFor(acct));
124:                 pendingEmail = null;
125:                 done.ok();
126:             } catch (Exception e) {
127:                 done.fail(e);
128:             }
129:             return;
130:         }
131:         if (!clientReady(a, () -> connect(a, acct, done))) return;
132:         final String key = keyFor(acct);
133:         // 11.3: رمز محفوظ يعمل فعلًا؟ (كان وجوده يكفي فلا تفعل «إعادة الدخول» شيئًا مع رمز منتهٍ)
134:         if (DeskDriveAuth.has(key)) {
135:             ((Ui) a).busy("جارٍ التحقق من دخول Google…", p -> {
136:                 try {
137:                     new DeskDriveAuth(SmartApp.db, key).token(true);
138:                     return Boolean.TRUE;
139:                 } catch (Exception e) {
140:                     return Boolean.FALSE;
141:                 }
142:             }, r -> {
143:                 if (Boolean.TRUE.equals(r)) done.ok();
144:                 else browserLogin(a, key, acct, done);
145:             });
146:             return;
147:         }
148:         browserLogin(a, key, acct, done);
149:     }
