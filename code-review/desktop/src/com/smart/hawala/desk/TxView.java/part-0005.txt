772:     void attach() {
773:         File f = Export.open(this, db, "اختر صورة الوصل", ".jpg", ".jpeg", ".png", ".heic", ".gif", ".bmp");
774:         if (f == null) return;
775:         try {
776:             BufferedImage im = ImageIO.read(f);
777:             if (im == null) throw new IllegalArgumentException("الصيغة غير مدعومة؛ اختر JPG أو PNG (صور HEIC: صدّرها JPG من «الصور»).");
778:             int max = 2000;
779:             double s = Math.min(1, (double) max / Math.max(im.getWidth(), im.getHeight()));
780:             int w = (int) (im.getWidth() * s), h = (int) (im.getHeight() * s);
781:             BufferedImage rgb = new BufferedImage(w, h, BufferedImage.TYPE_INT_RGB);
782:             Graphics2D g = rgb.createGraphics();
783:             g.setColor(Color.WHITE);
784:             g.fillRect(0, 0, w, h);
785:             g.setRenderingHint(RenderingHints.KEY_INTERPOLATION, RenderingHints.VALUE_INTERPOLATION_BICUBIC);
786:             g.drawImage(im, 0, 0, w, h, null);
787:             g.dispose();
788:             ByteArrayOutputStream o = new ByteArrayOutputStream();
789:             ImageIO.write(rgb, "jpg", o);
790:             imageData.add(o.toByteArray());
791:             renderImages();
792:         } catch (Exception e) {
793:             Ui.error(this, e);
794:         }
795:     }

814:     Db.TxInput input(boolean strict) throws Exception {
815:         Model.Office o = (Model.Office) office.getSelectedItem();
816:         if (o == null) throw new IllegalArgumentException("اختر المكتب");
817:         if (type.none()) throw new IllegalArgumentException("اختر «لنا» أو «علينا»");
818:         String c = (String) currency.getSelectedItem();
819:         if (c == null) throw new IllegalArgumentException("اختر العملة");
820:         Db.TxInput in = new Db.TxInput();
821:         in.office = o.id;
822:         in.type = type.selected == 0 ? Model.TYPE_DEBIT : Model.TYPE_CREDIT;
823:         in.currency = c;
824:         if (amount.val().isEmpty()) throw new IllegalArgumentException("اكتب المبلغ");
825:         in.amount = Db.minor(amount.getText(), db.decimals(c));
826:         if (in.amount <= 0) throw new IllegalArgumentException("المبلغ يجب أن يكون أكبر من صفر");
827:         in.person = person.val();
828:         in.description = desc.getText().trim();
829:         in.notes = notes.val();
830:         Object sv = service.getEditor().getItem();
831:         in.service = sv == null ? "" : sv.toString().trim();
832:         in.dest = dest.val();
833:         long day = date.start();
834:         long now = Clock.now();
835:         if (day == 0 || day == Ui.startOfDay(now)) in.when = editing != null && Ui.startOfDay(editing.createdAt) == Ui.startOfDay(now) ? editing.createdAt : now;
836:         else if (editing != null && Ui.startOfDay(editing.createdAt) == day) in.when = editing.createdAt;
837:         else {
838:             Calendar n = Model.cal();
839:             Calendar x = Model.cal();
840:             x.setTimeInMillis(day);
841:             x.set(Calendar.HOUR_OF_DAY, n.get(Calendar.HOUR_OF_DAY));
842:             x.set(Calendar.MINUTE, n.get(Calendar.MINUTE));
843:             x.set(Calendar.SECOND, n.get(Calendar.SECOND));
844:             in.when = Math.min(x.getTimeInMillis(), now);
845:         }
846:         if (feeMode.selected == 1) {
847:             in.feeType = feeSide.selected == 0 ? Model.FEE_US : Model.FEE_THEM;
848:             if (feeKind.selected == 1) {
849:                 BigDecimal r = Model.parseRate(rate.getText());
850:                 in.feeCurrency = c;
851:                 in.fee = Model.feeFromRate(in.amount, r, per());
852:                 if (in.fee <= 0) throw new IllegalArgumentException("الأجرة المحسوبة أصغر من أصغر وحدة في " + c + ". اكتبها كمبلغ محدد أو اختر «بدون أجرة».");
853:             } else {
854:                 String fc = (String) feeCur.getSelectedItem();
855:                 in.feeCurrency = fc == null ? c : fc;
856:                 if (fee.val().isEmpty()) throw new IllegalArgumentException("اكتب مبلغ الأجرة أو اختر «بدون أجرة»");
857:                 in.fee = Db.minor(fee.getText(), db.decimals(in.feeCurrency));
858:                 if (in.fee <= 0) throw new IllegalArgumentException("مبلغ الأجرة يجب أن يكون أكبر من صفر");
859:             }
860:         } else {
861:             in.feeType = Model.FEE_NONE;
862:             in.fee = 0;
863:             in.feeCurrency = c;
864:         }
865:         return in;
866:     }
