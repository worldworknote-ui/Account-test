510:     static void showImages(Component parent, Db db, List<Long> ids) {
511:         JPanel c = Ui.column();

513:             try {
514:                 BufferedImage im = ImageIO.read(new java.io.ByteArrayInputStream(db.attachment(id)));
515:                 if (im == null) continue;
516:                 double s = Math.min(1, 640.0 / im.getWidth());
517:                 Image sc = im.getScaledInstance((int) (im.getWidth() * s), (int) (im.getHeight() * s), Image.SCALE_SMOOTH);
518:                 JLabel l = new JLabel(new ImageIcon(sc));
519:                 l.setAlignmentX(Component.RIGHT_ALIGNMENT);
520:                 c.add(l);
521:                 c.add(Ui.gap(10));
522:             } catch (Exception ignored) {
523:                 c.add(Ui.label("(صورة غير قابلة للعرض)", 13, Ui.muted, false));
524:             }
525:         }
526:         JScrollPane sp = new JScrollPane(c);
527:         sp.setPreferredSize(new Dimension(700, 560));
528:         Ui.form(parent, "صور الوصل", sp, "إغلاق");
