252:     static java.math.BigDecimal amountOf(String s) {
253:         try {
254:             StringBuilder b = new StringBuilder();
255:             for (char ch : Model.nz(s).toCharArray()) {
256:                 if (ch >= '٠' && ch <= '٩') b.append((char) ('0' + ch - '٠'));
257:                 else if (ch >= '0' && ch <= '9' || ch == '.') b.append(ch);
258:                 else if (ch == '٫') b.append('.');
259:             }
260:             String v = b.toString();
261:             return v.isEmpty() ? null : new java.math.BigDecimal(v);
262:         } catch (Exception e) {
263:             return null;
264:         }
265:     }
