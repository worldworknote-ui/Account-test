157:     void currencies() {
158:         String keep = label(cur);
159:         common.clear();
160:         Model.Office f = from.get(), t = to.get();
161:         if (f != null && t != null) {
162:             List<String> y = db.officeCurrencies(t.id);
163:             for (String c : db.officeCurrencies(f.id)) if (y.contains(c)) common.add(c);
164:         }
165:         cur = chips(curHolder, common, Ui.primary, i -> update(), f == null || t == null ? "اختر المكتبين لتظهر عملاتهما المشتركة" : "لا عملة مشتركة مفعّلة للمكتبين");
166:         if (cur != null) {
167:             if (keep != null && common.contains(keep)) cur.set(common.indexOf(keep));
168:             else if (common.size() == 1) cur.set(0);
169:         }
170:         Ui.rtl(curHolder);
171:         from.showBalance();
172:         to.showBalance();
173:         update();

176:     Db.TransferInput input() {

183:         in.currency = c;

191:         in.toFeeType = fees ? FEE_KINDS[Math.max(0, toFee.selected)] : Model.FEE_NONE;
