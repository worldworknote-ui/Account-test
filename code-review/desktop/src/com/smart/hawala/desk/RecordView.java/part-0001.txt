140:     static List<Model.Office> myFirst(Db db, List<Model.Office> all) {
141:         java.util.Set<Long> mine = new java.util.HashSet<>();
142:         for (Model.Office o : db.myOffices()) mine.add(o.id);
143:         if (mine.isEmpty()) return all;
144:         List<Model.Office> out = new ArrayList<>();
145:         for (Model.Office o : all) if (mine.contains(o.id)) out.add(o);
146:         for (Model.Office o : all) if (!mine.contains(o.id)) out.add(o);
147:         return out;
148:     }
