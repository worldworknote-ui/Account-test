242:     /** يرسم الملخص (الناقص بالأحمر مع ✗) وسطر الحالة فوق زر التسجيل. */
243:     void render(List<String> lines, String okLine) {
244:         StringBuilder h = new StringBuilder("<html><div style='text-align:right;width:" + Ui.css(400) + "px'>");
245:         List<String> missing = new ArrayList<>();
246:         boolean inTable = false;
247:         for (String l : lines) {
248:             // 10.8: جدول المقاصة «▤عملة|قبل|الأثر|بعد»
249:             if (l.startsWith("▤")) {
250:                 if (!inTable) {
251:                     h.append("<table cellpadding='3' style='margin:4px 0'><tr style='color:").append(hex(Ui.muted)).append("'><td align='right'>بعد</td><td align='right'>أثر العملية</td>")
252:                             .append("<td align='right'>قبل</td><td align='right'>العملة</td></tr>");
253:                     inTable = true;
254:                 }
255:                 String[] c = l.substring(1).split("\\|", 4);
256:                 h.append("<tr><td align='right'><b>").append(Ui.esc(c[3])).append("</b></td><td align='right' style='color:").append(hex(c[2].startsWith("خصم") ? Ui.red : Ui.green))
257:                         .append("'>").append(Ui.esc(c[2])).append("</td><td align='right'>").append(Ui.esc(c[1])).append("</td><td align='right'><b>").append(Ui.esc(c[0])).append("</b></td></tr>");
258:                 continue;
259:             }
260:             if (inTable) {
261:                 h.append("</table>");
262:                 inTable = false;
263:             }
264:             if (l.startsWith("!")) {
265:                 // تنبيه غير مانع: سطر كامل بالأحمر، لا يُعد ناقصًا
266:                 h.append("<div style='margin:6px 0;color:").append(hex(Ui.red)).append("'><b>").append(Ui.esc(l)).append("</b></div>");
267:                 continue;
268:             }
269:             boolean bad = l.startsWith("×");
270:             String b = bad ? l.substring(1).trim() : l;
271:             int c = b.indexOf(": ");
272:             String label = c > 0 ? b.substring(0, c) : b, value = c > 0 ? b.substring(c + 2) : "";
273:             if (bad) missing.add(label);
274:             h.append("<div style='margin:3px 0'><span style='color:").append(hex(bad ? Ui.red : Ui.muted)).append("'>").append(bad ? "× " : "").append(Ui.esc(label))
275:                     .append(":</span>&nbsp; <b style='color:").append(hex(bad ? Ui.red : Ui.ink)).append("'>").append(Ui.esc(value)).append("</b></div>");
276:         }
277:         if (inTable) h.append("</table>");
278:         summary.setText(h.append("</div></html>").toString());
279:         if (missing.isEmpty()) {
280:             footLine.setText("✓ " + okLine);
281:             footLine.setForeground(Ui.primaryInk);
282:         } else {
283:             footLine.setText("ناقص: " + Model.join(missing, "، "));
284:             footLine.setForeground(Ui.red);
285:         }
286:     }
