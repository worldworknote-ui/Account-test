807:     String rateText(String code, Map<String, Db.Rate> rates) {
808:         if ("USD".equals(code)) return "العملة الأساس";
809:         Db.Rate r = rates.get(code);
810:         if (r == null || r.usd == null) return "بلا سعر صرف";
811:         if (r.usd.compareTo(BigDecimal.ONE) >= 0) return "1 " + code + " = " + r.usd.setScale(4, RoundingMode.HALF_EVEN).stripTrailingZeros().toPlainString() + " USD — " + r.source;
812:         return "1 USD = " + BigDecimal.ONE.divide(r.usd, 2, RoundingMode.HALF_EVEN).stripTrailingZeros().toPlainString() + " " + code + " — " + r.source;
813:     }
