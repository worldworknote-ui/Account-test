270:     void fill() {
271:         if (table == null || !db.cashConfig().on) return;
272:         long[] r = ReportBuilder.period(period.getSelectedIndex(), Clock.now());
273:         Map<String, Integer> dec = db.decimalsMap();
274:         Map<Long, String> names = db.officeNames();
275:         boolean hide = app.hidden();
276:         DefaultTableModel mdl = new DefaultTableModel(new Object[]{"التاريخ", "الحركة", "الشخص / البيان", "داخل", "خارج", "العملة", "أجور نقدًا", "الحالة"}, 0) {
277:             @Override public boolean isCellEditable(int a, int b) { return false; }

281:         Ui.Keep keep = Ui.Keep.of(table, shown, x -> x.id);   // 10.5 (§6)
282:         shown = new ArrayList<>();

284:         for (int i = all.size() - 1; i >= 0; i--) {

289:             String who = x.clientId != null ? "" : x.party;
290:             if (!x.description.isEmpty()) who = who.isEmpty() ? x.description : who + " — " + x.description;
291:             long fc = x.feeCash();
292:             Object in = hide ? (x.dir > 0 ? "••••" : "") : new Ui.Money(x.dir > 0 ? x.amount : 0, d, false);
293:             Object out = hide ? (x.dir < 0 ? "••••" : "") : new Ui.Money(x.dir < 0 ? x.amount : 0, d, false);
294:             mdl.addRow(new Object[]{Model.date(x.createdAt), title(x, names), who, in, out, x.currency,
295:                     fc == 0 || hide ? "" : (fc > 0 ? "لنا " : "علينا ") + Model.fmt(Math.abs(fc), d), x.stateText()});
296:         }
