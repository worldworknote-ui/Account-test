67:     void unlock() {
68:         long until = db.settingLong("cash_locked_until", 0);
69:         if (until > Clock.now()) {
70:             Ui.toast("محاولات خاطئة كثيرة. حاول بعد " + Math.max(1, (until - Clock.now()) / 1000) + " ثانية.");
71:             return;
72:         }
73:         JPasswordField f = Ui.password();
74:         JPanel q = Ui.column();
75:         q.add(Ui.label("كلمة سر الصندوق", 14, Ui.ink, true));
76:         q.add(Ui.gap(8));
77:         q.add(f);
78:         if (!Ui.form(this, "فتح الصندوق", q, "فتح")) return;
79:         if (Pin.verify(new String(f.getPassword()), db.setting("cash_pin_salt", ""), db.setting("cash_pin_hash", ""))) {
80:             db.setSettingQuiet("cash_fails", "0");
81:             openUntil = Clock.now() + OPEN_MS;
82:             refresh();
83:             return;
84:         }
85:         int fails = (int) db.settingLong("cash_fails", 0) + 1;
86:         db.setSettingQuiet("cash_fails", "" + fails);
87:         long wait = Pin.lockoutMillis(fails);
88:         if (wait > 0) {
89:             db.setSettingQuiet("cash_locked_until", "" + (Clock.now() + wait));
90:             Ui.toast("محاولات خاطئة كثيرة. انتظر " + wait / 1000 + " ثانية.");
91:         } else Ui.toast("كلمة السر غير صحيحة");
92:     }
