321:         DefaultComboBoxModel<String> m = new DefaultComboBoxModel<>();

324:         JSONObject st = stored();
325:         List<String> names = new ArrayList<>();
326:         for (Iterator<String> it = st.keys(); it.hasNext(); ) names.add(it.next());
327:         Collections.sort(names);
328:         for (String n : names) m.addElement(n);
329:         presets.setModel(m);

332:     void loadPreset(String name) {
333:         if (name.startsWith(READY)) {
334:             for (Object[] b : builtIn()) if (name.equals(READY + b[0])) {
335:                 apply((ReportBuilder.Spec) b[2]);
336:                 period.setSelectedIndex((Integer) b[1]);
337:                 run();
338:                 return;
339:             }
340:         }
341:         JSONObject j = stored().optJSONObject(name);
342:         if (j == null) return;
343:         ReportBuilder.Spec s = ReportBuilder.Spec.fromJson(j);
344:         int p = j.optInt("p", -1);
345:         if (p >= 0 && p < Ui.PERIODS.length - 1) {
346:             s.from = 0;
347:             s.to = 0;
348:         }
349:         apply(s);
350:         if (p >= 0 && p < Ui.PERIODS.length - 1) period.setSelectedIndex(p);
351:         run();
352:     }
