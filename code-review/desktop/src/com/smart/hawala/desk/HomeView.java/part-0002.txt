216:     /** تنبيهات حدود المكاتب (لكل عملة، ومقوّمة بالدولار) كما في الهاتف. */
217:     static List<String> limitAlerts(Db db, Model.Office o) {
218:         List<String> out = new ArrayList<>();
219:         Map<String, Long> b = db.balances(o.id);
220:         Map<String, Db.Limit> limits = db.officeLimits(o.id);
221:         for (Map.Entry<String, Long> e : b.entrySet()) {
222:             Db.Limit l = limits.get(e.getKey());
223:             if (l == null) continue;
224:             Ledger.LimitBreach br = Ledger.check(e.getKey(), e.getValue(), l.receivable, l.payable);
225:             if (br == null) continue;
226:             int d = db.decimals(e.getKey());
227:             out.add(o.name + " — " + e.getKey() + ": " + (br.receivable ? "لنا " : "علينا ") + Db.fmt(br.amount, d) + " (الحد " + Db.fmt(br.limit, d) + ")");
228:         }
229:         if (o.usdLimit != null || o.usdOweLimit != null) {
230:             Ledger.Valuation v = Ledger.valueOffice(b, db.usdRates(), db.decimalsMap());
231:             long us = v.receivableUsd.movePointRight(2).setScale(0, java.math.RoundingMode.HALF_UP).longValue();
232:             long them = v.payableUsd.movePointRight(2).setScale(0, java.math.RoundingMode.HALF_UP).longValue();
233:             if (o.usdLimit != null && o.usdLimit > 0 && us > o.usdLimit) out.add(o.name + " — مجموع «لنا» مقوّمًا ≈ " + Db.fmt(us, 2) + " USD (الحد " + Db.fmt(o.usdLimit, 2) + ")");
234:             if (o.usdOweLimit != null && o.usdOweLimit > 0 && them > o.usdOweLimit)
235:                 out.add(o.name + " — مجموع «علينا» مقوّمًا ≈ " + Db.fmt(them, 2) + " USD (الحد " + Db.fmt(o.usdOweLimit, 2) + ")");
236:         }
237:         return out;
238:     }
