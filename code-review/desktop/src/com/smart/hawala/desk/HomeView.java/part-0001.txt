146:     void totals(Map<String, Integer> dec) {

150:             for (Map.Entry<String, Long> e : m.entrySet()) {
151:                 long[] x = tot.computeIfAbsent(e.getKey(), k -> new long[2]);
152:                 if (e.getValue() > 0) x[0] += e.getValue();

155:         JPanel cards = new JPanel(new GridLayout(0, 3, 12, 12));
156:         cards.setOpaque(false);
157:         cards.setAlignmentX(Component.RIGHT_ALIGNMENT);
158:         for (Map.Entry<String, long[]> e : tot.entrySet()) {
159:             if (e.getValue()[0] == 0 && e.getValue()[1] == 0) continue;
160:             int d = dec.getOrDefault(e.getKey(), 2);
161:             Ui.Card c = Ui.card();
162:             c.add(Ui.label(e.getKey(), 15, Ui.goldInk, true));
163:             c.add(Ui.gap(4));
