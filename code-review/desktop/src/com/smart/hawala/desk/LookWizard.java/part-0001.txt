49:     /** يضع المعالج بجانب نافذة البرنامج (وعند أول تشغيل يزيح النافذة قليلًا ليظهر الاثنان معًا). */
50:     static void place(boolean firstRun) {
51:         Window main = current();
52:         Rectangle scr = GraphicsEnvironment.getLocalGraphicsEnvironment().getMaximumWindowBounds();
53:         int w = dlg.getWidth(), h = Math.min(dlg.getHeight(), scr.height - 20);
54:         dlg.setSize(w, h);
55:         if (main == null) { dlg.setLocationRelativeTo(null); return; }
56:         Rectangle m = main.getBounds();
57:         int gap = 16;
58:         if (firstRun && m.width + w + gap <= scr.width && (main instanceof Frame) && (((Frame) main).getExtendedState() & Frame.MAXIMIZED_BOTH) == 0) {
59:             int total = m.width + gap + w;
60:             int x0 = scr.x + (scr.width - total) / 2;
61:             // المعالج يسار النافذة (الكتابة من اليمين: البرنامج في مكانه المعتاد يمينًا)
62:             dlg.setLocation(x0, Math.max(scr.y, m.y + (m.height - h) / 2));
63:             main.setLocation(x0 + w + gap, m.y);
64:         } else if (m.x - scr.x >= w + gap) {
65:             dlg.setLocation(m.x - w - gap, Math.max(scr.y, m.y + (m.height - h) / 2));
66:         } else if (scr.x + scr.width - (m.x + m.width) >= w + gap) {
67:             dlg.setLocation(m.x + m.width + gap, Math.max(scr.y, m.y + (m.height - h) / 2));
68:         } else {
69:             dlg.setLocation(Math.max(scr.x, m.x + 24), Math.max(scr.y, m.y + m.height - h - 24));
70:         }
71:     }
