226:             final int k = i;
227:             JToggleButton b = new JToggleButton(labels[i], i == sel);
228:             b.setFont(Ui.font(13.5f, i == sel));
229:             b.putClientProperty("JButton.buttonType", "roundRect");
230:             b.setFocusable(false);
231:             b.setPreferredSize(new Dimension(150, 40));

240:         }
