117:     private static void hookUsage() {
118:         if (usageHooked) return;
119:         usageHooked = true;
120:         focusCheck.setRepeats(false);
121:         Toolkit.getDefaultToolkit().addAWTEventListener(ev -> {
122:             int id = ev.getID();
123:             if (id == WindowEvent.WINDOW_ACTIVATED || id == WindowEvent.WINDOW_DEACTIVATED || id == WindowEvent.WINDOW_LOST_FOCUS || id == WindowEvent.WINDOW_GAINED_FOCUS) {
124:                 focusCheck.restart();
125:                 return;
126:             }
127:             if (id == java.awt.event.MouseEvent.MOUSE_PRESSED || id == java.awt.event.MouseEvent.MOUSE_WHEEL || id == java.awt.event.KeyEvent.KEY_PRESSED
128:                     || id == java.awt.event.MouseEvent.MOUSE_MOVED) {
129:                 if (activity != null) activity.onUserInteraction();
130:             }
131:         }, AWTEvent.WINDOW_EVENT_MASK | AWTEvent.WINDOW_FOCUS_EVENT_MASK | AWTEvent.MOUSE_EVENT_MASK | AWTEvent.MOUSE_MOTION_EVENT_MASK
132:                 | AWTEvent.MOUSE_WHEEL_EVENT_MASK | AWTEvent.KEY_EVENT_MASK);
133:     }

135:     private static void checkFocus() {
136:         if (activity == null) return;

138:         if (!front && !away) {
139:             away = true;
140:             activity.usageAway();
141:         } else if (front && away) {
142:             away = false;
143:             activity.usageBack();
144:         }
