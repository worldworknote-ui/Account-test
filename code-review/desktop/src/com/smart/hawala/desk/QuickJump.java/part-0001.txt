67:             l.add(new Item(s[1], "قسم", () -> app.go(k)));
68:         }
69:         if (app.allowed("fx")) l.add(new Item("تسجيل حركة قص عملات", "إجراء", () -> app.go("fx")));
70:         if (app.allowed("xfer")) l.add(new Item("تسجيل حركة بين مكتبين", "إجراء", () -> app.go("xfer")));
71:         l.add(new Item("مزامنة الآن", "إجراء", () -> SyncView.now(app)));

111:     static void open(App app, String initial) {
112:         if (app.locked) return;
113:         final List<Item> all = items(app);
114:         final JDialog d = new JDialog(App.frame, "الانتقال السريع", Dialog.ModalityType.APPLICATION_MODAL);
115:         d.setUndecorated(true);
116:         final Ui.Field q = Ui.field("اكتب اسم قسم أو مكتب…  (↑ ↓ ثم Enter، وEsc للإغلاق)");
117:         final DefaultListModel<Item> m = new DefaultListModel<>();
118:         final JList<Item> list = new JList<>(m);
119:         list.setSelectionMode(ListSelectionModel.SINGLE_SELECTION);
120:         list.setFixedCellHeight(34);
121:         list.setCellRenderer((jl, it, i, sel, foc) -> {
122:             JPanel r = new JPanel(new BorderLayout(10, 0));
123:             r.setBorder(new EmptyBorder(0, 12, 0, 12));
124:             r.setBackground(sel ? Ui.primarySoft : Ui.surface);
125:             r.add(Ui.label(it.label, 14.5f, Ui.ink, sel), BorderLayout.CENTER);
126:             r.add(Ui.label(it.hint, 12, Ui.muted, false), BorderLayout.LINE_END);
127:             r.applyComponentOrientation(Ui.RTL);
128:             return r;
129:         });
130:         final Runnable fill = () -> {
131:             String f = OwnerReports.norm(q.getText());
132:             m.clear();
133:             List<Item> starts = new ArrayList<>(), has = new ArrayList<>();
134:             for (Item it : all) {
135:                 if (f.isEmpty() || it.norm.startsWith(f)) starts.add(it);
136:                 else {
137:                     boolean allW = true;
138:                     for (String wq : f.split(" ")) allW &= wq.isEmpty() || it.norm.contains(wq);
139:                     if (allW) has.add(it);
140:                 }
141:             }
142:             // بلا كتابة: الأقسام والإجراءات فقط (المكاتب تظهر بالكتابة)
143:             for (Item it : starts) if (!f.isEmpty() || !it.hint.contains("مكتب") && !it.hint.contains("كشف حساب") && !it.hint.contains("تسجيل حركة") && !it.hint.contains("إعداد")) m.addElement(it);
144:             for (Item it : has) m.addElement(it);
