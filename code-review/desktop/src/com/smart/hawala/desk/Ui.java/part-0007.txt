1219:         static <T> Keep of(JTable t, List<T> rows, java.util.function.Function<T, Object> key) {
1220:             Keep k = new Keep();

1222:             for (int v : t.getSelectedRows()) {

1231:             if (t == null) return;
1232:             for (int m = 0; m < rows.size() && m < t.getModel().getRowCount(); m++) {
1233:                 if (!keys.contains(key.apply(rows.get(m)))) continue;
1234:                 int v = t.convertRowIndexToView(m);
1235:                 if (v >= 0) t.addRowSelectionInterval(v, v);

1237:             final Point p = pos;
1238:             if (p != null && t.getParent() instanceof JViewport) {
1239:                 final JViewport vp = (JViewport) t.getParent();
1240:                 SwingUtilities.invokeLater(() -> {
1241:                     int maxY = Math.max(0, vp.getView().getHeight() - vp.getExtentSize().height);
1242:                     vp.setViewPosition(new Point(p.x, Math.min(p.y, maxY)));

1244:             }
