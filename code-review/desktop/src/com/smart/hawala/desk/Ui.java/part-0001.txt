136:     static void loadFonts() {
137:         if (base != null) return;
138:         try {
139:             base = load("/res/plex_regular.ttf");
140:             baseBold = load("/res/plex_bold.ttf");
141:             GraphicsEnvironment ge = GraphicsEnvironment.getLocalGraphicsEnvironment();
142:             ge.registerFont(base);
143:             ge.registerFont(baseBold);
144:             for (String f : new String[]{"cairo_regular", "cairo_bold", "kufi_regular", "kufi_bold"}) {
145:                 try {
146:                     ge.registerFont(load("/res/" + f + ".ttf"));
147:                 } catch (Exception ignored) {
148:                     // اختياري
149:                 }
150:             }
151:         } catch (Exception e) {
152:             base = new Font(Font.SANS_SERIF, Font.PLAIN, 13);
153:             baseBold = new Font(Font.SANS_SERIF, Font.BOLD, 13);

169:     static Color parse(String s, Color def) {
170:         try {
171:             s = s == null ? "" : s.trim();
172:             if (s.isEmpty()) return def;
173:             if (s.startsWith("#")) s = s.substring(1);
174:             long v = Long.parseLong(s, 16);
175:             if (s.length() == 8) return new Color((int) (v >> 16) & 255, (int) (v >> 8) & 255, (int) v & 255);
176:             return new Color((int) v);
177:         } catch (Exception e) {
178:             return def;
179:         }
180:     }

182:     static Color mix(Color a, Color b, float t) {
183:         return new Color((int) (a.getRed() + (b.getRed() - a.getRed()) * t), (int) (a.getGreen() + (b.getGreen() - a.getGreen()) * t),
184:                 (int) (a.getBlue() + (b.getBlue() - a.getBlue()) * t));
185:     }
