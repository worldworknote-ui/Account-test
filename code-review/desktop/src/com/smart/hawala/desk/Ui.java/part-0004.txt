582:         private void snap() {
583:             if (selected < 0 || selected >= buttons.size()) return;
584:             Btn b = buttons.get(selected);
585:             ix = b.getX();
586:             iw = b.getWidth();
587:             repaint();
588:         }

590:         void set(int i) {
591:             int prev = selected;
592:             selected = i;
593:             for (int k = 0; k < buttons.size(); k++) {
594:                 Btn b = buttons.get(k);
595:                 b.outline = false;
596:                 Color c = colorOf(k);
597:                 b.colors(new Color(0, 0, 0, 0), k == i ? onColor(c) : colors == null ? muted : c);
598:             }
599:             if (i < 0 || i >= buttons.size()) {
600:                 ix = -1;
601:                 repaint();
602:                 return;
603:             }
604:             Btn b = buttons.get(i);
605:             Color to = colorOf(i);
606:             if (b.getWidth() == 0 || prev < 0 || ix < 0) {
607:                 ic = to;
608:                 SwingUtilities.invokeLater(this::snap);
609:                 snap();
610:                 return;
611:             }
612:             double fx = ix, fw = iw, tx = b.getX(), tw = b.getWidth();
613:             Color fc = ic == null ? to : ic;
614:             if (anim != null) anim.stop();
615:             long t0 = com.smart.hawala.Clock.now();
616:             anim = new Timer(15, null);
617:             anim.addActionListener(e -> {
618:                 double x = Math.min(1, (com.smart.hawala.Clock.now() - t0) / 560.0), y = backOut(x);
619:                 ix = fx + (tx - fx) * y;
620:                 iw = fw + (tw - fw) * y;
621:                 ic = mix(fc, to, (float) Math.min(1, x * 1.3));
622:                 repaint();
623:                 if (x >= 1) anim.stop();
624:             });
625:             anim.start();
626:         }
