661:             Dimension min = frame.getMinimumSize();
662:             r.width = Math.max(r.width, min.width);
663:             r.height = Math.max(r.height, min.height);
664:             boolean visible = false;
665:             for (GraphicsDevice g : GraphicsEnvironment.getLocalGraphicsEnvironment().getScreenDevices()) {
666:                 Rectangle sb = g.getDefaultConfiguration().getBounds();

675:         }
