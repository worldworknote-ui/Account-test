873:     void startTimers() {
874:         // القفل بعد الخمول (عند وجود مستخدمين)
875:         Timer idle = new Timer(15000, e -> {
876:             long after = lockAfterMs();
877:             if (!locked && db.usersEnabled() && after > 0 && Clock.now() - lastInput > after) lock();
878:             else if (!locked && after > 0 && DeskLock.on(db) && Clock.now() - lastInput > after) DeskLock.lockNow(this);
879:         });
880:         idle.start();
881:         timers.add(idle);
882:         // 10.16: تذكيرات المكاتب والنسخ (كالهاتف): بعد دقيقتين من الفتح ثم كل 30 دقيقة
883:         Timer remind = new Timer(30 * 60000, e -> DeskNotify.check(this));
884:         remind.setInitialDelay(120000);
885:         remind.start();
886:         timers.add(remind);
887:         // النسخ التلقائي: فحص كل 10 دقائق
888:         Timer auto = new Timer(10 * 60000, e -> DeskBackup.runIfDueAsync(this));
889:         auto.setInitialDelay(60000);
890:         auto.start();
891:         timers.add(auto);
892:         // 10.0: المزامنة التلقائية كل دقيقة (حين تكون مفعّلة)
893:         // 10.4: الفترة يختارها المستخدم من شاشة المزامنة (30 ثانية إلى 15 دقيقة)
894:         Timer sync = new Timer(15000, e -> {
895:             if (!locked && Clock.now() - SyncView.lastAuto >= SyncEngine.everySeconds(db) * 1000L) SyncView.auto(this);
896:         });
897:         // 10.1: كل عملية مسجّلة تجدول مزامنة بعد 4 ثوانٍ (المتتالية تُجمع في جولة واحدة)
898:         final Timer soon = new Timer(4000, e -> {
899:             if (!locked && "1".equals(db.setting("sync_after_change", "1"))) SyncView.auto(this);
900:         });
901:         soon.setRepeats(false);
902:         Db.onLocalChange = () -> SwingUtilities.invokeLater(soon::restart);
903:         sync.setInitialDelay(5000);
904:         sync.start();
905:         timers.add(sync);
906:         timers.add(soon);
907:     }
