305:     /** يحرك الإطار إلى قسم الشاشة الحالية (من مكانه السابق) ويبدّل لونه تدريجيًا. */
306:     void slideNav() {
307:         if (navList == null) return;
308:         NavItem n = nav.get(navOf(current));
309:         if (n == null || n.getHeight() == 0) {
310:             navY = -1;
311:             navList.repaint();
312:             if (n != null) SwingUtilities.invokeLater(() -> {
313:                 if (n.getHeight() > 0) slideNav();
314:             });
315:             return;
316:         }
317:         double to = n.getY();
318:         Color cTo = navColor(n.key);
319:         if (navAnim != null) navAnim.stop();
320:         if (navY < 0 || navColorNow == null) {
321:             navY = to;
322:             navColorNow = cTo;
323:             navList.repaint();
324:             return;
325:         }
326:         double from = navY;
327:         Color cFrom = navColorNow;
328:         long t0 = Clock.now();
329:         navAnim = new Timer(15, null);
330:         navAnim.addActionListener(e -> {
331:             double x = Math.min(1, (Clock.now() - t0) / 560.0), y = Ui.backOut(x);
332:             navY = from + (to - from) * y;
333:             navColorNow = Ui.mix(cFrom, cTo, (float) Math.min(1, x * 1.2));
334:             navList.repaint();
335:             for (NavItem k : nav.values()) k.repaint();
336:             if (x >= 1) navAnim.stop();
337:         });
338:         navAnim.start();
339:     }
