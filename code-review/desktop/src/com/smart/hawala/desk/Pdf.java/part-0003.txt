471:     static boolean print(Report r, Opts o) throws Exception {
472:         final Pdf ex = new Pdf(r, o);
473:         final List<List<Block>> pages = ex.pages();
474:         PrinterJob job = PrinterJob.getPrinterJob();
475:         job.setJobName(r.title);
476:         PageFormat pf = job.defaultPage();
477:         Paper p = new Paper();
478:         p.setSize(595, 842);
479:         p.setImageableArea(0, 0, 595, 842);
480:         pf.setPaper(p);
481:         pf.setOrientation(o.style.landscape ? PageFormat.LANDSCAPE : PageFormat.PORTRAIT);
482:         job.setPrintable((g, f, i) -> {
483:             if (i >= pages.size()) return Printable.NO_SUCH_PAGE;
484:             Graphics2D g2 = (Graphics2D) g;
485:             g2.translate(f.getImageableX(), f.getImageableY());
486:             double s = Math.min(f.getImageableWidth() / ex.W, f.getImageableHeight() / ex.H);
487:             g2.scale(s, s);
488:             ex.drawPage(g2, pages.get(i), i, pages.size());
489:             return Printable.PAGE_EXISTS;
490:         }, pf);
491:         if (!job.printDialog()) return false;
492:         job.print();
493:         return true;
494:     }

496:     /** كاتب PDF بسيط: صفحة لكل صورة RGB مضغوطة Flate. */
497:     static final class PdfWriter {
498:         final OutputStream out;
499:         long pos;
500:         final List<Long> offsets = new ArrayList<>();
501:         final List<Integer> pageIds = new ArrayList<>();
502:         int next = 3;   // 1 = الكتالوج، 2 = شجرة الصفحات
503: 
504:         PdfWriter(OutputStream out) throws IOException {
505:             this.out = out;
506:             offsets.add(0L);
507:             offsets.add(0L);
508:             offsets.add(0L);
509:             write("%PDF-1.4\n%âãÏÓ\n".getBytes(StandardCharsets.ISO_8859_1));
510:         }
511: 
512:         void write(byte[] b) throws IOException {
513:             out.write(b);
514:             pos += b.length;
515:         }
516: 
517:         void w(String s) throws IOException { write(s.getBytes(StandardCharsets.ISO_8859_1)); }
518: 
519:         int obj() {
520:             int id = next++;
521:             while (offsets.size() <= id) offsets.add(0L);
522:             return id;
523:         }
524: 
525:         void begin(int id) throws IOException {
526:             offsets.set(id, pos);
527:             w(id + " 0 obj\n");
528:         }
529: 
530:         void page(BufferedImage im, float pw, float ph) throws IOException {
531:             int iw = im.getWidth(), ih = im.getHeight();
532:             byte[] raw = new byte[iw * ih * 3];
533:             int k = 0;
534:             int[] row = new int[iw];
535:             for (int y = 0; y < ih; y++) {
536:                 im.getRGB(0, y, iw, 1, row, 0, iw);
537:                 for (int x = 0; x < iw; x++) {
538:                     int c = row[x];
539:                     raw[k++] = (byte) (c >> 16);
540:                     raw[k++] = (byte) (c >> 8);
541:                     raw[k++] = (byte) c;
542:                 }
543:             }
544:             Deflater d = new Deflater(Deflater.BEST_COMPRESSION);
545:             d.setInput(raw);
546:             d.finish();
547:             ByteArrayOutputStream z = new ByteArrayOutputStream(raw.length / 8);
548:             byte[] buf = new byte[65536];
549:             while (!d.finished()) z.write(buf, 0, d.deflate(buf));
550:             d.end();
551:             byte[] data = z.toByteArray();
552:             int img = obj(), content = obj(), page = obj();
553:             begin(img);
554:             w("<< /Type /XObject /Subtype /Image /Width " + iw + " /Height " + ih + " /ColorSpace /DeviceRGB /BitsPerComponent 8 /Filter /FlateDecode /Length " + data.length + " >>\nstream\n");
555:             write(data);
556:             w("\nendstream\nendobj\n");
557:             String cs = "q " + f(pw) + " 0 0 " + f(ph) + " 0 0 cm /Im0 Do Q\n";
558:             begin(content);
559:             w("<< /Length " + cs.length() + " >>\nstream\n" + cs + "endstream\nendobj\n");
560:             begin(page);
561:             w("<< /Type /Page /Parent 2 0 R /MediaBox [0 0 " + f(pw) + " " + f(ph) + "] /Resources << /XObject << /Im0 " + img + " 0 R >> >> /Contents " + content + " 0 R >>\nendobj\n");
562:             pageIds.add(page);
563:         }
564: 
565:         static String f(float v) { return String.format(java.util.Locale.US, "%.2f", v); }
566: 
567:         void finish(String title) throws IOException {
568:             offsets.set(2, pos);
569:             StringBuilder kids = new StringBuilder();
570:             for (int id : pageIds) kids.append(id).append(" 0 R ");
571:             w("2 0 obj\n<< /Type /Pages /Kids [" + kids + "] /Count " + pageIds.size() + " >>\nendobj\n");
572:             int info = obj();
573:             begin(info);
574:             w("<< /Producer (Smart Accountant for Mac) /Title <" + utf16(title) + "> >>\nendobj\n");
575:             offsets.set(1, pos);
576:             w("1 0 obj\n<< /Type /Catalog /Pages 2 0 R >>\nendobj\n");
577:             long xref = pos;
578:             w("xref\n0 " + next + "\n0000000000 65535 f \n");
579:             for (int i = 1; i < next; i++) w(String.format(java.util.Locale.US, "%010d 00000 n \n", offsets.get(i)));
580:             w("trailer\n<< /Size " + next + " /Root 1 0 R /Info " + info + " 0 R >>\nstartxref\n" + xref + "\n%%EOF\n");
581:             out.flush();
582:         }
583: 
584:         static String utf16(String s) {
585:             StringBuilder b = new StringBuilder("FEFF");
586:             for (char c : (s == null ? "" : s).toCharArray()) b.append(String.format(java.util.Locale.ROOT, "%04X", (int) c));
587:             return b.toString();
588:         }
