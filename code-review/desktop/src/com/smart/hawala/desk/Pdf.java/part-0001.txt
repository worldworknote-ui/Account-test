54:     /** نص ملتف من اليمين لليسار. */
55:     static final class TB {
56:         final List<TextLayout> lines = new ArrayList<>();
57:         final List<Float> gaps = new ArrayList<>();
58:         final float width;
59:         final int align;
60:         final Color color;
61:         float h;
62: 
63:         TB(String s, Font f, Color color, float width, int align) {
64:             this.width = Math.max(8, width);
65:             this.align = align;
66:             this.color = color;
67:             FontRenderContext frc = new FontRenderContext(null, true, true);
68:             String t = clean(s == null ? "" : s);
69:             for (String p : t.split("\n", -1)) {
70:                 if (p.isEmpty()) {
71:                     lines.add(null);
72:                     float lh = f.getSize2D() * 1.2f;
73:                     gaps.add(lh);
74:                     h += lh;
75:                     continue;
76:                 }
77:                 AttributedString as = new AttributedString(p);
78:                 as.addAttribute(TextAttribute.FONT, f);
79:                 as.addAttribute(TextAttribute.RUN_DIRECTION, TextAttribute.RUN_DIRECTION_RTL);
80:                 LineBreakMeasurer m = new LineBreakMeasurer(as.getIterator(), frc);
81:                 while (m.getPosition() < p.length()) {
82:                     TextLayout l = m.nextLayout(this.width);
83:                     if (l == null) break;
84:                     lines.add(l);
85:                     float lh = (l.getAscent() + l.getDescent() + l.getLeading()) * 1.06f;
86:                     gaps.add(lh);
87:                     h += lh;
88:                 }
89:             }
90:         }
91: 
92:         void draw(Graphics2D g, float x, float y) {
93:             g.setColor(color);
94:             float yy = y;
95:             for (int i = 0; i < lines.size(); i++) {
96:                 TextLayout l = lines.get(i);
97:                 if (l != null) {
98:                     float adv = l.getAdvance();
99:                     float lx = align == CENTER ? x + (width - adv) / 2 : x + width - adv;
100:                     l.draw(g, lx, yy + l.getAscent());
101:                 }
102:                 yy += gaps.get(i);
103:             }
104:         }
105:     }
