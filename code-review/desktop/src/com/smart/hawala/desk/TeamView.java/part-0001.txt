69:     /** فريق مكتب واحد: لكل محاسب «—» أو «يسجّل» أو «يتحقق». */
70:     static void edit(Component parent, Db db, Db.Team t, Runnable after) {
71:         List<Model.User> acc = db.accountants();
72:         if (acc.isEmpty()) {
73:             Ui.toast("لا محاسبين بعد: أضفهم من «المستخدمون»");
74:             return;
75:         }
76:         JPanel c = Ui.column();
77:         c.add(Ui.para("لكل محاسب دور واحد في «" + t.office.name + "». من يسجّل لا يتحقق من المكتب نفسه.", 13, Ui.muted, 420));
78:         c.add(Ui.gap(6));
79:         Set<Long> own0 = new HashSet<>(), chk0 = new HashSet<>();
80:         for (Model.User u : t.owners) own0.add(u.id);
81:         for (Model.User u : t.checkers) chk0.add(u.id);
82:         Map<Long, JComboBox<String>> roles = new LinkedHashMap<>();
83:         for (Model.User u : acc) {
84:             boolean canCheck = Auth.can(u, Auth.Perm.VERIFY);
85:             JComboBox<String> cb = Ui.combo(canCheck ? new String[]{"—", "يسجّل", "يتحقق"} : new String[]{"—", "يسجّل"});
86:             cb.setSelectedIndex(own0.contains(u.id) ? 1 : chk0.contains(u.id) && canCheck ? 2 : 0);
87:             cb.setMaximumSize(new Dimension(150, 36));
88:             roles.put(u.id, cb);
89:             c.add(Ui.row(Ui.label(u.name + (canCheck ? "" : " (بلا صلاحية التحقق)"), 14, Ui.ink, true), Box.createHorizontalGlue(), cb));
90:             c.add(Ui.gap(4));
91:         }
92:         if (!Ui.form(parent, "فريق «" + t.office.name + "»", Ui.rtl(c), "حفظ")) return;
93:         Set<Long> own = new HashSet<>(), chk = new HashSet<>();
94:         for (Map.Entry<Long, JComboBox<String>> e : roles.entrySet()) {
95:             int i = e.getValue().getSelectedIndex();
96:             if (i == 1) own.add(e.getKey());
97:             if (i == 2) chk.add(e.getKey());
98:         }
99:         List<String> warn = new ArrayList<>();
100:         if (own.size() > 1) warn.add("أكثر من محاسب يسجّل هذا المكتب (" + own.size() + "): يجوز، لكن كل منهم يراه في «مكاتبي».");
101:         if (!own.isEmpty() && chk.isEmpty()) warn.add("لا أحد يتحقق من هذا المكتب: حركاته تبقى بلا تحقق.");
102:         if (!warn.isEmpty() && !Ui.confirm(parent, "تنبيه", String.join("\n", warn), "حفظ رغم ذلك")) return;
103:         try {
104:             db.setOfficeTeam(t.office.id, own, chk);
105:             Ui.toast("حُفظ فريق «" + t.office.name + "»");
106:             if (after != null) after.run();
107:         } catch (Exception ex) {
108:             Ui.error(parent, ex);
109:         }
110:     }
