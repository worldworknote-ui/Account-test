172:         @Override public Dimension getPreferredSize() {
173:             // ارتفاع الأسطر الملتفة بعرض النافذة
174:             Container par = getParent();

176:             if (width <= 0) width = 500;
177:             FlowLayout fl = (FlowLayout) getLayout();
178:             int x = fl.getHgap(), y = fl.getVgap(), rowH = 0;
179:             for (Component c : getComponents()) {
180:                 Dimension d = c.getPreferredSize();
181:                 if (x + d.width + fl.getHgap() > width && x > fl.getHgap()) {
182:                     x = fl.getHgap();
