192:     /** نافذة حفظ الماك الأصلية، مع تذكّر آخر مجلد. */
193:     static File save(Component parent, Db db, String name, String ext) {
194:         FileDialog fd = new FileDialog(frame(parent), "حفظ", FileDialog.SAVE);
195:         String dir = db.setting("last_export_dir", "");
196:         if (dir.isEmpty() || !new File(dir).isDirectory()) dir = new File(System.getProperty("user.home"), "Documents").getAbsolutePath();
197:         fd.setDirectory(dir);
198:         fd.setFile(safe(name) + "-" + Model.day(Clock.now()).replace('/', '-') + ext);
199:         fd.setVisible(true);
200:         if (fd.getFile() == null) return null;
201:         String fn = fd.getFile();
202:         if (!fn.toLowerCase().endsWith(ext)) fn += ext;
203:         db.setSettingQuiet("last_export_dir", fd.getDirectory());
204:         return new File(fd.getDirectory(), fn);
205:     }

207:     static File open(Component parent, Db db, String title, String... exts) {
208:         FileDialog fd = new FileDialog(frame(parent), title, FileDialog.LOAD);
209:         String dir = db.setting("last_export_dir", "");
210:         if (!dir.isEmpty() && new File(dir).isDirectory()) fd.setDirectory(dir);
211:         if (exts.length > 0) fd.setFilenameFilter((d, n) -> {
212:             for (String e : exts) if (n.toLowerCase().endsWith(e)) return true;
213:             return false;
214:         });
215:         fd.setVisible(true);
216:         if (fd.getFile() == null) return null;
217:         return new File(fd.getDirectory(), fd.getFile());
218:     }
