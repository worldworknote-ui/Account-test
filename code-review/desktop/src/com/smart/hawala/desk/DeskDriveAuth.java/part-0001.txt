89:     @Override public synchronized String token(boolean refresh) throws IOException {
90:         if (!refresh && access != null && Clock.now() < expires - 60000) return access;
91:         String rt;
92:         try {
93:             rt = SecretStore.get(key);
94:         } catch (Exception e) {
95:             throw new IOException("تعذر قراءة رمز Google المحفوظ: " + e.getMessage());
96:         }
97:         if (rt == null || rt.isEmpty()) throw new IOException("سجّل الدخول بحساب Google من شاشة «المزامنة» أولًا");
98:         String[] c = client(db);
99:         if (c == null) throw new IOException("معرّف عميل Google للكمبيوتر غير مضبوط");
100:         JSONObject r = post("client_id=" + enc(c[0]) + "&client_secret=" + enc(c[1]) + "&refresh_token=" + enc(rt) + "&grant_type=refresh_token");
101:         if (r.has("error")) {
102:             if ("invalid_grant".equals(r.optString("error"))) drop(key);
103:             throw new IOException("انتهى دخول Google (" + r.optString("error") + "). سجّل الدخول من جديد من «المزامنة».");
104:         }
105:         access = r.optString("access_token");
106:         expires = Clock.now() + r.optLong("expires_in", 3000) * 1000;
107:         return access;
108:     }

132:         InputStream in = code < 400 ? c.getInputStream() : c.getErrorStream();
