13: /**
14:  * 10.16: صورة مطابقة الرصيد على الكمبيوتر (كانت على الهاتف وحده): بطاقة بعرض 1080 بالقالب نفسه (التحية، العنوان، المكتب،
15:  * سطر لكل عملة بلون «لنا/لكم» وموضع الكلمة المختارين، التذييل، اسم مكتبك). تُنسخ إلى الحافظة لتُلصق في واتساب، أو تُحفظ PNG.
16:  */
17: final class ReconImage {
18:     private ReconImage() {}
19: 
20:     static final int W = 1080, PAD = 54, ROW_H = 104;
21: 
22:     private static Font f(String s, float size, boolean bold) {
23:         Font app = Ui.font(size, bold);
24:         // رموز لا يحويها خط البرنامج (🌹 🔍): خط النظام المنطقي يعرضها
25:         return s != null && app.canDisplayUpTo(s) >= 0 ? new Font(Font.DIALOG, bold ? Font.BOLD : Font.PLAIN, Math.round(size)) : app;
26:     }
27: 
28:     /** يرسم نصًّا عربيًّا محاذًى (-1 يسار، 0 وسط، 1 يمين) عند x ويرجع عرضه. */
29:     private static float text(Graphics2D g, String s, float size, boolean bold, Color c, float x, float y, int align) {
30:         if (s == null || s.isEmpty()) return 0;
31:         FontRenderContext frc = g.getFontRenderContext();
32:         TextLayout tl = new TextLayout(s, f(s, size, bold), frc);
33:         float w = tl.getAdvance();
34:         g.setColor(c);
35:         tl.draw(g, align < 0 ? x : align == 0 ? x - w / 2 : x - w, y);
36:         return w;
37:     }
38: 
39:     private static float width(Graphics2D g, String s, float size, boolean bold) {
40:         return s == null || s.isEmpty() ? 0 : new TextLayout(s, f(s, size, bold), g.getFontRenderContext()).getAdvance();
41:     }
42: 
43:     private static Color tint(Color c, float a) {
44:         return new Color(Math.round(255 + (c.getRed() - 255) * a), Math.round(255 + (c.getGreen() - 255) * a), Math.round(255 + (c.getBlue() - 255) * a));
45:     }
46: 
47:     static BufferedImage render(Reconcile.Template t, String office, String issuer, long at, List<Reconcile.Row> rows, Color accent) {
48:         String greeting = Model.nz(t.greeting).trim(), footer = Model.nz(t.footer).trim(), orn = Model.nz(t.ornament).trim();
49:         boolean hasIssuer = t.showIssuer && issuer != null && !issuer.trim().isEmpty();
50:         int h = PAD + (greeting.isEmpty() ? 0 : 64) + 150 + (t.showOffice ? 70 : 0) + (t.showDate ? 50 : 0) + 30
51:                 + Math.max(1, rows.size()) * (ROW_H + (t.gap ? 22 : 12)) + (footer.isEmpty() ? 0 : 80) + PAD + 20;
52:         BufferedImage im = new BufferedImage(W, h, BufferedImage.TYPE_INT_RGB);
53:         Graphics2D g = im.createGraphics();
54:         g.setRenderingHint(RenderingHints.KEY_ANTIALIASING, RenderingHints.VALUE_ANTIALIAS_ON);
55:         g.setRenderingHint(RenderingHints.KEY_TEXT_ANTIALIASING, RenderingHints.VALUE_TEXT_ANTIALIAS_ON);
56:         g.setRenderingHint(RenderingHints.KEY_FRACTIONALMETRICS, RenderingHints.VALUE_FRACTIONALMETRICS_ON);
57:         g.setColor(tint(accent, 0.10f));
58:         g.fillRect(0, 0, W, h);
59:         g.setColor(Color.WHITE);
60:         g.fill(new RoundRectangle2D.Float(24, 24, W - 48, h - 48, 44, 44));
61:         Color gray = new Color(120, 120, 120), ink = new Color(40, 40, 40);
62:         float left = PAD, right = W - PAD, y = PAD + 30;
63:         if (!greeting.isEmpty()) {
64:             text(g, greeting, 30, false, gray, W / 2f, y + 10, 0);
65:             y += 64;
66:         }
67:         // شريط العنوان
68:         g.setColor(accent);
69:         g.fill(new RoundRectangle2D.Float(left, y, right - left, 126, 28, 28));
70:         String title = (orn.isEmpty() ? "" : orn + "  ") + t.title.trim() + (orn.isEmpty() ? "" : "  " + orn);
71:         text(g, title, 44, true, Color.WHITE, W / 2f, y + (hasIssuer ? 58 : 78), 0);
72:         if (hasIssuer) text(g, issuer.trim(), 26, false, new Color(255, 255, 255, 225), W / 2f, y + 102, 0);
73:         y += 150;
74:         if (t.showOffice) {
75:             String who = (t.salutation.trim().isEmpty() ? "" : t.salutation.trim() + " ") + Model.nz(office) + (t.honorific.trim().isEmpty() ? "" : " " + t.honorific.trim());
76:             text(g, who, 38, true, ink, W / 2f, y + 38, 0);
77:             y += 70;
78:         }
79:         if (t.showDate) {
80:             String lbl = t.dateLabel.replaceAll("[^\\p{L}\\p{N}\\p{P}\\s]", "").trim();
81:             text(g, (lbl.isEmpty() ? "" : lbl + " ") + Model.date(at), 28, false, gray, W / 2f, y + 26, 0);
82:             y += 50;
83:         }
84:         y += 30;
85:         if (rows.isEmpty()) text(g, t.emptyText.replaceAll("[^\\p{L}\\p{N}\\p{P}\\s]", "").trim(), 34, true, ink, W / 2f, y + 60, 0);
86:         boolean before = t.wordBefore();
87:         for (Reconcile.Row r : rows) {
88:             Color col = new Color(t.color(r.sign()) & 0xFFFFFF);
89:             g.setColor(tint(col, 0.09f));
90:             g.fill(new RoundRectangle2D.Float(left, y, right - left, ROW_H, 24, 24));
91:             g.setColor(col);
92:             g.fill(new RoundRectangle2D.Float(right - 12, y + 18, 12, ROW_H - 36, 6, 6));
93:             float mid = y + ROW_H / 2f;
94:             String word = Reconcile.word(t, r), amount = t.amount(r);
95:             float ww = width(g, word, 28, true) + 44, aw = width(g, amount, 46, true);
96:             float amountX = before ? left + 26 : left + 26 + ww + 22, pillL = before ? left + 26 + aw + 22 : left + 26;
97:             g.setColor(col);
98:             g.fill(new RoundRectangle2D.Float(pillL, mid - 30, ww, 60, 60, 60));
99:             text(g, word, 28, true, Color.WHITE, pillL + ww / 2, mid + 10, 0);
100:             text(g, amount, 46, true, col, amountX, mid + 16, -1);
101:             String label = r.label(t.showCode);
102:             float avail = right - 34 - Math.max(pillL + ww, amountX + aw) - 28, size = 40;
103:             while (size > 24 && width(g, label, size, false) > avail) size -= 2;
104:             text(g, label, size, t.boldName, ink, right - 34, mid + 14, 1);
105:             y += ROW_H + (t.gap ? 22 : 12);
106:         }
107:         if (!footer.isEmpty()) {
108:             y += 14;
109:             float size = 30;
110:             while (size > 20 && width(g, footer, size, false) > right - left) size -= 2;
111:             text(g, footer, size, false, gray, W / 2f, y + 34, 0);
112:         }
113:         g.dispose();
114:         return im;
115:     }

117:     /** ينسخ الصورة إلى الحافظة (تُلصق مباشرة في واتساب للكمبيوتر أو واتساب ويب). */
118:     static void copy(final BufferedImage im) {
119:         Toolkit.getDefaultToolkit().getSystemClipboard().setContents(new Transferable() {
120:             @Override public DataFlavor[] getTransferDataFlavors() { return new DataFlavor[]{DataFlavor.imageFlavor}; }
121: 
122:             @Override public boolean isDataFlavorSupported(DataFlavor fl) { return DataFlavor.imageFlavor.equals(fl); }
123: 
124:             @Override public Object getTransferData(DataFlavor fl) { return im; }
125:         }, null);
126:     }
127: }
