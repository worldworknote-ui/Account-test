88:     static long remaining(Db db) {
89:         long until = db.settingLong("login_locked_until", 0);
90:         long now = Clock.now();
91:         if (until > now + 2 * 3600000L) until = now + 3600000L;   // ساعة مقدَّمة لا تمد الانتظار بلا حد
92:         return Math.max(0, until - now);
93:     }

95:     static int recordFail(Db db, String who, long userId) {
96:         int f = (int) db.settingLong("login_fails", 0) + 1;
97:         db.setSettingQuiet("login_fails", "" + f);
98:         long ms = Pin.lockoutMillis(f);
99:         if (ms > 0) db.setSettingQuiet("login_locked_until", "" + (Clock.now() + ms));
100:         String prev = Db.currentUserName;
101:         Db.currentUserName = who;
102:         db.audit("LOGIN_FAILED", "user", userId, "محاولة " + f);
103:         Db.currentUserName = prev;
104:         return f;
105:     }
