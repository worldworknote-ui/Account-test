926:     /**
927:      * بصمة صغيرة لصورة ما تحت الإطار: متوسط الإضاءة في خلايا 8×8 بكسل، والخلايا تحت الأرقام صفر.
928:      */
929:     static int[] signature(BufferedImage img, Rectangle in, List<Rectangle> mask) {
930:         int cell = 8;
931:         int cw = Math.max(1, img.getWidth() / cell), ch = Math.max(1, img.getHeight() / cell);
932:         double kx = img.getWidth() / (double) Math.max(1, in.width), ky = img.getHeight() / (double) Math.max(1, in.height);
933:         int[] sig = new int[cw * ch];
934:         for (int cy = 0; cy < ch; cy++) {
935:             for (int cx = 0; cx < cw; cx++) {
936:                 int x0 = cx * cell, y0 = cy * cell;
937:                 boolean masked = false;
938:                 for (Rectangle m : mask) {
939:                     int mx = (int) ((m.x - in.x) * kx), my = (int) ((m.y - in.y) * ky), mw = (int) (m.width * kx) + 1, mh = (int) (m.height * ky) + 1;
940:                     if (x0 + cell > mx && x0 < mx + mw && y0 + cell > my && y0 < my + mh) { masked = true; break; }
941:                 }
942:                 if (masked) continue;
943:                 long s = 0;
944:                 int n = 0;
945:                 for (int y = y0; y < y0 + cell && y < img.getHeight(); y += 2)
946:                     for (int x = x0; x < x0 + cell && x < img.getWidth(); x += 2) {
947:                         int p = img.getRGB(x, y);
948:                         s += ((p >> 16) & 255) * 3 + ((p >> 8) & 255) * 6 + (p & 255);
949:                         n++;
950:                     }
951:                 sig[cy * cw + cx] = n == 0 ? 0 : (int) (s / (n * 10));
952:             }
953:         }
954:         return sig;
955:     }
