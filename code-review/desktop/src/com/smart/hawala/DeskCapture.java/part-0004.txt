714:         /** سحب الإطار من شريطه (8)، وتغيير حجمه من حوافه: 1 يسار، 2 يمين، 4 أسفل. */
715:         private final class Mover extends MouseAdapter {
716:             final int base;
717:             Point start;
718:             Rectangle startA;
719:             int mode;
720: 
721:             Mover(int base) { this.base = base; }
722: 
723:             int modeAt(MouseEvent e) {
724:                 if (base != 4) return base;
725:                 int x = e.getX(), w = e.getComponent().getWidth();
726:                 if (x >= w - 24) return 4 | 2;
727:                 if (x <= 24) return 4 | 1;
728:                 return 4;
729:             }
730: 
731:             @Override public void mouseMoved(MouseEvent e) {
732:                 int m = modeAt(e);
733:                 int c = m == 8 ? Cursor.MOVE_CURSOR : m == 6 ? Cursor.SE_RESIZE_CURSOR : m == 5 ? Cursor.SW_RESIZE_CURSOR
734:                         : m == 1 ? Cursor.W_RESIZE_CURSOR : m == 2 ? Cursor.E_RESIZE_CURSOR : Cursor.S_RESIZE_CURSOR;
735:                 e.getComponent().setCursor(Cursor.getPredefinedCursor(c));
736:             }
737: 
738:             @Override public void mousePressed(MouseEvent e) {
739:                 start = e.getLocationOnScreen();
740:                 startA = new Rectangle(area);
741:                 mode = modeAt(e);
742:                 dragging = true;
743:             }
744: 
745:             @Override public void mouseDragged(MouseEvent e) {
746:                 if (start == null) return;
747:                 int dx = e.getXOnScreen() - start.x, dy = e.getYOnScreen() - start.y;
748:                 Rectangle b = new Rectangle(startA);
749:                 if (mode == 8) b.setLocation(startA.x + dx, startA.y + dy);
750:                 else {
751:                     if ((mode & 1) != 0) { b.x = startA.x + dx; b.width = startA.width - dx; }
752:                     if ((mode & 2) != 0) b.width = startA.width + dx;
753:                     if ((mode & 4) != 0) b.height = startA.height + dy;
754:                     if (b.width < minW()) { if ((mode & 1) != 0) b.x -= minW() - b.width; b.width = minW(); }
755:                     if (b.height < minH()) b.height = minH();
756:                 }
757:                 clearBadges();
758:                 setArea(b);
759:             }
760: 
761:             @Override public void mouseReleased(MouseEvent e) {
762:                 boolean was = start != null && !area.equals(startA);
763:                 start = null;
764:                 dragging = false;
765:                 if (was) moved();
766:             }
767:         }
