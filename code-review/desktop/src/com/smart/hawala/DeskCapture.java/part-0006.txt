1062:         /**
1063:          * إحداثيات إمكانية الوصول (بكسل حقيقي) إلى إحداثيات جافا (قد تكون مقسومة على تكبير الشاشة):
1064:          * شريط الإطار نفسه (F) يعطي التحويل الدقيق، وإلا نقسم على التكبير.
1065:          */
1066:         static List<Bubbles.Node> toJava(List<Bubbles.Node> raw, Rectangle bar, double scale) {
1067:             Bubbles.Node self = null;
1068:             for (Bubbles.Node n : raw) if (n.kind == 'F') self = n;
1069:             double sx = 1 / Math.max(0.5, scale), sy = sx, ox = 0, oy = 0;
1070:             if (self != null && self.w > 20 && self.h > 10 && bar != null) {
1071:                 sx = bar.width / (double) self.w;
1072:                 sy = bar.height / (double) self.h;
1073:                 ox = bar.x - self.x * sx;
1074:                 oy = bar.y - self.y * sy;
1075:             }
1076:             List<Bubbles.Node> o = new ArrayList<>();
1077:             for (Bubbles.Node n : raw) {
1078:                 if (n.kind == 'F') continue;
1079:                 o.add(new Bubbles.Node(n.kind, (int) Math.round(n.x * sx + ox), (int) Math.round(n.y * sy + oy),
1080:                         (int) Math.round(n.w * sx), (int) Math.round(n.h * sy), n.text));
1081:             }
1082:             return o;
1083:         }
