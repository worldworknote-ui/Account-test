70:     static String storedToken(Context ctx, Db db) {
71:         String a = db.setting("license_token", "").trim();
72:         String b = readFile(ctx).trim();
73:         if (a.equals(b)) return a;
74:         byte[] dk = null;
75:         try {
76:             dk = deviceKeyX509();
77:         } catch (Throwable ignored) {
78:             // يُعامل كغير صالح

81:         boolean bOk = !b.isEmpty() && LicenseToken.verify(b, dk) != null;
82:         String keep = aOk ? a : bOk ? b : (a.isEmpty() ? b : a);
83:         save(ctx, db, keep);
84:         return keep;
