256:     static String mac(String data) {
257:         try {
258:             String k = SecretStore.get(MAC);
259:             if (k == null) {
260:                 byte[] raw = new byte[32];
261:                 new SecureRandom().nextBytes(raw);
262:                 k = Base64.getEncoder().encodeToString(raw);
263:                 SecretStore.put(MAC, k);
264:             }
265:             Mac m = Mac.getInstance("HmacSHA256");
266:             m.init(new SecretKeySpec(Base64.getDecoder().decode(k), "HmacSHA256"));
267:             return BackupFormat.hex(m.doFinal(data.getBytes(StandardCharsets.UTF_8)));
268:         } catch (Throwable t) {
269:             return "";
270:         }
271:     }

284:     public static String checkOnline(Context ctx, Db db) {
285:         String base = db.setting("license_server", "").trim();
286:         if (base.isEmpty() || !current.pro) return "لا يوجد خادم ترخيص مضبوط أو لا رخصة مفعّلة.";
287:         HttpURLConnection c = null;
288:         try {
289:             byte[] nonce = new byte[16];
290:             new SecureRandom().nextBytes(nonce);
291:             String n = Base64.getUrlEncoder().withoutPadding().encodeToString(nonce);
292:             byte[] dk = deviceKeyX509();
293:             String body = "v=1;t=check;lid=" + current.lid + ";dk=" + LicenseToken.deviceFingerprint(dk) + ";n=" + n + ";at=" + Clock.now();
294:             JSONObject req = new JSONObject();
295:             req.put("body", body);
296:             req.put("sig", sign(body.getBytes(StandardCharsets.UTF_8)));
297:             req.put("pub", Base64.getEncoder().encodeToString(dk));
298:             req.put("ver", BuildInfo.VERSION);
299:             req.put("platform", "mac");
300:             c = (HttpURLConnection) new URL(base.replaceAll("/+$", "") + "/v1/check").openConnection();
301:             c.setConnectTimeout(15000);
302:             c.setReadTimeout(20000);
303:             c.setDoOutput(true);
304:             c.setRequestMethod("POST");
305:             c.setRequestProperty("Content-Type", "application/json; charset=utf-8");
306:             try (OutputStream o = c.getOutputStream()) {
307:                 o.write(req.toString().getBytes(StandardCharsets.UTF_8));
308:             }
309:             if (c.getResponseCode() != 200) return "رد الخادم: " + c.getResponseCode();
310:             JSONObject res = new JSONObject(new String(Db.readAll(c.getInputStream()), StandardCharsets.UTF_8));
311:             JSONArray rv = res.optJSONArray("revoked");
312:             StringBuilder list = new StringBuilder();
313:             if (rv != null) for (int i = 0; i < rv.length(); i++) list.append(i > 0 ? "," : "").append(rv.optString(i));
314:             db.setSettingQuiet("license_revoked", list.toString());
315:             db.setSettingQuiet("license_last_check", "" + Clock.now());
316:             load(ctx, db);
317:             return res.optString("message", "تم الفحص ✓");
318:         } catch (Throwable t) {
319:             return "تعذر الاتصال بخادم الترخيص (" + t.getClass().getSimpleName() + "). لا يتعطل شيء؛ يُعاد لاحقًا.";
320:         } finally {
321:             if (c != null) c.disconnect();
322:         }
323:     }
