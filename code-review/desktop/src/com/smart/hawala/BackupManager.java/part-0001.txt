58:     public static final class BackupException extends Exception {
59:         public final Stage stage;
60: 
61:         public BackupException(Stage stage, String message, Throwable cause) {
62:             super(message, cause);
63:             this.stage = stage;
64:         }
65: 
66:         public String userMessage() {
67:             String m = getMessage();
68:             return "المرحلة: " + stage.label + "\n\n" + (m == null ? "خطأ غير معروف" : m);
69:         }
70:     }

72:     public interface Progress {
73:         void update(String message);
74:     }

76:     static BackupException wrap(Stage stage, Throwable t) {
77:         if (t instanceof BackupException) return (BackupException) t;
78:         String m;
79:         if (t instanceof SecureStreams.WrongPasswordException || t instanceof SecureStreams.CorruptBackupException) m = t.getMessage();
80:         else if (t instanceof OutOfMemoryError) m = "الذاكرة لا تكفي. أغلق التطبيقات الأخرى ثم أعد المحاولة.";
81:         else if (t instanceof IOException && t.getMessage() != null && t.getMessage().toLowerCase(Locale.US).contains("space")) m = "لا توجد مساحة كافية في التخزين.";
82:         else if (t.getMessage() != null && !t.getMessage().trim().isEmpty()) m = t.getMessage();
83:         else m = t.getClass().getSimpleName();
84:         return new BackupException(stage, m, t);
85:     }

109:     /**
110:      * ينشئ نسخة مشفّرة داخل التطبيق ويتحقق منها بالكامل. لا يعدّل مصفوفة كلمة المرور الممرّرة؛
111:      * يعمل على نسخ مستقلة ويمسحها في النهاية. المستدعي مسؤول عن مسح مصفوفته.
112:      */
113:     public static Created createLocal(Context ctx, Db db, char[] password, String appVersion, Progress p) throws BackupException {
114:         if (password == null || password.length < 8) throw new BackupException(Stage.PREPARE, "كلمة المرور يجب ألا تقل عن 8 محارف", null);
115:         File dir = backupsDir(ctx);
116:         long free = dir.getUsableSpace();
117:         if (free > 0 && free < 5L * 1024 * 1024) throw new BackupException(Stage.PREPARE, "مساحة التخزين المتبقية في الهاتف قليلة جدًا. احذف بعض الملفات ثم أعد المحاولة.", null);
118:         String name = newName();
119:         File tmp = new File(dir, name + ".tmp");
120:         File out = new File(dir, name);
121:         char[] pwWrite = password.clone();
122:         char[] pwVerify = password.clone();
123:         Created c = new Created();
124:         try {
125:             p.update("جارٍ تجهيز البيانات…");
126:             c.counts = db.counts();
127:             BackupFormat.Source src = db.backupSource();
128: 
129:             p.update("جارٍ التشفير والكتابة…");
130:             SecureStreams.EncryptingOutputStream enc = null;
131:             try (FileOutputStream fos = new FileOutputStream(tmp)) {
132:                 BufferedOutputStream bos = new BufferedOutputStream(fos, 65536);
133:                 // 10.14: إن فُعّل «رمز استرداد النسخ» تُفتح النسخة بكلمة مرورها أو بالرمز
134:                 enc = new SecureStreams.EncryptingOutputStream(bos, pwWrite, BackupRecovery.key(db));
135:                 c.result = BackupFormat.write(src, enc, appVersion, Db.VER);
136:                 enc.close();
137:                 bos.flush();
138:                 fos.getFD().sync();
139:             } catch (Throwable t) {
140:                 if (enc != null) enc.abort();
141:                 tmp.delete();
142:                 throw wrap(Stage.ENCRYPT, t);
143:             }
144: 
145:             p.update("جارٍ التحقق من النسخة (فك كامل)…");
146:             try {
147:                 verifyFile(tmp, pwVerify, c.result);
148:             } catch (Throwable t) {
149:                 tmp.delete();
150:                 throw wrap(Stage.VERIFY, t);
151:             }
152:             if (!tmp.renameTo(out)) {
153:                 tmp.delete();
154:                 throw new BackupException(Stage.ENCRYPT, "تعذر حفظ ملف النسخة داخل التطبيق", null);
155:             }
156:             c.file = out;
157:             c.size = out.length();
158:             try {
159:                 c.sha256 = sha256(out);
160:             } catch (Exception e) {
161:                 throw wrap(Stage.VERIFY, e);
162:             }
163:             pruneLocal(ctx, 3);
164:             return c;
165:         } finally {
166:             Arrays.fill(pwWrite, '\0');
167:             Arrays.fill(pwVerify, '\0');
168:         }
169:     }
