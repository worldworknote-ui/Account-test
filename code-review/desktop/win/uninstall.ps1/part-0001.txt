1: # إزالة البرنامج والاختصارات فقط. البيانات (%APPDATA%\SmartAccountant) تبقى كما هي عمدًا:
2: # احذفها يدويًا فقط بعد أخذ نسخة احتياطية إن لم تعد تحتاجها.
3: $ErrorActionPreference = 'Continue'
4: $dst = Join-Path $env:LOCALAPPDATA 'Programs\SmartAccountant'
5: $name = 'المحاسب الذكي'
6: Get-Process -Name 'SmartAccountant' -ErrorAction SilentlyContinue | ForEach-Object { $_.CloseMainWindow() | Out-Null }
7: Start-Sleep -Seconds 1
8: Remove-Item -Force -LiteralPath (Join-Path ([Environment]::GetFolderPath('Programs')) ($name + '.lnk')) -ErrorAction SilentlyContinue
9: Remove-Item -Force -LiteralPath (Join-Path ([Environment]::GetFolderPath('Desktop')) ($name + '.lnk')) -ErrorAction SilentlyContinue
10: Set-Location $env:TEMP
11: Remove-Item -Recurse -Force $dst -ErrorAction SilentlyContinue
12: Write-Host 'Removed. Your data is still in:'
13: Write-Host (Join-Path $env:APPDATA 'SmartAccountant')
14: [void](Read-Host 'Press Enter to close')
