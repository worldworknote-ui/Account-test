188:     /**
189:      * يقرأ النسبة كما يكتبها المحاسب: أرقام عربية أو إنجليزية، والفاصلة العشرية «.» أو «,» أو «٫»،
190:      * مع تجاهل المسافات وعلامات ٪ و% و‰. يعيد null إن لم تكن رقمًا موجبًا صالحًا.

192:     public static BigDecimal parseRate(String text) {
193:         if (text == null) return null;
194:         StringBuilder sb = new StringBuilder();
195:         for (char ch : text.trim().toCharArray()) {
196:             if (ch >= '\u0660' && ch <= '\u0669') sb.append((char) ('0' + (ch - '\u0660')));
197:             else if (ch >= '\u06F0' && ch <= '\u06F9') sb.append((char) ('0' + (ch - '\u06F0')));
198:             else if (ch == '\u066B' || ch == ',' || ch == '\u060C') sb.append('.');
199:             else if (ch == ' ' || ch == '\u00A0' || ch == '%' || ch == '\u066A' || ch == '\u2030') continue;
200:             else sb.append(ch);
201:         }
202:         String s = sb.toString();
203:         if (!s.matches("\\d{1,4}(\\.\\d{1,4})?|\\.\\d{1,4}")) return null;
204:         BigDecimal r = new BigDecimal(s);
205:         return r.signum() > 0 ? r : null;
206:     }

210:      * مثال: 5000.00 دولار × 1٪ = 50.00؛ 2000.00 دولار × 1.5‰ = 3.00.
211:      */
212:     public static long feeFromRate(long amountMinor, BigDecimal rate, int per) {
213:         if (amountMinor <= 0) throw new IllegalArgumentException("المبلغ يجب أن يكون أكبر من صفر");
214:         if (rate == null || rate.signum() <= 0) throw new IllegalArgumentException("النسبة يجب أن تكون أكبر من صفر");
215:         if (per != PER_CENT && per != PER_MILLE) throw new IllegalArgumentException("per");
216:         if (rate.compareTo(BigDecimal.valueOf(per)) > 0) throw new IllegalArgumentException("النسبة أكبر من المبلغ نفسه");
217:         return BigDecimal.valueOf(amountMinor).multiply(rate)
218:                 .divide(BigDecimal.valueOf(per), 0, java.math.RoundingMode.HALF_UP).longValueExact();
219:     }
