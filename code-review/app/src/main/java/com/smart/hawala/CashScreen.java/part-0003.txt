445:     private boolean passes(CashBox.Move m, long dayStart) {
446:         switch (filter) {
447:             case 1: return m.createdAt >= dayStart;
448:             case 2: return CashBox.IN.equals(m.kind) && !m.cancelled();
449:             case 3: return CashBox.OUT.equals(m.kind) && !m.cancelled();
450:             case 4: return CashBox.EXPENSE.equals(m.kind) && !m.cancelled();
451:             case 5: return CashBox.EXCHANGE.equals(m.kind) && !m.cancelled();
452:             case 6: return m.clientId != null;
453:             case 7: return m.officeId != null;
454:             case 8: return CashBox.TRANSFER.equals(m.kind);
455:             case 9: return m.cancelled();
456:             default: return true;
457:         }
458:     }
