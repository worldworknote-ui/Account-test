384:     public static final long[] BUCKETS = {30, 90, 180};

386:     public static final class Aging {
387:         /** oldestDays: عمر أقدم جزء من الرصيد (الأقدم يُسدَّد أولًا). sinceDays: أيام منذ آخر مرة كان الحساب مسدَّدًا (صفرًا أو بالاتجاه المعاكس). */
388:         public long office, balance, oldestDays, sinceDays;
389:         public String currency;
390:         /** 0-30، 31-90، 91-180، أكثر من 180 يومًا. */
391:         public final long[] buckets = new long[4];
392: 
393:         public long over90() { return buckets[2] + buckets[3]; }
394:     }

396:     /** receivable=true: ما لك (رصيد موجب)؛ false: ما عليك. */
397:     public static List<Aging> aging(Input in, boolean receivable) {
398:         Map<String, List<long[]>> moves = new HashMap<>();   // office|currency ← {when, amount} في اتجاه الرصيد
399:         List<Object[]> fx = new ArrayList<>();
400:         for (Model.Tx t : in.txs) {
401:             if (!active(t)) continue;
402:             fx.clear();
403:             effects(t, fx);
404:             for (Object[] e : fx) {
405:                 long v = (Long) e[1];
406:                 if (receivable ? v > 0 : v < 0) moves.computeIfAbsent(t.officeId + "|" + e[0], k -> new ArrayList<>()).add(new long[]{t.createdAt, Math.abs(v)});
407:             }
408:         }
409:         for (Opening o : in.openings) {
410:             if (receivable ? o.signed > 0 : o.signed < 0) moves.computeIfAbsent(o.office + "|" + o.currency, k -> new ArrayList<>()).add(new long[]{o.when, Math.abs(o.signed)});
411:         }
412:         // كل الحركات بترتيبها الزمني لحساب «غير مسدَّد منذ»
413:         Map<String, List<long[]>> all = new HashMap<>();   // office|currency ← {when, id, signed}
414:         for (Model.Tx t : in.txs) {
415:             if (!active(t)) continue;
416:             fx.clear();
417:             effects(t, fx);
418:             for (Object[] e : fx) all.computeIfAbsent(t.officeId + "|" + e[0], k -> new ArrayList<>()).add(new long[]{t.createdAt, t.id, (Long) e[1]});
419:         }
420:         for (Opening o : in.openings) all.computeIfAbsent(o.office + "|" + o.currency, k -> new ArrayList<>()).add(new long[]{o.when, -1, o.signed});
421:         List<Aging> out = new ArrayList<>();
422:         for (Map.Entry<Long, Map<String, Long>> e : balances(in).entrySet()) {
423:             for (Map.Entry<String, Long> b : e.getValue().entrySet()) {
424:                 long bal = receivable ? b.getValue() : -b.getValue();
425:                 if (bal <= 0) continue;
426:                 Aging a = new Aging();
427:                 List<long[]> seq = all.getOrDefault(e.getKey() + "|" + b.getKey(), new ArrayList<>());
428:                 Collections.sort(seq, (x, y) -> x[0] != y[0] ? Long.compare(x[0], y[0]) : Long.compare(x[1], y[1]));
429:                 long run = 0, start = seq.isEmpty() ? in.now : seq.get(0)[0];
430:                 for (long[] m : seq) {
431:                     long before = run;
432:                     run += receivable ? m[2] : -m[2];
433:                     if (before <= 0 && run > 0) start = m[0];
434:                 }
435:                 a.sinceDays = Math.max(0, (in.now - start) / DAY);
436:                 a.office = e.getKey();
437:                 a.currency = b.getKey();
438:                 a.balance = bal;
439:                 List<long[]> l = moves.getOrDefault(a.office + "|" + a.currency, new ArrayList<>());
440:                 Collections.sort(l, (x, y) -> Long.compare(y[0], x[0]));
441:                 long left = bal;
442:                 for (long[] m : l) {
443:                     if (left <= 0) break;
444:                     long take = Math.min(left, m[1]);
445:                     long days = Math.max(0, (in.now - m[0]) / DAY);
446:                     a.buckets[days <= BUCKETS[0] ? 0 : days <= BUCKETS[1] ? 1 : days <= BUCKETS[2] ? 2 : 3] += take;
447:                     a.oldestDays = Math.max(a.oldestDays, days);
448:                     left -= take;
449:                 }
450:                 if (left > 0) a.buckets[3] += left;   // أقدم من كل حركة مسجّلة
451:                 out.add(a);
452:             }
453:         }
454:         Collections.sort(out, (x, y) -> Long.compare(y.over90(), x.over90()) != 0 ? Long.compare(y.over90(), x.over90()) : Long.compare(y.balance, x.balance));
455:         return out;
456:     }

459:     public static Map<String, long[]> agingTotals(List<Aging> l) {

466:         }
