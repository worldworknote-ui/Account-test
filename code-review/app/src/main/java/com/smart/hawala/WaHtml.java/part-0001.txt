9:  * 12.1: نسخة HTML من الحافظة. واتساب ديسكتوب على ويندوز وواتساب ويب في المتصفح يضعان مع النص العادي (الملزوق كرسالة واحدة)
10:  * نسخة HTML فيها كل فقاعة في عنصر مستقل، وغالبًا مع خاصية data-pre-plain-text="[الوقت، التاريخ] الاسم: ".
11:  * نعيد بناء النص بصيغة النسخ في أندرويد ثم نقسمه بـ {@link WaText#split} نفسها، فيبقى منطق التقسيم واحدًا.
12:  * منطق بحت بلا أندرويد حتى يُختبر على الكمبيوتر. لا نعتمد على شكل واتساب الداخلي إلا كتلميح: إن لم نجد علامات الفقاعات نعيد null.
13:  */

17:     private static final Pattern PRE = Pattern.compile("data-pre-plain-text\\s*=\\s*(\"([^\"]*)\"|'([^']*)')", Pattern.CASE_INSENSITIVE);
18:     /** بداية فقاعة في واتساب ويب (رسالة واردة أو صادرة). */
19:     private static final Pattern BUBBLE = Pattern.compile("<[a-z0-9]+[^>]*class\\s*=\\s*[\"'][^\"']*\\bmessage-(?:in|out)\\b[^\"']*[\"'][^>]*>", Pattern.CASE_INSENSITIVE);

21:     /**
22:      * يحوّل HTML الحافظة إلى رسائل، أو null إن لم يكن فيه ما يدل على فقاعات (فيُستعمل النص العادي كما كان).
23:      */
24:     public static List<WaText.Msg> messages(String html) {
25:         if (html == null || html.isEmpty()) return null;
26:         String h = stripHeader(html);
27:         List<int[]> starts = new ArrayList<>();
28:         List<String> prefixes = new ArrayList<>();
29:         Matcher m = PRE.matcher(h);
30:         while (m.find()) {
31:             String v = m.group(2) != null ? m.group(2) : m.group(3);
32:             int tagStart = h.lastIndexOf('<', m.start());
33:             int tagEnd = h.indexOf('>', m.end());
34:             if (tagStart < 0 || tagEnd < 0) continue;
35:             starts.add(new int[]{tagStart, tagEnd + 1});
36:             prefixes.add(decode(v));
37:         }
38:         if (!starts.isEmpty()) {
39:             StringBuilder sb = new StringBuilder();
40:             for (int k = 0; k < starts.size(); k++) {
41:                 int from = starts.get(k)[1];
42:                 int to = k + 1 < starts.size() ? starts.get(k + 1)[0] : h.length();
43:                 String body = WaCuts.dropMeta(text(h.substring(from, to)));
44:                 if (body.trim().isEmpty()) continue;
45:                 String pre = prefixes.get(k).trim();
46:                 if (sb.length() > 0) sb.append('\n');
47:                 sb.append(pre).append(pre.endsWith(":") ? " " : "").append(body.trim());
48:             }
49:             List<WaText.Msg> out = WaText.split(sb.toString());
50:             return out.isEmpty() ? null : out;
51:         }
52:         // بلا data-pre-plain-text: فقاعات واتساب ويب بأسماء صنفها
53:         Matcher b = BUBBLE.matcher(h);
54:         List<Integer> pos = new ArrayList<>();
55:         while (b.find()) pos.add(b.start());
56:         if (pos.size() < 1) return null;
57:         List<WaText.Msg> out = new ArrayList<>();
58:         for (int k = 0; k < pos.size(); k++) {
59:             int to = k + 1 < pos.size() ? pos.get(k + 1) : h.length();
60:             String t = text(h.substring(pos.get(k), to));
61:             String time = WaCuts.lastTime(t);
62:             String body = WaCuts.dropMeta(t).trim();
63:             if (time.isEmpty()) {
64:                 String[] tt = WaCuts.trailingTime(body);
65:                 body = tt[0];
66:                 time = tt[1];
67:             }
68:             if (body.isEmpty()) continue;
69:             out.addAll(asMsgs(body, time));
70:         }
71:         return out.isEmpty() ? null : out;
72:     }

74:     private static List<WaText.Msg> asMsgs(String body, String time) {
75:         List<WaText.Msg> l = WaText.split(body);
76:         if (l.size() == 1 && l.get(0).time.isEmpty()) {
77:             List<WaText.Msg> r = new ArrayList<>();
78:             r.add(new WaText.Msg(l.get(0).sender, time, l.get(0).body));
79:             return r;
80:         }
81:         return l;
82:     }

84:     /** ترويسة CF_HTML في ويندوز («Version:0.9 StartHTML:…») إن بقيت. */
85:     static String stripHeader(String html) {
86:         int i = html.indexOf('<');
87:         if (i > 0 && html.substring(0, i).contains("StartHTML")) return html.substring(i);
88:         return html;

91:     private static final Pattern BLOCK = Pattern.compile("(?i)<\\s*(br|/p|/div|/li|/tr|/h[1-6]|p|div|li|tr)\\b[^>]*>");
92:     private static final Pattern DROP = Pattern.compile("(?is)<(script|style|head)\\b.*?</\\1\\s*>");
93: 
94:     /** نص HTML: العناصر الكتلية تصبح أسطرًا، والصور الرمزية (الإيموجي) تُستبدل بنصها البديل. */
95:     public static String text(String html) {
96:         String s = DROP.matcher(html).replaceAll("");
97:         s = s.replaceAll("(?i)<img\\b[^>]*\\balt\\s*=\\s*\"([^\"]*)\"[^>]*>", "$1");
98:         s = BLOCK.matcher(s).replaceAll("\n");
99:         s = s.replaceAll("<[^>]*>", "");
100:         s = decode(s);
101:         StringBuilder out = new StringBuilder();
102:         for (String line : s.split("\n", -1)) {
103:             String t = line.replace('\u00a0', ' ').replaceAll("[ \\t]+", " ").trim();
104:             if (t.isEmpty()) {
105:                 if (out.length() > 0 && out.charAt(out.length() - 1) != '\n') out.append('\n');
106:                 continue;
107:             }
108:             if (out.length() > 0 && out.charAt(out.length() - 1) != '\n') out.append('\n');
109:             out.append(t);
110:         }
111:         return out.toString().trim();
112:     }

114:     static String decode(String s) {
115:         if (s.indexOf('&') < 0) return s;
116:         StringBuilder b = new StringBuilder(s.length());
117:         int i = 0;
118:         while (i < s.length()) {
119:             char c = s.charAt(i);
120:             if (c == '&') {
121:                 int semi = s.indexOf(';', i);
122:                 if (semi > i && semi - i <= 10) {
123:                     String e = s.substring(i + 1, semi);
124:                     String r = null;
125:                     switch (e) {
126:                         case "amp": r = "&"; break;
127:                         case "lt": r = "<"; break;
128:                         case "gt": r = ">"; break;
129:                         case "quot": r = "\""; break;
130:                         case "apos": r = "'"; break;
131:                         case "nbsp": r = " "; break;
132:                         default:
133:                             try {
134:                                 if (e.startsWith("#x") || e.startsWith("#X")) r = new String(Character.toChars(Integer.parseInt(e.substring(2), 16)));
135:                                 else if (e.startsWith("#")) r = new String(Character.toChars(Integer.parseInt(e.substring(1))));
136:                             } catch (Exception ignored) {
137:                                 r = null;
138:                             }
139:                     }
140:                     if (r != null) {
141:                         b.append(r);
142:                         i = semi + 1;
143:                         continue;
144:                     }
145:                 }
146:             }
147:             b.append(c);
148:             i++;
149:         }
150:         return b.toString();
151:     }
