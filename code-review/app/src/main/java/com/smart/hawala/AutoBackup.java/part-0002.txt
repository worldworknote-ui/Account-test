256:     static void monthlyReadable(Context ctx, Db db, ContentResolver cr, Uri tree, String treeDoc) throws Exception {
257:         if (!"1".equals(db.setting("readable_export", "0"))) return;
258:         java.util.Calendar c = Model.cal();
259:         String month = String.format(java.util.Locale.US, "%04d-%02d", c.get(java.util.Calendar.YEAR), c.get(java.util.Calendar.MONTH) + 1);
260:         if (month.equals(db.setting("readable_export_month", ""))) return;
261:         long now = Clock.now();
262:         Uri parent = DocumentsContract.buildDocumentUriUsingTree(tree, treeDoc);
263:         String issuer = db.setting("office_name", "");
264:         Entitlements ent = License.current;
265:         Report bal = Reports.balances(db.officeTotals(null), db.decimalsMap(), db.currencyNames(), false, issuer, now);
266:         bal.stamp(ent.watermark(), ent.lid);
267:         writeXlsx(cr, parent, prefix(ctx).replace("-auto-", "-readable-") + month + "-balances.xlsx", bal);
268:         c.set(java.util.Calendar.DAY_OF_MONTH, 1);
269:         c.set(java.util.Calendar.HOUR_OF_DAY, 0);
270:         c.set(java.util.Calendar.MINUTE, 0);
271:         c.set(java.util.Calendar.SECOND, 0);
272:         c.set(java.util.Calendar.MILLISECOND, 0);
273:         long thisMonth = c.getTimeInMillis();
274:         c.add(java.util.Calendar.MONTH, -1);
275:         Db.TxFilter f = new Db.TxFilter();
276:         f.from = c.getTimeInMillis();
277:         f.to = thisMonth;
278:         f.includeCancelled = true;
279:         String prev = String.format(java.util.Locale.US, "%04d-%02d", c.get(java.util.Calendar.YEAR), c.get(java.util.Calendar.MONTH) + 1);
280:         Report txs = Reports.txList("حركات شهر " + prev, new ArrayList<String>(), db.txLog(f), db.officeNames(), db.decimalsMap(), issuer, now);
281:         txs.stamp(ent.watermark(), ent.lid);
282:         writeXlsx(cr, parent, prefix(ctx).replace("-auto-", "-readable-") + prev + "-transactions.xlsx", txs);
283:         db.setSettingQuiet("readable_export_month", month);
284:         db.setSettingQuiet("readable_export_at", "" + now);
285:         db.deleteSetting("readable_export_error");
286:     }
