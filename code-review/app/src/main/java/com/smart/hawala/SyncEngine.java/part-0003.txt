354:     public static String tombText(JSONObject h) {
355:         if (h == null) return "";
356:         JSONObject x = h.optJSONObject(h.has("deleted") ? "deleted" : "moved");
357:         String who = x == null ? "" : x.optString("by"), when = x == null || x.optLong("at") <= 0 ? "" : Model.date(x.optLong("at"));
358:         if (h.has("deleted")) return "حذف المدير" + (who.isEmpty() ? "" : " «" + who + "»") + " ملفات المزامنة من هذا المكان" + (when.isEmpty() ? "" : " في " + when) + ".";
359:         String to = x == null ? "" : x.optString("to");
360:         boolean fresh = x != null && x.optBoolean("fresh");
361:         return (fresh ? "أُنشئت مزامنة جديدة" : "نُقلت المزامنة") + (to.isEmpty() ? " إلى مكان آخر" : " إلى «" + to + "»")
362:                 + (who.isEmpty() ? "" : " (بواسطة «" + who + "»)") + (when.isEmpty() ? "" : " في " + when) + ".";
363:     }

380:     /** 11.5: توقيع أمر الحذف/النقل: HMAC بمفتاح مشتق من مفتاح الخزنة على الخزنة والنوع وكل حقول الأمر (مرتبة). */
381:     static String tombTag(byte[] vaultKey, JSONObject t) {
382:         String kind = t.has("deleted") ? "deleted" : "moved";
383:         JSONObject x = t.optJSONObject(kind);
384:         StringBuilder b = new StringBuilder("SA11-tomb|").append(t.optString("vault")).append('|').append(kind);
385:         if (x != null) {
386:             java.util.List<String> ks = new java.util.ArrayList<>();
387:             java.util.Iterator<String> it = x.keys();
388:             while (it.hasNext()) ks.add(it.next());
389:             java.util.Collections.sort(ks);
390:             for (String k : ks) b.append('|').append(k).append('=').append(String.valueOf(x.opt(k)));
391:         }
392:         byte[] key = SyncCrypto.hkdf(vaultKey, null, "SA11-tombstone");
393:         return SyncCrypto.hex(SyncCrypto.hmac(key, b.toString().getBytes(StandardCharsets.UTF_8)));
394:     }

397:     static boolean tombVerified(JSONObject h, byte[] vaultKey) {
398:         if (h == null || vaultKey == null || !tombstone(h)) return false;
399:         String tag = h.optString("auth", "");
400:         return !tag.isEmpty() && java.security.MessageDigest.isEqual(tag.getBytes(StandardCharsets.UTF_8), tombTag(vaultKey, h).getBytes(StandardCharsets.UTF_8));

438:     private int wipeLocked(android.content.Context ctx, char[] secret, boolean recovery) throws Exception {
439:         requireAdmin("حذف ملفات المزامنة");
440:         JSONObject h = header();
441:         if (h == null) throw new IllegalStateException("لا توجد خزنة مزامنة في هذا المكان");
442:         if (tombstone(h)) throw new IllegalStateException(tombText(h) + " لا شيء آخر يُحذف هنا.");
443:         byte[] vk;
444:         try {
445:             vk = unlock(h, secret, recovery);
446:         } catch (GeneralSecurityException e) {
447:             throw new IllegalArgumentException(recovery ? "رمز الاسترداد غير صحيح" : "كلمة سر المزامنة غير صحيحة");
448:         }
449:         // نسخة أمان من بيانات هذا الجهاز قبل أي حذف (يمكن التراجع إليها من «النسخ الاحتياطي ← نسخ الأمان»)
450:         if (ctx != null) Snapshot.take(ctx, db, "قبل حذف ملفات المزامنة");
451:         String vault = h.optString("vault");
452:         int n = deleteFiles();
453:         folder.write(HEADER, tombHeader(vault, "deleted", new JSONObject(), vk).toString(1).getBytes(StandardCharsets.UTF_8));
