124:     private void usage(Entitlements e) {
125:         QuotaState q;
126:         try {
127:             q = a.db.quotaNow();
128:         } catch (Exception ex) {
129:             a.hint("تعذر حساب الاستهلاك الآن: " + Ui.friendly(ex));
130:             return;
131:         }
132:         a.section("الاستهلاك", null, null);
133:         LinearLayout g = a.group();
134:         long mo = e.maxOffices(), mt = e.maxTxPerOffice();
135:         a.groupRow(g, R.drawable.ic_users, a.primaryInk, "المكاتب", mo == Entitlements.UNLIMITED ? "بلا حد" : "حذف أي مكتب يعيد مكانه فارغًا",
136:                 mo == Entitlements.UNLIMITED ? "" + q.offices : q.offices + " من " + mo, null);
137:         if (mt != Entitlements.UNLIMITED) {
138:             List<long[]> top = new ArrayList<>();
139:             for (Map.Entry<Long, Long> x : q.tx.entrySet()) top.add(new long[]{x.getKey(), x.getValue()});
140:             Collections.sort(top, (p1, p2) -> Long.compare(p2[1], p1[1]));
141:             int shown = 0;
142:             for (long[] t : top) {
143:                 if (shown >= 5 || t[1] < mt * 6 / 10) break;
144:                 Model.Office o = a.db.office(t[0]);
145:                 if (o == null) continue;
146:                 long left = QuotaState.remaining(t[1], mt);
147:                 a.groupRow(g, R.drawable.ic_scale, left == 0 ? a.red : a.primaryInk, Model.nz(o.name),
148:                         left == 0 ? "بلغ السقف: التصحيح والكشف والتقارير تعمل" : "بقي " + left + " حركة", t[1] + " من " + mt, null);
149:                 shown++;
150:             }
