76:     /** الأرقام العربية والفارسية إلى 0-9، وفواصلها إلى , و . */
77:     static String digits(String s) {
78:         StringBuilder sb = new StringBuilder(s.length());
79:         for (char ch : s.toCharArray()) {
80:             if (ch >= '٠' && ch <= '٩') sb.append((char) ('0' + ch - '٠'));
81:             else if (ch >= '۰' && ch <= '۹') sb.append((char) ('0' + ch - '۰'));
82:             else if (ch == '٫') sb.append('.');
83:             else if (ch == '٬') sb.append(',');
84:             else sb.append(ch);
85:         }
86:         return sb.toString();
87:     }

89:     /** يحذف التطويل والتشكيل ويجمع المسافات، مع إبقاء الحروف كما كُتبت (للعرض). */
90:     static String clean(String s) {
91:         StringBuilder sb = new StringBuilder(s.length());
92:         for (char ch : digits(s).toCharArray()) {
93:             if (ch == 'ـ') continue;                         // تطويل
94:             if (ch >= 'ً' && ch <= 'ٟ') continue;       // تشكيل
95:             if (ch == 'ٰ' || ch == '\u200F' || ch == '\u200E' || ch == '​' || ch == '\u202A' || ch == '\u202B' || ch == '\u202C') continue;
96:             if (ch == '*' || ch == '~' || ch == '_') continue;     // تنسيق واتساب (عريض/مشطوب/مائل)
97:             if (ch == '\u00a0' || ch == '\t') ch = ' ';
98:             sb.append(ch);
99:         }
100:         return sb.toString().replaceAll(" {2,}", " ").trim();
101:     }

103:     /**
104:      * للتطابق: كالتنظيف + توحيد الألف والتاء المربوطة والياء، وحروف صغيرة بلا علامات:
105:      * Yılmaz ← yilmaz، Ștefan ← stefan، Müller ← muller، Señor ← senor، Ёлка ← елка.
106:      */
107:     static String norm(String s) {
108:         String c = clean(s).replace('İ', 'i').toLowerCase(Locale.ROOT);
109:         StringBuilder sb = new StringBuilder(c.length());
110:         for (char ch : c.toCharArray()) {
111:             if (ch == 'أ' || ch == 'إ' || ch == 'آ' || ch == 'ٱ') ch = 'ا';
112:             else if (ch == 'ة') ch = 'ه';
113:             else if (ch == 'ى' || ch == 'ی') ch = 'ي';    // ی الفارسية
114:             else if (ch == 'ک') ch = 'ك';
115:             else if (ch == 'ё') ch = 'е';
116:             else if (ch >= '̀' && ch <= 'ͯ') continue;   // علامات مركّبة (مثل نقطة i التركية)
117:             else if ((ch >= 'À' && ch <= 'ɏ') || (ch >= 'Ḁ' && ch <= 'ỿ')) {
118:                 sb.append(foldLatin(ch));
119:                 continue;
120:             }
121:             sb.append(ch);
122:         }
123:         return sb.toString();
124:     }

126:     private static String foldLatin(char ch) {
127:         switch (ch) {
128:             case 'ı': return "i";
129:             case 'ß': return "ss";
130:             case 'æ': return "ae";
131:             case 'œ': return "oe";
132:             case 'ø': return "o";
133:             case 'đ': return "d";
134:             case 'ł': return "l";
135:             case 'þ': return "th";
136:             default:
137:                 String d = java.text.Normalizer.normalize(String.valueOf(ch), java.text.Normalizer.Form.NFD);
138:                 StringBuilder o = new StringBuilder();
139:                 for (char x : d.toCharArray()) if (Character.getType(x) != Character.NON_SPACING_MARK) o.append(x);
140:                 return o.length() == 0 ? String.valueOf(ch) : o.toString();
141:         }
142:     }
