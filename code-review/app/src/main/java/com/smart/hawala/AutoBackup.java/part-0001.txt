78:     public static boolean due(Db db) {
79:         if (!enabled(db)) return false;
80:         long last = db.settingLong("last_auto_backup", 0);
81:         int every = everyTx(db);
82:         long since = every > 0 ? db.txCount() - db.settingLong("last_backup_tx", 0) : 0;
83:         return BackupPolicy.due(Clock.now(), last, intervalHours(db), since, every);
84:     }

86:     public static void schedule(Context c) {
87:         JobScheduler js = (JobScheduler) c.getSystemService(Context.JOB_SCHEDULER_SERVICE);
88:         if (js == null || js.getPendingJob(JOB_ID) != null) return;
89:         JobInfo job = new JobInfo.Builder(JOB_ID, new ComponentName(c, AutoBackupJob.class))
90:                 .setPeriodic(6L * 3600000L)
91:                 .setPersisted(true)
92:                 .setRequiresStorageNotLow(true)
93:                 .build();
94:         js.schedule(job);
95:     }

97:     public static void cancel(Context c) {
98:         JobScheduler js = (JobScheduler) c.getSystemService(Context.JOB_SCHEDULER_SERVICE);
99:         if (js != null) js.cancel(JOB_ID);
100:     }

103:     public static void runIfDueAsync(final Context ctx) {
104:         final Context app = ctx.getApplicationContext();
105:         new Thread(new Runnable() {
106:             @Override public void run() {
107:                 Db db = SmartApp.db;
108:                 if (db == null) return;

112:                     long now = Clock.now();
