56:     /**
57:      * 12.6: ومعه اليوم (yyyymmdd، لا يرجع للوراء: أكبر يوم رآه التطبيق) لاشتراك بتاريخ انتهاء. بعد الانتهاء يعمل بالمجاني
58:      * ولا يُقفل أي شيء من البيانات؛ تتوقف الإضافة عند سقوف المجاني فقط.
59:      */
60:     public static Entitlements of(LicenseToken.Grant g, int buildDate, long today) {
61:         if (g == null) return free("");
62:         long ex = g.num(K_EXPIRES);
63:         if (ex > 0 && today > ex) {
64:             return free("انتهى اشتراك رخصتك (" + g.lid() + ") في " + fmtDate(ex) + ". جدّده من البائع؛ بياناتك كلها متاحة، والإضافة عند حدود المجاني فقط.");
65:         }
66:         if (!g.coversBuild(buildDate)) {
67:             return free("رخصتك (" + g.lid() + ") تشمل التحديثات حتى " + fmtDate(g.num("uu")) + "، وهذا الإصدار أحدث منها. "
68:                     + "ارجع إلى آخر إصدار مشمول أو جدّد التحديثات؛ بياناتك كلها متاحة.");
69:         }
70:         return new Entitlements(true, g.lid(), "", g);
71:     }

73:     private boolean feature(String f) { return pro && grant != null && grant.has(f); }
74: 
75:     /**
76:      * 12.6: الحد من الرمز: رقم يكتبه البائع (أكبر من صفر) هو الحد؛ وإلا بلا حد إن كانت الميزة مفعّلة (أو الرخصة مدفوعة للحدود
77:      * التي كانت مفتوحة للجميع قبل 12.6: الصناديق والعملاء)؛ وإلا حد المجاني.
78:      */
79:     private long limit(String key, String flag, boolean openWhenPro, long free) {
80:         if (!pro || grant == null) return free;
81:         long n = grant.num(key);
82:         if (n > 0) return n;
83:         if ((flag != null && grant.has(flag)) || openWhenPro) return UNLIMITED;
84:         return free;
85:     }

135:     static String fmtDate(long yyyymmdd) {
136:         if (yyyymmdd <= 0) return "—";

138:         return s.length() == 8 ? s.substring(0, 4) + "/" + s.substring(4, 6) + "/" + s.substring(6) : s;
