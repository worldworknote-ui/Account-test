221:     public static String trimZeros(String a) {
222:         if (a == null || a.indexOf('.') < 0) return a;
223:         int e = a.length();
224:         while (e > 0 && a.charAt(e - 1) == '0') e--;
225:         if (e > 0 && a.charAt(e - 1) == '.') e--;
226:         return a.substring(0, e);

230:     public static String line(Template t, Row r) {
231:         String mark = r.sign() > 0 ? t.markPlus : r.sign() < 0 ? t.markMinus : t.markZero;
232:         String name = r.label(t.showCode);
233:         String out = validLine(t.lineFormat).replace("\\n", "\n")
234:                 .replace("{علامة}", Model.nz(mark).trim())
235:                 .replace("{مبلغ}", b(t, t.amount(r)))
236:                 .replace("{عملة}", t.boldName ? b(t, name) : name)
237:                 .replace("{رمز}", r.code)
238:                 .replace("{كلمة}", word(t, r));
239:         // خانة فارغة (علامة محذوفة مثلًا) لا تترك فراغين ولا فراغًا أول السطر
240:         StringBuilder sb = new StringBuilder();
241:         for (String ln : out.split("\n", -1)) {
242:             String x = ln.replaceAll(" {2,}", " ").trim();
243:             if (x.isEmpty()) continue;
244:             if (sb.length() > 0) sb.append("\n");
245:             sb.append(x);
246:         }
247:         return sb.toString();
248:     }
