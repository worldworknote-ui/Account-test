383:             last = e instanceof IOException ? (IOException) e : new IOException(e.getMessage(), e);
384:         }
385:         throw last != null ? last : new IOException("تعذر فتح المكان المختار للكتابة");

440:     /** يفك النسخة ويجهّز صورها (مشفّرة بمفتاح هذا الهاتف) ويفحصها كاملة، دون لمس البيانات الحالية. */
441:     public static Prepared prepareRestore(Context ctx, File file, char[] password, Progress p) throws BackupException {
442:         char[] pw = password.clone();
443:         clearStaging(ctx);
444:         final File dir = stagingDir(ctx);
445:         if (!dir.exists() && !dir.mkdirs()) throw new BackupException(Stage.READ, "تعذر تجهيز مجلد مؤقت", null);
446:         BackupFormat.AttachmentStore store = new BackupFormat.AttachmentStore() {
447:             @Override public Object store(long id, InputStream data) throws Exception {
448:                 byte[] raw = readLimited(data);
449:                 File f = new File(dir, UUID.randomUUID() + ".evd");
450:                 try (FileOutputStream o = new FileOutputStream(f)) {
451:                     o.write(Crypto.encBytes(raw));
452:                     o.getFD().sync();
453:                 }
454:                 Arrays.fill(raw, (byte) 0);
455:                 return f;
456:             }
457: 
458:             @Override public void discard(Object handle) {
459:                 if (handle instanceof File) ((File) handle).delete();
460:             }
461:         };
462:         Prepared out = new Prepared();
463:         try {
464:             byte[] head = new byte[8];
465:             try (InputStream in = new FileInputStream(file)) {
466:                 int n = 0;
467:                 while (n < 8) {
468:                     int r = in.read(head, n, 8 - n);
469:                     if (r < 0) break;
470:                     n += r;
471:                 }
472:                 if (n < 8) throw new SecureStreams.CorruptBackupException("الملف صغير جدًا وليس نسخة احتياطية");
473:             } catch (Throwable t) {
474:                 throw wrap(Stage.READ, t);
475:             }
476:             p.update("جارٍ فك التشفير وقراءة النسخة…");
477:             try {
478:                 if (SecureStreams.startsWith(head, SecureStreams.MAGIC)) {
479:                     try (InputStream in = new BufferedInputStream(new FileInputStream(file), 65536)) {
480:                         out.parsed = BackupFormat.readV5(new SecureStreams.DecryptingInputStream(in, pw), store, dir);
481:                     }
482:                 } else if (SecureStreams.startsWith(head, BackupFormat.MAGIC_V4)) {
483:                     if (file.length() > 300L * 1024 * 1024) throw new SecureStreams.CorruptBackupException("ملف الإصدار القديم كبير جدًا");
484:                     out.parsed = BackupFormat.readV4(Db.readAll(file), pw, store);
485:                 } else {
486:                     throw new SecureStreams.CorruptBackupException("هذا الملف ليس نسخة احتياطية من المحاسب الذكي (امتداد النسخ: .sba)");
487:                 }
488:             } catch (Throwable t) {
489:                 clearStaging(ctx);
490:                 throw wrap(Stage.DECRYPT, t);
491:             }
492:             p.update("جارٍ فحص القيود والعلاقات والصور…");
493:             out.report = BackupFormat.validate(out.parsed);
494:             if (!out.report.ok()) {
495:                 clearStaging(ctx);
496:                 throw new BackupException(Stage.VALIDATE, "النسخة لا تجتاز الفحص، ولن تُستعاد حتى لا تفسد بياناتك:\n• "
497:                         + join(out.report.errors, "\n• "), null);
498:             }
499:             return out;
500:         } finally {
501:             Arrays.fill(pw, '\0');
502:         }
503:     }

522:     /** نسخة أمان من البيانات الحالية ثم الاستبدال. */
523:     public static void applyRestore(Context ctx, Db db, Prepared prep, Progress p) throws BackupException {
524:         p.update("جارٍ حفظ نسخة أمان من بياناتك الحالية…");
525:         try {
526:             Snapshot.take(ctx, db);
527:         } catch (Throwable t) {
528:             throw wrap(Stage.SNAPSHOT, t);
529:         }
530:         p.update("جارٍ استبدال البيانات…");
531:         try {
532:             db.restoreKeepLook = prep.keepLook;
533:             db.importBackup(prep.parsed, prep.keep);
534:         } catch (Throwable t) {
535:             for (BackupFormat.StagedAttachment s : prep.parsed.attachments) {
536:                 if (s.handle instanceof File) ((File) s.handle).delete();
537:             }
538:             throw wrap(Stage.IMPORT, t);
539:         } finally {
540:             db.restoreKeepLook = false;
541:             clearStaging(ctx);
542:         }
543:     }
