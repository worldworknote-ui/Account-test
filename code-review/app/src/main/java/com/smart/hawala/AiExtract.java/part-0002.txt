206:     /** رقم المبلغ كما يقبله النموذج: «2500»، «1,500.50»، «1.500»؛ وإلا فارغ. */
207:     static String amount(String raw) {
208:         String s = MessageParser.digits(raw == null ? "" : raw).replaceAll("[\\s'’_]", "");
209:         if (s.isEmpty()) return "";
210:         BigDecimal v = MessageParser.number(s);
211:         if (v == null) {
212:             try {
213:                 v = new BigDecimal(s.replace(",", ""));
214:             } catch (NumberFormatException e) {
215:                 return "";
216:             }
217:         }
218:         return v.signum() > 0 ? v.stripTrailingZeros().toPlainString() : "";
219:     }

258:     /** رسالة خطأ مفهومة بالعربية، مع رمز HTTP ورسالة الخدمة المختصرة للتشخيص. */

260:         String detail = "";
261:         try {
262:             JSONObject e = new JSONObject(body).optJSONObject("error");

268:         if (detail.length() > 160) detail = detail.substring(0, 160) + "…";
269:         String low = detail.toLowerCase(Locale.ROOT);
270:         String m;
271:         if (code == 401) m = "مفتاح الخدمة غير صحيح أو منتهٍ. تحقق منه في الإعدادات.";

273:         else if (code == 404) m = "اسم النموذج أو عنوان الخدمة غير صحيح.";

278:         else if (code == 400) m = "لم تقبل الخدمة الطلب. تحقق من اسم النموذج.";
