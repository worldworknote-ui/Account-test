252:     private static String sign(byte[] data) throws Exception {
253:         KeyStore ks = KeyStore.getInstance("AndroidKeyStore");
254:         ks.load(null);
255:         PrivateKey k = (PrivateKey) ks.getKey(DEV, null);
256:         java.security.Signature s = java.security.Signature.getInstance("SHA256withECDSA");
257:         s.initSign(k);
258:         s.update(data);
259:         return Base64.getUrlEncoder().withoutPadding().encodeToString(s.sign());

302:     /** بصمة شهادة التوقيع تطابق شهادة المطوّر؟ نسخة معاد توقيعها تختلف. يرمي إن تعذّر الفحص. */
303:     static boolean signatureMatchesOrThrow(Context ctx) throws Exception {
304:         {
305:             PackageInfo pi = ctx.getPackageManager().getPackageInfo(ctx.getPackageName(), PackageManager.GET_SIGNING_CERTIFICATES);
306:             Signature[] sigs = pi.signingInfo.hasMultipleSigners() ? pi.signingInfo.getApkContentsSigners() : pi.signingInfo.getSigningCertificateHistory();
307:             if (sigs == null || sigs.length == 0) return false;
308:             for (Signature s : sigs) {
309:                 String h = BackupFormat.hex(MessageDigest.getInstance("SHA-256").digest(s.toByteArray())).toUpperCase(Locale.US);
310:                 if (h.equals(BuildInfo.SIGNING_CERT_SHA256)) return true;
311:             }
312:             return false;
313:         }
314:     }
