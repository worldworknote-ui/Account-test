43:     private void refresh() throws IOException {
44:         Uri children = DocumentsContract.buildChildDocumentsUriUsingTree(tree, treeDoc);
45:         boolean wait = !waited && android.os.Looper.myLooper() != android.os.Looper.getMainLooper();
46:         waited = true;
47:         if (wait) {
48:             try {
49:                 // يطلب من المزوّد جلب أحدث قائمة من السحابة (يتجاهله من لا يدعمه)
50:                 cr.refresh(children, null, null);
51:             } catch (Throwable ignored) {
52:                 // غير مدعوم
53:             }
54:         }
55:         for (int round = 0; ; round++) {
56:             Map<String, String[]> found = new HashMap<>();
57:             boolean loading;
58:             final java.util.concurrent.CountDownLatch changed = new java.util.concurrent.CountDownLatch(1);
59:             android.database.ContentObserver obs = null;
60:             Uri notify = null;
61:             try (Cursor c = cr.query(children, new String[]{DocumentsContract.Document.COLUMN_DOCUMENT_ID, DocumentsContract.Document.COLUMN_DISPLAY_NAME,
62:                     DocumentsContract.Document.COLUMN_LAST_MODIFIED, DocumentsContract.Document.COLUMN_SIZE}, null, null, null)) {
63:                 if (c == null) throw new IOException("مجلد المزامنة غير متاح. اختره من جديد من «المزامنة».");
64:                 while (c.moveToNext()) {
65:                     String n = c.getString(1);
66:                     if (n == null) continue;
67:                     long m = c.isNull(2) ? 0 : c.getLong(2);
68:                     String[] prev = found.get(n);
69:                     // أسماء مكررة (Google Drive يسمح بها): الأحدث يفوز
70:                     long sz = c.isNull(3) ? 0 : c.getLong(3);
71:                     if (prev == null || Long.parseLong(prev[1]) < m) found.put(n, new String[]{c.getString(0), "" + m, "" + sz});
72:                 }
73:                 android.os.Bundle ex = c.getExtras();
74:                 loading = ex != null && ex.getBoolean(DocumentsContract.EXTRA_LOADING, false);
75:                 if (loading && wait && round < 4) notify = c.getNotificationUri();
76:             } catch (SecurityException e) {
77:                 throw new IOException("انتهت صلاحية الوصول إلى مجلد المزامنة. اختره من جديد من «المزامنة».");
78:             }
79:             ids.clear();
80:             info.clear();
81:             for (Map.Entry<String, String[]> e : found.entrySet()) {
82:                 ids.put(e.getKey(), e.getValue()[0]);
83:                 info.put(e.getKey(), new long[]{Long.parseLong(e.getValue()[2]), Long.parseLong(e.getValue()[1])});
84:             }
85:             listedOnce = true;
86:             visible = ids.size();
87:             stillLoading = loading;
88:             if (!loading || !wait || round >= 4) return;
89:             // المزوّد ما زال يجلب القائمة: ننتظر إشعاره (أو 6 ثوانٍ) ثم نقرأ من جديد
90:             try {
91:                 if (notify != null) {
92:                     obs = new android.database.ContentObserver(null) {
93:                         @Override public void onChange(boolean self) { changed.countDown(); }
94:                     };
95:                     cr.registerContentObserver(notify, true, obs);
96:                 }
97:                 changed.await(6, java.util.concurrent.TimeUnit.SECONDS);
98:             } catch (InterruptedException ie) {
99:                 Thread.currentThread().interrupt();
100:                 return;
101:             } catch (Throwable ignored) {
102:                 // نكتفي بالانتظار
103:             } finally {
104:                 if (obs != null) cr.unregisterContentObserver(obs);
105:             }
106:         }
107:     }
