806:     private static List<JSONObject> readLines(InputStream in, File spillDir) throws IOException, JSONException {
807:         BufferedReader br = new BufferedReader(new InputStreamReader(new NonClosing(in), StandardCharsets.UTF_8));
808:         List<String> pending = new ArrayList<>();
809:         long chars = 0;

816:                 spill.append(line);

818:             }

839:     /**
840:      * صفوف جدول كبير في ملف مؤقت مشفّر (AES-CTR بمفتاح عابر في الذاكرة فقط، فلا يُقرأ الملف بعد انتهاء العملية)،
841:      * تُقرأ بالتتابع عند كل مرور. للقراءة فقط: get(i) يمر من البداية (للاختبارات)، والتعديل غير مدعوم.
842:      */
843:     public static final class SpilledRows extends java.util.AbstractList<JSONObject> {
844:         private final File file;
845:         private final javax.crypto.SecretKey key;
846:         private final byte[] iv = new byte[16];
847:         private java.io.Writer w;
848:         private int size;
849: 
850:         SpilledRows(File dir) throws IOException {
851:             if (!dir.exists() && !dir.mkdirs()) throw new IOException("تعذر إنشاء مجلد مؤقت للقراءة");
852:             file = new File(dir, "rows-" + UUID.randomUUID() + ".tmp");
853:             try {
854:                 javax.crypto.KeyGenerator kg = javax.crypto.KeyGenerator.getInstance("AES");
855:                 kg.init(256);
856:                 key = kg.generateKey();
857:                 new java.security.SecureRandom().nextBytes(iv);
858:                 javax.crypto.Cipher c = javax.crypto.Cipher.getInstance("AES/CTR/NoPadding");
859:                 c.init(javax.crypto.Cipher.ENCRYPT_MODE, key, new javax.crypto.spec.IvParameterSpec(iv));
860:                 w = new java.io.BufferedWriter(new java.io.OutputStreamWriter(
861:                         new javax.crypto.CipherOutputStream(new java.io.FileOutputStream(file), c), StandardCharsets.UTF_8), 65536);
862:             } catch (java.security.GeneralSecurityException e) {
863:                 throw new IOException(e);
864:             }
865:         }
866: 
867:         void append(String line) throws IOException {
868:             w.write(line);
869:             w.write('\n');
870:             size++;
871:         }
872: 
873:         void finish() throws IOException {
874:             w.close();
875:             w = null;
876:         }
877: 
878:         public File file() { return file; }
879: 
880:         @Override public int size() { return size; }
881: 
882:         @Override public JSONObject get(int i) {
883:             if (i < 0 || i >= size) throw new IndexOutOfBoundsException("" + i);
884:             int k = 0;
885:             for (JSONObject o : this) if (k++ == i) return o;
886:             throw new IndexOutOfBoundsException("" + i);
887:         }
888: 
889:         @Override public java.util.Iterator<JSONObject> iterator() {
890:             final BufferedReader r;
891:             try {
892:                 javax.crypto.Cipher c = javax.crypto.Cipher.getInstance("AES/CTR/NoPadding");
893:                 c.init(javax.crypto.Cipher.DECRYPT_MODE, key, new javax.crypto.spec.IvParameterSpec(iv));
894:                 r = new BufferedReader(new InputStreamReader(new javax.crypto.CipherInputStream(new java.io.FileInputStream(file), c), StandardCharsets.UTF_8), 65536);
895:             } catch (Exception e) {
896:                 throw new IllegalStateException("تعذرت قراءة الملف المؤقت للنسخة: " + e.getMessage(), e);
897:             }
898:             return new java.util.Iterator<JSONObject>() {
899:                 int read;
900:                 String next = advance();
901: 
902:                 private String advance() {
903:                     try {
904:                         String l;
905:                         while ((l = r.readLine()) != null) if (!l.trim().isEmpty()) return l;
906:                         r.close();
907:                         return null;
908:                     } catch (IOException e) {
909:                         throw new IllegalStateException("تعذرت قراءة الملف المؤقت للنسخة: " + e.getMessage(), e);
910:                     }
911:                 }
912: 
913:                 @Override public boolean hasNext() { return next != null && read < size; }
914: 
915:                 @Override public JSONObject next() {
916:                     if (!hasNext()) throw new java.util.NoSuchElementException();
917:                     String l = next;
918:                     read++;
919:                     next = read < size ? advance() : null;
920:                     try {
921:                         return new JSONObject(l);
922:                     } catch (JSONException e) {
923:                         throw new IllegalStateException(e);
924:                     }
925:                 }
926:             };
927:         }
928:     }
