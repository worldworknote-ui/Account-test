61:     /** يُستدعى عند الإقلاع بعد فتح قاعدة البيانات. لا يرمي أبدًا. */
62:     public static synchronized void load(Context ctx, Db db) {
63:         try {
64:             int sig = signatureState(ctx);
65:             signatureOk = sig != SIG_MISMATCH;
66:             signatureUnknown = sig == SIG_UNKNOWN;
67:             if (signatureUnknown) {
68:                 try { db.audit("LICENSE_SIG_UNKNOWN", "license", 0, "تعذّر التحقق من توقيع التطبيق عند الإقلاع"); } catch (Throwable ignored) { /* سجل */ }
69:             }
70:             signals = environmentSignals(ctx);
71:             String token = storedToken(ctx, db);
72:             if (token.isEmpty()) {
73:                 current = Entitlements.free("");
74:                 return;
75:             }
76:             byte[] dk = deviceKeyX509();
77:             LicenseToken.Grant g = LicenseToken.verify(token, dk);
78:             if (g == null) {
79:                 current = Entitlements.free(dk == null
80:                         ? "رمز الترخيص المحفوظ يخص هاتفًا آخر أو مفتاح الجهاز غير موجود هنا (مثلًا بعد نقل البيانات إلى هاتف جديد). اطلب نقل الرخصة."
81:                         : "رمز الترخيص المحفوظ لا يطابق مفتاح هذا الهاتف. اطلب رمزًا جديدًا من البائع.");
82:                 return;
83:             }
84:             if (revoked(db, g.lid())) {
85:                 current = Entitlements.free("أُلغيت الرخصة " + g.lid() + " (مسرّبة أو منقولة). تواصل مع البائع.");
86:                 return;
87:             }
88:             if (!signatureOk) {
89:                 current = Entitlements.free("هذه نسخة من التطبيق ليست موقّعة من مطوّره (معدَّلة أو من مصدر مجهول)؛ الرخصة لا تعمل عليها. "
90:                         + "ثبّت النسخة الأصلية؛ بياناتك تنتقل بنسخة احتياطية.");
91:                 return;
92:             }
93:             current = LicenseGuard.decide(ctx, db, g);   // 12.7: المكتب والوقت الموثوق
94:         } catch (Throwable t) {
95:             current = Entitlements.free("تعذر التحقق من الترخيص الآن (" + t.getClass().getSimpleName() + "). أعد فتح التطبيق؛ الإضافة متوقفة عند حدود المجاني فقط.");
96:         }

100:     static String storedToken(Context ctx, Db db) {
101:         String a = db.setting("license_token", "").trim();
102:         String b = readFile(ctx).trim();
103:         if (a.equals(b)) return a;
104:         byte[] dk = null;
105:         try {
106:             dk = deviceKeyX509();
107:         } catch (Throwable ignored) {
108:             // يُعامل كغير صالح
109:         }
110:         boolean aOk = !a.isEmpty() && LicenseToken.verify(a, dk) != null;
111:         boolean bOk = !b.isEmpty() && LicenseToken.verify(b, dk) != null;
112:         String keep = aOk ? a : bOk ? b : (a.isEmpty() ? b : a);
113:         save(ctx, db, keep);
114:         return keep;
115:     }

156:     /** يُنشأ مرة واحدة عند أول طلب؛ التحدّي = رقم الطلب، فيرتبط المفتاح المصدَّق ببيعة واحدة. */
157:     private static void createDeviceKey(byte[] challenge) throws Exception {
158:         KeyPairGenerator g = KeyPairGenerator.getInstance(KeyProperties.KEY_ALGORITHM_EC, "AndroidKeyStore");
159:         for (boolean strong : new boolean[]{true, false}) {
160:             try {
161:                 KeyGenParameterSpec.Builder b = new KeyGenParameterSpec.Builder(DEV, KeyProperties.PURPOSE_SIGN)
162:                         .setAlgorithmParameterSpec(new ECGenParameterSpec("secp256r1"))
163:                         .setDigests(KeyProperties.DIGEST_SHA256)
164:                         .setAttestationChallenge(challenge);
165:                 if (strong) b.setIsStrongBoxBacked(true);
166:                 g.initialize(b.build());
167:                 g.generateKeyPair();
168:                 return;
169:             } catch (StrongBoxUnavailableException e) {
170:                 // لا StrongBox في هذا الهاتف: العتاد الآمن العادي (TEE)
171:             }
172:         }
173:         throw new IllegalStateException("تعذر إنشاء مفتاح الجهاز");
174:     }
