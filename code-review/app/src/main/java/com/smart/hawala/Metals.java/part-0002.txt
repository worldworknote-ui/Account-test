313:     /** الوزن ووحدته المذكوران في النص («2.5 كيلو»، «غرام 50»)، أو null. */
314:     public static String[] weightIn(String text) {
315:         Matcher m = NUM_UNIT.matcher(norm(text));
316:         while (m.find()) {
317:             String num = m.group(1) != null ? m.group(1) : m.group(4);
318:             String w = m.group(2) != null ? m.group(2) : m.group(3);
319:             String u = UNIT_WORDS.get(w.toLowerCase(Locale.ROOT));
320:             if (u == null || num == null) continue;
321:             // رقم العيار ليس وزنًا («21 غ» بعد «عيار» نادر؛ «عيار 21 غرام 50»: الوزن 50)
322:             int at = m.start();
323:             String before = norm(text).substring(Math.max(0, at - 7), at);
324:             if (before.matches("(?s).*(?:عيار|العيار)\\s*$") && m.group(1) != null) continue;
325:             return new String[]{num.replace(',', '.'), u};
326:         }
327:         return null;
328:     }

334:     public static void apply(MessageParser.Result r, String text, List<MessageParser.Cur> curs) {
335:         if (r == null || text == null || curs == null) return;
336:         List<Metal> mine = new ArrayList<>();
337:         for (MessageParser.Cur c : curs) {
338:             Metal m = of(c.code);
339:             if (m != null) mine.add(m);
340:         }
341:         if (mine.isEmpty()) return;
342:         String base = baseIn(text);
343:         Metal picked = r.currency == null ? null : of(r.currency);
344:         if (base == null && picked == null) return;
345:         if (base == null) base = picked.base;
346:         int fin = finenessIn(text);
347:         List<Metal> same = new ArrayList<>();
348:         for (Metal m : mine) if (m.base.equals(base)) same.add(m);
349:         Metal chosen = null;
350:         if (fin > 0) {
351:             for (Metal m : same) if (m.fineness == fin) chosen = m;
352:         } else if (same.size() == 1) chosen = same.get(0);
353:         String[] w = weightIn(text);
354:         if (chosen == null) {
355:             if (!same.isEmpty()) {
356:                 // صرامة: لا نخمّن العيار
357:                 if (picked != null && picked.base.equals(base)) r.currency = null;
358:                 if (r.currency != null && of(r.currency) != null) r.currency = null;
359:                 r.amountNote = (r.amountNote.isEmpty() ? "" : r.amountNote + " — ") + (fin > 0
360:                         ? "لا يوجد حساب " + baseName(base) + " " + purityLabel(base, fin) + " لهذا المكتب: اختر الحساب بيدك"
361:                         : "لم يُذكر عيار " + baseName(base) + ": اختر الحساب بيدك");

363:             }
364:             if (w != null && r.amount.isEmpty()) r.amount = w[0];
365:             return;
366:         }
367:         r.currency = chosen.code;
368:         r.currencyWord = baseName(base);
