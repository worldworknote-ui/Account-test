344:      * يرجع null إذا لم يكن الرقم صالحًا.
345:      */
346:     public static String whatsappNumber(String phone, String countryCode) {
347:         if (phone == null) return null;
348:         StringBuilder d = new StringBuilder();
349:         String t = phone.trim();
350:         for (char ch : t.toCharArray()) {
351:             if (ch >= '0' && ch <= '9') d.append(ch);
352:             else if (ch >= 0x0660 && ch <= 0x0669) d.append((char) ('0' + ch - 0x0660));
353:             else if (ch >= 0x06F0 && ch <= 0x06F9) d.append((char) ('0' + ch - 0x06F0));
354:         }
355:         String n = d.toString();
356:         if (n.isEmpty()) return null;
357:         boolean intl = t.startsWith("+") || n.startsWith("00");
358:         if (n.startsWith("00")) n = n.substring(2);
359:         else if (!intl && n.startsWith("0")) {
360:             String cc = countryCode == null ? "" : countryCode.replaceAll("[^0-9]", "");
361:             if (cc.isEmpty()) return null;
362:             n = cc + n.substring(1);
363:         }
364:         return n.length() >= 8 && n.length() <= 15 ? n : null;
365:     }
