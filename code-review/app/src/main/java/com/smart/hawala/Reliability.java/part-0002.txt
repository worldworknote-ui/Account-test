225:         if (!a.require(Auth.Perm.BACKUP)) return;
226:         List<File> files = BackupManager.localBackups(a);
227:         if (files.isEmpty()) {

261:     private static void runDrill(final MainActivity a, final File f, final char[] pw, final boolean triedAuto) {

275:                     : "fail|" + r.message;

281:                             + r.attachments + " صورة سليمة. بياناتك لم تتغير." + (r.newerTx > 0 ? "\n\nسُجّلت بعدها " + r.newerTx + " حركة: أنشئ نسخة جديدة." : "")
