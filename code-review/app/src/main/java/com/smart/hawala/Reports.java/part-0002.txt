233:     public static Report balances(List<Model.OfficeTotals> offices, Map<String, Integer> decimals, Map<String, String> names,
234:                                   boolean includeZero, String issuer, long generatedAt) {
235:         Report r = new Report();
236:         r.title = "أرصدة كل المكاتب";
237:         if (issuer != null && !issuer.trim().isEmpty()) r.headerLines.add("صادر عن: " + issuer.trim());
238:         r.headerLines.add("تاريخ الإصدار: " + Model.date(generatedAt));
239:         r.fileBase = "أرصدة-المكاتب";
240: 
241:         Map<Long, Map<String, Long>> net = new LinkedHashMap<>();
242:         for (Model.OfficeTotals o : offices) {
243:             Map<String, Long> m = new TreeMap<>();
244:             for (Map.Entry<String, long[]> e : o.perCurrency.entrySet()) m.put(e.getKey(), e.getValue()[2]);
245:             net.put(o.officeId, m);
246:         }
247:         Map<String, Ledger.CurrencyTotal> totals = Ledger.totals(net);
248:         Report.Section g = r.section("الإجمالي حسب العملة (بدون تقاصّ بين المكاتب)", "الإجمالي");
249:         g.col("العملة", 1.4f, Report.TEXT).col("مجموع ما لنا على المكاتب", 1.6f, Report.MONEY).col("مجموع ما علينا للمكاتب", 1.6f, Report.MONEY)
250:                 .col("الصافي", 1.5f, Report.BAL).col("مكاتب لنا", 0.9f, Report.INT).col("مكاتب علينا", 0.9f, Report.INT);
251:         for (Ledger.CurrencyTotal t : totals.values()) {
252:             if (t.receivable == 0 && t.payable == 0) continue;
253:             int d = dec(decimals, t.code);
254:             g.row(Report.NORMAL, t.code + (names.get(t.code) == null ? "" : " — " + names.get(t.code)), new Report.Money(t.receivable, d),
255:                     new Report.Money(t.payable, d), new Report.Bal(t.net(), d), (long) t.officesReceivable, (long) t.officesPayable);
256:         }
257:         if (g.rows.isEmpty()) g.note = "لا توجد أرصدة.";
258: 
259:         Report.Section s = r.section("رصيد كل مكتب: كل ما لنا، كل ما علينا، ثم الرصيد", "المكاتب");
260:         s.col("المكتب", 2.2f, Report.TEXT).col("العملة", 0.9f, Report.TEXT).col("كل ما لنا", 1.4f, Report.MONEY)
261:                 .col("كل ما علينا", 1.4f, Report.MONEY).col("الرصيد", 1.6f, Report.BAL);
262:         boolean shade = false;
263:         for (Model.OfficeTotals o : offices) {
264:             boolean first = true;
265:             for (Map.Entry<String, long[]> e : o.perCurrency.entrySet()) {
266:                 long[] v = e.getValue();
267:                 if (!includeZero && v[0] == 0 && v[1] == 0 && v[2] == 0) continue;
268:                 int d = dec(decimals, e.getKey());
269:                 s.row(shade ? Report.SHADED : Report.NORMAL, first ? o.name + (o.active ? "" : " (موقوف)") : "", e.getKey(),
270:                         new Report.Money(v[0], d), new Report.Money(v[1], d), new Report.Bal(v[2], d));
271:                 first = false;
272:             }
273:             if (!first) shade = !shade;
274:         }
275:         if (s.rows.isEmpty()) s.note = "لا توجد مكاتب بأرصدة.";
276:         return r;
277:     }
