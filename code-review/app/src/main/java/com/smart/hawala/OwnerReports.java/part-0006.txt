819:     public static Map<String, Size> sizes(Input in, long from) {
820:         Map<String, List<Long>> m = new TreeMap<>();
821:         for (Model.Tx t : in.txs) if (active(t) && t.createdAt >= from && !feeRow(t)) m.computeIfAbsent(t.currency, k -> new ArrayList<>()).add(t.amount);
822:         Map<String, Size> out = new TreeMap<>();
823:         for (Map.Entry<String, List<Long>> e : m.entrySet()) {
824:             List<Long> l = e.getValue();
825:             Collections.sort(l);
826:             Size s = new Size();
827:             s.count = l.size();
828:             int n = l.size();
829:             s.median = n % 2 == 1 ? l.get(n / 2) : (l.get(n / 2 - 1) + l.get(n / 2)) / 2;
830:             s.p90 = l.get(Math.min(n - 1, (int) Math.ceil(n * 0.9) - 1));
831:             s.p99 = l.get(Math.min(n - 1, (int) Math.ceil(n * 0.99) - 1));
832:             s.max = l.get(n - 1);
833:             int top = Math.max(1, (int) Math.ceil(n * 0.01));
834:             for (int i = 0; i < n; i++) {
835:                 s.volume += l.get(i);
836:                 if (i >= n - top) s.top1Volume += l.get(i);
837:             }
838:             out.put(e.getKey(), s);
839:         }
840:         return out;
841:     }

877:     public static final class Quality {
878:         public final List<long[]> bulkDays = new ArrayList<>();   // {بداية اليوم، العمليات}
879:         public final List<String[]> similarNames = new ArrayList<>();
880:         public long distinctDescriptions, numericNames;
881:     }
882: 
883:     public static Quality quality(Input in) {
884:         Quality q = new Quality();
885:         Map<String, long[]> perDay = new TreeMap<>();
886:         Set<String> desc = new HashSet<>();
887:         for (Model.Tx t : in.txs) {
888:             if (!active(t)) continue;
889:             perDay.computeIfAbsent(Model.day(t.createdAt), k -> new long[]{t.createdAt, 0})[1]++;
890:             String d = norm(t.description);
891:             if (!d.isEmpty()) desc.add(d);
892:         }
893:         q.distinctDescriptions = desc.size();
894:         List<Long> counts = new ArrayList<>();
895:         for (long[] v : perDay.values()) counts.add(v[1]);
896:         Collections.sort(counts);
897:         long median = counts.isEmpty() ? 0 : counts.get(counts.size() / 2);
898:         for (long[] v : perDay.values()) if (v[1] >= Math.max(100, median * 10)) q.bulkDays.add(new long[]{v[0], v[1]});
899:         Collections.sort(q.bulkDays, (a, b) -> Long.compare(b[1], a[1]));
900:         List<Map.Entry<Long, String>> names = new ArrayList<>(in.names.entrySet());
901:         for (Map.Entry<Long, String> e : names) if (norm(e.getValue()).matches("[0-9٠-٩ ]+")) q.numericNames++;
902:         for (int i = 0; i < names.size(); i++) {
903:             String a = norm(names.get(i).getValue());
904:             if (a.length() < 3) continue;
905:             for (int j = 0; j < names.size(); j++) {
906:                 if (i == j) continue;
907:                 String b = norm(names.get(j).getValue());
908:                 if (b.length() > a.length() && (b.startsWith(a + " ") || b.endsWith(" " + a))) q.similarNames.add(new String[]{names.get(i).getValue(), names.get(j).getValue()});
909:             }
910:         }
911:         return q;
912:     }
