53:     static void healthIfDue(final Db db) {
54:         if (android.os.SystemClock.elapsedRealtime() < deferUntil) return;
55:         long last = db.settingLong("health_last", 0);
56:         if (Model.day(last).equals(Model.day(Clock.now()))) return;

60:     private static volatile boolean searchStarted;
61: 
62:     /** 12.7: يكمل نص البحث المحلي في الخلفية (دفعات قصيرة) فيكون البحث سريعًا من أول مرة. */
63:     static void searchIndexAsync(final Db db) {
64:         if (searchStarted) return;
65:         searchStarted = true;
66:         Thread t = new Thread(() -> {
67:             try {
68:                 Thread.sleep(8000);
69:                 db.fillSearchIndex();
70:             } catch (Throwable ignored) {
71:                 // يُكمَل عند أول بحث
72:             } finally {
73:                 searchStarted = false;
74:             }
75:         }, "search-index");
76:         t.setPriority(Thread.MIN_PRIORITY);
77:         t.start();
78:     }

81:     static void healthAsync(final Db db) {
82:         Thread t = new Thread(() -> {
83:             try {
84:                 db.runHealthAuto();
85:             } catch (Throwable ignored) {
86:                 // تُعاد في المرة التالية
87:             }
88:         }, "health");
89:         t.setPriority(Thread.MIN_PRIORITY);
90:         t.start();
91:     }

116:                     autoErr.isEmpty() ? "كل " + AutoBackup.intervalHours(db) + " ساعة" + (AutoBackup.everyTx(db) > 0 ? " أو بعد " + AutoBackup.everyTx(db) + " حركة" : "")
