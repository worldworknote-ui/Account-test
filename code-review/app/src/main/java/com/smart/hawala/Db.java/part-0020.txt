2881:     /**
2882:      * معاينة أثر عملية (ساق أو أكثر) على أرصدة مكاتبها، مع طرح أثر الحركة التي تُستبدل عند التعديل (replacing).
2883:      * تُحسب من قاعدة البيانات لحظة الطلب، فتُستدعى من جديد عند عرض التأكيد وعند الحفظ.
2884:      */
2885:     public List<Preview> preview(List<TxInput> legs, Long replacing) {
2886:         Map<Long, Preview> out = new LinkedHashMap<>();
2887:         SQLiteDatabase d = getReadableDatabase();
2888:         boolean pending = false;
2889:         for (TxInput x : legs) pending |= approvalRequired(x);
2890:         Map<Long, Map<String, Long>> delta = new LinkedHashMap<>();
2891:         for (TxInput x : legs) for (Map.Entry<String, Long> e : effect(x).entrySet()) delta.computeIfAbsent(x.office, k -> new TreeMap<>()).merge(e.getKey(), e.getValue(), Long::sum);
2892:         if (replacing != null) {
2893:             Model.Tx old = tx(replacing);
2894:             if (old != null && old.isActive())
2895:                 for (Map.Entry<String, Long> e : effect(inputOf(old)).entrySet()) delta.computeIfAbsent(old.officeId, k -> new TreeMap<>()).merge(e.getKey(), -e.getValue(), Long::sum);
2896:         }
2897:         for (Map.Entry<Long, Map<String, Long>> e : delta.entrySet()) {
2898:             Preview p = new Preview();
2899:             p.office = e.getKey();
2900:             Model.Office o = office(p.office);
2901:             p.officeName = o == null ? "" : Model.nz(o.name);
2902:             p.pending = pending;
2903:             Map<String, Long> now = officeBalances(d, p.office);
2904:             for (Map.Entry<String, Long> c : e.getValue().entrySet()) {
2905:                 long b = now.containsKey(c.getKey()) ? now.get(c.getKey()) : 0;
2906:                 p.before.put(c.getKey(), b);
2907:                 p.after.put(c.getKey(), b + c.getValue());
2908:             }
2909:             out.put(p.office, p);
2910:         }
2911:         return new ArrayList<>(out.values());
2912:     }
