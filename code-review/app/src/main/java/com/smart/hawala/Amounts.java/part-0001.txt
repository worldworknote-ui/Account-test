1: package com.smart.hawala;
2: 
3: import java.math.BigDecimal;
4: import java.math.RoundingMode;
5: 
6: /**
7:  * قراءة واحدة للمبالغ في كل التطبيق (نموذج الإدخال، رسائل واتساب، المرشّحات، الحدود).
8:  * قبل 8.0 كان «1,500» يُقرأ 1.5 في نموذج الإدخال و1500 في محلل الرسائل: خطأ بألف ضعف في حركة مالية.
9:  * القاعدة الآن تعتمد منازل العملة:
10:  *  - فاصل واحد يليه ثلاثة أرقام بالضبط = فاصل آلاف (1,500 = ألف وخمسمئة)، إلا في عملة بثلاث منازل فيُرفض بدل التخمين؛
11:  *  - فاصل متكرر = آلاف حصرًا (1,250,000)؛ فاصلان مختلفان = الأخير عشري (1.250,50 و1,250.50)؛
12:  *  - غير ذلك عشري (12,5 و12.50 و0,500).
13:  * منطق بحت بلا Android حتى يُختبر على الكمبيوتر.
14:  */
15: public final class Amounts {
16:     private Amounts() {}
17: 
18:     private static boolean digits(String p) {
19:         if (p.isEmpty()) return false;
20:         for (int i = 0; i < p.length(); i++) if (p.charAt(i) < '0' || p.charAt(i) > '9') return false;
21:         return true;
22:     }
23: 
24:     /** آلاف صحيحة: مجموعة أولى من 1-3 أرقام ثم مجموعات كل منها 3 أرقام يفصل بينها sep. */
25:     private static boolean grouped(String s, char sep) {
26:         String[] g = s.split(java.util.regex.Pattern.quote(String.valueOf(sep)), -1);
27:         if (!digits(g[0]) || g[0].length() > 3) return false;
28:         for (int i = 1; i < g.length; i++) if (!digits(g[i]) || g[i].length() != 3) return false;
29:         return true;
30:     }
31: 
32:     /** رقم قياسي (نقطة عشرية وبلا فواصل آلاف)، ويرفض الملتبس بدل أن يخمّن. «» = فارغ. */
33:     public static String normalize(String t, int dec) {
34:         if (t == null) return "";
35:         StringBuilder sb = new StringBuilder();
36:         for (char ch : t.trim().toCharArray()) {
37:             if (ch >= 0x0660 && ch <= 0x0669) sb.append((char) ('0' + (ch - 0x0660)));
38:             else if (ch >= 0x06F0 && ch <= 0x06F9) sb.append((char) ('0' + (ch - 0x06F0)));
39:             else if (ch == 0x066B) sb.append('.');   // ٫ عشرية عربية
40:             else if (ch == 0x060C) sb.append(',');   // ، فاصلة عربية
41:             else if (ch == 0x066C || ch == ' ' || ch == 0x00A0 || ch == 0x202F || ch == '\'' || ch == 0x2019 || ch == 0x200F || ch == 0x200E) continue;
42:             else sb.append(ch);
43:         }
44:         String s = sb.toString(), sign = "";
45:         if (s.startsWith("-") || s.startsWith("+")) {
46:             sign = s.startsWith("-") ? "-" : "";
47:             s = s.substring(1);
48:         }
49:         if (s.isEmpty()) return "";
50:         for (int i = 0; i < s.length(); i++) {
51:             char c = s.charAt(i);
52:             if ((c < '0' || c > '9') && c != '.' && c != ',') throw new NumberFormatException("المبلغ يحتوي محارف غير مسموحة");
53:         }
54:         if (s.indexOf('.') < 0 && s.indexOf(',') < 0) return sign + s;
55: 
56:         int last = Math.max(s.lastIndexOf('.'), s.lastIndexOf(','));
57:         char sep = s.charAt(last), other = sep == '.' ? ',' : '.';
58:         String head = s.substring(0, last), frac = s.substring(last + 1);
59:         if (head.isEmpty() && frac.isEmpty()) throw new NumberFormatException("المبلغ غير مفهوم");
60:         boolean hasOther = head.indexOf(other) >= 0, repeated = head.indexOf(sep) >= 0;
61: 
62:         boolean decimal;
63:         if (hasOther) decimal = true;              // 1.250,50 أو 1,250.50: الأخير عشري والآخر آلاف
64:         else if (repeated) decimal = false;        // 1,250,000: آلاف حصرًا
65:         else if (frac.length() == 3 && !head.isEmpty() && !head.equals("0")) {
66:             if (dec == 3) throw new NumberFormatException("المبلغ ملتبس: اكتبه بلا فواصل أو حدّد الكسور صراحة");
67:             decimal = false;                       // 1,500 بعملة منزلتيها 2 أو 0 = ألف وخمسمئة
68:         } else decimal = true;                     // 12,5 أو 12.50 أو 0,500
69: 
70:         if (decimal) {
71:             if (!frac.isEmpty() && !digits(frac)) throw new NumberFormatException("تنسيق المبلغ غير مفهوم");
72:             if (head.isEmpty()) head = "0";
73:             if (!(hasOther ? grouped(head, other) : digits(head))) throw new NumberFormatException("تنسيق المبلغ غير مفهوم");
74:             return sign + head.replace(String.valueOf(other), "") + (frac.isEmpty() ? "" : "." + frac);
75:         }
76:         if (!grouped(s, sep)) throw new NumberFormatException("تنسيق الآلاف غير مفهوم");
77:         return sign + s.replace(String.valueOf(sep), "");
78:     }
79: 
80:     /** المبلغ بالوحدة الصغرى؛ كسور أكثر من منازل العملة تُرفض (لا تقريب أبدًا). */
81:     public static long minor(String text, int dec) {
82:         String n = normalize(text, dec);
83:         if (n.isEmpty()) throw new NumberFormatException("empty");
84:         return new BigDecimal(n).movePointRight(dec).setScale(0, RoundingMode.UNNECESSARY).longValueExact();
85:     }
86: 
87:     /** رقم عشري عام (سعر صرف، نسبة) بقواعد منزلتين: يرجع null للفارغ. */
88:     public static BigDecimal decimal(String text) {
89:         String n = normalize(text, 2);
90:         return n.isEmpty() ? null : new BigDecimal(n);
91:     }
92: }
