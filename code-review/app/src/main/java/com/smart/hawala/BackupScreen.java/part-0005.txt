603:     void passwordDialog(String title, String message, String okLabel, final PwCb cb) {
604:         LinearLayout l = new LinearLayout(a);
605:         l.setOrientation(LinearLayout.VERTICAL);
606:         l.setPadding(a.dp(18), a.dp(8), a.dp(18), a.dp(4));

609:         final EditText p2 = new EditText(a);
610:         p2.setHint("أعد كتابة كلمة المرور للتأكيد");
611:         for (EditText e : new EditText[]{p1, p2}) {
612:             Ui.secret(e, false);
613:             l.addView(e);
614:         }
615:         final CheckBox showPw = new CheckBox(a);
616:         showPw.setText("إظهار كلمة المرور");
617:         showPw.setOnCheckedChangeListener((b, c) -> {
618:             for (EditText e : new EditText[]{p1, p2}) {
619:                 e.setTransformationMethod(c ? HideReturnsTransformationMethod.getInstance() : PasswordTransformationMethod.getInstance());
620:                 e.setSelection(e.getText().length());
621:             }
622:         });
623:         l.addView(showPw);
624:         final TextView hint = a.text("", 13, a.muted, false);
625:         l.addView(hint);

631:                 hint.setTextColor(a.red);
632:             } else if (p2.getText().length() > 0 && !p1.getText().toString().equals(p2.getText().toString())) {
633:                 hint.setText("الكلمتان غير متطابقتين");
634:                 hint.setTextColor(a.red);
635:             } else if (p2.getText().length() > 0) {
636:                 hint.setText("✓ الكلمتان متطابقتان");
637:                 hint.setTextColor(a.primary);
638:             } else {
639:                 hint.setText("أعد كتابتها في الحقل الثاني");
640:                 hint.setTextColor(a.muted);
641:             }
642:         };
643:         p1.addTextChangedListener(Ui.watcher(check));
644:         p2.addTextChangedListener(Ui.watcher(check));
645:         final AlertDialog dlg = new AlertDialog.Builder(a).setTitle(title).setMessage(message)
646:                 .setView(l).setPositiveButton(okLabel, null).setNegativeButton("إلغاء", null).create();
647:         dlg.setOnShowListener(d -> dlg.getButton(AlertDialog.BUTTON_POSITIVE).setOnClickListener(v -> {
648:             Editable e1 = p1.getText(), e2 = p2.getText();

651:                 hint.setTextColor(a.red);
652:                 return;
653:             }
654:             if (!e1.toString().equals(e2.toString())) {

656:                 hint.setTextColor(a.red);

662:             p2.setText("");

664:             cb.on(pw);
665:         }));

667:     }
