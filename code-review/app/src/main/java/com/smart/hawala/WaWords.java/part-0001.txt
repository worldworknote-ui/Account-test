77:     public static boolean systemText(String t) {
78:         if (t == null) return true;
79:         String s = clean(t);
80:         if (s.isEmpty()) return true;
81:         // كل الأسطر تنبيهات (فاصل يوم فوق «رسائل غير مقروءة» مثلًا)
82:         for (String line : s.split("\n")) {
83:             String l = line.trim();
84:             if (l.isEmpty()) continue;
85:             if (!systemLine(l)) return false;
86:         }
87:         return true;
88:     }

98:     static boolean systemLine(String l) {
99:         String s = clean(l);
100:         if (s.isEmpty() || DATE_PILL.matcher(s).matches() || WaCuts.timeLine(s)) return true;
101:         for (Pattern p : SYSTEM) if (p.matcher(s).matches()) return true;
102:         if (s.length() <= 120 && !PHONES.matcher(s).replaceAll(" ").matches(".*\\d.*") && MEMBERS.matcher(s).matches()) return true;
103:         return false;
