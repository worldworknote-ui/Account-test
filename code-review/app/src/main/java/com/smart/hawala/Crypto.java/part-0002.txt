196:     public static byte[] encBytes(byte[] raw) {
197:         try {
198:             Cipher c = Cipher.getInstance("AES/GCM/NoPadding");
199:             c.init(Cipher.ENCRYPT_MODE, key());
200:             byte[] b = c.doFinal(raw);
201:             byte[] iv = c.getIV();
202:             byte[] o = new byte[1 + iv.length + b.length];
203:             o[0] = (byte) iv.length;
204:             System.arraycopy(iv, 0, o, 1, iv.length);
205:             System.arraycopy(b, 0, o, 1 + iv.length, b.length);
206:             return o;
207:         } catch (Exception e) {
208:             throw new IllegalStateException("تعذر تشفير الصورة على هذا الهاتف", e);
209:         }
210:     }

212:     public static byte[] decBytes(byte[] in) {
213:         try {
214:             int n = in[0] & 255;
215:             byte[] iv = new byte[n];
216:             System.arraycopy(in, 1, iv, 0, n);
217:             Cipher c = Cipher.getInstance("AES/GCM/NoPadding");
218:             c.init(Cipher.DECRYPT_MODE, key(), new GCMParameterSpec(128, iv));
219:             return c.doFinal(in, 1 + n, in.length - 1 - n);
220:         } catch (Exception e) {
221:             throw new IllegalStateException("تعذر فك تشفير الصورة", e);
222:         }
223:     }
