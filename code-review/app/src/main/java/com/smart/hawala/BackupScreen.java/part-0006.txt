785:     private void mergePreview(final File file, final boolean temporary, final BackupManager.Prepared prep, final MergePlan plan) {
786:         final Runnable cancel = () -> {
787:             BackupManager.clearStaging(a);
788:             if (temporary) file.delete();
789:             a.restoreInput = null;
790:             show();
791:         };
792:         a.base("إضافة ملف النقل", cancel);
793:         LinearLayout c = a.card();
794:         c.addView(a.text("تُضاف أرصدة الملف وحركاته إلى بياناتك الحالية. لا يُحذف شيء.", 15, a.ink, true));
795:         c.addView(a.text("المكاتب: " + plan.matchedOffices + " موجود تُضاف إليه، و " + plan.newOffices + " جديد"
796:                 + "\nالحركات: " + plan.txs.size() + "   •   الأرصدة السابقة: " + plan.openings.size()
797:                 + (plan.newCurrencies.isEmpty() ? "" : "\nعملات جديدة: " + newCodes(plan)), 14, a.dark, false));
798:         for (String e : plan.errors) a.notice(R.drawable.ic_warn, e, true, null, null);
799:         for (String w : plan.warnings) a.notice(R.drawable.ic_warn, w, false, null, null);

803:         final boolean over = maxOff != Entitlements.UNLIMITED && live + plan.newOffices > maxOff;
804:         if (over) {
805:             a.notice(R.drawable.ic_warn, "الملف يضيف " + plan.newOffices + " مكتبًا جديدًا وعندك " + live + "، والنسخة المجانية تتسع لـ " + maxOff
806:                     + ". اختر المكاتب الجديدة التي تُضاف (حتى " + Math.max(0, maxOff - live) + ")، أو فعّل النسخة الكاملة.", true, "اختيار المكاتب",
807:                     v -> pickNew(file, temporary, prep, plan, (int) Math.max(0, maxOff - live)));
808:         } else if (plan.droppedOffices > 0) {
809:             a.notice(R.drawable.ic_info, "استُبعد " + plan.droppedOffices + " مكتبًا جديدًا باختيارك؛ لا تُضاف حركاتها.", false, "إعادة الاختيار",
810:                     v -> mergePlan(file, temporary, prep));
811:         }
812:         // 10.16: اعتماد رصيد الملف كما هو بدل جمعه مع الرصيد الحالي
813:         if (plan.ok() && plan.adoptable() > 0) {
814:             final android.widget.CheckBox ad = new android.widget.CheckBox(a);
815:             ad.setText("اعتماد رصيد الملف كما هو (بدون جمعه مع الرصيد الحالي)");
816:             ad.setChecked(plan.adopt);
817:             ad.setOnCheckedChangeListener((b, on) -> {
818:                 plan.setAdopt(on);
819:                 mergePreview(file, temporary, prep, plan);
820:             });
821:             LinearLayout ac = a.card();
822:             ac.addView(ad);
823:             ac.addView(a.text(plan.adopt ? "للمكاتب الموجودة: يُلغى رصيدها الحالي بقيد تسوية ظاهر في الكشف، ثم يُكتب رصيد الملف، فيصبح الرصيد = رصيد الملف تمامًا. لا تُحذف أي حركة."
