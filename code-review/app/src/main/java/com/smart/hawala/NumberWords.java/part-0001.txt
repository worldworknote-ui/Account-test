24:     static final class W {
25:         final int kind;
26:         final long value;
27:         /** للمثنى العربي «ألفين» و«مليونين»: العدد المضمَّن (2). */
28:         final int count;
29:         final Set<String> langs;
30: 
31:         W(int kind, long value, int count, Set<String> langs) {
32:             this.kind = kind;
33:             this.value = value;
34:             this.count = count;
35:             this.langs = langs;
36:         }
37:     }
38: 
39:     private static final Map<String, W> DICT = new HashMap<>();
40:     private static final Map<String, Set<String>> CONNECTORS = new HashMap<>();
41:     /** كلمات رقمية تشبه كلمات عادية أو أسماء («bin» في Ahmad bin Ali، «on» الإنجليزية، «ست» بمعنى سيدة…): لا تكفي وحدها. */
42:     private static final Set<String> WEAK = new HashSet<>();
43: 
44:     private static void put(String lang, int kind, long value, int count, String... words) {
45:         for (String w : words) {
46:             String k = MessageParser.norm(w);
47:             W old = DICT.get(k);
48:             if (old != null && old.kind == kind && old.value == value && old.count == count) {
49:                 old.langs.add(lang);
50:                 continue;
51:             }
52:             if (old != null) continue;   // أول تعريف هو المعتمد
53:             DICT.put(k, new W(kind, value, count, new HashSet<>(Arrays.asList(lang))));
54:         }
55:     }
56: 
57:     private static void small(String lang, long v, String... w) { put(lang, SMALL, v, 0, w); }
58: 
59:     private static void conn(String lang, String... w) {
60:         for (String x : w) {
61:             String k = MessageParser.norm(x);
62:             Set<String> s = CONNECTORS.get(k);
63:             if (s == null) CONNECTORS.put(k, s = new HashSet<>());
64:             s.add(lang);
65:         }
66:     }
