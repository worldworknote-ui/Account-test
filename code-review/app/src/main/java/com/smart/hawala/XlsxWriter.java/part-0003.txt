351:     static String esc(String s) {
352:         if (s == null) return "";
353:         StringBuilder b = new StringBuilder(s.length() + 16);
354:         for (int i = 0; i < s.length(); i++) {
355:             char ch = s.charAt(i);
356:             switch (ch) {
357:                 case '&': b.append("&amp;"); break;
358:                 case '<': b.append("&lt;"); break;
359:                 case '>': b.append("&gt;"); break;
360:                 case '"': b.append("&quot;"); break;
361:                 default:
362:                     if (ch < 0x20 && ch != '\n' && ch != '\t' && ch != '\r') continue; // محارف غير مسموحة في XML
363:                     if (ch == 0xFFFE || ch == 0xFFFF) continue;
364:                     b.append(ch);
365:             }
366:         }
367:         return b.toString();
368:     }
