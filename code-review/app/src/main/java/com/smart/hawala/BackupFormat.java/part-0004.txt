559:             if (uuid == null || uuid.isEmpty()) {

561:                 break;
562:             }

566:             }
567:             Long amount = lng(row, "amount_minor");
568:             if (amount == null || amount < 0) {
569:                 r.errors.add("مبلغ غير صالح في الحركة رقم " + id);
570:                 break;

572:             String cur = str(row, "currency");

574:             if ("CANCELLED".equals(str(row, "status"))) r.txCancelled++;

576:         }

608:                 break;
609:             }

612:             balance.put(key, (b == null ? 0 : b) + debit - credit);

619:                 break;
620:             }
621:         }

623:         for (Long e : entries) if (!linesPerEntry.containsKey(e)) emptyEntries++;
624:         if (emptyEntries > 0) r.warnings.add(emptyEntries + " قيد بلا أسطر (لا يؤثر على الأرصدة)");
625:

627:             Long o = lng(row, "office_id");
628:             if (o != null && !offices.contains(o)) {
629:                 r.errors.add("رصيد سابق يشير إلى مكتب غير موجود (" + o + ")");
630:                 break;

664:                 break;
665:             }
666:             Long tx = lng(row, "tx_id");
667:             if (tx != null && !txs.contains(tx)) {
668:                 r.errors.add("حركة صندوق تشير إلى حركة مكتب غير موجودة (" + tx + ")");
669:                 break;
670:             }
671:             String u = str(row, "uuid");
672:             if (u == null || u.isEmpty() || !cashUuids.add(u)) {
673:                 r.errors.add("حركة صندوق بلا مرجع أو بمرجع مكرر (رقم " + id + ")");
674:                 break;
675:             }
676:             String cur = str(row, "currency");
677:             if (cur == null || !currencies.contains(cur)) r.warnings.add("حركة صندوق بعملة غير معرّفة: " + cur);
678:             r.cashMoves++;
679:         }

691:                 r.errors.add("حركة حساب عميل غير صالحة (رقم " + lng(row, "id") + ")");
692:                 break;
693:             }
694:             r.clientMoves++;
695:         }
696:         r.clients = clients.size();

703:             }

706:                 break;
707:             }

709:                 r.errors.add("صورة فارغة رقم " + s.id);
710:                 break;
711:             }
712:         }
713:         r.attachments = p.attachments.size();
