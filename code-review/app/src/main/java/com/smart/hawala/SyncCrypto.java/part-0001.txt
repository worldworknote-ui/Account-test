23:  * <p>القرارات:
24:  * <ul>
25:  *   <li>مفتاح الخزنة (Vault Key) عشوائي 32 بايت، لا يخرج من الأجهزة إلا مغلّفًا: بمفتاح مشتق من كلمة سر المزامنة
26:  *       (PBKDF2-HMAC-SHA256، 600 ألف دورة، ملح 16 بايت) وبرمز استرداد عشوائي. تغيير كلمة السر يعيد التغليف فقط.
27:  *       (Argon2id غير متاح مدمجًا في Java/Android 9 دون مكتبة خارجية؛ PBKDF2 بهذا العدد هو توصية OWASP البديلة.)</li>
28:  *   <li>من مفتاح الخزنة تُشتق بـ HKDF-SHA256 مفاتيح فرعية: للتشفير، ولتعمية أسماء الملفات.</li>
29:  *   <li>كل ملف: AES-256-GCM، ورقم Nonce عشوائي 96 بت لكل ملف (لا عدّاد: استعادة نسخة قديمة للجهاز قد تعيد العدّاد فيتكرر
30:  *       الرقم مع المفتاح نفسه وهو خطأ قاتل في GCM؛ العشوائي آمن حتى مليارات الملفات). البيانات الإضافية (AAD) تربط
31:  *       الملف بالخزنة ونوعه واسمه فلا يُنقل ملف مكان آخر.</li>
32:  *   <li>المحتوى يُضغط ثم يُحشى ليكون حجم الملف مضاعفًا لـ 4KB فلا يكشف الحجم عدد الحركات.</li>
33:  *   <li>كل الأجهزة أعضاء موثوقون في الخزنة نفسها (أجهزة المكتب الواحد)، فالتوثيق بـ GCM بمفتاح الخزنة؛ وسلسلة التجزئة بين
34:  *       حزم كل جهاز تكشف الحذف وإعادة الترتيب.</li>
35:  * </ul>

74:     public static byte[] hmac(byte[] key, byte[] data) {
75:         try {
76:             Mac m = Mac.getInstance("HmacSHA256");
77:             m.init(new SecretKeySpec(key, "HmacSHA256"));
78:             return m.doFinal(data);
79:         } catch (GeneralSecurityException e) {
80:             throw new IllegalStateException(e);
81:         }
82:     }

84:     /** HKDF-SHA256 (RFC 5869) بطول 32 بايت. */
85:     public static byte[] hkdf(byte[] ikm, byte[] salt, String info) {
86:         byte[] prk = hmac(salt == null ? new byte[32] : salt, ikm);
87:         byte[] in = (info + "\u0001").getBytes(StandardCharsets.UTF_8);
88:         return hmac(prk, in);

91:     public static byte[] pbkdf2(char[] password, byte[] salt, int iter) {
92:         try {
93:             PBEKeySpec spec = new PBEKeySpec(password, salt, iter, 256);
94:             try {
95:                 return SecretKeyFactory.getInstance("PBKDF2WithHmacSHA256").generateSecret(spec).getEncoded();
96:             } finally {
97:                 spec.clearPassword();
98:             }
99:         } catch (GeneralSecurityException e) {
100:             throw new IllegalStateException(e);
101:         }
102:     }

104:     /** AES-256-GCM: nonce(12) || ciphertext+tag. */
105:     public static byte[] seal(byte[] key, byte[] plain, byte[] aad) {
106:         try {
107:             byte[] nonce = random(12);
108:             Cipher c = Cipher.getInstance("AES/GCM/NoPadding");
109:             c.init(Cipher.ENCRYPT_MODE, new SecretKeySpec(key, "AES"), new GCMParameterSpec(128, nonce));
110:             if (aad != null) c.updateAAD(aad);
111:             byte[] ct = c.doFinal(plain);
112:             return ByteBuffer.allocate(12 + ct.length).put(nonce).put(ct).array();
113:         } catch (GeneralSecurityException e) {
114:             throw new IllegalStateException(e);
115:         }
116:     }

118:     /** يفك ويتحقق؛ أي تغيير ولو بايت واحد (أو مفتاح/AAD خاطئ) = استثناء. */
119:     public static byte[] open(byte[] key, byte[] sealed, byte[] aad) throws GeneralSecurityException {
120:         if (sealed == null || sealed.length < 28) throw new GeneralSecurityException("ملف قصير أو تالف");
121:         Cipher c = Cipher.getInstance("AES/GCM/NoPadding");
122:         c.init(Cipher.DECRYPT_MODE, new SecretKeySpec(key, "AES"), new GCMParameterSpec(128, sealed, 0, 12));
123:         if (aad != null) c.updateAAD(aad);
124:         return c.doFinal(sealed, 12, sealed.length - 12);
125:     }
