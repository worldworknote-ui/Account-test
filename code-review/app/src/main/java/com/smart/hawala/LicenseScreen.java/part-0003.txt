290:     void onQrImage(final android.net.Uri uri) {
291:         if (uri == null) return;
292:         a.busy("جارٍ قراءة رمز QR…", p -> {
293:             android.graphics.BitmapFactory.Options o = new android.graphics.BitmapFactory.Options();
294:             o.inJustDecodeBounds = true;
295:             try (java.io.InputStream in = a.getContentResolver().openInputStream(uri)) {
296:                 android.graphics.BitmapFactory.decodeStream(in, null, o);
297:             }
298:             int big = Math.max(o.outWidth, o.outHeight), sample = 1;
299:             while (big / sample > 2400) sample *= 2;
300:             android.graphics.BitmapFactory.Options o2 = new android.graphics.BitmapFactory.Options();
301:             o2.inSampleSize = sample;
302:             android.graphics.Bitmap bm;
303:             try (java.io.InputStream in = a.getContentResolver().openInputStream(uri)) {
304:                 bm = android.graphics.BitmapFactory.decodeStream(in, null, o2);
305:             }
306:             if (bm == null) throw new java.io.IOException("تعذر فتح الصورة");
307:             String txt = QrRead.decode(bm);
308:             return txt == null ? "" : txt;
309:         }, res -> {
310:             String txt = String.valueOf(res);
311:             if (txt.isEmpty() || LicenseToken.extract(txt).indexOf('.') < 0) {
312:                 a.toast("لم يُقرأ رمز ترخيص في الصورة. قرّب الصورة على الرمز أو استعمل ملف المفتاح.");
313:                 return;
314:             }
315:             activateText(txt);
316:         });
317:     }

329:         } catch (Exception e) {
