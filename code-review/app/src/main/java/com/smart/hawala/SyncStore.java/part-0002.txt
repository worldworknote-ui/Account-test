223:     static String upTo(String col, long cutoff, String marks) {
224:         if (marks == null || marks.trim().isEmpty() || marks.trim().equals("{}")) return "(" + col + "<=" + cutoff + ")";
225:         try {
226:             JSONObject o = new JSONObject(marks);
227:             StringBuilder b = new StringBuilder("(");
228:             java.util.Iterator<String> it = o.keys();
229:             boolean first = true;
230:             while (it.hasNext()) {
231:                 String g = it.next();
232:                 long base = Long.parseLong(g) << 32, max = o.getLong(g);
233:                 if (!first) b.append(" OR ");
234:                 b.append("(").append(col).append(">=").append(base).append(" AND ").append(col).append("<=").append(max).append(")");
235:                 first = false;
236:             }
237:             if (first) return "(" + col + "<=" + cutoff + ")";
238:             return b.append(")").toString();
239:         } catch (Exception e) {
240:             return "(" + col + "<=" + cutoff + ")";
241:         }
242:     }
