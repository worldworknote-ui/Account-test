296:     private List<Block> build() {
297:         List<Block> out = new ArrayList<>();
298:         out.add(header());
299:         float maxImg = H - 2 * M - FOOTER - RUNNING - 60;
300:         for (Report.Section s : r.sections) {
301:             if (s.heading != null && !s.heading.isEmpty()) out.add(band(s.heading, s.accent));
302:             for (int i = 0; i < s.summary.size(); i++) {
303:                 Block k = kv(s.summary.get(i));
304:                 if (i == 0) {
305:                     Block spacer = new Block() {
306:                         @Override void draw(Canvas c, float top) {}
307:                     };
308:                     spacer.h = 6;
309:                     spacer.keepWithNext = true;
310:                     out.add(spacer);
311:                 }
312:                 k.keepWithNext = i < s.summary.size() - 1;
313:                 out.add(k);
314:             }
315:             if (s.note != null && !s.note.isEmpty()) out.add(note(s.note));
316:             if (!s.cols.isEmpty() && !s.rows.isEmpty()) {
317:                 float[] w = widths(s.cols);
318:                 Block spacer = new Block() {
319:                     @Override void draw(Canvas c, float top) {}
320:                 };
321:                 spacer.h = 6;
322:                 spacer.keepWithNext = true;
323:                 out.add(spacer);
324:                 Block head = row(s.cols, w, new Object[0], Report.NORMAL, true, s.accent, 0);
325:                 head.keepWithNext = true;
326:                 out.add(head);
327:                 for (Report.Row rw : s.rows) {
328:                     Block b = row(s.cols, w, rw.cells, rw.style, false, s.accent, rw.tint);
329:                     b.repeatHeader = head;
330:                     out.add(b);
331:                     if (rw.images != null && o.db != null && o.style != null) {
332:                         for (Long id : rw.images) {
333:                             Block im = image(id, maxImg);
334:                             if (im != null) {
335:                                 im.repeatHeader = head;
336:                                 out.add(im);
337:                             }
338:                         }
339:                     }
340:                 }
341:             }
342:         }
343:         if (r.signature) out.add(signature());
344:         return out;
345:     }

347:     private List<List<Block>> paginate(List<Block> blocks) {
348:         List<List<Block>> pages = new ArrayList<>();
349:         List<Block> page = new ArrayList<>();
350:         float avail = H - 2 * M - FOOTER;
351:         float used = 0;
352:         for (int i = 0; i < blocks.size(); i++) {
353:             Block b = blocks.get(i);
354:             float need = b.h;
355:             int j = i;
356:             while (blocks.get(j).keepWithNext && j + 1 < blocks.size() && need < avail * 0.5f) {
357:                 j++;
358:                 need += blocks.get(j).h;
359:             }
360:             if (used + need > avail && !page.isEmpty()) {
361:                 pages.add(page);
362:                 page = new ArrayList<>();
363:                 avail = H - 2 * M - FOOTER - RUNNING;
364:                 used = 0;
365:                 if (b.repeatHeader != null) {
366:                     page.add(b.repeatHeader);
367:                     used += b.repeatHeader.h;
368:                 }
369:             }
370:             page.add(b);
371:             used += b.h;
372:         }
373:         if (!page.isEmpty()) pages.add(page);
374:         return pages;
375:     }
