71:     public static Grant verify(String token, byte[] deviceKeyX509, Map<String, String> trusted) {
72:         try {
73:             if (token == null || deviceKeyX509 == null) return null;
74:             token = token.replaceAll("\\s+", "");
75:             int dot = token.indexOf('.');
76:             if (dot < 1 || dot != token.lastIndexOf('.') || dot == token.length() - 1) return null;
77:             Base64.Decoder u = Base64.getUrlDecoder();
78:             byte[] payload = u.decode(token.substring(0, dot));
79:             byte[] sig = u.decode(token.substring(dot + 1));
80:             Map<String, String> f = parse(new String(payload, StandardCharsets.UTF_8));
81:             if (f == null) return null;
82:             String pub = f.containsKey("kid") ? trusted.get(f.get("kid")) : null;
83:             if (pub == null || !"1".equals(f.get("v"))) return null;
84:             Signature s = Signature.getInstance("SHA256withECDSA");
85:             s.initVerify(KeyFactory.getInstance("EC").generatePublic(new X509EncodedKeySpec(Base64.getDecoder().decode(pub))));
86:             s.update(payload);
87:             if (!s.verify(sig)) return null;
88:             byte[] want = deviceFingerprint(deviceKeyX509).getBytes(StandardCharsets.US_ASCII);
89:             if (!MessageDigest.isEqual(want, f.getOrDefault("dk", "").getBytes(StandardCharsets.US_ASCII))) return null;
90:             if (f.getOrDefault("lid", "").isEmpty()) return null;
91:             return new Grant(f);
92:         } catch (Exception e) {
93:             return null;
94:         }
95:     }

110:      * 11.5: طلب التفعيل مضغوطًا (SA8-REQZ:): JSON مضغوط deflate ثم Base64url — أقصر بنحو الثلث من SA8-REQ: فلا ينقسم بسهولة.

113:     public static String packRequest(String json) {
114:         java.util.zip.Deflater d = new java.util.zip.Deflater(java.util.zip.Deflater.BEST_COMPRESSION, true);
115:         d.setInput(json.getBytes(StandardCharsets.UTF_8));
116:         d.finish();
117:         java.io.ByteArrayOutputStream out = new java.io.ByteArrayOutputStream();
118:         byte[] buf = new byte[4096];
119:         while (!d.finished()) out.write(buf, 0, d.deflate(buf));
120:         d.end();
121:         return "SA8-REQZ:" + Base64.getUrlEncoder().withoutPadding().encodeToString(out.toByteArray());
122:     }
