321:     /** نسخة جديدة في الحافظة أثناء التجميع. يعيد true إن أُضيف شيء. */
322:     boolean collectNow() {
323:         if (!collecting) return false;
324:         WaClip.Read r = WaClip.read(a);
325:         if (r.text.trim().isEmpty() || r.text.equals(lastCollected)) return false;
326:         lastCollected = r.text;
327:         List<WaText.Msg> msgs = WaClip.messages(r.text, r.html);
328:         int added = 0;
329:         long left = waRemaining(false);   // 12.6: المجانية لا تجمع أكثر من المتبقي اليوم
330:         boolean capped = false;
331:         for (WaText.Msg m : msgs) {
332:             if (collected.size() >= left) {
333:                 capped = true;
334:                 break;
335:             }
336:             String fp = WaText.fingerprint(m.body);
337:             boolean dup = false;
338:             for (WaText.Msg c : collected) if (WaText.fingerprint(c.body).equals(fp)) { dup = true; break; }
339:             if (dup) continue;
340:             collected.add(m);
341:             added++;
342:         }
343:         if (capped) a.toast("النسخة المجانية: بقي لك اليوم " + left + " رسالة للاستخراج، وجُمع العدد كله. انقل ما جُمع، والباقي بالإدخال اليدوي.");
344:         if (added > 0) {
345:             CollectView cv = collectView;
346:             if (cv != null) cv.show(new ArrayList<>(collected), this::moveCollected, this::stopCollect);
347:             else a.toast("جُمعت " + collected.size() + (collected.size() == 1 ? " رسالة" : " رسائل"));
348:             if (a.onHome) a.homeScreen.refreshClipCard();
349:         }
350:         return added > 0;
351:     }
