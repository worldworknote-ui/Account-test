205:

207:             zip.putNextEntry(new ZipEntry("tables/" + table + ".ndjson"));
208:             final long[] n = {0};
209:             final OutputStream z = zip;
210:             src.rows(table, new RowSink() {
211:                 @Override public void row(JSONObject row) throws Exception {
212:                     z.write(row.toString().getBytes(StandardCharsets.UTF_8));
213:                     z.write('\n');
214:                     n[0]++;

216:             });
217:             zip.closeEntry();
218:             res.rows.put(table, n[0]);

221:         StringBuilder index = new StringBuilder();

223:             byte[] data;
224:             try {
225:                 data = src.attachmentData(ref);
226:             } catch (Throwable t) {
227:                 data = null;
228:             }
229:             if (data == null || data.length == 0) {
230:                 res.skippedAttachments++;
231:                 continue;
232:             }
233:             zip.putNextEntry(new ZipEntry("attachments/" + ref.id + ".bin"));
234:             zip.write(data);
235:             zip.closeEntry();
236:             JSONObject meta = new JSONObject();
237:             meta.put("id", ref.id);
238:             meta.put("tx_id", ref.txId);
239:             meta.put("mime", ref.mime == null ? "image/jpeg" : ref.mime);
240:             meta.put("created_at", ref.createdAt);
241:             meta.put("size", data.length);
242:             meta.put("sha256", sha256Hex(data));
243:             index.append(meta.toString()).append('\n');
244:             res.attachments++;
245:             Arrays.fill(data, (byte) 0);
246:         }
247:         putText(zip, "attachments/index.ndjson", index.toString());
248: 
249:         res.warnings.addAll(src.warnings());
250:         res.unreadable = src.unreadable();
251:         JSONObject summary = new JSONObject();
252:         JSONObject rows = new JSONObject();
253:         for (Map.Entry<String, Long> e : res.rows.entrySet()) rows.put(e.getKey(), e.getValue().longValue());
254:         summary.put("rows", rows);
255:         summary.put("attachments", res.attachments);
256:         summary.put("skippedAttachments", res.skippedAttachments);
257:         summary.put("unreadableFields", res.unreadable);
258:         summary.put("warnings", new JSONArray(res.warnings));
259:         summary.put("completedAt", Clock.now());
260:         putText(zip, "summary.json", summary.toString(1));
261:         zip.finish();
262:         zip.flush();
263:         return res;
264:     }
