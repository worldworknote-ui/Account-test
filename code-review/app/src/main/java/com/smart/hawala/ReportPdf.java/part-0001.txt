93:     static int blend(int c, float a) {
94:         return Color.rgb((int) (255 - (255 - Color.red(c)) * a), (int) (255 - (255 - Color.green(c)) * a), (int) (255 - (255 - Color.blue(c)) * a));
95:     }

101:     private static StaticLayout layout(String s, TextPaint p, float width, Layout.Alignment align) {
102:         String t = s == null ? "" : s;

104:                 .setAlignment(align)
105:                 .setTextDirection(TextDirectionHeuristics.RTL)
106:                 .setIncludePad(false)
107:                 .setLineSpacing(0f, 1.06f)
108:                 .build();

120:     private Block header() {
121:         final float logoSize = o.style.logo && o.logo != null ? 54f : 0f;
122:         final float gap = logoSize > 0 ? 10f : 0f;
123:         final StaticLayout t = layout(r.title, title, CW - logoSize - gap, Layout.Alignment.ALIGN_NORMAL);
124:         final StaticLayout lines = layout(Model.join(r.headerLines, "\n"), small, CW - logoSize - gap, Layout.Alignment.ALIGN_NORMAL);
125:         Block b = new Block() {
126:             @Override void draw(Canvas c, float top) {
127:                 if (logoSize > 0) {
128:                     RectF dst = new RectF(W - M - logoSize, top, W - M, top + logoSize);
129:                     c.drawBitmap(o.logo, new Rect(0, 0, o.logo.getWidth(), o.logo.getHeight()), dst, null);
130:                 }
131:                 drawLayout(c, t, M, top);
132:                 drawLayout(c, lines, M, top + t.getHeight() + 4);
133:                 Paint p = new Paint(Paint.ANTI_ALIAS_FLAG);
134:                 p.setColor(o.accent);
135:                 p.setStrokeWidth(1.6f);
136:                 c.drawLine(M, top + h - 5, W - M, top + h - 5, p);
137:             }
138:         };
139:         b.h = Math.max(logoSize, t.getHeight() + 4 + lines.getHeight()) + 12;
140:         return b;
141:     }

143:     private Block band(String s, final int accent) {
144:         TextPaint hp = heading;
145:         if (accent != 0) {
146:             hp = new TextPaint(heading);
147:             hp.setColor(darker(accent));
148:         }
149:         final StaticLayout l = layout(s, hp, CW - 2 * PAD, Layout.Alignment.ALIGN_NORMAL);
150:         Block b = new Block() {
151:             @Override void draw(Canvas c, float top) {
152:                 fill.setColor(accent != 0 ? blend(accent, 0.13f) : light);
153:                 c.drawRect(M, top + 8, W - M, top + h, fill);
154:                 fill.setColor(accent != 0 ? accent : o.accent);
155:                 c.drawRect(W - M - 3, top + 8, W - M, top + h, fill);
156:                 drawLayout(c, l, M + PAD, top + 8 + PAD);
157:             }
158:         };
159:         b.h = l.getHeight() + 8 + 2 * PAD;
160:         b.keepWithNext = true;
161:         return b;

164:     private Block kv(final Report.Kv kv) {
165:         final float boxW = Math.min(CW, 330f);
166:         final float labelW = boxW * 0.58f;
167:         final StaticLayout lab = layout(kv.label, kv.emphasis ? bold : text, labelW - 2 * PAD, Layout.Alignment.ALIGN_NORMAL);
168:         final StaticLayout val = layout(Report.text(kv.value), kv.emphasis ? bold : text, boxW - labelW - 2 * PAD, Layout.Alignment.ALIGN_CENTER);
169:         Block b = new Block() {
170:             @Override void draw(Canvas c, float top) {
171:                 float right = W - M, left = right - boxW;
172:                 fill.setColor(kv.emphasis ? total : Color.rgb(246, 246, 246));
173:                 c.drawRect(right - labelW, top, right, top + h, fill);
174:                 if (kv.emphasis) {
175:                     fill.setColor(total);
176:                     c.drawRect(left, top, right - labelW, top + h, fill);
177:                 }
178:                 drawLayout(c, lab, right - labelW + PAD, top + PAD);
179:                 drawLayout(c, val, left + PAD, top + PAD);
180:                 c.drawLine(left, top + h, right, top + h, line);
181:                 c.drawLine(left, top, left, top + h, line);
182:                 c.drawLine(right, top, right, top + h, line);
183:                 c.drawLine(right - labelW, top, right - labelW, top + h, line);
184:             }
185:         };
186:         b.h = Math.max(lab.getHeight(), val.getHeight()) + 2 * PAD;
187:         return b;

199:     private float[] widths(List<Report.Col> cols) {
200:         float sum = 0;
201:         for (Report.Col c : cols) sum += c.weight;
202:         float[] w = new float[cols.size()];
