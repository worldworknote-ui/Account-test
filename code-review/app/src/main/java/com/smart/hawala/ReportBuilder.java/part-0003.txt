333:         public final Map<String, Map<String, Agg>> groups = new LinkedHashMap<>();
334:         public final Map<String, String> labels = new HashMap<>();
335:         public final Map<String, Agg> totals = new TreeMap<>();
336:         public final Map<String, Map<String, Agg>> previous = new HashMap<>();
337:         public final Map<String, Agg> previousTotals = new TreeMap<>();
338:         public final List<Model.Tx> rows = new ArrayList<>();
339:         /** الجدول المتقاطع: المجموعة ← العمود ← العملة ← المجاميع. */
340:         public final Map<String, Map<String, Map<String, Agg>>> pivot = new HashMap<>();

345:     public static Result compute(Spec s, Data d) {
346:         Result r = new Result();
347:         r.from = s.from;
348:         r.to = s.to > 0 ? s.to : Long.MAX_VALUE;
349:         Map<String, Map<String, Agg>> groups = new HashMap<>();
350:         for (Model.Tx t : d.txs) {
351:             if (!matches(s, d, t, r.from, r.to)) continue;
352:             r.rows.add(t);
353:             String[] k = key(s.group, d, t);
354:             r.labels.put(k[0], k[1]);
355:             accumulate(groups.computeIfAbsent(k[0], z -> new TreeMap<>()), t, s.currencies);
356:             accumulate(r.totals, t, s.currencies);
357:             if (s.group2 != G_NONE) {
358:                 String[] k2 = key(s.group2, d, t);
359:                 r.pivotLabels.put(k2[0], k2[1]);
360:                 accumulate(r.pivot.computeIfAbsent(k[0], z -> new HashMap<>()).computeIfAbsent(k2[0], z -> new TreeMap<>()), t, s.currencies);
361:             }
362:         }
363:         if (s.compare && s.from > 0) {
364:             long len = r.to == Long.MAX_VALUE ? d.now - s.from : r.to - s.from;
365:             r.prevFrom = s.from - len;
366:             r.prevTo = s.from;
367:             for (Model.Tx t : d.txs) {
368:                 if (!matches(s, d, t, r.prevFrom, r.prevTo)) continue;
369:                 String[] k = key(s.group, d, t);
370:                 if (!r.labels.containsKey(k[0])) r.labels.put(k[0], k[1]);
371:                 accumulate(r.previous.computeIfAbsent(k[0], z -> new TreeMap<>()), t, s.currencies);
372:                 accumulate(r.previousTotals, t, s.currencies);
373:             }
374:             for (String k : r.previous.keySet()) groups.computeIfAbsent(k, z -> new TreeMap<>());
375:         }
376:         List<String> keys = new ArrayList<>(groups.keySet());
377:         final Map<String, Map<String, Agg>> g = groups;
378:         Comparator<String> byKey = timeGroup(s.group) || s.group == G_DIRECTION || s.group == G_KIND || s.group == G_SIZE || s.group == G_YEAR
379:                 ? Comparator.naturalOrder() : (a, b) -> r.labels.get(a).compareTo(r.labels.get(b));
380:         Comparator<String> cmp;
381:         switch (s.sort) {
382:             case SORT_COUNT: cmp = Comparator.comparingLong((String k) -> -sum(g.get(k), M_COUNT)); break;
383:             case SORT_VOLUME: cmp = Comparator.comparingLong((String k) -> -sum(g.get(k), M_VOLUME)); break;
384:             case SORT_NET: cmp = Comparator.comparingLong((String k) -> -Math.abs(sum(g.get(k), M_NET))); break;
385:             case SORT_FEES: cmp = Comparator.comparingLong((String k) -> -sum(g.get(k), M_FEES_NET)); break;
386:             default: cmp = byKey;
387:         }
388:         Collections.sort(keys, cmp.thenComparing(byKey));
389:         if (s.top > 0 && keys.size() > s.top) keys = keys.subList(0, s.top);
390:         for (String k : keys) r.groups.put(k, groups.get(k));
391:         return r;
392:     }

394:     /**
395:      * مجموع مقياس عبر العملات. للعدد يُجمع مباشرة؛ للمبالغ يُجمع بلا تحويل (للترتيب فقط، ضمن عملة واحدة في الغالب).
396:      */
397:     static long sum(Map<String, Agg> m, int measure) {
398:         long s = 0;
399:         if (m == null) return 0;
400:         for (Agg a : m.values()) s += a.measure(measure);
401:         return s;
402:     }

404:     /** رصيد كل مكتب|عملة قبل لحظة (حركات نشطة + أرصدة سابقة)، للرصيد المتراكم في الكشف التفصيلي. */
405:     static Map<String, Long> balanceBefore(Data d, long before) {
406:         Map<String, Long> b = new HashMap<>();
407:         for (Model.Tx t : d.txs) {
408:             if (!t.isActive() || t.createdAt >= before) continue;
409:             b.merge(t.officeId + "|" + t.currency, isDebit(t) ? t.amount : -t.amount, Long::sum);
410:             if (t.hasFee()) b.merge(t.officeId + "|" + t.feeCurrency, Model.FEE_US.equals(t.feeType) ? t.fee : -t.fee, Long::sum);
411:         }
412:         for (OwnerReports.Opening o : d.openings) if (o.when < before) b.merge(o.office + "|" + o.currency, o.signed, Long::sum);
413:         return b;
414:     }
