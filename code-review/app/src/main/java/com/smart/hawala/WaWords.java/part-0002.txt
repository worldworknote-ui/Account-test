129:     private static final String PHONE = "\\+?\\d{1,4}(?:[ \\-]?\\d{2,4}){2,5}";
130:     private static final Pattern SENDER_NAME_PHONE = Pattern.compile("^\\s*~?\\s*([^\\d+~\\n]{1,40}?)\\s*~?\\s*(" + PHONE + ")\\s*$");
131:     private static final Pattern SENDER_PHONE_NAME = Pattern.compile("^\\s*(" + PHONE + ")\\s*~\\s*([^\\d+~\\n]{1,40}?)\\s*$");
132:     private static final Pattern SENDER_TILDE = Pattern.compile("^\\s*~\\s*([^\\d~\\n]{1,40}?)\\s*$");
133:     private static final Pattern PHONE_ONLY = Pattern.compile("^\\s*(" + PHONE + ")\\s*$");

135:     /**
136:      * أول سطر في فقاعة الغروب: اسم المرسل ورقمه («~ Abo Nour +40 765 450 476»، «+90 545 965 72 40 ~زهرة حلب»، «~ أبو عمر»).
137:      * يعيد {الاسم، الرقم} أو null إن لم يكن سطر مرسل.
138:      */
139:     public static String[] sender(String line) {
140:         if (line == null) return null;
141:         String s = MessageParser.digits(WaText.stripMarks(line)).trim();
142:         if (s.isEmpty() || s.length() > 70) return null;
143:         Matcher m = SENDER_NAME_PHONE.matcher(s);
144:         if (m.matches() && (s.contains("~") || s.contains("+"))) return new String[]{m.group(1).trim(), m.group(2).trim()};
145:         m = SENDER_PHONE_NAME.matcher(s);
146:         if (m.matches()) return new String[]{m.group(2).trim(), m.group(1).trim()};
147:         m = SENDER_TILDE.matcher(s);
148:         if (m.matches()) return new String[]{m.group(1).trim(), ""};
149:         m = PHONE_ONLY.matcher(s);
150:         if (m.matches() && s.startsWith("+")) return new String[]{"", m.group(1).trim()};
151:         return null;
152:     }

154:     /**
155:      * نص فقاعة مقروءة من الشاشة ← {المرسل، النص}: يُفصل سطر المرسل في أولها، وتُحذف أسطر «Forwarded» و«Edited» وحجم الملف أينما كانت،
156:      * وتنبيهات البرنامج الملتصقة بها.
157:      */
158:     public static String[] split(String text) {
159:         List<String> l = new ArrayList<>(Arrays.asList(WaText.stripMarks(text == null ? "" : text).replace("\r\n", "\n").split("\n", -1)));
160:         String who = "";
161:         while (!l.isEmpty() && l.get(0).trim().isEmpty()) l.remove(0);
162:         if (l.size() > 1) {
163:             String[] s = sender(l.get(0));
164:             if (s != null) {
165:                 who = s[0].isEmpty() ? s[1] : s[0];
166:                 l.remove(0);
167:             }
168:         }
169:         List<String> out = new ArrayList<>();
170:         for (String x : l) {
171:             if (metaLine(x)) continue;
172:             if (systemLine(x) && !WaCuts.timeLine(x) && DATE_PILL.matcher(clean(x)).matches()) continue;
173:             out.add(x);
174:         }
175:         while (!out.isEmpty() && out.get(0).trim().isEmpty()) out.remove(0);
176:         while (!out.isEmpty() && out.get(out.size() - 1).trim().isEmpty()) out.remove(out.size() - 1);
177:         return new String[]{who, String.join("\n", out)};
178:     }
