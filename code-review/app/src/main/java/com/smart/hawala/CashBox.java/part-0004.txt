482:             if (to > 0 && x.createdAt >= to) continue;
483:             if (!x.counts()) continue;
484:             s.count++;
485:             long fc = x.feeCash();
486:             if (fc > 0) s.feesIn += fc;
487:             else s.feesOut -= fc;
488:             switch (x.kind) {
489:                 case OPENING: s.openingsInPeriod += x.amount; break;
490:                 case IN:
491:                     s.in += x.amount;
492:                     if (x.officeId != null) s.officeIn += x.amount;
493:                     break;
494:                 case OUT:
495:                     s.out += x.amount;
496:                     if (x.officeId != null) s.officeOut += x.amount;
497:                     break;
498:                 case EXPENSE:
499:                     s.expenses += x.amount;
500:                     String cat = Model.nz(x.category).trim().isEmpty() ? "أخرى" : x.category.trim();
501:                     Long v = s.byCategory.get(cat);
502:                     s.byCategory.put(cat, (v == null ? 0 : v) + x.amount);
503:                     break;
504:                 case EXCHANGE:
505:                     if (x.dir > 0) s.exchangeIn += x.amount;
506:                     else s.exchangeOut += x.amount;
507:                     break;
508:                 case ADJUST:
509:                     if (x.dir > 0) s.adjustUp += x.amount;
510:                     else s.adjustDown += x.amount;
511:                     break;
512:                 case TRANSFER:
513:                     if (x.dir > 0) s.transferIn += x.amount;
514:                     else s.transferOut += x.amount;
515:                     break;
516:                 default: break;
517:             }
518:         }
519:         for (Summary s : out.values()) s.closing = s.computedClosing();
520:         return out;
521:     }
