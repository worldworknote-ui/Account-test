461:     private void fill(final Model.Tx t, Runnable after, String title) {
462:         editing = t;
463:         afterEdit = after;
464:         a.pendingImages.clear();
465:         existingAtt.clear();

472:         f.amount.setText(Model.plain(t.amount, a.db.decimals(t.currency)));

499:             f.specialType = t.special;

502:             setSpecial(true);
