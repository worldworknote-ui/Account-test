94:     /**
95:      * تدوير آمن: الأقدم يُنقل جانبًا (لا يُحذف بعد)، ثم تُزاح الخانات، ثم تدخل الجديدة. فشل أي خطوة يعيد كل شيء.
96:      * الأقدم يُحذف في النهاية فقط بعد نجاح الإدخال.
97:      */
98:     static void rotateIn(Context ctx, File fresh) throws IOException {
99:         File drop = new File(ctx.getFilesDir(), "pre_restore_drop");
100:         deleteTree(drop);
101:         List<File[]> done = new ArrayList<>();   // {من، إلى} لكل نقل نجح، للتراجع بالعكس
102:         try {
103:             File last = slot(ctx, SLOTS - 1);
104:             if (last.exists()) move(last, drop, done);
105:             for (int i = SLOTS - 2; i >= 0; i--) {
106:                 File s = slot(ctx, i);
107:                 if (s.exists()) move(s, slot(ctx, i + 1), done);
108:             }
109:             move(fresh, slot(ctx, 0), done);
110:         } catch (IOException e) {
111:             for (int i = done.size() - 1; i >= 0; i--) done.get(i)[1].renameTo(done.get(i)[0]);
112:             throw e;
113:         }
114:         deleteTree(drop);
115:     }

117:     private static void move(File from, File to, List<File[]> done) throws IOException {
118:         deleteTree(to);
119:         if (!from.renameTo(to)) throw new IOException("تعذر حفظ نسخة الأمان (نقل " + from.getName() + ")");
120:         done.add(new File[]{from, to});
121:     }
