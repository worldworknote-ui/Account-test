363:     private void restoreScroll(final String title, boolean sameScreen) {
364:         final long[] m = scrollMemory.get(title);
365:         if (m == null || m[0] <= 0 || Clock.now() - m[1] > 20 * 60_000L) return;
366:         if (!sameScreen && !returning) return;
367:         final ScrollView sv = scroll;
368:         Runnable go = () -> { if (sv == scroll) sv.scrollTo(0, (int) m[0]); };
369:         sv.post(go);
370:         sv.postDelayed(go, 150);
371:     }

478:     /** 10.13: شاشة عريضة (تابلت، أو هاتف بالعرض): النصفان متقابلان بدل التمرير. */
479:     boolean wide() { return getResources().getConfiguration().screenWidthDp >= 720; }

481:     /**
482:      * 10.13: على الشاشة العريضة يصير محتوى الشاشة عمودين متقابلين: ما قبل at في العمود الأول (يمين) وما بعده في الثاني.
483:      * على الهاتف بالطول لا يتغير شيء. ما يُضاف لاحقًا في أعلى الشاشة (رأس واتساب مثلًا) يبقى بعرضها كله.
484:      */
485:     void splitBody(View at) {
486:         if (!wide() || at == null) return;
487:         int k = body.indexOfChild(at);
488:         int n = body.getChildCount();
489:         if (k <= 0 || k >= n) return;
490:         List<View> all = new ArrayList<>();
491:         for (int i = 0; i < n; i++) all.add(body.getChildAt(i));
492:         body.removeAllViews();
493:         LinearLayout row = row();
494:         row.setBaselineAligned(false);
495:         row.setGravity(Gravity.TOP);
496:         LinearLayout c1 = new LinearLayout(this), c2 = new LinearLayout(this);
497:         c1.setOrientation(LinearLayout.VERTICAL);
498:         c2.setOrientation(LinearLayout.VERTICAL);
499:         for (int i = 0; i < n; i++) {
500:             View v = all.get(i);
501:             ViewGroup.LayoutParams lp = v.getLayoutParams();
502:             (i < k ? c1 : c2).addView(v, lp != null ? lp : new LinearLayout.LayoutParams(-1, -2));
503:         }
504:         LinearLayout.LayoutParams p1 = new LinearLayout.LayoutParams(0, -2, 1), p2 = new LinearLayout.LayoutParams(0, -2, 1);
505:         p1.setMarginEnd(dp(10));
506:         p2.setMarginStart(dp(10));
507:         row.addView(c1, p1);
508:         row.addView(c2, p2);
509:         body.addView(row, new LinearLayout.LayoutParams(-1, -2));
510:     }
