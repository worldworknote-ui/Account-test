653:             Bitmap s = Bitmap.createScaledBitmap(bm, Math.max(1, Math.round(w * f)), Math.max(1, Math.round(h * f)), true);
654:             if (s != bm) bm.recycle();
655:             bm = s;
656:         }
657:         ByteArrayOutputStream out = new ByteArrayOutputStream();
658:         if (bm.hasAlpha()) bm.compress(Bitmap.CompressFormat.PNG, 100, out);
659:         else bm.compress(Bitmap.CompressFormat.JPEG, 88, out);
660:         bm.recycle();
661:         return Base64.encodeToString(out.toByteArray(), Base64.NO_WRAP);
662:     }
