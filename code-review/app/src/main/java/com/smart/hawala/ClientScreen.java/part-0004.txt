513:         final Map<String, Long> b = ClientBook.balances(a.db.clientMoves(k.id));
514:         if (b.isEmpty()) {

517:         }

522:         a.add(a.label("من رصيده بـ"));
523:         final List<String> have = new ArrayList<>(b.keySet());
524:         final Ui.Seg from = a.new Seg(have.toArray(new String[0]), null, have.size() == 1 ? 0 : -1, have.size() > 4);

528:         a.add(a.label("إلى"));
529:         final List<String> curs = new ArrayList<>();
530:         for (Model.Currency x : a.db.currencies(false)) curs.add(x.code);
531:         final Ui.Seg to = a.new Seg(curs.toArray(new String[0]), null, -1, curs.size() > 4);
532:         a.body.addView(to.view, a.lp(-1, -2, 2));
533:         final EditText rate = a.field("السعر: 1 من الأولى = كم من الثانية");
534:         Ui.numeric(rate);
535:         a.body.addView(rate, a.lp(-1, 50, 2));

537:         final EditText toAmt = a.amountInput("المبلغ بالعملة الجديدة (يُحسب تلقائيًا)");

541:             String f = from.label(), t = to.label();
542:             if (f != null) {
543:                 long v = b.get(f);
544:                 fromBal.setText("رصيده: " + c.sideText(v, f));

546:                     amount.setText(BigDecimal.valueOf(Math.abs(v), CashBox.dec(dec, f)).stripTrailingZeros().toPlainString());
547:             }
548:             if (f == null || t == null || f.equals(t)) {
549:                 market.setText("");
550:                 return;
551:             }
552:             BigDecimal mk = CashBox.crossRate(f, t, a.db.usdRates());
553:             market.setText(mk == null ? "لا يوجد سعر سوق لهما" : "سعر السوق: " + CashScreen.iso("1 " + f + " = " + mk.toPlainString() + " " + t) + "   (اضغط لاستخدامه)");
554:             market.setOnClickListener(mk == null ? null : v -> rate.setText(mk.toPlainString()));
555:             try {
556:                 long am = Db.minor(amount.getText().toString(), CashBox.dec(dec, f));
557:                 Long ta = CashBox.counterAmount(am, CashBox.dec(dec, f), rate.getText().toString(), CashBox.dec(dec, t));
558:                 if (ta != null) toAmt.setText(BigDecimal.valueOf(ta, CashBox.dec(dec, t)).stripTrailingZeros().toPlainString());
559:             } catch (Exception ignored) {

561:             }

569:         rate.addTextChangedListener(Ui.watcher(calc));

571:         a.footer(a.makeButton("مراجعة التبديل", c.teal(), v -> {

576:                 final long ta = Db.minor(toAmt.getText().toString(), CashBox.dec(dec, t));

578:                 if (err != null) throw new Exception(err);

580:                 final long when = c.now();
581:                 long bf = b.get(f), sideNow = bf > 0 ? 1 : -1;
582:                 long bt = b.containsKey(t) ? b.get(t) : 0;
