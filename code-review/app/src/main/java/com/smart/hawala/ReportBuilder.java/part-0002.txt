180:     public static boolean matches(Spec s, Data d, Model.Tx t, long from, long to) {
181:         if (t.createdAt < from || (to > 0 && t.createdAt >= to)) return false;
182:         if (!s.offices.isEmpty() && !s.offices.contains(t.officeId)) return false;
183:         if (!s.currencies.isEmpty() && !s.currencies.contains(t.currency) && !(t.hasFee() && s.currencies.contains(t.feeCurrency))) return false;
184:         if (s.direction == DIR_US && !isDebit(t)) return false;
185:         if (s.direction == DIR_THEM && isDebit(t)) return false;
186:         boolean feeRow = OwnerReports.feeRow(t);
187:         switch (s.kind) {
188:             case K_TRANSFERS: if (feeRow) return false; break;
189:             case K_FEE_ROWS: if (!feeRow) return false; break;
190:             case K_RAJE: if (!OwnerReports.rajeRow(t)) return false; break;
191:             case K_WITH_FEE: if (feeRow || !t.hasFee()) return false; break;
192:             case K_NO_FEE: if (feeRow || t.hasFee()) return false; break;
193:             case K_FX: if (!t.isFx()) return false; break;
194:             case K_XFER: if (!t.isTransfer()) return false; break;
195:             default: break;
196:         }
197:         switch (s.status) {
198:             case S_ALL: break;
199:             case S_CANCELLED: if (!"CANCELLED".equals(t.status)) return false; break;
200:             case S_EDITED: if (t.replacesId == null && !replaced(d, t.id)) return false; break;
201:             case S_PENDING: if (!t.isPending()) return false; break;
202:             default: if (!t.isActive()) return false;
203:         }
204:         if (!s.text.trim().isEmpty()) {
205:             String q = OwnerReports.norm(s.text);
206:             String hay = OwnerReports.norm(Model.nz(t.person) + " " + Model.nz(t.description) + " " + Model.nz(t.notes) + " " + Model.nz(t.dest) + " " + Model.nz(t.service));
207:             for (String w : q.split(" ")) if (!w.isEmpty() && !hay.contains(w)) return false;
208:         }
209:         if (!s.service.trim().isEmpty() && !OwnerReports.norm(t.service).contains(OwnerReports.norm(s.service))) return false;
210:         if (!s.dest.trim().isEmpty() && !OwnerReports.norm(t.dest).contains(OwnerReports.norm(s.dest))) return false;
211:         if (!s.user.trim().isEmpty() && !OwnerReports.norm(t.userName).contains(OwnerReports.norm(s.user))) return false;
212:         int dec = dec(d, t.currency);
213:         if (s.minAmount != null && BigDecimal.valueOf(t.amount, dec).compareTo(s.minAmount) < 0) return false;
214:         if (s.maxAmount != null && BigDecimal.valueOf(t.amount, dec).compareTo(s.maxAmount) > 0) return false;
215:         return true;
216:     }

235:     /** يضيف أثر الحركة إلى مجاميع عملاتها (الأصل في عملته، والأجرة في عملتها). */
236:     static void accumulate(Map<String, Agg> m, Model.Tx t, Set<String> onlyCurrencies) {
237:         boolean feeRow = OwnerReports.feeRow(t);
238:         if (onlyCurrencies.isEmpty() || onlyCurrencies.contains(t.currency)) {
239:             Agg a = m.computeIfAbsent(t.currency, k -> new Agg());
240:             a.count++;
241:             if (feeRow) {
242:                 if (isDebit(t)) a.feeUs += t.amount;
243:                 else a.feeThem += t.amount;
244:             } else {
245:                 if (isDebit(t)) a.us += t.amount;
246:                 else a.them += t.amount;
247:                 a.max = Math.max(a.max, t.amount);
248:             }
249:         }
250:         if (!feeRow && t.hasFee() && (onlyCurrencies.isEmpty() || onlyCurrencies.contains(t.feeCurrency))) {
251:             Agg f = m.computeIfAbsent(t.feeCurrency, k -> new Agg());
252:             if (Model.FEE_US.equals(t.feeType)) f.feeUs += t.fee;
253:             else f.feeThem += t.fee;
254:         }
255:     }

268:             case G_CURRENCY: return new String[]{t.currency, t.currency};
269:             case G_DAY: return new String[]{String.format(Locale.US, "%04d%02d%02d", y, mo, da), String.format(Locale.US, "%04d/%02d/%02d", y, mo, da)};
270:             case G_WEEK: {
271:                 // الأسبوع يبدأ السبت
272:                 int back = (c.get(Calendar.DAY_OF_WEEK) - Calendar.SATURDAY + 7) % 7;
