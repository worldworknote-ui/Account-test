136:     static String exactKey(String s) {
137:         if (s == null) return "";
138:         StringBuilder b = new StringBuilder();
139:         for (char ch : s.trim().toCharArray()) {
140:             if (ch == 'ـ' || (ch >= 'ً' && ch <= 'ْ') || ch == '\u200E' || ch == '\u200F') continue;
141:             b.append(Character.isWhitespace(ch) ? ' ' : Character.toLowerCase(ch));
142:         }
143:         return b.toString().replaceAll(" +", " ").trim();
144:     }

155:     /** يحوّل مبلغًا من منازل الملف إلى منازل التطبيق؛ null إن لم يكن ممكنًا دون تقريب. */
156:     static Long convert(long minor, int from, int to) {
157:         if (from == to) return minor;
158:         BigInteger v = BigInteger.valueOf(minor);
159:         if (to > from) {
160:             BigInteger r = v.multiply(BigInteger.TEN.pow(to - from));
161:             return r.bitLength() < 63 ? r.longValue() : null;
162:         }
163:         BigInteger[] qr = v.divideAndRemainder(BigInteger.TEN.pow(from - to));
164:         return qr[1].signum() == 0 ? qr[0].longValue() : null;
165:     }
