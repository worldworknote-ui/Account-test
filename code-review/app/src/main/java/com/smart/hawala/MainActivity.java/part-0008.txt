1336:         String k = foldAr(f);

1340:     static String foldAr(String s) {
1341:         StringBuilder b = new StringBuilder(s.length());
1342:         for (char c : s.toLowerCase(java.util.Locale.ROOT).toCharArray()) {
1343:             if (c == 'أ' || c == 'إ' || c == 'آ' || c == 'ٱ') c = 'ا';
1344:             else if (c == 'ة') c = 'ه';
1345:             else if (c == 'ى') c = 'ي';
1346:             else if (c == 'ؤ') c = 'و';
1347:             else if (c == 'ئ') c = 'ي';
1348:             else if (c == 'ـ' || (c >= '\u064B' && c <= '\u0652')) continue;
1349:             b.append(c);
1350:         }
1351:         return b.toString();
1352:     }
