149:     public static String fmt(long minor, int dec) {
150:         StringBuilder p = new StringBuilder("#,##0");
151:         if (dec > 0) {
152:             p.append('.');
153:             for (int i = 0; i < dec; i++) p.append('#');
154:         }
155:         DecimalFormat f = new DecimalFormat(p.toString(), DecimalFormatSymbols.getInstance(Locale.US));
156:         return f.format(BigDecimal.valueOf(minor, dec));
157:     }
