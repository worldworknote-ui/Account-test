83:     static List<String[]> layout(Db db) {
84:         List<String[]> out = new ArrayList<>();
85:         java.util.Set<String> seen = new java.util.HashSet<>();
86:         for (String part : db.setting("home_widgets", "").split(",")) {
87:             String p = part.trim();
88:             if (p.isEmpty()) continue;
89:             boolean on = !p.startsWith("-");
90:             String k = on ? p : p.substring(1);
91:             if (widget(k) == null || !seen.add(k)) continue;
92:             out.add(new String[]{k, on ? "1" : "0"});
93:         }
94:         for (Widget w : WIDGETS) if (!seen.contains(w.key)) out.add(new String[]{w.key, w.defOn ? "1" : "0"});
95:         return out;
96:     }
