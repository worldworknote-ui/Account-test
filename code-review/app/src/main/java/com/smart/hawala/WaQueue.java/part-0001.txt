60:     /** يقرأ النص من نية المشاركة (رسالة أو عدة رسائل). يعيد null إن لم تكن مشاركة نصية. */
61:     List<WaText.Msg> fromIntent(Intent i) {
62:         if (i == null) return null;
63:         String act = i.getAction();
64:         if (!Intent.ACTION_SEND.equals(act) && !Intent.ACTION_SEND_MULTIPLE.equals(act)) return null;
65:         List<String> texts = new ArrayList<>();
66:         if (Intent.ACTION_SEND.equals(act)) {
67:             CharSequence cs = i.getCharSequenceExtra(Intent.EXTRA_TEXT);
68:             if (cs != null) texts.add(cs.toString());
69:             else {
70:                 Uri u = i.getParcelableExtra(Intent.EXTRA_STREAM);
71:                 String s = readText(u);
72:                 if (s != null) texts.add(s);
73:             }
74:         } else {
75:             ArrayList<CharSequence> l = i.getCharSequenceArrayListExtra(Intent.EXTRA_TEXT);
76:             if (l != null) for (CharSequence cs : l) if (cs != null) texts.add(cs.toString());
77:             if (texts.isEmpty()) {
78:                 ArrayList<Uri> us = i.getParcelableArrayListExtra(Intent.EXTRA_STREAM);
79:                 if (us != null) for (Uri u : us) {
80:                     String s = readText(u);
81:                     if (s != null) texts.add(s);
82:                 }
83:             }
84:         }
85:         List<WaText.Msg> out = new ArrayList<>();
86:         for (String t : texts) out.addAll(WaText.split(t));
87:         return out;
88:     }

90:     private String readText(Uri u) {
91:         if (u == null) return null;
92:         try (InputStream in = a.getContentResolver().openInputStream(u)) {
93:             if (in == null) return null;
94:             byte[] raw = Db.readAll(in);
95:             if (raw.length > 2_000_000) return null;
96:             return new String(raw, java.nio.charset.StandardCharsets.UTF_8);
97:         } catch (Exception e) {
98:             return null;
99:         }
100:     }

127:     State load() {
128:         String raw = a.db.secretSetting(KEY);
129:         if (raw.isEmpty()) return null;
130:         try {
131:             JSONObject o = new JSONObject(raw);
132:             State s = new State();
133:             s.index = o.optInt("i");
134:             s.saved = o.optInt("saved");
135:             s.skipped = o.optInt("skipped");
136:             s.office = o.optLong("office", -1);
137:             JSONArray arr = o.getJSONArray("items");
138:             for (int k = 0; k < arr.length(); k++) {
139:                 JSONObject m = arr.getJSONObject(k);
140:                 s.items.add(new WaText.Msg(m.optString("s"), m.optString("t"), m.optString("b")));
141:             }
142:             return s.items.isEmpty() ? null : s;
143:         } catch (Exception e) {
144:             a.db.deleteSetting(KEY);
145:             return null;
146:         }
147:     }
148: 
149:     void store(State s) {
150:         try {
151:             if (s == null || s.done()) {
152:                 a.db.deleteSetting(KEY);
153:                 return;
154:             }
155:             JSONObject o = new JSONObject();
156:             o.put("i", s.index);
157:             o.put("saved", s.saved);
158:             o.put("skipped", s.skipped);
159:             o.put("office", s.office);
160:             JSONArray arr = new JSONArray();
161:             for (WaText.Msg m : s.items) arr.put(new JSONObject().put("s", m.sender).put("t", m.time).put("b", m.body));
162:             o.put("items", arr);
163:             a.db.setSecretSetting(KEY, o.toString());
164:         } catch (Exception ignored) {
165:             // الدفعة ليست حرجة: الرسائل باقية في واتساب
166:         }
167:     }

181:     private Map<String, Long> senders() {
182:         Map<String, Long> m = new HashMap<>();
183:         for (String line : a.db.setting(SENDERS, "").split("\n")) {
184:             int t = line.lastIndexOf('\t');
185:             if (t <= 0) continue;
186:             try {
187:                 m.put(line.substring(0, t), Long.parseLong(line.substring(t + 1)));
188:             } catch (NumberFormatException ignored) {
189:                 // سطر تالف
190:             }
191:         }
192:         return m;
193:     }

195:     static String senderKey(String sender) { return MessageParser.norm(WaText.stripMarks(sender)).trim(); }
196: 
197:     void rememberSender(String sender, long office) {
198:         String k = senderKey(sender);
199:         if (k.isEmpty() || k.contains("\n") || k.contains("\t")) return;
200:         Map<String, Long> m = senders();
201:         Long old = m.get(k);
202:         if (old != null && old == office) return;
203:         m.put(k, office);
204:         StringBuilder sb = new StringBuilder();
205:         int n = 0;
206:         for (Map.Entry<String, Long> e : m.entrySet()) {
207:             if (n++ > 300) break;
208:             sb.append(e.getKey()).append('\t').append(e.getValue()).append('\n');
209:         }
210:         a.db.setSettingQuiet(SENDERS, sb.toString());
211:     }
