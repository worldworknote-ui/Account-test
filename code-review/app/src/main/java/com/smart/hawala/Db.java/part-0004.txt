445:     static Set<String> columnsOf(SQLiteDatabase d, String table) {
446:         Set<String> out = new HashSet<>();
447:         try (Cursor c = d.rawQuery("PRAGMA table_info(" + table + ")", null)) {
448:             while (c.moveToNext()) out.add(c.getString(1).toLowerCase(Locale.US));
449:         }
450:         return out;
451:     }
