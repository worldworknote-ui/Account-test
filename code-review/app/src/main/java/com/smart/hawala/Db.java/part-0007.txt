845:      * أرضية الحركات: الأصلية لكل مكتب (الملغاة تُحتسب حتى لا يكون الإلغاء طريقًا لتصفير السقف،
846:      * والحركة المصحَّحة replaces_id لا تُحتسب لأن التصحيح مجاني دائمًا). office&lt;0: لا شيء، 0: كل المكاتب.
847:      */
848:     private static Map<Long, Long> liveTx(SQLiteDatabase d, long office) {
849:         Map<Long, Long> m = new HashMap<>();
850:         if (office < 0) return m;
851:         try (Cursor c = d.rawQuery("SELECT office_id,COUNT(*) FROM tx WHERE replaces_id IS NULL"
852:                 + (office > 0 ? " AND office_id=" + office : "") + " GROUP BY office_id", null)) {
853:             while (c.moveToNext()) m.put(c.getLong(0), c.getLong(1));
854:         }
855:         return m;
856:     }

858:     /**
859:      * العدّادات كما تُحفظ مرفوعة إلى أرضية البيانات الحية. البصمة المختلفة تعني عبثًا بالإعدادات:
860:      * تُسجَّل ملاحظة، وأرضية المكاتب تؤخذ أيضًا من سجل التدقيق (المكاتب المنشأة ولو حُذفت).
861:      */
862:     private QuotaState quota(SQLiteDatabase d, long office) {
863:         String raw = rawSetting(d, "quota");
864:         QuotaState q = QuotaState.parse(raw);
865:         String stored = rawSetting(d, "quota_mac");
866:         String want = raw == null ? "" : quotaMac.apply(q.serialize());
867:         if (raw != null && !want.isEmpty() && !want.equals(stored)) {
868:             d.execSQL("INSERT OR REPLACE INTO settings(k,v) VALUES('risk_note',?)", new Object[]{"عدّادات النسخة المجانية عُدّلت خارج التطبيق؛ أُعيد بناؤها من البيانات."});
869:         }
870:         // 9.0: حذف المكتب يعيد مكانه، فعدّاد المكاتب هو الموجود فعلًا دائمًا (حتى لو حمل العدّاد القديم مكاتب محذوفة)
871:         q.offices = 0;
872:         q.raiseTo(liveOffices(d), liveTx(d, office));
873:         return q;
874:     }
