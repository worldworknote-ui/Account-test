113:     public static SecretKeySpec deriveKey(char[] password, byte[] salt, int iterations) throws GeneralSecurityException {
114:         PBEKeySpec spec = new PBEKeySpec(password, salt, iterations, 256);
115:         try {
116:             byte[] raw = SecretKeyFactory.getInstance("PBKDF2WithHmacSHA256").generateSecret(spec).getEncoded();
117:             try {
118:                 return new SecretKeySpec(raw, "AES");
119:             } finally {
120:                 Arrays.fill(raw, (byte) 0);
121:             }
122:         } finally {
