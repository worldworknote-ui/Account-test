200:         public EncryptingOutputStream(OutputStream out, char[] password, int iterations, int chunkSize, RecoveryKey rk) throws IOException {
201:             if (password == null || password.length == 0) throw new IOException("كلمة المرور فارغة");
202:             if (chunkSize < 1024 || chunkSize > MAX_CHUNK) throw new IllegalArgumentException("chunk");
203:             this.out = out;
204:             SecureRandom rnd = new SecureRandom();
205:             byte[] salt = new byte[SALT_BYTES];
206:             rnd.nextBytes(salt);
207:             prefix = new byte[PREFIX_BYTES];
208:             rnd.nextBytes(prefix);
209:             ByteArrayOutputStream h = new ByteArrayOutputStream();
210:             DataOutputStream d = new DataOutputStream(h);
211:             d.write(MAGIC);
212:             d.writeInt(rk == null ? VERSION : VERSION_RC);
213:             d.writeInt(iterations);
214:             d.writeInt(salt.length);
215:             d.write(salt);
216:             d.writeInt(prefix.length);
217:             d.write(prefix);
218:             d.writeInt(chunkSize);
219:             try {
220:                 if (rk == null) {
221:                     d.flush();
222:                     header = h.toByteArray();
223:                     key = deriveKey(password, salt, iterations);
224:                 } else {
225:                     // الإصدار 2: مفتاح بيانات عشوائي مغلّف بكلمة المرور وبرمز الاسترداد
226:                     d.writeInt(rk.salt.length);
227:                     d.write(rk.salt);
228:                     d.writeInt(rk.iterations);
229:                     d.flush();
230:                     byte[] base = h.toByteArray();
231:                     byte[] dk = new byte[32];
232:                     rnd.nextBytes(dk);
233:                     byte[] wp = gcm(Cipher.ENCRYPT_MODE, deriveKey(password, salt, iterations), iv(prefix, 0xFFFFFFFE), base, dk, 0, dk.length);
234:                     byte[] wr = gcm(Cipher.ENCRYPT_MODE, rk.key, iv(prefix, 0xFFFFFFFD), base, dk, 0, dk.length);
235:                     d.writeInt(wp.length);
236:                     d.write(wp);
237:                     d.writeInt(wr.length);
238:                     d.write(wr);
239:                     d.flush();
240:                     header = h.toByteArray();
241:                     key = new SecretKeySpec(dk, "AES");
242:                     Arrays.fill(dk, (byte) 0);
243:                 }
244:                 byte[] v = gcm(Cipher.ENCRYPT_MODE, key, iv(prefix, 0xFFFFFFFF), concat(header, VERIFY_AAD), new byte[16], 0, 16);
245:                 out.write(header);
246:                 writeInt(out, v.length);
247:                 out.write(v);
248:             } catch (GeneralSecurityException e) {
249:                 throw new IOException("تعذر تجهيز التشفير على هذا الهاتف: " + e.getMessage(), e);
250:             } catch (IOException e) {
251:                 throw e;
252:             }
253:             buf = new byte[chunkSize];
254:         }

258:         @Override public void write(int b) throws IOException {
259:             if (closed) throw new IOException("closed");
260:             if (pos == buf.length) emit(false);
261:             buf[pos++] = (byte) b;
262:             plainBytes++;
263:         }
264: 
265:         @Override public void write(byte[] b, int off, int len) throws IOException {
266:             if (closed) throw new IOException("closed");
267:             while (len > 0) {
268:                 if (pos == buf.length) emit(false);
269:                 int n = Math.min(len, buf.length - pos);
270:                 System.arraycopy(b, off, buf, pos, n);
271:                 pos += n;
272:                 off += n;
273:                 len -= n;
274:                 plainBytes += n;
275:             }
276:         }

278:         private void emit(boolean last) throws IOException {
279:             int flag = last ? 1 : 0;
280:             try {
281:                 byte[] ct = gcm(Cipher.ENCRYPT_MODE, key, iv(prefix, index), chunkAad(header, index, flag), buf, 0, pos);
282:                 out.write(flag);
283:                 writeInt(out, ct.length);
284:                 out.write(ct);
285:             } catch (GeneralSecurityException e) {
286:                 throw new IOException("فشل التشفير: " + e.getMessage(), e);
287:             }
288:             index++;
289:             pos = 0;
290:         }

292:         @Override public void flush() throws IOException { out.flush(); }

295:         @Override public void close() throws IOException {
296:             if (closed) return;
297:             emit(true);
298:             Arrays.fill(buf, (byte) 0);
299:             out.flush();
300:             closed = true;
301:         }

304:         public void abort() {

306:             Arrays.fill(buf, (byte) 0);
307:         }
