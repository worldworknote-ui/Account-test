426:     public static List<File> png(Report r, Opts o, File dir, String base) throws IOException {
427:         ReportPdf ex = new ReportPdf(r, o);
428:         List<List<Block>> pages = ex.paginate(ex.build());
429:         List<File> files = new ArrayList<>();
430:         float scale = 2f;
431:         for (int i = 0; i < pages.size(); i++) {
432:             Bitmap bm = Bitmap.createBitmap((int) (ex.W * scale), (int) (ex.H * scale), Bitmap.Config.ARGB_8888);
433:             try {
434:                 Canvas c = new Canvas(bm);
435:                 c.scale(scale, scale);
436:                 ex.drawPage(c, pages.get(i), i, pages.size());
437:                 File f = new File(dir, base + (pages.size() > 1 ? "-" + (i + 1) : "") + ".png");
438:                 try (OutputStream os = new FileOutputStream(f)) {
439:                     bm.compress(Bitmap.CompressFormat.PNG, 100, os);
440:                 }
441:                 files.add(f);
442:             } finally {
443:                 bm.recycle();
444:             }
445:         }
446:         return files;
447:     }
