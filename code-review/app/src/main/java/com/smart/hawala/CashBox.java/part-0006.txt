853:     /** قيمة مبلغ بالدولار أو null بلا سعر. */
854:     public static BigDecimal usd(long minor, String cur, Map<String, BigDecimal> rates, Map<String, Integer> decimals) {
855:         BigDecimal r = "USD".equals(cur) ? BigDecimal.ONE : rates.get(cur);
856:         if (r == null || r.signum() <= 0) return null;
857:         return BigDecimal.valueOf(minor, dec(decimals, cur)).multiply(r).setScale(2, RoundingMode.HALF_EVEN);
858:     }

860:     /**
861:      * يقارن كل عملة فيها زيادة بكل عملة فيها عجز: إن كانت قيمتاهما بالدولار متقاربتين (الفرق ≤ 3٪ أو دولار واحد)
862:      * فالأرجح أنها عملية بيع (خرجت عملة العجز ودخلت عملة الزيادة) لم تُسجَّل. الأقرب أولًا، وكل عملة في اقتراح واحد.
863:      */
864:     public static List<Suggestion> analyze(List<CountLine> lines, Map<String, BigDecimal> rates, Map<String, Integer> decimals) {
865:         List<Suggestion> all = new ArrayList<>();
866:         for (CountLine up : lines) {
867:             if (up.diff() <= 0) continue;
868:             for (CountLine down : lines) {
869:                 if (down.diff() >= 0 || down == up) continue;
870:                 BigDecimal gu = usd(up.diff(), up.currency, rates, decimals), su = usd(-down.diff(), down.currency, rates, decimals);
871:                 if (gu == null || su == null || gu.signum() == 0 || su.signum() == 0) continue;
872:                 BigDecimal gap = gu.subtract(su).abs(), tol = gu.max(su).multiply(new BigDecimal("0.03")).max(BigDecimal.ONE);
873:                 if (gap.compareTo(tol) > 0) continue;
874:                 Suggestion s = new Suggestion();
875:                 s.soldCur = down.currency;
876:                 s.sold = -down.diff();
877:                 s.gotCur = up.currency;
878:                 s.got = up.diff();
879:                 s.soldUsd = su;
880:                 s.gotUsd = gu;
881:                 s.rate = BigDecimal.valueOf(s.got, dec(decimals, s.gotCur)).divide(BigDecimal.valueOf(s.sold, dec(decimals, s.soldCur)), 6, RoundingMode.HALF_EVEN).stripTrailingZeros();
882:                 all.add(s);
883:             }
884:         }
885:         Collections.sort(all, (x, y) -> x.gapUsd().abs().compareTo(y.gapUsd().abs()));
886:         List<Suggestion> out = new ArrayList<>();
887:         Set<String> used = new java.util.HashSet<>();
888:         for (Suggestion s : all) {
889:             if (used.contains(s.soldCur) || used.contains(s.gotCur)) continue;
890:             used.add(s.soldCur);
891:             used.add(s.gotCur);
892:             out.add(s);
893:         }
894:         return out;
895:     }

897:     /** صافي فروق الجرد بالدولار (زيادة − عجز) والعملات بلا سعر. */
898:     public static BigDecimal netUsd(List<CountLine> lines, Map<String, BigDecimal> rates, Map<String, Integer> decimals, List<String> missing) {
899:         BigDecimal n = BigDecimal.ZERO;
900:         for (CountLine l : lines) {
901:             if (l.diff() == 0) continue;
902:             BigDecimal v = usd(Math.abs(l.diff()), l.currency, rates, decimals);
903:             if (v == null) {
904:                 if (missing != null) missing.add(l.currency);
905:                 continue;
906:             }
907:             n = l.diff() > 0 ? n.add(v) : n.subtract(v);
908:         }
909:         return n;
910:     }
