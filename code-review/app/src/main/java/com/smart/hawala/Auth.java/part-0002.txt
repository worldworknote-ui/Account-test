231:     /**
232:      * 12.0: قائمة بلا علامة {@link #DETAILED} (من إصدار أقدم أو من جهاز لم يُحدَّث): تُشتق الصلاحيات التفصيلية الجديدة
233:      * من القديمة كما كانت تعمل، فلا يفقد مستخدم شيئًا. ومع العلامة تُؤخذ كما اختارها المدير.
234:      */
235:     public static Set<String> expand(Set<String> s) {
236:         if (s.contains(DETAILED)) {
237:             if (s.contains(DETAILED126)) return s;
238:             // 12.0–12.5: «العملات» كانت تشمل المعادن، و«أسعار الصرف» كانت تشمل التحديث أونلاين
239:             Set<String> o = new LinkedHashSet<>(s);
240:             if (s.contains("CURRENCIES")) o.add(Perm.METALS.name());
241:             if (s.contains("RATES")) o.add(Perm.RATES_ONLINE.name());
242:             if (s.contains("CASH")) for (Perm p : new Perm[]{Perm.CASH_EXPENSE, Perm.CASH_FX, Perm.CASH_COUNT}) o.add(p.name());
243:             return o;
244:         }
245:         Set<String> o = new LinkedHashSet<>(s);
246:         if (s.contains("TX_ADD")) for (Perm p : new Perm[]{Perm.XFER, Perm.TX_WA, Perm.BACKDATE, Perm.HOLDS, Perm.SPECIAL}) o.add(p.name());
247:         if (s.contains("TX_ADD") && s.contains("TX_EDIT")) o.add(Perm.FX.name());   // كان القص يحتاج التسجيل والتعديل
248:         if (s.contains("VIEW")) for (Perm p : new Perm[]{Perm.STATEMENT, Perm.RECON, Perm.REPORTS, Perm.OWNER}) o.add(p.name());
249:         if (s.contains("CASH")) for (Perm p : new Perm[]{Perm.CLIENTS, Perm.CASH_EXPENSE, Perm.CASH_FX, Perm.CASH_COUNT}) o.add(p.name());
250:         if (s.contains("CURRENCIES")) for (Perm p : new Perm[]{Perm.RATES, Perm.METALS, Perm.RATES_ONLINE}) o.add(p.name());
251:         if (s.contains("SETTINGS")) o.add(Perm.LICENSE.name());
252:         return o;
253:     }
