70:     public static String norm(String s) {

72:         String x = Normalizer.normalize(s, Normalizer.Form.NFKC).replaceAll("[\\u064B-\\u065F\\u0670\\u0640]", "");

96:     private static final java.util.regex.Pattern RAJE_BASE = java.util.regex.Pattern.compile(
97:             "راجع\\s*(\\d+(?:[.,]\\d+)?)\\s*(الف|الاف|اف|الفين)?((?:\\s*\\+\\s*\\d+(?:[.,]\\d+)?\\s*(?:الف|الاف|اف)?)*)");
98:     private static final java.util.regex.Pattern PLUS_TERM = java.util.regex.Pattern.compile("\\+\\s*(\\d+(?:[.,]\\d+)?)\\s*(الف|الاف|اف)?");
99: 
100:     /**
101:      * المبلغ المذكور بعد «راجع» (حجم الحوالة التي هذه أجرتها) بوحدات العملة، أو null إن لم يُذكر.
102:      * «راجع154 الف» = 154,000 · «راجع 5000» = 5,000 · «راجع35 الف+ 50125» = 85,125.
103:      */
104:     public static BigDecimal rajeBase(String description) {
105:         String d = norm(description);
106:         StringBuilder ascii = new StringBuilder(d.length());
107:         for (char ch : d.toCharArray()) ascii.append(ch >= '٠' && ch <= '٩' ? (char) ('0' + ch - '٠') : ch);
108:         java.util.regex.Matcher m = RAJE_BASE.matcher(ascii);
109:         if (!m.find()) return null;
110:         BigDecimal v = term(m.group(1), m.group(2));
111:         java.util.regex.Matcher p = PLUS_TERM.matcher(m.group(3) == null ? "" : m.group(3));
112:         while (p.find()) v = v.add(term(p.group(1), p.group(2)));
113:         return v.signum() > 0 ? v : null;
114:     }

116:     private static BigDecimal term(String num, String unit) {
117:         BigDecimal v = new BigDecimal(num.replace(',', '.'));
118:         return unit != null && !unit.isEmpty() ? v.multiply(BigDecimal.valueOf(1000)) : v;
119:     }

121:     /**
122:      * نسبة الراجع الفعلية (بالمئة) = الأجرة ÷ المبلغ المذكور. إن جاءت أكبر من 10% دون كلمة «ألف»
123:      * فالأرجح أن المذكور بالآلاف («راجع33» على أجرة 66) فيُضرب بألف؛ وإن بقيت غير معقولة تُهمل.
124:      */
125:     public static BigDecimal rajeRate(Model.Tx t, int decimals) {
126:         BigDecimal base = rajeBase(t.description);
127:         if (base == null) return null;
128:         BigDecimal fee = BigDecimal.valueOf(t.amount, decimals);
129:         BigDecimal r = fee.multiply(BigDecimal.valueOf(100)).divide(base, 4, java.math.RoundingMode.HALF_UP);
130:         if (r.compareTo(BigDecimal.TEN) > 0) r = r.divide(BigDecimal.valueOf(1000), 4, java.math.RoundingMode.HALF_UP);
131:         return r.compareTo(BigDecimal.TEN) > 0 || r.signum() <= 0 ? null : r;
132:     }
