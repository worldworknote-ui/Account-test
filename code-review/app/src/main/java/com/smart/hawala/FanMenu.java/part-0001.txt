135:     /** المروحة أقل من نصف دائرة (من 195° إلى 345°) حتى لا تختفي الأزرار الطرفية خلف الشريط. */
136:     static final float START = 195f, SWEEP = 150f, END = START + SWEEP;

138:     @Override
139:     protected void onDraw(Canvas c) {
140:         int n = items.size();
141:         float[] cc = center.get();
142:         cx = cc[0];
143:         cy = cc[1];
144:         scrim.setColor(Color.argb((int) (175 * Math.min(1f, t * 1.4f)), 10, 22, 42));
145:         c.drawRect(0, 0, getWidth(), getHeight(), scrim);
146:         float R = outerR(), r0 = innerR(), sp = span();
147:         outer.set(cx - R, cy - R, cx + R, cy + R);
148:         inner.set(cx - r0, cy - r0, cx + r0, cy + r0);
149:         // ظل المروحة كاملة
150:         float allSweep = SWEEP * Math.min(1f, t);
151:         if (allSweep > 1) {
152:             p.reset();
153:             p.arcTo(outer, START, allSweep);
154:             p.arcTo(inner, START + allSweep, -allSweep);
155:             p.close();
156:             c.drawPath(p, shadow);
157:         }
158:         float[] starts = new float[n], lts = new float[n];
159:         for (int k = 0; k < n; k++) {
160:             // الشرائح تنفتح كالمروحة من اليسار: الشريحة تدور من 180° إلى موضعها، بتتابع خفيف
161:             int i = n - 1 - k; // i = ترتيب العنصر (0 على اليمين)
162:             float delay = k * 0.06f;
163:             float lt = Math.max(0f, Math.min(1f, (t - delay) / (1f - delay)));
164:             lts[i] = lt;
165:             float home = END - (i + 1) * sp; // موضعها النهائي
166:             starts[i] = START + (home - START) * lt;
167:         }
168:         // الشرائح أولًا ثم الأيقونات والكلمات فوقها كلها (حتى لا تغطي شريحةٌ كلمةَ جارتها)
169:         for (int k = 0; k < n; k++) {
170:             int i = n - 1 - k;
171:             float lt = lts[i];
172:             if (lt <= 0) continue;
173:             float alpha = Math.min(1f, lt * 1.6f);
174:             p.reset();
175:             p.arcTo(outer, starts[i], sp);
176:             p.arcTo(inner, starts[i] + sp, -sp);
177:             p.close();
178:             int[] pal = st.fanStyle == 1 ? new int[]{Color.rgb(28, 58, 102), Color.rgb(36, 70, 120), Color.rgb(52, 92, 150)}
179:                     : st.fanStyle == 2 ? new int[]{Color.rgb(36, 39, 46), Color.rgb(48, 52, 60), Color.rgb(70, 76, 88)}
180:                     : new int[]{Color.rgb(250, 251, 253), Color.rgb(236, 241, 248), Color.rgb(222, 231, 245)};
181:             int base = i == pressed ? pal[2] : pal[i % 2];
182:             wedge.setColor(base);
183:             wedge.setAlpha((int) (255 * alpha));
184:             c.drawPath(p, wedge);
185:             edge.setAlpha((int) (120 * alpha));
186:             c.drawPath(p, edge);
187:         }
188:         for (int i = 0; i < n; i++) {
189:             float lt = lts[i];
190:             if (lt <= 0) continue;
191:             float alpha = Math.min(1f, lt * 1.6f);
192:             Item it = items.get(i);
193:             // الأيقونة في دائرة ملوّنة والكلمة تحتها (معتدلة دائمًا للقراءة)
194:             double mid = Math.toRadians(starts[i] + sp / 2f);
