730:     /** سطر مقارنة سعرك بسعر السوق (أسعار الإنترنت أو اليدوية)، أو null بلا سعر. */
731:     String marketLine(String cur, String counter, long amount, long counterAmount, boolean buy) {
732:         BigDecimal mk = CashBox.crossRate(cur, counter, a.db.usdRates());
733:         if (mk == null || amount <= 0 || counterAmount <= 0) return null;
734:         int d1 = a.db.decimals(cur), d2 = a.db.decimals(counter);
735:         BigDecimal mine = BigDecimal.valueOf(counterAmount, d2).divide(BigDecimal.valueOf(amount, d1), 10, RoundingMode.HALF_EVEN);
736:         BigDecimal atMarket = BigDecimal.valueOf(amount, d1).multiply(mk);
737:         BigDecimal diff = BigDecimal.valueOf(counterAmount, d2).subtract(atMarket);
738:         if (buy) diff = diff.negate();
739:         BigDecimal pct = mine.subtract(mk).multiply(BigDecimal.valueOf(100)).divide(mk, 2, RoundingMode.HALF_EVEN);
740:         String sign = diff.signum() >= 0 ? "لصالحك ≈ " : "عليك ≈ ";
741:         return "السوق " + iso("1 " + cur + " = " + mk.toPlainString() + " " + counter) + "\n" + sign
742:                 + iso(diff.abs().setScale(Math.min(d2, 2), RoundingMode.HALF_EVEN).toPlainString() + " " + counter) + " (" + iso(pct.abs().toPlainString() + "%") + ")";
743:     }
