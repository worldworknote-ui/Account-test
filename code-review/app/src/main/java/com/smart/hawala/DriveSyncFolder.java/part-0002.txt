323:         c.setRequestProperty("Content-Type", "application/json; charset=UTF-8");
324:         c.setRequestProperty("X-Upload-Content-Type", "application/octet-stream");
325:         c.setRequestProperty("X-Upload-Content-Length", "" + data.length);
