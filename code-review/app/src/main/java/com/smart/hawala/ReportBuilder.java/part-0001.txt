74:         public JSONObject toJson() {
75:             JSONObject j = new JSONObject();
76:             try {
77:                 j.put("title", title).put("from", from).put("to", to).put("direction", direction).put("kind", kind).put("status", status)
78:                         .put("text", text).put("service", service).put("dest", dest).put("user", user).put("group", group).put("group2", group2)
79:                         .put("measure", measure).put("sort", sort).put("top", top).put("compare", compare).put("details", details).put("running", runningBalance);
80:                 if (minAmount != null) j.put("min", minAmount.toPlainString());
81:                 if (maxAmount != null) j.put("max", maxAmount.toPlainString());
82:                 JSONArray o = new JSONArray();
83:                 for (long x : offices) o.put(x);
84:                 j.put("offices", o);
85:                 JSONArray c = new JSONArray();
86:                 for (String x : currencies) c.put(x);
87:                 j.put("currencies", c);
88:             } catch (Exception ignored) {
89:                 // لا يحدث مع هذه الأنواع
90:             }
91:             return j;
92:         }

94:         public static Spec fromJson(JSONObject j) {
95:             Spec s = new Spec();
96:             s.title = j.optString("title", "");
97:             s.from = j.optLong("from", 0);
98:             s.to = j.optLong("to", 0);
99:             s.direction = clamp(j.optInt("direction", 0), DIR_NAMES.length);
100:             s.kind = clamp(j.optInt("kind", 0), KIND_NAMES.length);
101:             s.status = clamp(j.optInt("status", 0), STATUS_NAMES.length);
102:             s.text = j.optString("text", "");
103:             s.service = j.optString("service", "");
104:             s.dest = j.optString("dest", "");
105:             s.user = j.optString("user", "");
106:             s.group = clamp(j.optInt("group", G_OFFICE), GROUP_NAMES.length);
107:             s.group2 = clamp(j.optInt("group2", 0), GROUP_NAMES.length);
108:             s.measure = clamp(j.optInt("measure", M_NET), MEASURE_NAMES.length);
109:             s.sort = clamp(j.optInt("sort", 0), SORT_NAMES.length);
110:             s.top = Math.max(0, j.optInt("top", 0));
111:             s.compare = j.optBoolean("compare", false);
112:             s.details = j.optBoolean("details", false);
113:             s.runningBalance = j.optBoolean("running", true);
114:             try {
115:                 if (j.has("min")) s.minAmount = new BigDecimal(j.getString("min"));
116:                 if (j.has("max")) s.maxAmount = new BigDecimal(j.getString("max"));
117:             } catch (Exception ignored) {
118:                 // حد غير صالح يُهمل
119:             }
120:             JSONArray o = j.optJSONArray("offices");
121:             for (int i = 0; o != null && i < o.length(); i++) s.offices.add(o.optLong(i));
122:             JSONArray c = j.optJSONArray("currencies");
123:             for (int i = 0; c != null && i < c.length(); i++) s.currencies.add(c.optString(i));
124:             return s;
125:         }
