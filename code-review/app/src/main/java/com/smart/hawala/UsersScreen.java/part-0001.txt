49:     static void startLock(Db db, long ms) {
50:         db.setSettingQuiet("lock_start_rt", "" + SystemClock.elapsedRealtime());
51:         db.setSettingQuiet("lock_ms", "" + ms);
52:     }

54:     static long remaining(Db db) {
55:         long ms = db.settingLong("lock_ms", 0);
56:         if (ms <= 0) return 0;
57:         long now = SystemClock.elapsedRealtime(), start = db.settingLong("lock_start_rt", 0);
58:         if (now < start) {
59:             start = now;
60:             db.setSettingQuiet("lock_start_rt", "" + start);
61:         }
62:         return Math.max(0, ms - (now - start));
63:     }
64: 
65:     private long lockRemaining() {
66:         return remaining(a.db);
67:     }

69:     /** يسجّل محاولة خاطئة ويعيد عدد المحاولات المتتالية. */
70:     private int recordFail(String who, long userId) {
71:         int f = (int) a.db.settingLong("login_fails", 0) + 1;
72:         a.db.setSettingQuiet("login_fails", "" + f);
73:         long ms = Pin.lockoutMillis(f);
74:         if (ms > 0) startLock(a.db, ms);
75:         String prev = Db.currentUserName;
76:         Db.currentUserName = who;
77:         a.db.audit("LOGIN_FAILED", "user", userId, "محاولة " + f);
78:         Db.currentUserName = prev;
79:         return f;
80:     }
