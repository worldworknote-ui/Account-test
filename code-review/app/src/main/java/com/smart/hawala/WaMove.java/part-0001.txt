19: /**
20:  * 12.1: نقل الرسائل من واتساب (أو أي برنامج) إلى التسجيل بأسهل طريقة على كل جهاز:
21:  * <ul>
22:  * <li>الحافظة بكل صيغها (رؤوس واتساب، فقاعات HTML، أسطر الوقت) — {@link WaClip}.</li>
23:  * <li>«قص هنا» يدوي حين يبقى النص ملزوقًا.</li>
24:  * <li>«وضع التجميع»: كل نسخ رسالة تصير بندًا مستقلًا، ثم «نقل N رسائل».</li>
25:  * <li>مراقبة الحافظة على الكمبيوتر: بطاقة «في الحافظة N رسائل من واتساب».</li>
26:  * <li>«إطار الالتقاط» على ويندوز وماك ({@link Capture}): يرقّم الفقاعات تحت الإطار، فتضغط الرقم فينتقل نصها.</li>
27:  * <li>«فحص الحافظة»: شكل ما نُسخ دون محتواه، للدعم.</li>
28:  * </ul>
29:  * لا يُسجَّل شيء تلقائيًا: كل رسالة تمر بنموذج التسجيل وتأكيد المحاسب، وصلاحية «التسجيل من واتساب» (TX_WA) شرط في كل طريق.

83:     static volatile Capture capture;

87:     /**
88:      * 12.6: في المجانية الالتقاط والتجميع والقص ضمن الرسائل اليومية المسموحة (20): بعدها لا تبدأ، ويبقى الإدخال اليدوي.
89:      * يعيد عدد المتبقي اليوم (Long.MAX_VALUE في الكاملة)، أو 0 مع رسالة.
90:      */
91:     long waRemaining(boolean tell) {
92:         long[] u = a.db.waUsage();
93:         if (u[1] == Entitlements.UNLIMITED) return Long.MAX_VALUE;
94:         long left = Math.max(0, u[1] - u[0]);
95:         if (left == 0 && tell) a.error(new Entitlements.LimitReached(Entitlements.waLimitMessage(u[1])));
96:         return left;
97:     }

99:     void openCapture(boolean whole) {
100:         if (!a.require(Auth.Perm.TX_WA)) return;
101:         if (waRemaining(true) == 0) return;
102:         Capture c = capture;
103:         if (c == null) {
104:             a.toast("إطار الالتقاط متاح على ويندوز وماك فقط. على الهاتف: حدّد الرسائل في واتساب ← مشاركة ← المحاسب الذكي.");
105:             return;
106:         }
107:         String why = c.unavailable();
108:         if (why != null && !why.isEmpty()) {
109:             new AlertDialog.Builder(a).setTitle("إطار الالتقاط غير متاح").setMessage(why).setPositiveButton("حسنًا", null).show();
110:             return;
111:         }
112:         c.open(whole, this::fromCapture, prefs());
113:     }

115:     /** فقاعات من الإطار: إلى خانة النموذج المفتوح إن وُجد، وإلا إلى نموذج جديد أو دفعة. */
116:     void fromCapture(List<WaText.Msg> msgs, String chat) {
117:         if (msgs == null || msgs.isEmpty()) return;
118:         if (!a.require(Auth.Perm.TX_WA)) return;
119:         long left = waRemaining(true);   // 12.6: المجانية ضمن المتبقي اليوم
120:         if (left == 0) return;
121:         if (msgs.size() > left) {
122:             msgs = new ArrayList<>(msgs.subList(0, (int) left));
123:             a.toast("النسخة المجانية: بقي لك اليوم " + left + " رسالة للاستخراج، فنُقل هذا العدد فقط.");
124:         }
125:         Model.Office sug = officeFromChat(chat);
126:         if (msgs.size() == 1 && a.txScreen.formOpen()) {
127:             a.txScreen.captured(msgs.get(0), sug);
128:             return;
129:         }
130:         a.beforeScreenChange();
131:         if (msgs.size() == 1) a.txScreen.showCaptured(msgs.get(0), sug);
132:         else {
133:             a.waQueue.suggested = sug;
134:             a.waQueue.suggestedFrom = chat;
135:             a.waQueue.overview(msgs);
136:         }
137:     }
138: 
139:     Model.Office officeFromChat(String chat) {
140:         if (chat == null || chat.trim().isEmpty()) return null;
141:         List<Model.Office> offices = new ArrayList<>();
142:         List<String> names = new ArrayList<>();
143:         for (Model.Office o : a.db.offices()) {
144:             if (!o.active) continue;
145:             offices.add(o);
146:             names.add(o.name);
147:         }
148:         int i = Bubbles.matchOffice(chat, names);
149:         return i < 0 ? null : offices.get(i);
150:     }

164:     /** رسالة واحدة ← نموذج (مع عرض «القص» إن بدا النص ملزوقًا)، عدة رسائل ← دفعة. */
165:     void route(List<WaText.Msg> msgs, String raw) {
166:         if (msgs.isEmpty()) {
167:             a.txScreen.showNew();
168:             return;
169:         }
170:         if (msgs.size() > 1) {
171:             if (!a.can(Auth.Perm.TX_WA)) {
172:                 a.txScreen.showShared(new WaText.Msg("", "", raw.trim()));
173:                 return;
174:             }
175:             a.waQueue.overview(msgs);
176:             return;
177:         }
178:         if (a.can(Auth.Perm.TX_WA) && WaClip.maybeGlued(msgs.get(0).body)) {
179:             final WaText.Msg one = msgs.get(0);
180:             new AlertDialog.Builder(a).setTitle("هل هذا النص أكثر من رسالة؟")
181:                     .setMessage("يبدو أن النص فيه أكثر من رسالة ملزوقة (يحصل عند النسخ من واتساب على ويندوز). اقطعه بنفسك عند حدود الرسائل، أو سجّله كما هو.")
182:                     .setPositiveButton("✂ قص الرسائل", (d, w) -> cutScreen(one.body))
183:                     .setNeutralButton("رسالة واحدة", (d, w) -> a.txScreen.showShared(one))
184:                     .setNegativeButton("إلغاء", null).show();
185:             return;
186:         }
187:         a.txScreen.showShared(msgs.get(0));
188:     }
