649:     /** المفتاح الأصلي لمفتاح شخصي («u12.tabs» ← «tabs»)، وإلا المفتاح نفسه. */
650:     public static String baseKey(String k) {
651:         if (k == null || k.length() < 3 || k.charAt(0) != 'u' || !Character.isDigit(k.charAt(1))) return k;
652:         int dot = k.indexOf('.');
653:         if (dot < 0) return k;
654:         for (int i = 1; i < dot; i++) if (!Character.isDigit(k.charAt(i))) return k;
655:         return k.substring(dot + 1);
656:     }

704:      * 10.18: يحمّل «توقيت المكتب» (إعداد عام يتزامن: office_tz) وفرق تصحيح الساعة المحفوظ على هذا الجهاز (clock_offset_ms)،

706:      */
707:     void loadClock(SQLiteDatabase d) {
708:         try (Cursor c = d.rawQuery("SELECT k,v FROM settings WHERE k IN ('office_tz','clock_offset_ms')", null)) {
709:             String tz = "";
710:             long off = 0;
711:             while (c.moveToNext()) {
712:                 if ("office_tz".equals(c.getString(0))) tz = Model.nz(c.getString(1));
713:                 else try {
714:                     off = Long.parseLong(c.getString(1));
715:                 } catch (Exception ignored) {
716:                     // قيمة تالفة = بلا تصحيح
717:                 }
718:             }
719:             Model.setZone(tz);
720:             if (Clock.learnedAt() == 0) Clock.load(off);
721:         } catch (Exception ignored) {
722:             // قاعدة قيد الإنشاء
723:         }
724:         Clock.saver = o -> {
725:             try {
726:                 getWritableDatabase().execSQL("INSERT OR REPLACE INTO settings(k,v) VALUES('clock_offset_ms',?)", new Object[]{"" + o});
727:             } catch (Throwable ignored) {
728:                 // يُحفظ في المرة التالية
729:             }
730:         };
731:     }

742:         long off = Clock.offset();

745:         else {
746:             long sec = Math.abs(off) / 1000;
747:             String d = sec >= 3600 ? (sec / 3600) + " ساعة و" + (sec % 3600 / 60) + " دقيقة" : sec >= 60 ? (sec / 60) + " دقيقة" : sec + " ثانية";
748:             l.add("ساعة هذا الجهاز " + (off > 0 ? "متأخرة" : "متقدمة") + " " + d + " عن الوقت الصحيح: يصحّحها البرنامج تلقائيًا في كل ما يسجّله ويعرضه.");
