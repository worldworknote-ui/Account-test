687:     static ContentValues values(Db db, SQLiteDatabase d, String table, JSONObject r, Map<String, Set<String>> cache) throws Exception {
688:         Set<String> local = columns(d, table, cache);
689:         List<String> sens = Db.SENSITIVE.get(table);
690:         ContentValues v = new ContentValues();
691:         java.util.Iterator<String> it = r.keys();
692:         while (it.hasNext()) {
693:             String col = it.next();
694:             if (col.startsWith("_") || !local.contains(col.toLowerCase(Locale.US))) continue;
695:             Object val = r.get(col);
696:             if (val == JSONObject.NULL) v.putNull(col);
697:             else if (val instanceof Number) {
698:                 if (val instanceof Double || val instanceof Float || val instanceof java.math.BigDecimal) {
699:                     double dv = ((Number) val).doubleValue();
700:                     if (dv == Math.rint(dv) && Math.abs(dv) < 9.0E15) v.put(col, (long) dv);
701:                     else v.put(col, dv);
702:                 } else v.put(col, ((Number) val).longValue());
703:             } else if (val instanceof Boolean) v.put(col, ((Boolean) val) ? 1L : 0L);
704:             else {
705:                 String sv = String.valueOf(val);
706:                 if (sv.startsWith("b64:")) v.put(col, Base64.getDecoder().decode(sv.substring(4)));
707:                 else if (table.equals("settings") && col.equals("v") && Db.ENCRYPTED_SETTINGS.contains(r.optString("k"))) v.put(col, Crypto.enc(sv));
708:                 else v.put(col, sens != null && sens.contains(col) ? Crypto.enc(sv) : sv);
709:             }
710:         }
711:         if (table.equals("attachments")) {
712:             String path = null;
713:             try (Cursor c = d.rawQuery("SELECT path FROM attachments WHERE id=?", new String[]{"" + r.getLong("id")})) {
714:                 if (c.moveToFirst()) path = c.getString(0);
715:             }
716:             if (path != null && !new File(path).isFile() && db.attachmentFile(path).isFile()) path = db.attachmentFile(path).getAbsolutePath();
717:             if (path == null || !new File(path).isFile()) {
718:                 if (!r.has("_file")) return null;
719:                 File dir = db.evidenceDir();
720:                 if (!dir.exists() && !dir.mkdirs()) throw new IOException("تعذر إنشاء مجلد الصور");
721:                 File f = new File(dir, UUID.randomUUID() + ".evd");
722:                 try (FileOutputStream o = new FileOutputStream(f)) {
723:                     o.write(Crypto.encBytes(Base64.getDecoder().decode(r.getString("_file"))));
724:                     o.getFD().sync();
725:                 }
726:                 path = f.getAbsolutePath();
727:             }
728:             v.put("path", path);
729:         }
730:         return v;
731:     }
