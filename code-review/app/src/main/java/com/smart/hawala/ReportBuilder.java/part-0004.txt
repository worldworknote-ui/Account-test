522:     static void pivotSections(Report r, Spec s, Data d, Result res) {
523:         List<String> cols = new ArrayList<>(res.pivotLabels.keySet());
524:         int maxCols = 24;
525:         boolean more = cols.size() > maxCols;
526:         if (more) {
527:             // أكبر الأعمدة حسب العدد
528:             final Map<String, Long> weight = new HashMap<>();
529:             for (Map<String, Map<String, Agg>> row : res.pivot.values())
530:                 for (Map.Entry<String, Map<String, Agg>> c : row.entrySet()) weight.merge(c.getKey(), sum(c.getValue(), M_COUNT), Long::sum);
531:             cols.sort((a, b) -> Long.compare(weight.getOrDefault(b, 0L), weight.getOrDefault(a, 0L)));
532:             cols = new ArrayList<>(cols.subList(0, maxCols));
533:             Collections.sort(cols);
534:         }
535:         boolean count = s.measure == M_COUNT;
536:         Set<String> curs = count ? Collections.singleton("") : res.totals.keySet();
537:         for (String c : curs) {
538:             String head = MEASURE_NAMES[s.measure] + (count ? "" : " بـ" + c) + " — " + GROUP_NAMES[s.group] + " × " + GROUP_NAMES[s.group2];
539:             Report.Section p = r.section(head, "جدول " + (count ? "العدد" : c));
540:             p.col(GROUP_NAMES[s.group], 1.4f, Report.TEXT);
541:             int type = count ? Report.INT : (s.measure == M_US || s.measure == M_THEM || s.measure == M_VOLUME ? Report.MONEY : Report.BAL);
542:             for (String k : cols) p.col(res.pivotLabels.get(k), 0.9f, type);
543:             if (more) p.col("أخرى", 0.9f, type);
544:             p.col("المجموع", 1f, type);
545:             int dc = count ? 0 : dec(d, c);
546:             long[] colTotals = new long[cols.size() + 2];
547:             for (String g : res.groups.keySet()) {
548:                 Map<String, Map<String, Agg>> row = res.pivot.getOrDefault(g, new HashMap<>());
549:                 Object[] cells = new Object[cols.size() + 2 + (more ? 1 : 0)];
550:                 cells[0] = res.labels.get(g);
551:                 long total = 0, other = 0;
552:                 for (Map.Entry<String, Map<String, Agg>> e : row.entrySet()) {
553:                     long v = count ? sum(e.getValue(), M_COUNT) : e.getValue().getOrDefault(c, new Agg()).measure(s.measure);
554:                     total += v;
555:                     int i = cols.indexOf(e.getKey());
556:                     if (i < 0) other += v;
557:                 }
558:                 for (int i = 0; i < cols.size(); i++) {
559:                     Map<String, Agg> m = row.get(cols.get(i));
560:                     long v = m == null ? 0 : count ? sum(m, M_COUNT) : m.getOrDefault(c, new Agg()).measure(s.measure);
561:                     cells[1 + i] = cell(type, v, dc);
562:                     colTotals[i] += v;
563:                 }
564:                 int at = 1 + cols.size();
565:                 if (more) {
566:                     cells[at++] = cell(type, other, dc);
567:                     colTotals[cols.size()] += other;
568:                 }
569:                 cells[at] = cell(type, total, dc);
570:                 colTotals[cols.size() + 1] += total;
571:                 p.row(Report.NORMAL, cells);
572:             }
573:             Object[] tot = new Object[cols.size() + 2 + (more ? 1 : 0)];
574:             tot[0] = "المجموع";
575:             for (int i = 0; i < cols.size(); i++) tot[1 + i] = cell(type, colTotals[i], dc);
576:             int at = 1 + cols.size();
577:             if (more) tot[at++] = cell(type, colTotals[cols.size()], dc);
578:             tot[at] = cell(type, colTotals[cols.size() + 1], dc);
579:             p.row(Report.TOTAL, tot);
580:             if (more) p.note = "أكبر " + maxCols + " عمودًا بعدد العمليات، والباقي في «أخرى».";
581:         }
582:     }
