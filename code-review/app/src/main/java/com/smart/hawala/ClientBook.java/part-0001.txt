13: /**
14:  * 9.0.1: «حسابات العملاء» داخل الصندوق — منطق بحت مشترك.
15:  *
16:  * <p>العميل شخص يتعامل مع الصندوق مباشرة وله حساب لدينا مستقل تمامًا عن حسابات المكاتب ودفتر قيودها.
17:  * لكل عميل رصيد بكل عملة = مجموع (اتجاه × مبلغ) لحركاته النشطة: الموجب «لنا» (العميل مدين لنا)، والسالب «علينا» (له عندنا).
18:  *
19:  * <ul>
20:  *   <li>استلام نقد منه (يدخل الصندوق) = علينا له (−).</li>
21:  *   <li>تسليم نقد له (يخرج من الصندوق) = لنا عليه (+).</li>
22:  *   <li>قد يُقيَّد على حسابه بعملة غير عملة النقد (له 100 يورو فيأخذ دولارًا): النقد بالدولار، وحركة حسابه باليورو، والسعر محفوظ.</li>
23:  *   <li>قيد بلا نقد (رصيد سابق أو تسوية متفق عليها) بجهة يختارها المستخدم.</li>
24:  *   <li>تبديل عملة الرصيد بلا نقد: سطران مرتبطان (يُطفأ جزء من رصيد عملة ويُنشأ مقابله بعملة أخرى بالجهة نفسها).</li>
25:  * </ul>

60:     /** الرصيد لكل عملة (الموجب لنا والسالب علينا)، بلا الأصفار. */
61:     public static Map<String, Long> balances(List<Move> moves) {
62:         Map<String, Long> m = new TreeMap<>();
63:         for (Move x : moves) {
64:             long e = x.effect();
65:             if (e == 0) continue;
66:             Long b = m.get(x.currency);
67:             m.put(x.currency, (b == null ? 0 : b) + e);
68:         }
69:         m.values().removeIf(v -> v == 0);
70:         return m;
71:     }

73:     /** أرصدة كل العملاء: العميل ← (العملة ← الرصيد). */
74:     public static Map<Long, Map<String, Long>> allBalances(List<Move> moves) {
75:         Map<Long, List<Move>> by = new TreeMap<>();
76:         for (Move x : moves) {
77:             List<Move> l = by.get(x.clientId);
78:             if (l == null) {
79:                 l = new ArrayList<>();
80:                 by.put(x.clientId, l);
81:             }
82:             l.add(x);
83:         }
84:         Map<Long, Map<String, Long>> out = new TreeMap<>();
85:         for (Map.Entry<Long, List<Move>> e : by.entrySet()) out.put(e.getKey(), balances(e.getValue()));
86:         return out;
87:     }

89:     /** مجموع «لنا» و«علينا» لكل عملة عبر العملاء: [لنا، علينا، عدد عملاء لنا، عدد عملاء علينا]. */
90:     public static Map<String, long[]> totals(Map<Long, Map<String, Long>> all) {
91:         Map<String, long[]> t = new TreeMap<>();
92:         for (Map<String, Long> b : all.values())
93:             for (Map.Entry<String, Long> e : b.entrySet()) {
94:                 long[] v = t.get(e.getKey());
95:                 if (v == null) {
96:                     v = new long[4];
97:                     t.put(e.getKey(), v);
98:                 }
99:                 if (e.getValue() > 0) {
100:                     v[0] += e.getValue();
101:                     v[2]++;
102:                 } else if (e.getValue() < 0) {
103:                     v[1] -= e.getValue();
104:                     v[3]++;
105:                 }
106:             }
107:         return t;
108:     }

116:     /** iso: عزل الأرقام باتجاهها (للشاشة). */
117:     public static String text(Move x, Map<String, Integer> decimals, boolean iso) {
118:         String l = iso ? "\u2066" : "", r = iso ? "\u2069" : "";
119:         switch (x.kind) {
120:             case CASH: {
121:                 boolean got = x.cashDir > 0;
122:                 String t = got ? "استلمنا منه نقدًا" : "سلّمناه نقدًا";
123:                 if (x.cashCurrency != null && !x.cashCurrency.equals(x.currency))
124:                     t += " " + l + Model.fmt(x.cashAmount, CashBox.dec(decimals, x.cashCurrency)) + " " + x.cashCurrency + r
125:                             + (Model.nz(x.rate).isEmpty() ? "" : iso ? "\nبسعر " + l + "1 " + x.currency + " = " + x.rate + " " + x.cashCurrency + r
126:                             : " (1 " + x.currency + " = " + x.rate + " " + x.cashCurrency + ")");
127:                 return t;
128:             }
129:             case SWAP: return "تبديل عملة الرصيد";
130:             default: return x.dir > 0 ? "قيد لنا عليه" : "قيد علينا له";
131:         }

146:     public static String validateLink(String cashCurrency, long cashAmount, String clientCurrency, long clientAmount) {
147:         if (clientCurrency == null || clientCurrency.isEmpty()) return "اختر عملة حساب العميل";
148:         if (clientAmount <= 0) return "اكتب المبلغ الذي يُقيَّد على حساب العميل";
149:         if (clientCurrency.equals(cashCurrency) && clientAmount != cashAmount) return "بالعملة نفسها يُقيَّد على العميل المبلغ نفسه";

153:     /** التبديل: يُسمح بتبديل ما لا يزيد عن رصيد العملة، وبالجهة نفسها. يرجع null أو السبب. */
154:     public static String validateSwap(Map<String, Long> bal, String from, long amount, String to, long toAmount) {
155:         if (from == null || to == null) return "اختر العملتين";
156:         if (from.equals(to)) return "التبديل بين عملتين مختلفتين";
157:         Long b = bal.get(from);
158:         if (b == null || b == 0) return "لا رصيد للعميل بـ" + from + " لتبديله";
159:         if (amount <= 0 || toAmount <= 0) return "اكتب المبلغ والسعر";
160:         if (amount > Math.abs(b)) return "المبلغ أكبر من رصيد العميل بـ" + from;
161:         return null;
162:     }

164:     /** نص سعر: كم من cur2 لكل 1 من cur1 (6 منازل كحد أقصى). */
165:     public static String rateOf(long a1, int d1, long a2, int d2) {
166:         if (a1 <= 0 || a2 <= 0) return "";
167:         return BigDecimal.valueOf(a2, d2).divide(BigDecimal.valueOf(a1, d1), 6, RoundingMode.HALF_EVEN).stripTrailingZeros().toPlainString();
