498:     /**
499:      * رصيد متوقف: رصيد بعملة لم تتحرك (بتلك العملة) منذ أكثر من minDays يومًا، ولو كان المكتب نشطًا بعملة أخرى.
500:      * days = أيام أقدم رصيد متوقف في المكتب.
501:      */
502:     public static List<Dormant> dormant(Input in, int minDays) {
503:         Map<String, Long> last = new HashMap<>();
504:         List<Object[]> fx = new ArrayList<>();
505:         for (Model.Tx t : in.txs) {
506:             if (!active(t)) continue;
507:             fx.clear();
508:             effects(t, fx);
509:             for (Object[] e : fx) last.merge(t.officeId + "|" + e[0], t.createdAt, Math::max);
510:         }
511:         for (Opening o : in.openings) last.merge(o.office + "|" + o.currency, o.when, Math::max);
512:         List<Dormant> out = new ArrayList<>();
513:         for (Map.Entry<Long, Map<String, Long>> e : balances(in).entrySet()) {
514:             Dormant d = new Dormant();
515:             d.office = e.getKey();
516:             for (Map.Entry<String, Long> b : e.getValue().entrySet()) {
517:                 Long l = last.get(e.getKey() + "|" + b.getKey());
518:                 if (b.getValue() == 0 || l == null) continue;
519:                 long days = (in.now - l) / DAY;
520:                 if (days <= minDays) continue;
521:                 d.balances.put(b.getKey(), b.getValue());
522:                 d.days = Math.max(d.days, days);
523:             }
524:             if (!d.balances.isEmpty()) out.add(d);
525:         }
526:         Collections.sort(out, (a, b) -> Long.compare(b.days, a.days));
527:         return out;
528:     }

549:     public static final class Trend {
550:         public long office, last, prev;
551:         /** rising، declining، new، stopped، back. */
552:         public String kind;
553:     }
554: 
555:     public static List<Trend> trends(Input in, int window) {
556:         long w = window * DAY;
557:         Map<Long, long[]> m = new HashMap<>();   // {آخر، قبلها، أول حركة}
558:         for (Model.Tx t : in.txs) {
559:             if (!active(t)) continue;
560:             long[] x = m.computeIfAbsent(t.officeId, k -> new long[]{0, 0, Long.MAX_VALUE});
561:             long age = in.now - t.createdAt;
562:             if (age >= 0 && age < w) x[0]++;
563:             else if (age >= w && age < 2 * w) x[1]++;
564:             x[2] = Math.min(x[2], t.createdAt);
565:         }
566:         List<Trend> out = new ArrayList<>();
567:         for (Map.Entry<Long, long[]> e : m.entrySet()) {
568:             long[] x = e.getValue();
569:             Trend t = new Trend();
570:             t.office = e.getKey();
571:             t.last = x[0];
572:             t.prev = x[1];
573:             if (x[1] == 0 && x[0] > 0) t.kind = in.now - x[2] < w ? "new" : "back";
574:             else if (x[1] > 0 && x[0] == 0) t.kind = "stopped";
575:             else if (x[1] > 0 && x[0] * 3 >= x[1] * 4) t.kind = "rising";
576:             else if (x[1] > 0 && x[0] * 3 <= x[1] * 2) t.kind = "declining";
577:             else continue;
578:             out.add(t);
579:         }
580:         Collections.sort(out, (a, b) -> Long.compare(Math.abs(b.last - b.prev), Math.abs(a.last - a.prev)));
581:         return out;
582:     }
