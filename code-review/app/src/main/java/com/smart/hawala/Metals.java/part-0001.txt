15: /**
16:  * 12.5: المعادن الثمينة (ذهب، فضة، بلاتين، بلاديوم…) حسابات بالوزن كالعملات تمامًا.
17:  * <p>
18:  * المنطق الصارم:
19:  * <ul>
20:  * <li>كل معدن بعياره حساب مستقل (ذهب 21 غير ذهب 24): لا تحويل خفي بينها؛ التحويل بـ«قص» بنسبة ظاهرة.</li>
21:  * <li>الوحدة (غرام، كيلو، أونصة، مثقال…) ومنازلها تُختار عند الإنشاء ولا تتغير بعد أول حركة (تغييرها يغيّر معنى كل وزن مسجّل).</li>
22:  * <li>الوزن في الرسالة بوحدة أخرى («2 كيلو») يُحوَّل إلى وحدة الحساب بحساب دقيق، ويُنبَّه المحاسب إن احتاج تقريبًا.</li>
23:  * <li>العيار غير الواضح حين يكون للمعدن أكثر من حساب: لا يُخمَّن (يختاره المحاسب).</li>
24:  * <li>التقويم بالدولار اختياري لكل معدن (يختاره المحاسب)، بسعر يدخله لوحدة واحدة، وسعر أونلاين للاستئناس إن توفر.</li>
25:  * </ul>

81:     /** العيار بالألف من القيراط الذهبي (24 = 999، 21 = 875…)، أو ‰ نفسه إن كتبه هكذا. */
82:     public static int finenessOfKarat(int k) {
83:         switch (k) {
84:             case 24: return 999;
85:             case 23: return 958;
86:             case 22: return 916;
87:             case 21: return 875;
88:             case 20: return 833;
89:             case 18: return 750;
90:             case 16: return 666;
91:             case 14: return 585;
92:             case 12: return 500;
93:             case 10: return 417;
94:             case 9: return 375;
95:             case 8: return 333;
96:             default: return 0;
97:         }
98:     }

100:     public static int karatOf(int fineness) {
101:         for (int k = 24; k >= 8; k--) if (finenessOfKarat(k) == fineness) return k;
102:         return 0;
103:     }

105:     /** وصف العيار للعرض: «عيار 21» للذهب، «925» لغيره. */
106:     public static String purityLabel(String base, int fineness) {
107:         if (fineness <= 0) return "";
108:         int k = karatOf(fineness);
109:         return GOLD.equals(base) && k > 0 ? "عيار " + k : String.valueOf(fineness);
110:     }

112:     /** رمز مقترح: XAU21، XAG925، XPT950… */
113:     public static String suggestCode(String base, int fineness) {
114:         if (fineness <= 0) return base;
115:         int k = karatOf(fineness);
116:         return base + (GOLD.equals(base) && k > 0 ? String.valueOf(k) : String.valueOf(fineness));
117:     }

143:      * 12.6: رمز مقترح لا يتكرر مع الموجود: XAU21، ثم XAU21K لوحدة أخرى، ثم XAU21K2… حتى لا يحتاج المحاسب أن يفكر بالرمز
144:      * ليضيف عيارًا ثانيًا.
145:      */
146:     public static String uniqueCode(String base, int fineness, String unitId, java.util.Collection<String> taken) {
147:         java.util.Set<String> t = new java.util.HashSet<>();
148:         for (String x : taken) t.add(x.toUpperCase(Locale.US));
149:         String c = suggestCode(base, fineness);
150:         if (!t.contains(c)) return c;
151:         String c2 = c + unitSuffix(unitId);
152:         if (!t.contains(c2) && c2.length() <= 10) return c2;
153:         for (int i = 2; i < 100; i++) {
154:             String c3 = (c2.length() <= 8 ? c2 : c2.substring(0, 8)) + i;
155:             if (!t.contains(c3)) return c3;
156:         }
157:         return c2;
158:     }
