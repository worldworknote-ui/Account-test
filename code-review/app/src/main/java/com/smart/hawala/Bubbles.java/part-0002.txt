179:     static List<Bubble> byShapes(List<Node> shapes, List<Node> lines) {
180:         List<Node> rest = new ArrayList<>();
181:         List<List<Node>> in = new ArrayList<>();
182:         for (int i = 0; i < shapes.size(); i++) in.add(new ArrayList<>());
183:         for (Node t : lines) {
184:             int best = -1;
185:             long area = Long.MAX_VALUE;
186:             for (int i = 0; i < shapes.size(); i++) {
187:                 Node s = shapes.get(i);
188:                 int cx = t.cx(), cy = t.cy();
189:                 if (cx < s.x || cx > s.x + s.w || cy < s.y || cy > s.y + s.h) continue;
190:                 long a = (long) s.w * s.h;
191:                 if (a < area) { area = a; best = i; }
192:             }
193:             if (best >= 0) {
194:                 // 12.5: أسطر داخل فقاعة صورة/ملصق (تعليق الصورة، أزرار التنزيل) لا تُنقل
195:                 if (shapes.get(best).kind != 'M') in.get(best).add(t);
196:             } else rest.add(t);
197:         }
198:         List<Bubble> out = new ArrayList<>();
199:         for (int i = 0; i < shapes.size(); i++) {
200:             List<Node> l = in.get(i);
201:             if (l.isEmpty()) continue;
202:             l.sort(Comparator.comparingInt((Node t) -> t.y).thenComparingInt(t -> -t.x));
203:             // أسطر على الارتفاع نفسه (اسم الملف والوقت مثلًا) تبقى على سطر واحد بترتيبها من اليمين
204:             StringBuilder sb = new StringBuilder();
205:             Node prev = null;
206:             for (Node t : l) {
207:                 if (prev != null) sb.append(Math.abs(t.cy() - prev.cy()) < Math.max(4, Math.min(t.h, prev.h) / 2) ? " " : "\n");
208:                 sb.append(t.text.trim());
209:                 prev = t;
210:             }
211:             Node s = shapes.get(i);
212:             out.add(bubble(s.x, s.y, s.w, s.h, sb.toString()));

214:         // 12.3: فقاعة لم يُعرف شكلها (لونها قريب جدًا من الخلفية مثلًا) لا تضيع: أسطرها خارج كل الأشكال تُجمَّع كما في 12.1
215:         if (!rest.isEmpty())
216:             for (Bubble b : cluster(rest)) if (letters(b.text) >= 3) out.add(b);
217:         out.sort(Comparator.comparingInt((Bubble b) -> b.y));
218:         return out;

228:     /** أسطر قراءة الصورة إلى فقاعات: فجوة عمودية كبيرة، أو تغيّر الجهة (وارد/صادر)، أو سطر وقت، تبدأ فقاعة جديدة. */
229:     static List<Bubble> cluster(List<Node> lines) {
230:         List<Node> l = new ArrayList<>(lines);
231:         l.sort(Comparator.comparingInt((Node n) -> n.y).thenComparingInt(n -> -n.x));
232:         List<Bubble> out = new ArrayList<>();
233:         List<Node> cur = new ArrayList<>();
234:         int avgH = 0;
235:         for (Node n : l) avgH += n.h;
236:         avgH = l.isEmpty() ? 16 : Math.max(8, avgH / l.size());
237:         for (Node n : l) {
238:             if (!cur.isEmpty()) {
239:                 Node last = cur.get(cur.size() - 1);
240:                 int gap = n.y - (last.y + last.h);
241:                 int bx = Integer.MAX_VALUE, bx2 = Integer.MIN_VALUE;
242:                 for (Node c : cur) {
243:                     bx = Math.min(bx, c.x);
244:                     bx2 = Math.max(bx2, c.x + c.w);
245:                 }
246:                 int overlap = Math.min(bx2, n.x + n.w) - Math.max(bx, n.x);
247:                 // 12.2: سطر على ارتفاع السطر السابق نفسه (الوقت بجانب النص) لا يبدأ فقاعة جديدة مهما بعُد أفقيًا
248:                 boolean sameRow = Math.abs(n.cy() - last.cy()) < Math.max(4, Math.min(n.h, last.h) * 0.6);
249:                 boolean side = !sameRow && overlap < Math.min(n.w, bx2 - bx) * 0.3;
250:                 if ((!sameRow && gap > avgH * 0.75) || side || (!sameRow && WaCuts.timeLine(last.text))) {
251:                     out.add(fromLines(cur));
252:                     cur = new ArrayList<>();
253:                 }
254:             }
255:             cur.add(n);
256:         }
257:         if (!cur.isEmpty()) out.add(fromLines(cur));
258:         return out;
259:     }

269:             if (sb.length() > 0) sb.append('\n');
270:             sb.append(n.text.trim());
271:         }
272:         return bubble(x, y, x2 - x, y2 - y, sb.toString());

299:     public static int matchOffice(String chat, List<String> names) {
300:         String c = key(chat);

307:             double s;

310:             else s = words(c, n);
311:             if (s > score) {
