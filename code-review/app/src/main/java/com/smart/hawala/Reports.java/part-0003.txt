333:     static Report.Section txSection(Report r, String heading, String sheet, List<Model.Tx> txs, Map<Long, String> officeNames,
334:                                     Map<String, Integer> decimals, Map<Long, List<Long>> attachments) {
335:         Report.Section s = r.section(heading, sheet);
336:         s.col("التاريخ", 1.15f, Report.DATE).col("المكتب", 1.5f, Report.TEXT).col("البيان", 2.6f, Report.TEXT)
337:                 .col("لنا", 1.1f, Report.MONEY).col("علينا", 1.1f, Report.MONEY).col("العملة", 0.7f, Report.TEXT)
338:                 .col("الأجرة", 1.3f, Report.TEXT).col("الحالة", 0.8f, Report.TEXT);
339:         boolean shade = false;
340:         for (Model.Tx t : txs) {
341:             int d = dec(decimals, t.currency);
342:             String desc = Model.join(Model.list(Model.nz(t.person), MessageParser.compact(t.description, t.person), "مرجع: " + Model.shortRef(t.uuid)), "\n");
343:             String fee = t.hasFee() ? (Model.FEE_US.equals(t.feeType) ? "لنا " : "علينا ") + Model.fmt(t.fee, dec(decimals, t.feeCurrency)) + " " + t.feeCurrency : "";
344:             boolean us = Model.TYPE_DEBIT.equals(t.type);
345:             String name = officeNames.get(t.officeId);
346:             Report.Row row = s.row(t.isActive() ? (shade ? Report.SHADED : Report.NORMAL) : Report.MUTED, new Report.DateV(t.createdAt), name == null ? "—" : name, desc,
347:                     us ? new Report.Money(t.amount, d) : "", us ? "" : new Report.Money(t.amount, d), t.currency, fee,
348:                     t.isActive() ? (t.replacesId != null ? "معدّلة" : "نشطة") : t.isPending() ? "بانتظار الاعتماد" : t.isRejected() ? "مرفوضة" : "ملغاة");
349:             if (attachments != null) {
350:                 List<Long> ids = attachments.get(t.id);
351:                 if (ids != null && !ids.isEmpty()) row.images = ids;
352:             }
353:             shade = !shade;
354:         }
355:         return s;
356:     }

362:     /**
363:      * يلوّن كل عملة بلونها: أقسام العملة الواحدة (رأس الجدول والعنوان)، والصفوف التي فيها عمود «العملة»
364:      * (خلفية خفيفة وأرقام بلون العملة). لا يغيّر أي رقم أو نص.
365:      */
366:     public static void colorize(Report r, ExportStyle st) {
367:         if (r == null || st == null || !st.curColors) return;
368:         for (Report.Section s : r.sections) {
369:             if (s.accent == 0 && s.heading != null && s.heading.startsWith("العملة: ") && isCode(s.sheetName)) s.accent = st.colorFor(s.sheetName);
370:             int ci = -1;
371:             for (int i = 0; i < s.cols.size(); i++) if ("العملة".equals(s.cols.get(i).title)) ci = i;
372:             if (ci < 0) continue;
373:             for (Report.Row row : s.rows) {
374:                 if (row.style == Report.MUTED || ci >= row.cells.length) continue;
375:                 String code = codeOf(Report.text(row.cells[ci]));
376:                 if (code != null) row.tint = st.colorFor(code);
377:             }
378:         }
379:     }
380: 
381:     static String codeOf(String cell) {
382:         String t = cell == null ? "" : cell.trim();
383:         int i = t.indexOf(' ');
384:         if (i > 0) t = t.substring(0, i);
385:         return isCode(t) ? t : null;
386:     }
387: 
388:     static boolean isCode(String s) {
389:         return s != null && s.matches("[A-Z0-9]{2,10}");
390:     }
