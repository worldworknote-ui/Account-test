693:     public static String change(long now, long before) {
694:         if (before == 0) return now == 0 ? "—" : "جديد";
695:         BigDecimal p = BigDecimal.valueOf(now - before).multiply(BigDecimal.valueOf(100)).divide(BigDecimal.valueOf(Math.abs(before)), 0, RoundingMode.HALF_UP);
696:         return (p.signum() > 0 ? "+" : p.signum() < 0 ? "−" : "") + p.abs().toPlainString() + "%";
697:     }

699:     /** CSV (UTF-8 مع BOM ليفتحه Excel بالعربية): كل قسم جدول، بقيم رقمية خام. */
700:     public static String csv(Report r) {
701:         StringBuilder b = new StringBuilder("\uFEFF");
702:         for (Report.Section s : r.sections) {
703:             if (s.cols.isEmpty()) continue;
704:             b.append(q(s.heading)).append("\r\n");
705:             List<String> h = new ArrayList<>();
706:             for (Report.Col c : s.cols) h.add(q(c.title));
707:             b.append(Model.join(h, ",")).append("\r\n");
708:             for (Report.Row row : s.rows) {
709:                 List<String> cells = new ArrayList<>();
710:                 for (int i = 0; i < s.cols.size(); i++) {
711:                     Object v = i < row.cells.length ? row.cells[i] : "";
712:                     if (v instanceof Report.Money) cells.add(BigDecimal.valueOf(((Report.Money) v).v, ((Report.Money) v).dec).toPlainString());
713:                     else if (v instanceof Report.Bal) cells.add(BigDecimal.valueOf(((Report.Bal) v).v, ((Report.Bal) v).dec).toPlainString());
714:                     else if (v instanceof Report.DateV) cells.add(Model.day(((Report.DateV) v).t));
715:                     else cells.add(q(Report.text(v)));
716:                 }
717:                 b.append(Model.join(cells, ",")).append("\r\n");
718:             }
719:             b.append("\r\n");
720:         }
721:         return b.toString();
722:     }

724:     static String q(String s) {
725:         String v = s == null ? "" : s.replace("\u2066", "").replace("\u2069", "");
726:         if (v.contains(",") || v.contains("\"") || v.contains("\n") || v.startsWith("=") || v.startsWith("+") || v.startsWith("-") || v.startsWith("@"))
727:             v = "\"" + (v.startsWith("=") || v.startsWith("+") || v.startsWith("@") ? "'" : "") + v.replace("\"", "\"\"") + "\"";
728:         return v;
729:     }

770:     public static long[] period(int i, long now) {
771:         long today = startOfDay(now), tomorrow = addDays(today, 1);
772:         switch (i) {
773:             case P_TODAY: return new long[]{today, tomorrow};
774:             case P_YESTERDAY: return new long[]{addDays(today, -1), today};
775:             case P_WEEK: return new long[]{addDays(today, -6), tomorrow};
776:             case P_MONTH: return new long[]{startOfMonth(today), tomorrow};
777:             case P_LAST_MONTH: return new long[]{addMonths(startOfMonth(today), -1), startOfMonth(today)};
778:             case P_3M: return new long[]{addMonths(today, -3), tomorrow};
779:             case P_12M: return new long[]{addMonths(today, -12), tomorrow};
780:             case P_YEAR: {
781:                 Calendar c = Model.cal();
782:                 c.setTimeInMillis(today);
783:                 c.set(Calendar.DAY_OF_YEAR, 1);
784:                 return new long[]{c.getTimeInMillis(), tomorrow};

786:             default: return new long[]{0, 0};
