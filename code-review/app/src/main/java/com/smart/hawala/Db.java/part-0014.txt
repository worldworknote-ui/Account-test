1928:     /** ساقا التحويل كما تُسجَّلان (للمعاينة والتسجيل بالقواعد نفسها). */
1929:     public TxInput[] transferLegs(TransferInput in) {
1930:         if (in.from == in.to) throw new IllegalArgumentException("المكتب طالب الحركة والمكتب المنفّذ يجب أن يكونا مختلفين");
1931:         TxInput a = new TxInput(), b = new TxInput();
1932:         Model.Office fo = office(in.from), to = office(in.to);
1933:         for (TxInput x : new TxInput[]{a, b}) {
1934:             x.currency = in.currency;
1935:             x.amount = in.amount;
1936:             x.person = Model.nz(in.person);
1937:             x.notes = Model.nz(in.notes);
1938:             x.when = in.when;

1942:         // طالب الحركة: مدين لنا بالمبلغ (وأجرتنا عليه إن وُجدت «الأجرة لنا»)
