63:     public static final class BackupException extends Exception {
64:         public final Stage stage;
65: 
66:         public BackupException(Stage stage, String message, Throwable cause) {
67:             super(message, cause);
68:             this.stage = stage;
69:         }
70: 
71:         public String userMessage() {
72:             String m = getMessage();
73:             return "المرحلة: " + stage.label + "\n\n" + (m == null ? "خطأ غير معروف" : m);
74:         }

81:     static BackupException wrap(Stage stage, Throwable t) {
82:         if (t instanceof BackupException) return (BackupException) t;
83:         String m;
84:         if (t instanceof SecureStreams.WrongPasswordException || t instanceof SecureStreams.CorruptBackupException) m = t.getMessage();
85:         else if (t instanceof OutOfMemoryError) m = "الذاكرة لا تكفي. أغلق التطبيقات الأخرى ثم أعد المحاولة.";
86:         else if (t instanceof IOException && t.getMessage() != null && t.getMessage().toLowerCase(Locale.US).contains("space")) m = "لا توجد مساحة كافية في التخزين.";
87:         else if (t.getMessage() != null && !t.getMessage().trim().isEmpty()) m = t.getMessage();
88:         else m = t.getClass().getSimpleName();
89:         return new BackupException(stage, m, t);
90:     }

114:     /**
115:      * ينشئ نسخة مشفّرة داخل التطبيق ويتحقق منها بالكامل. لا يعدّل مصفوفة كلمة المرور الممرّرة؛
116:      * يعمل على نسخ مستقلة ويمسحها في النهاية. المستدعي مسؤول عن مسح مصفوفته.
117:      */
118:     public static Created createLocal(Context ctx, Db db, char[] password, String appVersion, Progress p) throws BackupException {
119:         if (password == null || password.length < 8) throw new BackupException(Stage.PREPARE, "كلمة المرور يجب ألا تقل عن 8 محارف", null);
120:         File dir = backupsDir(ctx);
121:         long free = dir.getUsableSpace();
122:         if (free > 0 && free < 5L * 1024 * 1024) throw new BackupException(Stage.PREPARE, "مساحة التخزين المتبقية في الهاتف قليلة جدًا. احذف بعض الملفات ثم أعد المحاولة.", null);
123:         String name = newName();
124:         File tmp = new File(dir, name + ".tmp");
125:         File out = new File(dir, name);
126:         char[] pwWrite = password.clone();
127:         char[] pwVerify = password.clone();
128:         Created c = new Created();
129:         try {
130:             p.update("جارٍ تجهيز البيانات…");
131:             c.counts = db.counts();
132:             BackupFormat.Source src = db.backupSource();
133: 
134:             p.update("جارٍ التشفير والكتابة…");
135:             SecureStreams.EncryptingOutputStream enc = null;
136:             try (FileOutputStream fos = new FileOutputStream(tmp)) {
137:                 BufferedOutputStream bos = new BufferedOutputStream(fos, 65536);
138:                 // 10.14: إن فُعّل «رمز استرداد النسخ» تُفتح النسخة بكلمة مرورها أو بالرمز
139:                 enc = new SecureStreams.EncryptingOutputStream(bos, pwWrite, BackupRecovery.key(db));
140:                 c.result = BackupFormat.write(src, enc, appVersion, Db.VER);
141:                 enc.close();
142:                 bos.flush();
143:                 fos.getFD().sync();
144:             } catch (Throwable t) {
145:                 if (enc != null) enc.abort();
146:                 tmp.delete();
147:                 throw wrap(Stage.ENCRYPT, t);
148:             }
149: 
150:             p.update("جارٍ التحقق من النسخة (فك كامل)…");
151:             try {
152:                 verifyFile(tmp, pwVerify, c.result);
153:             } catch (Throwable t) {
154:                 tmp.delete();
155:                 throw wrap(Stage.VERIFY, t);
156:             }
157:             if (!tmp.renameTo(out)) {
158:                 tmp.delete();
159:                 throw new BackupException(Stage.ENCRYPT, "تعذر حفظ ملف النسخة داخل التطبيق", null);
160:             }
161:             c.file = out;
162:             c.size = out.length();
163:             try {
164:                 c.sha256 = sha256(out);
165:             } catch (Exception e) {
166:                 throw wrap(Stage.VERIFY, e);
167:             }
168:             pruneLocal(ctx, 3);
169:             return c;
170:         } finally {
171:             Arrays.fill(pwWrite, '\0');
172:             Arrays.fill(pwVerify, '\0');
173:         }
174:     }
