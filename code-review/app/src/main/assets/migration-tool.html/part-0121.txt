750:     sheets.forEach(function (sh) {
751:       var rows = sh.rows;
752:       for (var r = sec.headerRow + 1; r < rows.length; r++) {
753:         var row = rows[r];

755:         if (isBlankRow(row)) continue;
756:         model.rowsRead++;
757:         var name = text(row[sec.cols.office]);
758:         var label = rowLabel(row);
759:         if (TOTAL_RE.test(name) || (!name && TOTAL_RE.test(label))) { skip(model, src, "صف مجموع"); continue; }
760:         var values = [];
761:         if (sec.layout === "wide") {
762:           (sec.wide || []).forEach(function (w) {
763:             if (w.code && w.code !== "__skip") values.push({ code: w.code, v: row[w.col], signed: true });
764:           });
765:         } else {
766:           var code = rowCurrency(cfg, sec, row, model, src);
767:           if (code === null) continue;
768:           if (code === "__skip") { skip(model, src, "عملة مستبعدة"); continue; }
769:           if (sec.layout === "split") {
770:             values.push({ code: code, v: row[sec.cols.debit], dir: "D" });
771:             values.push({ code: code, v: row[sec.cols.credit], dir: "C" });
772:           } else values.push({ code: code, v: row[sec.cols.amount], signed: true });
773:         }
774:         var any = false, bad = false, net = {};
775:         values.forEach(function (x) {
776:           var p = parseAmount(x.v);
777:           if (p.empty) return;
778:           if (!p.ok) { err(model, src, "مبلغ غير مفهوم: «" + text(x.v) + "» — " + p.reason); bad = true; return; }
779:           if (p.zero) return;
780:           if (!x.signed && p.neg) { err(model, src, "قيمة سالبة في عمود " + (x.dir === "D" ? "لنا/مدين" : "علينا/دائن") + ": «" + text(x.v) + "»"); bad = true; return; }
781:           var m = amountMinor(cfg, model, src, p, x.code, "المبلغ");
782:           if (m === null) { bad = true; return; }
783:           var pos = sec.positiveIs || "D";
784:           var dir = x.signed ? (p.neg ? (pos === "D" ? "C" : "D") : pos) : x.dir;
785:           dir = flip(dir, cfg);
786:           net[x.code] = (net[x.code] || 0n) + (dir === "D" ? m : -m);
787:           any = true;
788:         });
789:         if (bad) continue;
790:         if (!name) {
791:           if (any) err(model, src, "يوجد رصيد بلا اسم مكتب");
792:           else skip(model, src, "صف بلا مكتب ولا رصيد");
793:           continue;
794:         }
795:         var oi = officeFor(model, name, sec.cols.phone !== undefined ? text(row[sec.cols.phone]) : "", cfg);
796:         if (!any) { skip(model, src, "مكتب بلا رصيد (أُضيف المكتب فقط)"); model.offices[oi].emptyOk = true; continue; }
797:         Object.keys(net).forEach(function (code) {
798:           if (net[code] === 0n) return;
799:           model.openings.push({ office: oi, code: code, minor: net[code], when: sec.date || null, src: src });
800:           model.offices[oi].openings++;
801:           model.used[code] = true;
802:         });
803:       }
804:     });
805:   }
