1499:     for (var n = 0; n < 256; n++) {
1500:       var c = n;
1501:       for (var k = 0; k < 8; k++) c = c & 1 ? 0xedb88320 ^ (c >>> 1) : c >>> 1;
1502:       t[n] = c >>> 0;
1503:     }
1504:     return t;

1508:     var c = 0xffffffff;
1509:     for (var i = 0; i < b.length; i++) c = CRC_TABLE[(c ^ b[i]) & 0xff] ^ (c >>> 8);
1510:     return (c ^ 0xffffffff) >>> 0;

1515:   function zipStored(files, when) {

1517:     var dosTime = (d.getHours() << 11) | (d.getMinutes() << 5) | (d.getSeconds() >> 1);
1518:     var dosDate = ((d.getFullYear() - 1980) << 9) | ((d.getMonth() + 1) << 5) | d.getDate();
1519:     var parts = [], central = [], offset = 0;
1520:     files.forEach(function (f) {
1521:       var name = utf8(f.name), data = f.bytes, crc = crc32(data);
1522:       var h = new DataView(new ArrayBuffer(30));
1523:       h.setUint32(0, 0x04034b50, true); h.setUint16(4, 20, true); h.setUint16(6, 0x0800, true); h.setUint16(8, 0, true);
1524:       h.setUint16(10, dosTime, true); h.setUint16(12, dosDate, true); h.setUint32(14, crc, true);
1525:       h.setUint32(18, data.length, true); h.setUint32(22, data.length, true); h.setUint16(26, name.length, true); h.setUint16(28, 0, true);
1526:       parts.push(new Uint8Array(h.buffer), name, data);
1527:       var c = new DataView(new ArrayBuffer(46));
1528:       c.setUint32(0, 0x02014b50, true); c.setUint16(4, 20, true); c.setUint16(6, 20, true); c.setUint16(8, 0x0800, true); c.setUint16(10, 0, true);
1529:       c.setUint16(12, dosTime, true); c.setUint16(14, dosDate, true); c.setUint32(16, crc, true); c.setUint32(20, data.length, true);
1530:       c.setUint32(24, data.length, true); c.setUint16(28, name.length, true); c.setUint16(30, 0, true); c.setUint16(32, 0, true);
1531:       c.setUint16(34, 0, true); c.setUint16(36, 0, true); c.setUint32(38, 0, true); c.setUint32(42, offset, true);
1532:       central.push(new Uint8Array(c.buffer), name);
1533:       offset += 30 + name.length + data.length;
1534:     });
1535:     var cdSize = central.reduce(function (s, x) { return s + x.length; }, 0);
1536:     var e = new DataView(new ArrayBuffer(22));
1537:     e.setUint32(0, 0x06054b50, true); e.setUint16(4, 0, true); e.setUint16(6, 0, true); e.setUint16(8, files.length, true);
1538:     e.setUint16(10, files.length, true); e.setUint32(12, cdSize, true); e.setUint32(16, offset, true); e.setUint16(20, 0, true);
1539:     return concat(parts.concat(central, [new Uint8Array(e.buffer)]));

1583:   async function encrypt(plain, password, opts, subtle, rand) {
1584:     opts = opts || {};
1585:     var iterations = opts.iterations || 310000;
1586:     var chunk = opts.chunk || 262144;
1587:     var salt = opts.salt || rand(16);
1588:     var prefix = opts.prefix || rand(8);
1589:     var header = concat([utf8("SMARTBK5"), be32(1), be32(iterations), be32(16), salt, be32(8), prefix, be32(chunk)]);
1590:     var base = await subtle.importKey("raw", utf8(password), "PBKDF2", false, ["deriveKey"]);
1591:     var key = await subtle.deriveKey({ name: "PBKDF2", hash: "SHA-256", salt: salt, iterations: iterations }, base,
1592:       { name: "AES-GCM", length: 256 }, false, ["encrypt"]);
1593:     var out = [header];
1594:     var v = new Uint8Array(await subtle.encrypt({ name: "AES-GCM", iv: iv(prefix, 0xffffffff), additionalData: concat([header, utf8("VERIFY")]), tagLength: 128 },
1595:       key, new Uint8Array(16)));
1596:     out.push(be32(v.length), v);
1597:     var count = Math.max(1, Math.ceil(plain.length / chunk));
1598:     for (var i = 0; i < count; i++) {
1599:       var last = i === count - 1 ? 1 : 0;
1600:       var part = plain.subarray(i * chunk, Math.min(plain.length, (i + 1) * chunk));
1601:       var aad = new Uint8Array(header.length + 9);
1602:       aad.set(header, 0);
1603:       // رقم الجزء 8 بايت (long) ثم علامة «الأخير»
1604:       aad.set(be32(0), header.length);
1605:       aad.set(be32(i), header.length + 4);
1606:       aad[header.length + 8] = last;
1607:       var ct = new Uint8Array(await subtle.encrypt({ name: "AES-GCM", iv: iv(prefix, i), additionalData: aad, tagLength: 128 }, key, part));
1608:       out.push(new Uint8Array([last]), be32(ct.length), ct);
1609:     }
1610:     return concat(out);
1611:   }
