1035:   function stmtMeta(lines) {
1036:     var meta = { account: null, curText: "", from: null, to: null, title: [] };
1037:     var texts = [];
1038:     lines.forEach(function (row) {
1039:       var cells = row.filter(function (c) { return c && text(c); });

1055:     var curIdx = -1;

1057:     var accIdx = -1;
1058:     for (var j = 0; j < texts.length; j++) {

1064:     if (accIdx >= 0) meta.account = splitAccount(texts[accIdx]);
1065:     if (curIdx >= 0) meta.curText = clean(texts[curIdx]);

1068:       for (var n = accIdx + 1; n < texts.length; n++) if (n !== accIdx) { meta.curText = clean(texts[n]); break; }

1070:     meta.title = texts.filter(function (x, idx) { return idx !== accIdx && idx !== curIdx; });

1086:       else if (/متوازن|متعادل|صفر|balanced/i.test(t)) side = side || "0";

1113:             var m0 = stmtMeta(since);
