1528:     var c = 0xffffffff;

1535:   function zipStored(files, when) {
1536:     var d = new Date(when);
1537:     var dosTime = (d.getHours() << 11) | (d.getMinutes() << 5) | (d.getSeconds() >> 1);
1538:     var dosDate = ((d.getFullYear() - 1980) << 9) | ((d.getMonth() + 1) << 5) | d.getDate();
1539:     var parts = [], central = [], offset = 0;
1540:     files.forEach(function (f) {
1541:       var name = utf8(f.name), data = f.bytes, crc = crc32(data);
1542:       var h = new DataView(new ArrayBuffer(30));
1543:       h.setUint32(0, 0x04034b50, true); h.setUint16(4, 20, true); h.setUint16(6, 0x0800, true); h.setUint16(8, 0, true);
1544:       h.setUint16(10, dosTime, true); h.setUint16(12, dosDate, true); h.setUint32(14, crc, true);
1545:       h.setUint32(18, data.length, true); h.setUint32(22, data.length, true); h.setUint16(26, name.length, true); h.setUint16(28, 0, true);
1546:       parts.push(new Uint8Array(h.buffer), name, data);
1547:       var c = new DataView(new ArrayBuffer(46));
1548:       c.setUint32(0, 0x02014b50, true); c.setUint16(4, 20, true); c.setUint16(6, 20, true); c.setUint16(8, 0x0800, true); c.setUint16(10, 0, true);
1549:       c.setUint16(12, dosTime, true); c.setUint16(14, dosDate, true); c.setUint32(16, crc, true); c.setUint32(20, data.length, true);
1550:       c.setUint32(24, data.length, true); c.setUint16(28, name.length, true); c.setUint16(30, 0, true); c.setUint16(32, 0, true);
1551:       c.setUint16(34, 0, true); c.setUint16(36, 0, true); c.setUint32(38, 0, true); c.setUint32(42, offset, true);
1552:       central.push(new Uint8Array(c.buffer), name);
1553:       offset += 30 + name.length + data.length;
1554:     });
1555:     var cdSize = central.reduce(function (s, x) { return s + x.length; }, 0);
1556:     var e = new DataView(new ArrayBuffer(22));
1557:     e.setUint32(0, 0x06054b50, true); e.setUint16(4, 0, true); e.setUint16(6, 0, true); e.setUint16(8, files.length, true);
1558:     e.setUint16(10, files.length, true); e.setUint32(12, cdSize, true); e.setUint32(16, offset, true); e.setUint16(20, 0, true);
1559:     return concat(parts.concat(central, [new Uint8Array(e.buffer)]));
1560:   }

1603:   async function encrypt(plain, password, opts, subtle, rand) {
1604:     opts = opts || {};
1605:     var iterations = opts.iterations || 310000;
1606:     var chunk = opts.chunk || 262144;
1607:     var salt = opts.salt || rand(16);
1608:     var prefix = opts.prefix || rand(8);
1609:     var header = concat([utf8("SMARTBK5"), be32(1), be32(iterations), be32(16), salt, be32(8), prefix, be32(chunk)]);
1610:     var base = await subtle.importKey("raw", utf8(password), "PBKDF2", false, ["deriveKey"]);
1611:     var key = await subtle.deriveKey({ name: "PBKDF2", hash: "SHA-256", salt: salt, iterations: iterations }, base,
1612:       { name: "AES-GCM", length: 256 }, false, ["encrypt"]);
1613:     var out = [header];
1614:     var v = new Uint8Array(await subtle.encrypt({ name: "AES-GCM", iv: iv(prefix, 0xffffffff), additionalData: concat([header, utf8("VERIFY")]), tagLength: 128 },
1615:       key, new Uint8Array(16)));
1616:     out.push(be32(v.length), v);
1617:     var count = Math.max(1, Math.ceil(plain.length / chunk));
1618:     for (var i = 0; i < count; i++) {
1619:       var last = i === count - 1 ? 1 : 0;
1620:       var part = plain.subarray(i * chunk, Math.min(plain.length, (i + 1) * chunk));
1621:       var aad = new Uint8Array(header.length + 9);
1622:       aad.set(header, 0);
1623:       // رقم الجزء 8 بايت (long) ثم علامة «الأخير»
1624:       aad.set(be32(0), header.length);
1625:       aad.set(be32(i), header.length + 4);
1626:       aad[header.length + 8] = last;
1627:       var ct = new Uint8Array(await subtle.encrypt({ name: "AES-GCM", iv: iv(prefix, i), additionalData: aad, tagLength: 128 }, key, part));
1628:       out.push(new Uint8Array([last]), be32(ct.length), ct);
1629:     }
1630:     return concat(out);
1631:   }
