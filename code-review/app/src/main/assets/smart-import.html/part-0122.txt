967:   function recordBalance(model, cfg, sec, officeName, row, balCell, src, curFn) {
968:     var p = parseAmount(balCell);
969:     if (!p.ok || p.empty) return;
970:     var cur = curFn();

972:     var def = curDef(cfg, cur);
973:     if (!def) return;
974:     var m = toMinor(p, def.dec);
975:     if (m === null) {
976:       var f = p.frac.slice(0, def.dec);
977:       m = toMinor({ int: p.int, frac: f }, def.dec);
978:     }
979:     var pos = sec.balancePositiveIs || "D";
980:     var signed = (p.neg ? (pos === "D" ? -m : m) : (pos === "D" ? m : -m));
981:     if (cfg.reverse) signed = -signed;
982:     var key = keyOf(cfg, officeName) + "|" + cur;
983:     model.lastBalance = model.lastBalance || {};
984:     model.lastBalance[key] = { minor: signed, src: src };
985:   }

1016:     return model;
