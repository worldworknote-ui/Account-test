780:         var values = [];

784:           });
785:         } else {
786:           var code = rowCurrency(cfg, sec, row, model, src);
787:           if (code === null) continue;
788:           if (code === "__skip") { skip(model, src, "عملة مستبعدة"); continue; }
789:           if (sec.layout === "split") {

791:             values.push({ code: code, v: row[sec.cols.credit], dir: "C" });
792:           } else values.push({ code: code, v: row[sec.cols.amount], signed: true });
793:         }
794:         var any = false, bad = false, net = {};
795:         values.forEach(function (x) {
796:           var p = parseAmount(x.v);
797:           if (p.empty) return;

799:           if (p.zero) return;
800:           if (!x.signed && p.neg) { err(model, src, "قيمة سالبة في عمود " + (x.dir === "D" ? "لنا/مدين" : "علينا/دائن") + ": «" + text(x.v) + "»"); bad = true; return; }
801:           var m = amountMinor(cfg, model, src, p, x.code, "المبلغ");
802:           if (m === null) { bad = true; return; }

807:           any = true;
808:         });
809:         if (bad) continue;
810:         if (!name) {
811:           if (any) err(model, src, "يوجد رصيد بلا اسم مكتب");

813:           continue;
814:         }
815:         var oi = officeFor(model, name, sec.cols.phone !== undefined ? text(row[sec.cols.phone]) : "", cfg);
816:         if (!any) { skip(model, src, "مكتب بلا رصيد (أُضيف المكتب فقط)"); model.offices[oi].emptyOk = true; continue; }
817:         Object.keys(net).forEach(function (code) {
818:           if (net[code] === 0n) return;
819:           model.openings.push({ office: oi, code: code, minor: net[code], when: sec.date || null, src: src });
820:           model.offices[oi].openings++;

822:         });
823:       }
824:     });
825:   }
