2453:   function balanceMinor(cell, dec) {
2454:     if (!cell) return null;
2455:     if (cell.n !== undefined && typeof cell.n === "number") {
2456:       if (!isFinite(cell.n)) return null;
2457:       var f = Math.pow(10, dec), x = cell.n * f, r = Math.round(x);
2458:       if (Math.abs(x - r) > 1e-3 || Math.abs(r) > 9e15) return null;
2459:       return BigInt(r);
2460:     }
2461:     var p = C.parseAmount(cell);
2462:     if (!p.ok || p.empty) return null;
2463:     var m = C.toMinor(p, dec);
2464:     if (m === null) return null;
2465:     var neg = p.neg;
2466:     if (p.hadText) {
2467:       var g = guessDir(clean(normDigits(text(cell))).replace(/[\d.,()\-+]/g, " "));
2468:       if (!g) return null;
2469:       if (g === "C") neg = !neg;
2470:     }
2471:     return neg ? -m : m;
2472:   }
