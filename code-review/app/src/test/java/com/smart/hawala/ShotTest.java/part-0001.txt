154:     static View find(View v, java.util.function.Predicate<View> p) {
155:         if (v == null) return null;
156:         if (p.test(v)) return v;
157:         if (v instanceof android.view.ViewGroup) {
158:             android.view.ViewGroup g = (android.view.ViewGroup) v;
159:             for (int i = 0; i < g.getChildCount(); i++) {
160:                 View r = find(g.getChildAt(i), p);
161:                 if (r != null) return r;
162:             }
163:         }
164:         return null;
165:     }
